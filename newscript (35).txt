--------------------------------------------------
-- QCS V4.0 LINE ONLY MOBILE
-- CELULAR + NOTEBOOK
-- TENDENCIA + FORCA + LATERALIZACAO
-- SOMENTE LINHAS | SEM SETAS | SEM TEXTOS
-- SEM MARCADORES | SEM ORDENS AUTOMATICAS
--------------------------------------------------

instrument {
    name = "QCS V4.0 LINE ONLY MOBILE",
    short_name = "QCS LINE V4",
    overlay = true
}

--------------------------------------------------
-- CONFIGURACAO
--------------------------------------------------

periodo = input(5, "EMA Rapida", input.integer, 1, 50)
minScore = input(55, "Score Direcional", input.integer, 1, 100)

--------------------------------------------------
-- CORES
--------------------------------------------------

verde = "#00FF00"
vermelho = "#FF2020"
dourado = "#FFD700"
laranja = "#CC5500"
cinza = "#808080"
azul = "#00BFFF"

--------------------------------------------------
-- MEDIAS MOVEIS
--------------------------------------------------

ema5 = ema(close, periodo)
ema13 = ema(close, 13)
ema21 = ema(close, 21)
ema50 = ema(close, 50)

--------------------------------------------------
-- INDICADORES
--------------------------------------------------

rsi7 = rsi(close, 7)
atr7 = sma(tr, 7)

--------------------------------------------------
-- FORCA DA VELA
--------------------------------------------------

amplitude = high - low
corpo = abs(close - open)

forcaVela = 0

if amplitude > 0 then
    forcaVela = corpo / amplitude * 100
end

--------------------------------------------------
-- INCLINACAO DAS MEDIAS
--------------------------------------------------

inclinacao5 = ema5 - ema5[1]
inclinacao13 = ema13 - ema13[1]
inclinacao21 = ema21 - ema21[1]
inclinacao50 = ema50 - ema50[1]

anguloAlta =
    inclinacao13 > 0 and
    inclinacao21 > 0

anguloBaixa =
    inclinacao13 < 0 and
    inclinacao21 < 0

--------------------------------------------------
-- CRUZAMENTO REAL EMA 5 / EMA 13
-- USADO COMO GATILHO NO SISTEMA INTEGRADO
--------------------------------------------------

cruzouAlta =
    ema5 > ema13 and
    ema5[1] <= ema13[1]

cruzouBaixa =
    ema5 < ema13 and
    ema5[1] >= ema13[1]

--------------------------------------------------
-- TENDENCIA PRINCIPAL
--------------------------------------------------

tendAlta =
    ema13 > ema21 and
    ema21 > ema50 and
    anguloAlta

tendBaixa =
    ema13 < ema21 and
    ema21 < ema50 and
    anguloBaixa

--------------------------------------------------
-- MICRO TENDENCIA
--------------------------------------------------

microAlta =
    ema5 > ema13 and
    inclinacao5 > 0

microBaixa =
    ema5 < ema13 and
    inclinacao5 < 0

--------------------------------------------------
-- MOMENTUM E ACELERACAO
--------------------------------------------------

momentumAlta = close > close[1]
momentumBaixa = close < close[1]

acelAlta =
    close - close[1] >
    close[1] - close[2]

acelBaixa =
    close[1] - close >
    close[2] - close[1]

--------------------------------------------------
-- VOLATILIDADE
--------------------------------------------------

volatilidadeValida = atr7 > 0

--------------------------------------------------
-- SUPORTE E RESISTENCIA
-- NIVEIS CALCULADOS SEM USAR A VELA ATUAL
--------------------------------------------------

resistencia = highest(high[1], 10)
suporte = lowest(low[1], 10)

rompeAlta = close > resistencia
rompeBaixa = close < suporte

--------------------------------------------------
-- FILTRO DE LATERALIZACAO
--------------------------------------------------

mediasJuntas =
    atr7 > 0 and
    abs(ema5 - ema13) < atr7 * 0.10

inclinacaoFraca =
    atr7 > 0 and
    abs(inclinacao21) < atr7 * 0.03

rsiNeutro =
    rsi7 >= 45 and
    rsi7 <= 55

lateral =
    volatilidadeValida and
    mediasJuntas and
    inclinacaoFraca and
    rsiNeutro

--------------------------------------------------
-- PONTUACAO DIRECIONAL
--------------------------------------------------

scoreCall = 0
scorePut = 0

-- Tendencia principal

if tendAlta then
    scoreCall = scoreCall + 20
end

if tendBaixa then
    scorePut = scorePut + 20
end

-- Micro tendencia

if microAlta then
    scoreCall = scoreCall + 15
end

if microBaixa then
    scorePut = scorePut + 15
end

-- Inclinacao das medias

if anguloAlta then
    scoreCall = scoreCall + 10
end

if anguloBaixa then
    scorePut = scorePut + 10
end

-- Momentum

if momentumAlta then
    scoreCall = scoreCall + 10
end

if momentumBaixa then
    scorePut = scorePut + 10
end

-- Aceleracao

if acelAlta then
    scoreCall = scoreCall + 10
end

if acelBaixa then
    scorePut = scorePut + 10
end

-- RSI

if rsi7 > 52 then
    scoreCall = scoreCall + 10
end

if rsi7 < 48 then
    scorePut = scorePut + 10
end

-- Forca do candle

if forcaVela >= 60 and close > open then
    scoreCall = scoreCall + 5
end

if forcaVela >= 60 and close < open then
    scorePut = scorePut + 5
end

-- Rompimento de estrutura

if rompeAlta then
    scoreCall = scoreCall + 15
end

if rompeBaixa then
    scorePut = scorePut + 15
end

-- Cruzamento real

if cruzouAlta then
    scoreCall = scoreCall + 10
end

if cruzouBaixa then
    scorePut = scorePut + 10
end

--------------------------------------------------
-- FILTRO DE CONFLITO
--------------------------------------------------

conflito =
    (tendAlta and microBaixa) or
    (tendBaixa and microAlta)

--------------------------------------------------
-- CONTEXTO DIRECIONAL
--------------------------------------------------

callContexto =
    scoreCall >= minScore and
    scoreCall > scorePut and
    not lateral and
    not conflito

putContexto =
    scorePut >= minScore and
    scorePut > scoreCall and
    not lateral and
    not conflito

--------------------------------------------------
-- MOVIMENTO FORTE
--------------------------------------------------

forteAlta =
    tendAlta and
    microAlta and
    momentumAlta and
    acelAlta and
    forcaVela >= 60 and
    scoreCall >= 70 and
    not lateral and
    not conflito

forteBaixa =
    tendBaixa and
    microBaixa and
    momentumBaixa and
    acelBaixa and
    forcaVela >= 60 and
    scorePut >= 70 and
    not lateral and
    not conflito

--------------------------------------------------
-- LINHAS DO GRAFICO
-- SEM SETAS, TEXTOS OU MARCADORES
--------------------------------------------------

-- Referencia principal

plot(
    ema21,
    "EMA21 BASE",
    cinza,
    1
)

-- Referencia de tendencia longa

plot(
    ema50,
    "EMA50 FILTRO",
    azul,
    1
)

-- Tendencia de alta

plot(
    callContexto and ema5 or nil,
    "CONTEXTO ALTA",
    verde,
    2
)

-- Tendencia de baixa

plot(
    putContexto and ema5 or nil,
    "CONTEXTO BAIXA",
    vermelho,
    2
)

-- Movimento forte de alta ou baixa

plot(
    (forteAlta or forteBaixa) and ema5 or nil,
    "MOVIMENTO FORTE",
    dourado,
    5
)

-- Lateralizacao

plot(
    lateral and ema13 or nil,
    "MERCADO LATERAL",
    laranja,
    4
)

--------------------------------------------------
-- FIM QCS V4.0 LINE ONLY MOBILE
--------------------------------------------------
