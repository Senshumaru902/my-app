//@version=6
strategy(
     "NEXT CANDLE PRO V37.1 - MOTOR PREDITIVO CORRIGIDO",
     overlay=true,
     max_labels_count=500,
     pyramiding=0,
     calc_on_every_tick=true,
     process_orders_on_close=true)

// =====================================================
// NEXT CANDLE PRO V37.1
// MOTOR:
// ESTRUTURA
// MICRO TENDENCIA
// SUPORTE / RESISTENCIA
// PADROES
// ROMPIMENTO
// FORCA
// MOMENTUM
// ESPACO
// RISCO
// MEMORIA DE ESTABILIDADE
// DECISAO
// =====================================================


// =====================================================
// 1. CONFIGURACOES
// =====================================================

grupoModo = "MODO"

sensibilidade = input.string(
     "Equilibrada",
     "Sensibilidade",
     options=["Cautelosa", "Equilibrada", "Rapida"],
     group=grupoModo)

usarMemoria = input.bool(true, "Memoria de estabilidade", group=grupoModo)
estabilidadeMinima = input.int(2, "Estabilidade minima", minval=1, maxval=10, group=grupoModo)
forcaInversao = input.float(15.0, "Forca minima para inverter", minval=1.0, maxval=50.0, group=grupoModo)
toleranciaRuido = input.float(0.10, "Tolerancia de ruido", minval=0.01, maxval=1.0, step=0.01, group=grupoModo)

grupoVisual = "VISUAL"

mostrarPainel = input.bool(true, "Mostrar painel", group=grupoVisual)
mostrarSinais = input.bool(true, "Mostrar sinais", group=grupoVisual)
mostrarPadroes = input.bool(true, "Mostrar padroes", group=grupoVisual)
mostrarMedias = input.bool(true, "Mostrar medias", group=grupoVisual)

grupoSR = "SUPORTE E RESISTENCIA"

periodoSR = input.int(20, "Periodo SR", minval=5, maxval=200, group=grupoSR)
periodoMicro = input.int(5, "Periodo micro", minval=2, maxval=30, group=grupoSR)
pivotLen = input.int(3, "Pivot", minval=1, maxval=10, group=grupoSR)
distanciaMinTestes = input.int(3, "Distancia minima entre testes", minval=1, maxval=50, group=grupoSR)
validadePadrao = input.int(15, "Validade do padrao", minval=3, maxval=100, group=grupoSR)
toleranciaPadraoATR = input.float(0.35, "Tolerancia do padrao ATR", minval=0.05, maxval=2.0, step=0.05, group=grupoSR)

grupoCandle = "CANDLE"

corpoMinATR = input.float(0.35, "Corpo minimo ATR", minval=0.05, maxval=2.0, step=0.05, group=grupoCandle)
posicaoFechoCALL = input.float(0.65, "Fecho forte alta", minval=0.50, maxval=0.95, step=0.05, group=grupoCandle)
posicaoFechoPUT = input.float(0.35, "Fecho forte baixa", minval=0.05, maxval=0.50, step=0.05, group=grupoCandle)
pavioMax = input.float(1.5, "Pavio maximo / corpo", minval=0.5, maxval=5.0, step=0.1, group=grupoCandle)

grupoIndicadores = "INDICADORES"

rsiAlto = input.float(52.0, "RSI minimo alta", minval=40, maxval=70, group=grupoIndicadores)
rsiBaixo = input.float(48.0, "RSI maximo baixa", minval=30, maxval=60, group=grupoIndicadores)
adxMin = input.float(18.0, "ADX minimo", minval=5, maxval=50, group=grupoIndicadores)

grupoEspaco = "ESPACO E RISCO"

minSpaceATR = input.float(0.8, "Espaco minimo ATR", minval=0.1, maxval=10.0, step=0.1, group=grupoEspaco)
maxExtensionATR = input.float(3.0, "Extensao maxima ATR", minval=0.5, maxval=10.0, step=0.1, group=grupoEspaco)

grupoOperacao = "OPERACAO"

cooldownBars = input.int(2, "Cooldown", minval=0, maxval=50, group=grupoOperacao)
barraExpiracao = input.int(1, "Expiracao em velas", minval=1, maxval=20, group=grupoOperacao)


// =====================================================
// 2. INDICADORES
// =====================================================

ema5 = ta.ema(close, 5)
ema13 = ta.ema(close, 13)
ema21 = ta.ema(close, 21)
ema50 = ta.ema(close, 50)

rsi = ta.rsi(close, 7)

atr = ta.atr(7)
atr14 = ta.atr(14)

volumeMedia = ta.sma(volume, 20)

[diPlus, diMinus, adx] = ta.dmi(14, 14)


// =====================================================
// 3. SEGURANCA ATR
// =====================================================

atrSeguro = na(atr) or atr <= 0 ? syminfo.mintick : atr
atr14Seguro = na(atr14) or atr14 <= 0 ? atrSeguro : atr14

// =====================================================
// 4. ESTRUTURA DO CANDLE
// =====================================================

rangeCandle = high - low
corpo = math.abs(close - open)

rangeSeguro = rangeCandle > syminfo.mintick ? rangeCandle : syminfo.mintick
corpoSeguro = corpo > syminfo.mintick ? corpo : syminfo.mintick

fechoPosicao = (close - low) / rangeSeguro

pavioSuperior = high - math.max(open, close)
pavioInferior = math.min(open, close) - low

pavioSuperiorRel = pavioSuperior / corpoSeguro
pavioInferiorRel = pavioInferior / corpoSeguro

corpoForte = corpo >= atrSeguro * corpoMinATR

candleAltaForte =
     close > open and
     corpoForte and
     fechoPosicao >= posicaoFechoCALL and
     pavioSuperiorRel <= pavioMax

candleBaixaForte =
     close < open and
     corpoForte and
     fechoPosicao <= posicaoFechoPUT and
     pavioInferiorRel <= pavioMax

candleAltaNormal = close > open
candleBaixaNormal = close < open


// =====================================================
// 5. TENDENCIA PRINCIPAL
// =====================================================

tendenciaAlta =
     ema5 > ema13 and
     ema13 > ema21 and
     ema21 > ema50

tendenciaBaixa =
     ema5 < ema13 and
     ema13 < ema21 and
     ema21 < ema50

tendenciaAltaModerada =
     ema5 > ema13 and
     ema13 > ema21

tendenciaBaixaModerada =
     ema5 < ema13 and
     ema13 < ema21


// =====================================================
// 6. MICRO TENDENCIA
// =====================================================

microAltaEMA = ema5 > ema13
microBaixaEMA = ema5 < ema13

microAltaPreco = close > ema5
microBaixaPreco = close < ema5

microHH = high > high[1]
microLL = low < low[1]

microHL = low > low[1]
microLH = high < high[1]

microAlta =
     microAltaEMA and
     microAltaPreco and
     (microHH or microHL)

microBaixa =
     microBaixaEMA and
     microBaixaPreco and
     (microLL or microLH)

microAltaForte =
     microAltaEMA and
     microAltaPreco and
     microHH and
     microHL

microBaixaForte =
     microBaixaEMA and
     microBaixaPreco and
     microLL and
     microLH


// =====================================================
// 7. SUPORTE E RESISTENCIA
// IMPORTANTE:
// highest/lowest sao calculados sempre.
// Depois usamos [1].
// =====================================================

resistencia = ta.highest(high, periodoSR)
suporte = ta.lowest(low, periodoSR)

resistenciaAnterior = ta.highest(high, periodoSR)[1]
suporteAnterior = ta.lowest(low, periodoSR)[1]

microMax = ta.highest(high, periodoMicro)
microMin = ta.lowest(low, periodoMicro)

microMaxAnterior = ta.highest(high, periodoMicro)[1]
microMinAnterior = ta.lowest(low, periodoMicro)[1]


// =====================================================
// 8. ESPACO CORRIGIDO
// =====================================================

distResistenciaATR =
     atrSeguro > 0 ? (resistenciaAnterior - close) / atrSeguro : 999.0

distSuporteATR =
     atrSeguro > 0 ? (close - suporteAnterior) / atrSeguro : 999.0

espacoAltaOK =
     distResistenciaATR >= minSpaceATR

espacoBaixaOK =
     distSuporteATR >= minSpaceATR

extensaoAltaOK =
     distResistenciaATR <= maxExtensionATR

extensaoBaixaOK =
     distSuporteATR <= maxExtensionATR

espacoAltaFinal =
     espacoAltaOK and extensaoAltaOK

espacoBaixaFinal =
     espacoBaixaOK and extensaoBaixaOK


// =====================================================
// 9. PIVOTS
// =====================================================

pivLow = ta.pivotlow(low, pivotLen, pivotLen)
pivHigh = ta.pivothigh(high, pivotLen, pivotLen)

ultimoPivLow = ta.valuewhen(not na(pivLow), pivLow, 0)
penultimoPivLow = ta.valuewhen(not na(pivLow), pivLow, 1)

ultimoPivHigh = ta.valuewhen(not na(pivHigh), pivHigh, 0)
penultimoPivHigh = ta.valuewhen(not na(pivHigh), pivHigh, 1)

barraUltimoPivLow = ta.valuewhen(not na(pivLow), bar_index - pivotLen, 0)
barraPenultimoPivLow = ta.valuewhen(not na(pivLow), bar_index - pivotLen, 1)

barraUltimoPivHigh = ta.valuewhen(not na(pivHigh), bar_index - pivotLen, 0)
barraPenultimoPivHigh = ta.valuewhen(not na(pivHigh), bar_index - pivotLen, 1)

idadePivLow =
     na(barraUltimoPivLow) ? 999999 : bar_index - barraUltimoPivLow

idadePivHigh =
     na(barraUltimoPivHigh) ? 999999 : bar_index - barraUltimoPivHigh

distTestesLow =
     na(barraUltimoPivLow) or na(barraPenultimoPivLow) ?
     999999 :
     math.abs(barraUltimoPivLow - barraPenultimoPivLow)

distTestesHigh =
     na(barraUltimoPivHigh) or na(barraPenultimoPivHigh) ?
     999999 :
     math.abs(barraUltimoPivHigh - barraPenultimoPivHigh)

toleranciaPattern =
     atrSeguro * toleranciaPadraoATR

duploFundo =
     not na(ultimoPivLow) and
     not na(penultimoPivLow) and
     math.abs(ultimoPivLow - penultimoPivLow) <= toleranciaPattern and
     distTestesLow >= distanciaMinTestes and
     idadePivLow <= validadePadrao

duploTopo =
     not na(ultimoPivHigh) and
     not na(penultimoPivHigh) and
     math.abs(ultimoPivHigh - penultimoPivHigh) <= toleranciaPattern and
     distTestesHigh >= distanciaMinTestes and
     idadePivHigh <= validadePadrao


// =====================================================
// 10. LINHA DE CONFIRMACAO DO PADRAO DUPLO
// =====================================================

necklineAlta =
     ta.highest(high, periodoSR)[1]

necklineBaixa =
     ta.lowest(low, periodoSR)[1]

duploFundoConfirmado =
     duploFundo and
     close > necklineAlta and
     candleAltaForte

duploTopoConfirmado =
     duploTopo and
     close < necklineBaixa and
     candleBaixaForte


// =====================================================
// 11. PADROES DE CANDLE
// =====================================================

doji =
     rangeCandle > syminfo.mintick and
     corpo / rangeSeguro <= 0.12

spinningTop =
     rangeCandle > syminfo.mintick and
     corpo / rangeSeguro <= 0.30 and
     pavioSuperior > corpo and
     pavioInferior > corpo

martelo =
     pavioInferior >= corpoSeguro * 1.8 and
     pavioSuperior <= corpoSeguro * 0.8 and
     fechoPosicao >= 0.55

marteloInvertido =
     pavioSuperior >= corpoSeguro * 1.8 and
     pavioInferior <= corpoSeguro * 0.8 and
     fechoPosicao >= 0.55

estrelaCadente =
     pavioSuperior >= corpoSeguro * 1.8 and
     pavioInferior <= corpoSeguro * 0.8 and
     fechoPosicao <= 0.55

pinAlta =
     pavioInferior >= corpoSeguro * 2.0 and
     fechoPosicao >= 0.60

pinBaixa =
     pavioSuperior >= corpoSeguro * 2.0 and
     fechoPosicao <= 0.40

engolfoAlta =
     close > open and
     close[1] < open[1] and
     open <= close[1] and
     close >= open[1]

engolfoBaixa =
     close < open and
     close[1] > open[1] and
     open >= close[1] and
     close <= open[1]

haramiAlta =
     close[1] < open[1] and
     close > open and
     math.max(open, close) < math.max(open[1], close[1]) and
     math.min(open, close) > math.min(open[1], close[1])

haramiBaixa =
     close[1] > open[1] and
     close < open and
     math.max(open, close) < math.max(open[1], close[1]) and
     math.min(open, close) > math.min(open[1], close[1])

rejeicaoAlta =
     martelo or
     pinAlta or
     engolfoAlta or
     haramiAlta

rejeicaoBaixa =
     estrelaCadente or
     pinBaixa or
     engolfoBaixa or
     haramiBaixa


// =====================================================
// 12. PROXIMIDADE DE SUPORTE / RESISTENCIA
// =====================================================

pertoSuporte =
     math.abs(close - suporte) <= atrSeguro * toleranciaPadraoATR

pertoResistencia =
     math.abs(close - resistencia) <= atrSeguro * toleranciaPadraoATR

reversaoAlta =
     pertoSuporte and
     rejeicaoAlta

reversaoBaixa =
     pertoResistencia and
     rejeicaoBaixa


// =====================================================
// 13. ROMPIMENTO PRINCIPAL
// =====================================================

rompimentoAlta =
     close > resistenciaAnterior

rompimentoBaixa =
     close < suporteAnterior

rompimentoAltaForte =
     rompimentoAlta and
     candleAltaForte

rompimentoBaixaForte =
     rompimentoBaixa and
     candleBaixaForte


// =====================================================
// 14. MICRO ROMPIMENTO
// =====================================================

microRompimentoAlta =
     close > microMaxAnterior and
     close > open and
     corpo >= atrSeguro * 0.20

microRompimentoBaixa =
     close < microMinAnterior and
     close < open and
     corpo >= atrSeguro * 0.20


// =====================================================
// 15. PRE-ROMPIMENTO
// =====================================================

distResistenciaAtual =
     atrSeguro > 0 ? (resistencia - close) / atrSeguro : 999.0

distSuporteAtual =
     atrSeguro > 0 ? (close - suporte) / atrSeguro : 999.0

preRompimentoAlta =
     distResistenciaAtual >= 0 and
     distResistenciaAtual <= 0.35 and
     close > ema5 and
     microAlta

preRompimentoBaixa =
     distSuporteAtual >= 0 and
     distSuporteAtual <= 0.35 and
     close < ema5 and
     microBaixa


// =====================================================
// 16. FALSO ROMPIMENTO
// =====================================================

falsoRompimentoAlta =
     high > resistenciaAnterior and
     close <= resistenciaAnterior and
     pavioSuperior > corpoSeguro

falsoRompimentoBaixa =
     low < suporteAnterior and
     close >= suporteAnterior and
     pavioInferior > corpoSeguro


// =====================================================
// 17. COMPRESSAO
// =====================================================

atrCurto = ta.atr(5)
atrMedio = ta.atr(14)

compressao =
     atrCurto < atrMedio * 0.75 and
     rangeCandle < atrSeguro * 0.80

expansao =
     atrCurto > atrMedio * 1.15 and
     rangeCandle > atrSeguro * 1.10

consolidacao =
     math.abs(ema5 - ema21) <= atrSeguro * 0.20 and
     adx < adxMin


// =====================================================
// 18. CANAL
// =====================================================

canalAlta =
     microAlta and
     not tendenciaBaixa and
     close < resistencia

canalBaixa =
     microBaixa and
     not tendenciaAlta and
     close > suporte

canal =
     canalAlta or canalBaixa


// =====================================================
// 19. GAP / ANOMALIA
// =====================================================

gapAtual = math.abs(open - close[1])

gapGrande =
     gapAtual > atrSeguro * 1.5

anomaliaCandle =
     rangeCandle > atrSeguro * 3.0

riscoAnomalia =
     gapGrande or anomaliaCandle


// =====================================================
// 20. MOMENTUM
// =====================================================

momentumAlta =
     rsi >= rsiAlto and
     diPlus > diMinus

momentumBaixa =
     rsi <= rsiBaixo and
     diMinus > diPlus

adxForte =
     adx >= adxMin

volumeForte =
     volume > volumeMedia


// =====================================================
// 21. MICRO MOMENTUM
// =====================================================

microMomentumAlta =
     close > close[1] and
     ema5 > ema5[1] and
     rsi >= 50

microMomentumBaixa =
     close < close[1] and
     ema5 < ema5[1] and
     rsi <= 50


// =====================================================
// 22. MERCADO LATERAL
// =====================================================

rangeATR =
     atrSeguro > 0 ?
     (resistencia - suporte) / atrSeguro :
     999.0

mercadoLateral =
     rangeATR <= 3.0 and
     adx < 20 and
     math.abs(ema5 - ema21) < atrSeguro * 0.25

mercadoMuitoFraco =
     adx < 12 and
     math.abs(ema5 - ema21) < atrSeguro * 0.15

bloqueioLateral =
     mercadoLateral and
     not rompimentoAlta and
     not rompimentoBaixa and
     not duploFundoConfirmado and
     not duploTopoConfirmado


// =====================================================
// 23. FORCA DA ALTA
// =====================================================

forcaAlta = 0.0

forcaAlta += tendenciaAlta ? 25 : 0
forcaAlta += tendenciaAltaModerada and not tendenciaAlta ? 12 : 0
forcaAlta += microAlta ? 15 : 0
forcaAlta += microAltaForte ? 5 : 0
forcaAlta += momentumAlta ? 15 : 0
forcaAlta += adxForte and diPlus > diMinus ? 10 : 0
forcaAlta += volumeForte and close > open ? 5 : 0
forcaAlta += candleAltaForte ? 10 : 0
forcaAlta += rejeicaoAlta ? 5 : 0
forcaAlta += duploFundo ? 15 : 0
forcaAlta += duploFundoConfirmado ? 15 : 0
forcaAlta += reversaoAlta ? 12 : 0
forcaAlta += rompimentoAlta ? 18 : 0
forcaAlta += rompimentoAltaForte ? 10 : 0
forcaAlta += microRompimentoAlta ? 8 : 0
forcaAlta += expansao and close > open ? 5 : 0
forcaAlta += preRompimentoAlta ? 4 : 0

forcaAlta := math.min(forcaAlta, 100)


// =====================================================
// 24. FORCA DA BAIXA
// =====================================================

forcaBaixa = 0.0

forcaBaixa += tendenciaBaixa ? 25 : 0
forcaBaixa += tendenciaBaixaModerada and not tendenciaBaixa ? 12 : 0
forcaBaixa += microBaixa ? 15 : 0
forcaBaixa += microBaixaForte ? 5 : 0
forcaBaixa += momentumBaixa ? 15 : 0
forcaBaixa += adxForte and diMinus > diPlus ? 10 : 0
forcaBaixa += volumeForte and close < open ? 5 : 0
forcaBaixa += candleBaixaForte ? 10 : 0
forcaBaixa += rejeicaoBaixa ? 5 : 0
forcaBaixa += duploTopo ? 15 : 0
forcaBaixa += duploTopoConfirmado ? 15 : 0
forcaBaixa += reversaoBaixa ? 12 : 0
forcaBaixa += rompimentoBaixa ? 18 : 0
forcaBaixa += rompimentoBaixaForte ? 10 : 0
forcaBaixa += microRompimentoBaixa ? 8 : 0
forcaBaixa += expansao and close < open ? 5 : 0
forcaBaixa += preRompimentoBaixa ? 4 : 0

forcaBaixa := math.min(forcaBaixa, 100)


// =====================================================
// 25. CONFLUENCIA
// =====================================================

confluenciaAlta =
     (
     tendenciaAlta or
     tendenciaAltaModerada or
     microAlta or
     momentumAlta or
     reversaoAlta or
     duploFundo or
     rompimentoAlta
     )

confluenciaBaixa =
     (
     tendenciaBaixa or
     tendenciaBaixaModerada or
     microBaixa or
     momentumBaixa or
     reversaoBaixa or
     duploTopo or
     rompimentoBaixa
     )


// =====================================================
// 26. CONFLITO
// =====================================================

conflitoAlta =
     tendenciaBaixa and
     microBaixa and
     momentumBaixa and
     forcaBaixa > forcaAlta

conflitoBaixa =
     tendenciaAlta and
     microAlta and
     momentumAlta and
     forcaAlta > forcaBaixa


// =====================================================
// 27. FILTROS DE RISCO
// =====================================================

riscoAlta =
     falsoRompimentoAlta or
     riscoAnomalia or
     not espacoAltaFinal

riscoBaixa =
     falsoRompimentoBaixa or
     riscoAnomalia or
     not espacoBaixaFinal


// =====================================================
// 28. SENSIBILIDADE
// =====================================================

scoreMinimo = 48.0
vantagemMinima = 10.0

if sensibilidade == "Cautelosa"
    scoreMinimo := 60.0
    vantagemMinima := 15.0

if sensibilidade == "Rapida"
    scoreMinimo := 38.0
    vantagemMinima := 7.0


// =====================================================
// 29. DECISAO BASE
// =====================================================

vantagemAlta =
     forcaAlta - forcaBaixa

vantagemBaixa =
     forcaBaixa - forcaAlta

baseAlta =
     forcaAlta >= scoreMinimo and
     vantagemAlta >= vantagemMinima and
     confluenciaAlta

baseBaixa =
     forcaBaixa >= scoreMinimo and
     vantagemBaixa >= vantagemMinima and
     confluenciaBaixa


// =====================================================
// 30. PADROES PRIORITARIOS
// =====================================================

padraoAltaForte =
     duploFundoConfirmado or
     rompimentoAltaForte or
     reversaoAlta

padraoBaixaForte =
     duploTopoConfirmado or
     rompimentoBaixaForte or
     reversaoBaixa


// =====================================================
// 31. BLOQUEIOS
// =====================================================

bloqueioAlta =
     conflitoAlta or
     riscoAlta or
     falsoRompimentoAlta or
     bloqueioLateral or
     mercadoMuitoFraco

bloqueioBaixa =
     conflitoBaixa or
     riscoBaixa or
     falsoRompimentoBaixa or
     bloqueioLateral or
     mercadoMuitoFraco


// =====================================================
// 32. CANDIDATOS
// =====================================================

candidatoAlta =
     baseAlta and
     not bloqueioAlta

candidatoBaixa =
     baseBaixa and
     not bloqueioBaixa


// =====================================================
// 33. MEMORIA DE ESTABILIDADE
// =====================================================

var int direcaoMemoria = 0
var int estabilidade = 0

diferencaForca = math.abs(forcaAlta - forcaBaixa)

direcaoAtual = 0

if candidatoAlta and forcaAlta > forcaBaixa
    direcaoAtual := 1

if candidatoBaixa and forcaBaixa > forcaAlta
    direcaoAtual := -1

if direcaoAtual != 0
    if direcaoMemoria == 0
        direcaoMemoria := direcaoAtual
        estabilidade := 1
    else
        if direcaoAtual == direcaoMemoria
            estabilidade += 1
        else
            if diferencaForca >= forcaInversao
                direcaoMemoria := direcaoAtual
                estabilidade := 1
else
    if estabilidade > 0
        estabilidade -= 1

if estabilidade < 0
    estabilidade := 0


// =====================================================
// 34. ESTABILIDADE FINAL
// =====================================================

memoriaAlta =
     direcaoMemoria == 1 and
     estabilidade >= estabilidadeMinima

memoriaBaixa =
     direcaoMemoria == -1 and
     estabilidade >= estabilidadeMinima

// Quando a memoria esta desligada,
// a direcao atual pode entrar diretamente.

permitirAlta =
     not usarMemoria or memoriaAlta

permitirBaixa =
     not usarMemoria or memoriaBaixa


// =====================================================
// 35. DECISAO FINAL
// =====================================================

comprarFinal =
     candidatoAlta and
     permitirAlta and
     forcaAlta > forcaBaixa

venderFinal =
     candidatoBaixa and
     permitirBaixa and
     forcaBaixa > forcaAlta


// =====================================================
// 36. PROTECAO CONTRA TROCA RAPIDA
// =====================================================

ruidoDirecional =
     diferencaForca <= toleranciaRuido * 100.0

if ruidoDirecional
    comprarFinal := false
