f_limitar(valor, minimo, maximo) =>
    math.max(minimo, math.min(maximo, valor))


f_textoForca(valor) =>
    string resultado = "SEM FORCA"
    if valor >= 85
        resultado := "EXTREMA"
    else if valor >= 70
        resultado := "MUITO FORTE"
    else if valor >= 55
        resultado := "FORTE"
    else if valor >= 40
        resultado := "MODERADA"
    else if valor >= 25
        resultado := "FRACA"
    else if valor >= 10
        resultado := "MUITO FRACA"
    resultado


f_textoQualidade(valor) =>
    string resultado = "MUITO BAIXA"
    if valor >= 85
        resultado := "EXCELENTE"
    else if valor >= 70
        resultado := "MUITO BOA"
    else if valor >= 55
        resultado := "BOA"
    else if valor >= 40
        resultado := "REGULAR"
    else if valor >= 25
        resultado := "BAIXA"
    resultado


f_textoDirecao(direcao) =>
    string resultado = "NEUTRO"
    if direcao == 1
        resultado := "COMPRA"
    else if direcao == -1
        resultado := "VENDA"
    resultado


f_corDirecao(direcao) =>
    color resultado = color.white
    if direcao == 1
        resultado := color.lime
    else if direcao == -1
        resultado := color.red
    resultado


f_corQualidade(valor) =>
    color resultado = color.white
    if valor >= 70
        resultado := color.lime
    else if valor >= 50
        resultado := color.yellow
    else
        resultado := color.red
    resultado


f_expiracaoBarras(opcao) =>
    int resultado = 5
    if opcao == "1"
        resultado := 1
    else if opcao == "2"
        resultado := 2
    else if opcao == "5"
        resultado := 5
    else
        resultado := 15
    resultado


f_kvo(srcHigh, srcLow, srcClose, srcVolume, rapido, lento, sinal) =>
    float mediaPreco = (srcHigh + srcLow + srcClose) / 3.0
    float mediaAnterior = nz(mediaPreco[1], mediaPreco)
    float tendencia = mediaPreco > mediaAnterior ? 1.0 : mediaPreco < mediaAnterior ? -1.0 : 0.0
    float forca = srcVolume * tendencia * math.abs(mediaPreco - mediaAnterior)
    float linhaRapida = ta.ema(forca, rapido)
    float linhaLenta = ta.ema(forca, lento)
    float linha = linhaRapida - linhaLenta
    float sinalLinha = ta.ema(linha, sinal)
    [linha, sinalLinha]


f_scoreMTF(tf) =>
    float fechamento = request.security(syminfo.tickerid, tf, close)
    float ema9TF = request.security(syminfo.tickerid, tf, ta.ema(close, 9))
    float ema21TF = request.security(syminfo.tickerid, tf, ta.ema(close, 21))
    float rsiTF = request.security(syminfo.tickerid, tf, ta.rsi(close, 10))

    float compra = 0.0
    float venda = 0.0

    if fechamento > ema9TF
        compra += 35.0
    else
        venda += 35.0

    if ema9TF > ema21TF
        compra += 35.0
    else
        venda += 35.0

    if rsiTF > 50.0
        compra += 30.0
    else
        venda += 30.0

    [compra, venda]


// ============================================================================
// 14 - CALCULOS PRINCIPAIS

emaRapida = ta.ema(close, emaRapidaPeriodo)

rsi = ta.rsi(close, rsiPeriodo)

atr = ta.atr(atrPeriodo)
atrMedio = ta.sma(atr, atrMediaPeriodo)

[linhaKVO, linhaKVOSinal] = f_kvo(
     high,
     low,
     close,
     volume,
     kvoRapido,
     kvoLento,
     kvoSinal)

histogramaKVO = linhaKVO - linhaKVOSinal

volumeMedio = ta.sma(volume, 20)
volumeRelativo = volumeMedio > 0 ? volume / volumeMedio : 1.0

momento = ta.roc(close, 5)
momentoNormalizado = f_limitar(50.0 + momento * 5.0, 0.0, 100.0)

corpo = math.abs(close - open)
amplitudeCandle = math.max(high - low, syminfo.mintick)
forcaCandle = f_limitar((corpo / amplitudeCandle) * 100.0, 0.0, 100.0)

pressaoCandle = close > open ? forcaCandle : close < open ? -forcaCandle : 0.0


// ============================================================================
// 15 - SUPORTE E RESISTENCIA
// ============================================================================

resistencia = ta.highest(high, estruturaLookback)
suporte = ta.lowest(low, estruturaLookback)

distanciaResistencia = math.max(resistencia - close, 0.0)
distanciaSuporte = math.max(close - suporte, 0.0)

resistenciaATR = atr > 0 ? distanciaResistencia / atr : 999.0
suporteATR = atr > 0 ? distanciaSuporte / atr : 999.0

proximoDaResistencia = resistenciaATR <= distanciaMaximaSR
proximoDoSuporte = suporteATR <= distanciaMaximaSR


// ============================================================================
// 16 - PADROES
// ============================================================================

engolfoCompra = close > open and close[1] < open[1] and close >= open[1] and open <= close[1]
engolfoVenda = close < open and close[1] > open[1] and close <= open[1] and open >= close[1]

rejeicaoCompra = low < low[1] and close > open and close > (high + low) / 2.0
rejeicaoVenda = high > high[1] and close < open and close < (high + low) / 2.0

rompimentoCompra = close > ta.highest(high, microLookback)[1]
rompimentoVenda = close < ta.lowest(low, microLookback)[1]

padraoCompra = engolfoCompra or rejeicaoCompra or rompimentoCompra
padraoVenda = engolfoVenda or rejeicaoVenda or rompimentoVenda


// ============================================================================
// 17 - MTF
// ============================================================================

[mtf1Compra, mtf1Venda] = f_scoreMTF(tf1)
[mtf2Compra, mtf2Venda] = f_scoreMTF(tf2)
[mtf3Compra, mtf3Venda] = f_scoreMTF(tf3)
[mtf4Compra, mtf4Venda] = f_scoreMTF(tf4)
[mtf5Compra, mtf5Venda] = f_scoreMTF(tf5)

mtfCompra = (mtf1Compra + mtf2Compra + mtf3Compra + mtf4Compra + mtf5Compra) / 5.0
mtfVenda = (mtf1Venda + mtf2Venda + mtf3Venda + mtf4Venda + mtf5Venda) / 5.0


// ============================================================================
// 18 - SCORE PRINCIPAL
// ============================================================================

float scoreCompra = 0.0
float scoreVenda = 0.0

if close > emaRapida
    scoreCompra += 12.0
else
    scoreVenda += 12.0

if emaRapida > emaMedia
    scoreCompra += 12.0
else
    scoreVenda += 12.0

if emaMedia > emaLenta
    scoreCompra += 10.0
else
    scoreVenda += 10.0

if pressaoCandle > 0
    scoreCompra += 10.0
else if pressaoCandle < 0
    scoreVenda += 10.0

if histogramaKVO > 0
    scoreCompra += 10.0
else
    scoreVenda += 10.0

if linhaKVO > linhaKVOSinal
    scoreCompra += 8.0
else
    scoreVenda += 8.0

if momentoNormalizado > 50.0
    scoreCompra += 8.0
else
    scoreVenda += 8.0

if rsi > 50.0
    scoreCompra += 6.0
else
    scoreVenda += 6.0

if volumeRelativo > 1.0
    if close > open
        scoreCompra += 5.0
    else if close < open
        scoreVenda += 5.0

if proximoDoSuporte
    scoreCompra += 5.0

if proximoDaResistencia
    scoreVenda += 5.0

if usarPadroes
    if padraoCompra
        scoreCompra += pesoPadroes

    if padraoVenda
        scoreVenda += pesoPadroes

if usarMTF
    scoreCompra := scoreCompra * 0.65 + mtfCompra * 0.35
    scoreVenda := scoreVenda * 0.65 + mtfVenda * 0.35

scoreCompra := f_limitar(scoreCompra * sensibilidade, 0.0, 100.0)
scoreVenda := f_limitar(scoreVenda * sensibilidade, 0.0, 100.0)

vantagemPrincipal = math.abs(scoreCompra - scoreVenda)

int direcaoPrincipal = 0

if scoreCompra > scoreVenda
    direcaoPrincipal := 1
else if scoreVenda > scoreCompra
    direcaoPrincipal := -1


// ============================================================================
// 19 - FORCA PRINCIPAL
// ============================================================================

forcaPrincipal = f_limitar(
     math.max(scoreCompra, scoreVenda) * 0.65 +
     forcaCandle * 0.15 +
     math.min(volumeRelativo * 10.0, 10.0) +
     math.min(math.abs(histogramaKVO) * 0.01, 10.0),
     0.0,
     100.0)


// ============================================================================
// 20 - CONFLUENCIA E QUALIDADE
// ============================================================================

float confluenciaPrincipal = 0.0

if emaRapida > emaMedia and emaMedia > emaLenta
    confluenciaPrincipal += 20.0

if emaRapida < emaMedia and emaMedia < emaLenta
    confluenciaPrincipal += 20.0

if histogramaKVO > 0 and direcaoPrincipal == 1
    confluenciaPrincipal += 20.0

if histogramaKVO < 0 and direcaoPrincipal == -1
    confluenciaPrincipal += 20.0

if momentoNormalizado > 50.0 and direcaoPrincipal == 1
    confluenciaPrincipal += 15.0

if momentoNormalizado < 50.0 and direcaoPrincipal == -1
    confluenciaPrincipal += 15.0

if volumeRelativo > 1.0
    confluenciaPrincipal += 10.0

if vantagemPrincipal >= vantagemMinima
    confluenciaPrincipal += 15.0

qualidadePrincipal = f_limitar(
     confluenciaPrincipal * 0.60 +
     vantagemPrincipal * 2.0 +
     forcaPrincipal * 0.10,
     0.0,
     100.0)


// ============================================================================
// 21 - OPORTUNIDADE
// ============================================================================

float oportunidadePrincipal = 0.0

if direcaoPrincipal == 1
    if distanciaResistencia > atr * espacoMinimoAlvo
        oportunidadePrincipal += 45.0

    if not proximoDaResistencia
        oportunidadePrincipal += 20.0

    if rsi < rsiSobreCompra
        oportunidadePrincipal += 15.0

    if padraoCompra
        oportunidadePrincipal += 20.0

else if direcaoPrincipal == -1
    if distanciaSuporte > atr * espacoMinimoAlvo
        oportunidadePrincipal += 45.0

    if not proximoDoSuporte
        oportunidadePrincipal += 20.0

    if rsi > rsiSobreVenda
        oportunidadePrincipal += 15.0

    if padraoVenda
        oportunidadePrincipal += 20.0

oportunidadePrincipal := f_limitar(oportunidadePrincipal, 0.0, 100.0)


// ============================================================================
// 22 - MICRO MOTOR
// ============================================================================

microEMARapidaValor = ta.ema(close, microEMARapida)
microEMALentaValor = ta.ema(close, microEMALenta)

microROC1 = ta.roc(close, 1)
microROC2 = ta.roc(close, 2)
microROC3 = ta.roc(close, 3)

microAceleracao = microROC1 - microROC2

microVolume = volumeMedio > 0 ? volume / volumeMedio : 1.0

float microScoreCompra = 0.0
float microScoreVenda = 0.0

if close > microEMARapidaValor
    microScoreCompra += 15.0
else
    microScoreVenda += 15.0

if microEMARapidaValor > microEMALentaValor
    microScoreCompra += 15.0
else
    microScoreVenda += 15.0

if microROC1 > 0
    microScoreCompra += 10.0
else
    microScoreVenda += 10.0

if microROC2 > 0
    microScoreCompra += 10.0
else
    microScoreVenda += 10.0

if microROC3 > 0
    microScoreCompra += 10.0
else
    microScoreVenda += 10.0

if microAceleracao > 0
    microScoreCompra += 10.0
else
    microScoreVenda += 10.0

if histogramaKVO > 0
    microScoreCompra += 10.0
else
    microScoreVenda += 10.0

if microVolume > 1.0
    if close > open
        microScoreCompra += 5.0
    else if close < open
        microScoreVenda += 5.0

if rompimentoCompra or rejeicaoCompra
    microScoreCompra += 10.0

if rompimentoVenda or rejeicaoVenda
    microScoreVenda += 10.0

if forcaCandle >= 50.0
    if close > open
        microScoreCompra += 5.0
    else if close < open
        microScoreVenda += 5.0

microScoreCompra := f_limitar(microScoreCompra, 0.0, 100.0)
microScoreVenda := f_limitar(microScoreVenda, 0.0, 100.0)

microVantagem = math.abs(microScoreCompra - microScoreVenda)

int microDirecao = 0

if microScoreCompra > microScoreVenda
    microDirecao := 1
else if microScoreVenda > microScoreCompra
    microDirecao := -1

microForca = f_limitar(
     math.max(microScoreCompra, microScoreVenda) * 0.70 +
     forcaCandle * 0.20 +
     math.min(microVolume * 10.0, 10.0),
     0.0,
     100.0)

microQualidade = f_limitar(
     microForca * 0.50 +
     microVantagem * 2.0 +
     (
         microAceleracao > 0 and microDirecao == 1 ?
         10.0 :
         microAceleracao < 0 and microDirecao == -1 ?
         10.0 :
         0.0
     ),
     0.0,
     100.0)


// ============================================================================
// 23 - MICRO VALIDACAO
// ============================================================================

bool microValido = true

if usarMicro
    microValido := microDirecao != 0 and
         math.max(microScoreCompra, microScoreVenda) >= microScoreMinimo and
         microVantagem >= microVantagemMinima

bool confluenciaDirecao = true

if usarMicro and microValido
    confluenciaDirecao := microDirecao == direcaoPrincipal

previsaoValida = habilitarPrevisao and
     direcaoPrincipal != 0 and
     math.max(scoreCompra, scoreVenda) >= scoreMinimoPrevisao and
     vantagemPrincipal >= vantagemMinima and
     confluenciaDirecao


// ============================================================================
// 24 - CALCULO DO ALVO MICRO
// ============================================================================

float fatorAlvo = alvoMicroATRBase

fatorAlvo += (microForca - 50.0) / 200.0
fatorAlvo += (microQualidade - 50.0) / 250.0

if atrMedio > 0
    volatilidadeRelativa = atr / atrMedio

    if volatilidadeRelativa > 1.20
        fatorAlvo += 0.10
    else if volatilidadeRelativa < 0.80
        fatorAlvo -= 0.05

fatorAlvo := f_limitar(
     fatorAlvo,
     alvoMicroATRMin,
     alvoMicroATRMax)

float espacoDisponivelPreco = 0.0

if direcaoPrincipal == 1
    espacoDisponivelPreco := distanciaResistencia
else if direcaoPrincipal == -1
    espacoDisponivelPreco := distanciaSuporte

espacoDisponivelATR = atr > 0 ? espacoDisponivelPreco / atr : 0.0

float fatorAlvoFinal = fatorAlvo

if filtroEspaco and espacoDisponivelATR > 0
    fatorAlvoFinal := math.min(fatorAlvoFinal, espacoDisponivelATR)

alvoValido = usarAlvoMicro and
     direcaoPrincipal != 0 and
     fatorAlvoFinal >= alvoMicroATRMin

float alvoCompraPrecoCalculado = na
float alvoVendaPrecoCalculado = na

if alvoValido
    alvoCompraPrecoCalculado := close + atr * fatorAlvoFinal
    alvoVendaPrecoCalculado := close - atr * fatorAlvoFinal


// ============================================================================
// 25 - QUALIDADE DO ALVO
// ============================================================================

float qualidadeAlvo = 0.0

if alvoValido
    qualidadeAlvo += 35.0

if espacoDisponivelATR >= fatorAlvoFinal
    qualidadeAlvo += 25.0

if microForca >= 55.0
    qualidadeAlvo += 15.0

if microQualidade >= 55.0
    qualidadeAlvo += 15.0

if vantagemPrincipal >= vantagemMinima
    qualidadeAlvo += 10.0

qualidadeAlvo := f_limitar(qualidadeAlvo, 0.0, 100.0)


// ============================================================================
// 26 - SINAIS DE ENTRADA
// ============================================================================

sinalCompra = permitirCompra and
     previsaoValida and
     direcaoPrincipal == 1 and
     scoreCompra >= entradaScoreMinimo and
     vantagemPrincipal >= entradaVantagemMinima and
     forcaPrincipal >= 40.0 and
     qualidadePrincipal >= 45.0 and
     oportunidadePrincipal >= 40.0 and
     (not usarMicro or microDirecao == 1) and
     (not usarAlvoMicro or alvoValido)

sinalVenda = permitirVenda and
     previsaoValida and
     direcaoPrincipal == -1 and
     scoreVenda >= entradaScoreMinimo and
     vantagemPrincipal >= entradaVantagemMinima and
     forcaPrincipal >= 40.0 and
     qualidadePrincipal >= 45.0 and
     oportunidadePrincipal >= 40.0 and
     (not usarMicro or microDirecao == -1) and
     (not usarAlvoMicro or alvoValido)


// ============================================================================
// 27 - COOLDOWN
// ============================================================================

var int ultimaEntradaBarra = na

cooldownOK = na(ultimaEntradaBarra) or
     bar_index - ultimaEntradaBarra >= cooldownBarras


// ============================================================================
// 28 - ESTADO PERSISTENTE DA OPERACAO
// ============================================================================

var bool operacaoAtiva = false
var int operacaoDirecao = 0
var int operacaoBarraEntrada = na
var int operacaoBarraExpiracao = na
var int operacaoBarrasDecorridas = 0
var int operacaoBarrasRestantes = 0
var float operacaoPrecoEntrada = na
var float operacaoAlvo = na
var float operacaoStop = na
var string operacaoStatus = "AGUARDANDO"
var string ultimoResultado = "NENHUM"

expiracaoBarras = f_expiracaoBarras(expiracaoSelecionada)


// ============================================================================
// 29 - NOVA ENTRADA
// ============================================================================

novaEntradaCompra = usarEntrada and
     cooldownOK and
     not operacaoAtiva and
     sinalCompra

novaEntradaVenda = usarEntrada and
     cooldownOK and
     not operacaoAtiva and
     sinalVenda


// ============================================================================
// 30 - CONGELAMENTO DA COMPRA
// ============================================================================

if barstate.isconfirmed and novaEntradaCompra
    operacaoAtiva := true
    operacaoDirecao := 1
    operacaoBarraEntrada := bar_index
    operacaoBarraExpiracao := bar_index + (usarExpiracao ? expiracaoBarras : 999999)
    operacaoBarrasDecorridas := 0
    operacaoBarrasRestantes := usarExpiracao ? expiracaoBarras : 999999
    operacaoPrecoEntrada := close

    if usarAlvoMicro and alvoValido
        operacaoAlvo := alvoCompraPrecoCalculado
    else
        operacaoAlvo := close + atr * alvoStrategyATR

    operacaoStop := close - atr * stopATR
    operacaoStatus := "SEGURANDO COMPRA"
    ultimoResultado := "ENTRADA COMPRA"
    ultimaEntradaBarra := bar_index

    if usarOrdensStrategy
        strategy.entry("COMPRA", strategy.long)


// ============================================================================
// 31 - CONGELAMENTO DA VENDA
// ============================================================================

if barstate.isconfirmed and novaEntradaVenda
    operacaoAtiva := true
    operacaoDirecao := -1
    operacaoBarraEntrada := bar_index
    operacaoBarraExpiracao := bar_index + (usarExpiracao ? expiracaoBarras : 999999)
    operacaoBarrasDecorridas := 0
    operacaoBarrasRestantes := usarExpiracao ? expiracaoBarras : 999999
    operacaoPrecoEntrada := close

    if usarAlvoMicro and alvoValido
        operacaoAlvo := alvoVendaPrecoCalculado
    else
        operacaoAlvo := close - atr * alvoStrategyATR

    operacaoStop := close + atr * stopATR
    operacaoStatus := "SEGURANDO VENDA"
    ultimoResultado := "ENTRADA VENDA"
    ultimaEntradaBarra := bar_index

    if usarOrdensStrategy
        strategy.entry("VENDA", strategy.short)


// ============================================================================
// 32 - CONTROLE DO HOLDING
// ============================================================================

bool alvoAtingido = false
bool stopAtingido = false
bool operacaoExpirada = false

if operacaoAtiva
    operacaoBarrasDecorridas := bar_index - operacaoBarraEntrada

    if usarExpiracao
        operacaoBarrasRestantes := math.max(
             operacaoBarraExpiracao - bar_index,
             0)
    else
        operacaoBarrasRestantes := 999999

    if operacaoDirecao == 1
        alvoAtingido := bar_index > operacaoBarraEntrada and
             high >= operacaoAlvo

        stopAtingido := bar_index > operacaoBarraEntrada and
             low <= operacaoStop

    else if operacaoDirecao == -1
        alvoAtingido := bar_index > operacaoBarraEntrada and
             low <= operacaoAlvo

        stopAtingido := bar_index > operacaoBarraEntrada and
             high >= operacaoStop

    operacaoExpirada := usarExpiracao and
         bar_index >= operacaoBarraExpiracao


// ============================================================================
// 33 - FINALIZACAO DA OPERACAO
// ============================================================================

if operacaoAtiva and alvoAtingido
    operacaoAtiva := false
    operacaoStatus := "ALVO ATINGIDO"
    ultimoResultado := "ALVO ATINGIDO"

    if usarOrdensStrategy
        if operacaoDirecao == 1
            strategy.close("COMPRA", comment="ALVO")
        else if operacaoDirecao == -1
            strategy.close("VENDA", comment="ALVO")

else if operacaoAtiva and stopAtingido
    operacaoAtiva := false
    operacaoStatus := "STOP"
    ultimoResultado := "STOP"

    if usarOrdensStrategy
        if operacaoDirecao == 1
            strategy.close("COMPRA", comment="STOP")
        else if operacaoDirecao == -1
            strategy.close("VENDA", comment="STOP")

else if operacaoAtiva and operacaoExpirada
    operacaoAtiva := false
    operacaoStatus := "EXPIRADA"
    ultimoResultado := "EXPIRADA"

    if usarOrdensStrategy
        if operacaoDirecao == 1
            strategy.close("COMPRA", comment="EXPIRACAO")
        else if operacaoDirecao == -1
            strategy.close("VENDA", comment="EXPIRACAO")


// ============================================================================
// 34 - ORDENS FIXAS
// ============================================================================

if usarOrdensStrategy and operacaoAtiva
    if operacaoDirecao == 1
        strategy.exit(
             "SAIDA COMPRA",
             from_entry="COMPRA",
             stop=operacaoStop,
             limit=operacaoAlvo)

    else if operacaoDirecao == -1
        strategy.exit(
             "SAIDA VENDA",
             from_entry="VENDA",
             stop=operacaoStop,
             limit=operacaoAlvo)


// ============================================================================
// 35 - TEXTOS DO PAINEL
// ============================================================================

     f_textoDirecao(operacaoDirecao) :
     "NENHUMA"

precoEntradaTexto = operacaoAtiva and not na(operacaoPrecoEntrada) ?
     str.tostring(operacaoPrecoEntrada, format.mintick) :
     "-"

alvoTexto = operacaoAtiva and not na(operacaoAlvo) ?
     str.tostring(operacaoAlvo, format.mintick) :
     "-"

stopTexto = operacaoAtiva and not na(operacaoStop) ?
     str.tostring(operacaoStop, format.mintick) :
     "-"

distanciaAlvoAtual = operacaoAtiva and not na(operacaoAlvo) ?
     math.abs(operacaoAlvo - close) :
     na

distanciaAlvoATRAtual = operacaoAtiva and atr > 0 and not na(distanciaAlvoAtual) ?
     distanciaAlvoAtual / atr :
     na

distanciaAlvoTexto = not na(distanciaAlvoATRAtual) ?
     str.tostring(distanciaAlvoATRAtual, "#.##") + " ATR" :
     "-"

barrasTexto = operacaoAtiva ?
     str.tostring(operacaoBarrasDecorridas) + "/" + str.tostring(expiracaoBarras) :
     "-"

restanteTexto = operacaoAtiva ?
     str.tostring(operacaoBarrasRestantes) + " BARRAS" :
     "-"

expiracaoTexto = usarExpiracao ?
     str.tostring(expiracaoBarras) + " BARRAS" :
     "DESATIVADA"


// ============================================================================
// 36 - TEXTO DA PREVISAO
// ============================================================================

if previsaoValida
    if direcaoPrincipal == 1
        previsaoTexto := "COMPRA"
    else if direcaoPrincipal == -1
        previsaoTexto := "VENDA"

if operacaoAtiva
    if operacaoDirecao == 1
        previsaoTexto := "COMPRA SEGURANDO"
    else if operacaoDirecao == -1
        previsaoTexto := "VENDA SEGURANDO"


// ============================================================================
// 37 - TEXTO DO PADRAO
// ============================================================================

padraoTexto = "SEM PADRAO"

if padraoCompra and padraoVenda
    padraoTexto := "CONFLITO"
else if padraoCompra
    padraoTexto := "COMPRA"
else if padraoVenda
    padraoTexto := "VENDA"


// ============================================================================
// 38 - TEXTO MICRO
// ============================================================================

microTexto = "NEUTRO"

if usarMicro
    if microValido
        if microDirecao == 1
            microTexto := "COMPRA"
        else if microDirecao == -1
            microTexto := "VENDA"
    else
        microTexto := "SEM CONFIRMACAO"
else
    microTexto := "DESATIVADO"


// ============================================================================
// 39 - STATUS DO ALVO
// ============================================================================

alvoStatusTexto = "SEM ALVO"

if operacaoAtiva
    if not na(operacaoAlvo)
        if operacaoDirecao == 1
            if close >= operacaoAlvo
                alvoStatusTexto := "ALVO"
            else
                alvoStatusTexto := "BUSCANDO"

        else if operacaoDirecao == -1
            if close <= operacaoAlvo
                alvoStatusTexto := "ALVO"
            else
                alvoStatusTexto := "BUSCANDO"

else if alvoValido
    alvoStatusTexto := "DISPONIVEL"
