f_limitar(valor, minimo, maximo) =>
    math.max(minimo, math.min(maximo, valor))


f_scoreRetorno(retorno, escala) =>
    math.max(0.0, math.min(100.0, 50.0 + retorno / escala * 50.0))


f_status(direcao) =>
    direcao == 1 ? "COMPRA" : direcao == -1 ? "VENDA" : "AGUARDAR"

     vol >= 5.0 ? "ALTA" :
     vol >= 3.0 ? "MEDIA" :
     vol >= 1.5 ? "BAIXA" :
     "MUITO BAIXA"

f_pacote() =>
    atr = ta.atr(atrLen)

    atrPct = close != 0 ? atr / close * 100.0 : 0.0

    atrMedia = ta.sma(atrPct, lookbackVol)

    volatilidadeRelativa = atrMedia > 0 ? atrPct / atrMedia * 100.0 : 100.0

    ema9 = ta.ema(close, emaRapida)

    ema21 = ta.ema(close, emaMedia)

    ema50 = ta.ema(close, emaLenta)

    rsi = ta.rsi(close, rsiLen)

    volumeMedia = ta.sma(volume, volumeLen)

    volumeRel = volumeMedia > 0 ? volume / volumeMedia : 1.0

    corpo = math.abs(close - open)

    rangeCandle = math.max(high - low, syminfo.mintick)

    forcaCandle = corpo / rangeCandle * 100.0

    retorno1 = close / close[1] - 1.0

    retorno3 = close / close[3] - 1.0

    tendenciaLong = ema9 > ema21 and ema21 > ema50

    tendenciaShort = ema9 < ema21 and ema21 < ema50

    tendenciaLongScore = tendenciaLong ? 100.0 : ema9 > ema21 ? 65.0 : 30.0

    tendenciaShortScore = tendenciaShort ? 100.0 : ema9 < ema21 ? 65.0 : 30.0

    rsiLong = rsi > 50.0 ? f_limitar((rsi - 50.0) * 2.0, 0.0, 100.0) : 0.0

    rsiShort = rsi < 50.0 ? f_limitar((50.0 - rsi) * 2.0, 0.0, 100.0) : 0.0

    kvo = ta.ema(volume * (close - close[1]), 34)

    kvoSignal = ta.ema(kvo, 55)

    kvoLong = kvo > kvoSignal ? 100.0 : 25.0

    kvoShort = kvo < kvoSignal ? 100.0 : 25.0

    volumeScore = volumeRel > 1.0 ? f_limitar(volumeRel * 50.0, 0.0, 100.0) : 35.0

    retornoLong1 = f_scoreRetorno(retorno1, 0.01)

    retornoShort1 = f_scoreRetorno(-retorno1, 0.01)

    retornoLong3 = f_scoreRetorno(retorno3, 0.02)

    retornoShort3 = f_scoreRetorno(-retorno3, 0.02)

    scoreLong = (
         forcaCandle * 0.15 +
         tendenciaLongScore * 0.20 +
         rsiLong * 0.15 +
         kvoLong * 0.15 +
         volumeScore * 0.10 +
         retornoLong1 * 0.10 +
         retornoLong3 * 0.05 +
         (tendenciaLong ? 100.0 : 40.0) * 0.10)

    scoreShort = (
         forcaCandle * 0.15 +
         tendenciaShortScore * 0.20 +
         rsiShort * 0.15 +
         kvoShort * 0.15 +
         volumeScore * 0.10 +
         retornoShort1 * 0.10 +
         retornoShort3 * 0.05 +
         (tendenciaShort ? 100.0 : 40.0) * 0.10)

    microMax = ta.highest(high[1], microLookback)

    microMin = ta.lowest(low[1], microLookback)

    rompLong = close > microMax

    rompShort = close < microMin

    microEma3 = ta.ema(close, 3)

    microEma8 = ta.ema(close, 8)

    microLong = rompLong ? 100.0 : microEma3 > microEma8 ? 65.0 : 25.0

    microShort = rompShort ? 100.0 : microEma3 < microEma8 ? 65.0 : 25.0

    confluenciaLong = (
         forcaCandle +
         tendenciaLongScore +
         kvoLong +
         rsiLong +
         retornoLong1 +
         retornoLong3) / 6.0

    confluenciaShort = (
         forcaCandle +
         tendenciaShortScore +
         kvoShort +
         rsiShort +
         retornoShort1 +
         retornoShort3) / 6.0

    vantagemLong = scoreLong - scoreShort

    vantagemShort = scoreShort - scoreLong

    [scoreLong[1], scoreShort[1], atrPct[1], volatilidadeRelativa[1], tendenciaLongScore[1], tendenciaShortScore[1], confluenciaLong[1], confluenciaShort[1], microLong[1], microShort[1], volumeScore[1], vantagemLong[1], vantagemShort[1]]

     l1,
     l5,
     l15,
     l30,
     l60,
     l240,
     ld) =>
    (l1 * 10.0 +
     l5 * 15.0 +
     l15 * 20.0 +
     l30 * 20.0 +
     l60 * 15.0 +
     l240 * 10.0 +
     ld * 10.0) / 100.0


f_mtfShort(
     s1,

     s15,
     s30,
     s60,
     s240,
     sd) =>
    (s1 * 10.0 +
     s5 * 15.0 +
     s15 * 20.0 +
     s30 * 20.0 +
     s60 * 15.0 +
     s240 * 10.0 +
     sd * 10.0) / 100.0
