// ============================================================================
// FUNCOES BASICAS
// ============================================================================

f_limitar(valor, minimo, maximo) =>
    math.max(minimo, math.min(maximo, valor))


f_nz(valor, fallback) =>
    na(valor) ? fallback : valor


f_scoreRetorno(retorno, escala) =>
    math.max(0.0, math.min(100.0, 50.0 + retorno / escala * 50.0))


f_dirTexto(direcao) =>
    direcao == 1 ? "COMPRA" : direcao == -1 ? "VENDA" : "AGUARDAR"


f_scoreTexto(valor) =>
    str.tostring(math.round(valor)) + "%"


f_corDirecao(direcao) =>
    direcao == 1 ? corVerde : direcao == -1 ? corVermelho : corCinza


f_min4(a, b, c, d) =>
    math.min(math.min(a, b), math.min(c, d))


f_max4(a, b, c, d) =>
    math.max(math.max(a, b), math.max(c, d))

f_mtfPacote() =>
    atrLocal = ta.atr(atrLen)
    atrPctLocal = atrLocal / close * 100.0

    ema9 = ta.ema(close, emaRapida)
    ema21 = ta.ema(close, emaMedia)
    ema50 = ta.ema(close, emaLenta)

    rsiLocal = ta.rsi(close, rsiLen)

    volumeMedia = ta.sma(volume, volumeLen)
    volumeRelativo = volumeMedia > 0 ? volume / volumeMedia : 1.0

    corpo = math.abs(close - open)
    rangeCandle = math.max(high - low, syminfo.mintick)
    forcaCandle = corpo / rangeCandle * 100.0

    retorno1 = close / close[1] - 1.0
    retorno3 = close / close[3] - 1.0

    tendenciaLong = ema9 > ema21 and ema21 > ema50
    tendenciaShort = ema9 < ema21 and ema21 < ema50

    tendenciaScoreLong = tendenciaLong ? 100.0 : ema9 > ema21 ? 65.0 : 30.0
    tendenciaScoreShort = tendenciaShort ? 100.0 : ema9 < ema21 ? 65.0 : 30.0

    rsiLong = rsiLocal > 50 ? f_limitar((rsiLocal - 50.0) * 2.0, 0.0, 100.0) : 0.0
    rsiShort = rsiLocal < 50 ? f_limitar((50.0 - rsiLocal) * 2.0, 0.0, 100.0) : 0.0

    kvoBase = ta.ema(volume * (close - close[1]), 34)
    kvoSignal = ta.ema(kvoBase, 55)

    kvoLong = kvoBase > kvoSignal ? 100.0 : 25.0
    kvoShort = kvoBase < kvoSignal ? 100.0 : 25.0

    volumeLong = volumeRelativo > 1.0 ? f_limitar(volumeRelativo * 50.0, 0.0, 100.0) : 35.0
    volumeShort = volumeRelativo > 1.0 ? f_limitar(volumeRelativo * 50.0, 0.0, 100.0) : 35.0

    retornoLong1 = f_scoreRetorno(retorno1, 0.01)
    retornoShort1 = f_scoreRetorno(-retorno1, 0.01)

    retornoLong3 = f_scoreRetorno(retorno3, 0.02)
    retornoShort3 = f_scoreRetorno(-retorno3, 0.02)

    scoreLong = (
         forcaCandle * 0.15 +
         tendenciaScoreLong * 0.20 +
         rsiLong * 0.15 +
         kvoLong * 0.15 +
         volumeLong * 0.10 +
         retornoLong1 * 0.10 +
         retornoLong3 * 0.05 +
         (tendenciaLong ? 100.0 : 40.0) * 0.10)

    scoreShort = (
         forcaCandle * 0.15 +
         tendenciaScoreShort * 0.20 +
         rsiShort * 0.15 +
         kvoShort * 0.15 +
         volumeShort * 0.10 +
         retornoShort1 * 0.10 +
         retornoShort3 * 0.05 +
         (tendenciaShort ? 100.0 : 40.0) * 0.10)

    resistencia = ta.highest(high, lookbackEstrutura)
    suporte = ta.lowest(low, lookbackEstrutura)

    [
         nz(scoreLong[1], 50.0),
         nz(scoreShort[1], 50.0),
         nz(atrPctLocal[1], 0.0),
         nz(resistencia[1], high),
         nz(suporte[1], low)
     ]

     short30M * peso30 +

ema50Local = ta.ema(close, emaLenta)

rsiLocal = ta.rsi(close, rsiLen)

volumeMediaLocal = ta.sma(volume, volumeLen)

volumeRelativoLocal = volumeMediaLocal > 0 ? volume / volumeMediaLocal : 1.0

forcaCandleLocal = corpoLocal / rangeLocal * 100.0

retornoShort1 = f_scoreRetorno(-retorno1Local, 0.01)

retornoShort3 = f_scoreRetorno(-retorno3Local, 0.02)

// TENDENCIA LOCAL
// ============================================================================

// ============================================================================

// ============================================================================
// KVO
// ============================================================================

kvoBaseLocal = ta.ema(volume * (close - close[1]), 34)

kvoSignalLocal = ta.ema(kvoBaseLocal, 55)

kvoLongScore = kvoBaseLocal > kvoSignalLocal ? 100.0 : 25.0

kvoShortScore = kvoBaseLocal < kvoSignalLocal ? 100.0 : 25.0


// ============================================================================
// VOLUME
// ============================================================================

volumeScore = volumeRelativoLocal > 1.0 ? f_limitar(volumeRelativoLocal * 50.0, 0.0, 100.0) : 35.0


// ============================================================================

     forcaCandleLocal * 0.15 +
     tendenciaLongScore * 0.20 +
     rsiLongScore * 0.15 +
     kvoLongScore * 0.15 +
     volumeScore * 0.10 +
     retornoLong1 * 0.10 +
     retornoLong3 * 0.05 +
     (tendenciaLongLocal ? 100.0 : 40.0) * 0.10)

scoreLocalShort = (
     forcaCandleLocal * 0.15 +
     tendenciaShortScore * 0.20 +
     rsiShortScore * 0.15 +
     kvoShortScore * 0.15 +
     volumeScore * 0.10 +
     retornoShort1 * 0.10 +
     retornoShort3 * 0.05 +
     (tendenciaShortLocal ? 100.0 : 40.0) * 0.10)


// ============================================================================
// SCORE FINAL
// ============================================================================

scoreFinalLong = scoreLocalLong * 0.50 + mtfLong * 0.50

scoreFinalShort = scoreLocalShort * 0.50 + mtfShort * 0.50

scoreFinalLong := f_limitar(scoreFinalLong, 0.0, 100.0)

scoreFinalShort := f_limitar(scoreFinalShort, 0.0, 100.0)


// ============================================================================
// INTENSIDADE
// ============================================================================

intensidadeLong = (
     scoreFinalLong * 0.60 +
     tendenciaLongScore * 0.20 +
     forcaCandleLocal * 0.10 +
     volumeScore * 0.10)

intensidadeShort = (
     scoreFinalShort * 0.60 +
     tendenciaShortScore * 0.20 +
     forcaCandleLocal * 0.10 +
     volumeScore * 0.10)

intensidadeLong := f_limitar(intensidadeLong, 0.0, 100.0)

intensidadeShort := f_limitar(intensidadeShort, 0.0, 100.0)


// ============================================================================
// CONFLUENCIA
// ============================================================================

confluenciaLong = (
     forcaCandleLocal +
     tendenciaLongScore +
     kvoLongScore +
     rsiLongScore +
     retornoLong1 +
     retornoLong3 +
     mtfLong) / 7.0

confluenciaShort = (
     forcaCandleLocal +
     tendenciaShortScore +
     kvoShortScore +
     rsiShortScore +
     retornoShort1 +
     retornoShort3 +
     mtfShort) / 7.0

confluenciaLong := f_limitar(confluenciaLong, 0.0, 100.0)

confluenciaShort := f_limitar(confluenciaShort, 0.0, 100.0)


// ============================================================================
// VANTAGEM
// ============================================================================

vantagemLong = scoreFinalLong - scoreFinalShort

vantagemShort = scoreFinalShort - scoreFinalLong

vantagemAbsoluta = math.abs(scoreFinalLong - scoreFinalShort)


// ============================================================================
// MICROESTRUTURA
// ============================================================================

microMaxAnterior = ta.highest(high[1], microLookback)

microMinAnterior = ta.lowest(low[1], microLookback)

rompimentoLong = close > microMaxAnterior

rompimentoShort = close < microMinAnterior

microEma3 = ta.ema(close, 3)

microEma8 = ta.ema(close, 8)

microTendenciaLong = microEma3 > microEma8

microTendenciaShort = microEma3 < microEma8

microScoreLong = (
     (rompimentoLong ? 100.0 : 0.0) * 0.60 +
     (microTendenciaLong ? 100.0 : 0.0) * 0.40)

microScoreShort = (
     (rompimentoShort ? 100.0 : 0.0) * 0.60 +
     (microTendenciaShort ? 100.0 : 0.0) * 0.40)

microConfirmadoLong = microScoreLong >= confirmacaoMicroMin

microConfirmadoShort = microScoreShort >= confirmacaoMicroMin


// ============================================================================
// ESTRUTURA MTF
// ============================================================================

resistencia30 = res30M
resistencia1H = res1H
resistencia4H = res4H
resistenciaD = resD

suporte30 = sup30M
suporte1H = sup1H
suporte4H = sup4H
suporteD = supD

resistenciaPrincipal = f_min4(
     resistencia30,
     resistencia1H,
     resistencia4H,
     resistenciaD)

suportePrincipal = f_max4(
     suporte30,
     suporte1H,
     suporte4H,
     suporteD)


// ============================================================================
// DISTANCIA E ESTRUTURA
// ============================================================================

distanciaResistencia = resistenciaPrincipal - close

distanciaSuporte = close - suportePrincipal

espacoResistenciaATR = distanciaResistencia / atrSeguro

espacoSuporteATR = distanciaSuporte / atrSeguro

rompimentoEstruturalLong = close > resistenciaPrincipal

rompimentoEstruturalShort = close < suportePrincipal

estruturaLongValida = (
     usarRompimento and rompimentoEstruturalLong) or
     (espacoResistenciaATR >= espacoMinLong)

estruturaShortValida = (
     usarRompimento and rompimentoEstruturalShort) or
     (espacoSuporteATR >= espacoMinShort)


// ============================================================================
// CONDICOES COMPLETAS
// ============================================================================

condLongScore = scoreFinalLong >= scoreMinLong

condShortScore = scoreFinalShort >= scoreMinShort

condLongIntensidade = intensidadeLong >= intensidadeMinLong

condShortIntensidade = intensidadeShort >= intensidadeMinShort

condLongConfluencia = confluenciaLong >= confluenciaMinLong

condShortConfluencia = confluenciaShort >= confluenciaMinShort

condLongMTF = mtfLong >= mtfMinLong

condShortMTF = mtfShort >= mtfMinShort

condLongVantagem = vantagemLong >= vantagemMinima

condShortVantagem = vantagemShort >= vantagemMinima

condLongMicro = microConfirmadoLong

condShortMicro = microConfirmadoShort


// ============================================================================
// TODAS AS CONDICOES LONG
// ============================================================================

sinalLongBruto = (
     ativarLong and
     condLongScore and
     condLongIntensidade and
     condLongConfluencia and
     condLongMTF and
     condLongVantagem and
     condLongMicro and
     estruturaLongValida)


// ============================================================================
// TODAS AS CONDICOES SHORT
// ============================================================================

sinalShortBruto = (
     ativarShort and
     condShortScore and
     condShortIntensidade and
     condShortConfluencia and
     condShortMTF and
     condShortVantagem and
     condShortMicro and
     estruturaShortValida)


// ============================================================================
// DECISAO FINAL
// ============================================================================
//
// Se apenas COMPRA passar:
// COMPRA
//
// Se apenas VENDA passar:
// VENDA
//
// Se ambos passarem:
// escolhe somente se houver vantagem suficiente
//
// Se nenhum passar:
// AGUARDAR
// ============================================================================

direcaoAtual = 0

if sinalLongBruto and not sinalShortBruto
    direcaoAtual := 1
else if sinalShortBruto and not sinalLongBruto
    direcaoAtual := -1
else if sinalLongBruto and sinalShortBruto
    if vantagemLong >= vantagemMinima and vantagemLong > vantagemShort
        direcaoAtual := 1
    else if vantagemShort >= vantagemMinima and vantagemShort > vantagemLong
        direcaoAtual := -1
    else
        direcaoAtual := 0
else
    direcaoAtual := 0


// ============================================================================
// CONFIRMACAO FINAL
// ============================================================================

compraConfirmada = direcaoAtual == 1

vendaConfirmada = direcaoAtual == -1

aguardar = direcaoAtual == 0


// ============================================================================
// MUDANCA DE DIRECAO
// ============================================================================

mudouParaCompra = compraConfirmada and direcaoAtual[1] != 1

mudouParaVenda = vendaConfirmada and direcaoAtual[1] != -1


// ============================================================================
// ENTRADA PROJETADA
// ============================================================================

forcaEntradaLong = f_limitar(scoreFinalLong / 100.0, 0.0, 1.0)

forcaEntradaShort = f_limitar(scoreFinalShort / 100.0, 0.0, 1.0)

distEntradaLong = atrSeguro * 0.10 * (1.0 - forcaEntradaLong)

distEntradaShort = atrSeguro * 0.10 * (1.0 - forcaEntradaShort)

entradaProjetadaLong = close + distEntradaLong

entradaProjetadaShort = close - distEntradaShort

entradaProjetada = compraConfirmada ? entradaProjetadaLong : vendaConfirmada ? entradaProjetadaShort : na


// ============================================================================
// ESTADO FIXO DA OPERACAO
// ============================================================================

var float entradaFixada = na
var float atrFixado = na
var float scoreFixado = na
var float intensidadeFixada = na
var float vantagemFixada = na
var int direcaoFixada = 0
var int barraEntrada = na

novaEntrada = mudouParaCompra or mudouParaVenda

if novaEntrada
    entradaFixada := entradaProjetada
    atrFixado := atrSeguro
    scoreFixado := direcaoAtual == 1 ? scoreFinalLong : scoreFinalShort
    intensidadeFixada := direcaoAtual == 1 ? intensidadeLong : intensidadeShort
    vantagemFixada := direcaoAtual == 1 ? vantagemLong : vantagemShort
    direcaoFixada := direcaoAtual
    barraEntrada := bar_index


// ============================================================================
// RESET QUANDO PERDE O SINAL
// ============================================================================

perdeuDirecao = direcaoAtual == 0 and direcaoFixada != 0

if perdeuDirecao
    direcaoFixada := 0


// ============================================================================
// ALVOS 1 A 8 ATR
// ============================================================================

alvo1Long = entradaFixada + atrFixado * 1.0
alvo2Long = entradaFixada + atrFixado * 2.0
alvo3Long = entradaFixada + atrFixado * 3.0
alvo4Long = entradaFixada + atrFixado * 4.0
alvo5Long = entradaFixada + atrFixado * 5.0
alvo6Long = entradaFixada + atrFixado * 6.0
alvo7Long = entradaFixada + atrFixado * 7.0
alvo8Long = entradaFixada + atrFixado * 8.0

alvo1Short = entradaFixada - atrFixado * 1.0
alvo2Short = entradaFixada - atrFixado * 2.0
alvo3Short = entradaFixada - atrFixado * 3.0
alvo4Short = entradaFixada - atrFixado * 4.0
alvo5Short = entradaFixada - atrFixado * 5.0
alvo6Short = entradaFixada - atrFixado * 6.0
alvo7Short = entradaFixada - atrFixado * 7.0
alvo8Short = entradaFixada - atrFixado * 8.0


// ============================================================================
// ALVOS ATUAIS
// ============================================================================

alvo1 = direcaoFixada == 1 ? alvo1Long : direcaoFixada == -1 ? alvo1Short : na
alvo2 = direcaoFixada == 1 ? alvo2Long : direcaoFixada == -1 ? alvo2Short : na
alvo3 = direcaoFixada == 1 ? alvo3Long : direcaoFixada == -1 ? alvo3Short : na
alvo4 = direcaoFixada == 1 ? alvo4Long : direcaoFixada == -1 ? alvo4Short : na
alvo5 = direcaoFixada == 1 ? alvo5Long : direcaoFixada == -1 ? alvo5Short : na
alvo6 = direcaoFixada == 1 ? alvo6Long : direcaoFixada == -1 ? alvo6Short : na
alvo7 = direcaoFixada == 1 ? alvo7Long : direcaoFixada == -1 ? alvo7Short : na
alvo8 = direcaoFixada == 1 ? alvo8Long : direcaoFixada == -1 ? alvo8Short : na


// ============================================================================
// DETECCAO DE ALVOS
// ============================================================================

alvo1Atingido = direcaoFixada == 1 ? high >= alvo1Long : direcaoFixada == -1 ? low <= alvo1Short : false

alvo2Atingido = direcaoFixada == 1 ? high >= alvo2Long : direcaoFixada == -1 ? low <= alvo2Short : false

alvo3Atingido = direcaoFixada == 1 ? high >= alvo3Long : direcaoFixada == -1 ? low <= alvo3Short : false

alvo4Atingido = direcaoFixada == 1 ? high >= alvo4Long : direcaoFixada == -1 ? low <= alvo4Short : false
