// O algoritmo força todos os cálculos para o timeframe Diário.
// Pode ser colocado em outros gráficos, mas a análise operacional continua
// sendo feita exclusivamente com candles diários.

// ============================================================================
// CONFIGURAÇÕES
// ============================================================================

groupConfig = "V11.0 DIÁRIO - CONFIGURAÇÕES"

corpoMinimo = input.float(0.60, "Corpo mínimo", minval=0.10, maxval=1.00, step=0.05, group=groupConfig)

atrPeriodo = input.int(14, "ATR período", minval=2, group=groupConfig)

suavizacaoPeriodo = input.int(5, "Suavização", minval=2, maxval=30, group=groupConfig)

estruturaLookback = input.int(30, "Estrutura diária", minval=5, maxval=200, group=groupConfig)

confirmacaoLookback = input.int(8, "Confirmação diária", minval=2, maxval=50, group=groupConfig)

microLookback = input.int(5, "Micro-rompimento diário", minval=2, maxval=30, group=groupConfig)

microTrendFast = input.int(5, "Micro tendência rápida", minval=2, maxval=20, group=groupConfig)

microTrendSlow = input.int(13, "Micro tendência lenta", minval=5, maxval=50, group=groupConfig)

kvoFast = input.int(34, "KVO Fast", minval=2, group=groupConfig)

kvoSlow = input.int(55, "KVO Slow", minval=3, group=groupConfig)

kvoSignalPeriodo = input.int(13, "KVO Signal", minval=2, group=groupConfig)

volumePeriodo = input.int(20, "Volume média diária", minval=5, group=groupConfig)

boostMaximo = input.float(25.0, "Boost máximo", minval=0, maxval=50, group=groupConfig)

penaltyMaximo = input.float(20.0, "Penalty máximo", minval=0, maxval=50, group=groupConfig)

mostrarTabela = input.bool(true, "Mostrar painel", group=groupConfig)

mostrarProbabilidadeGrafico = input.bool(true, "Plotar probabilidade no gráfico", group=groupConfig)

mostrarMicroBreakout = input.bool(true, "Marcar micro-rompimento", group=groupConfig)

mostrarCandleForca = input.bool(true, "Colorir candle de força", group=groupConfig)

usarWebhook = input.bool(true, "Ativar webhook", group=groupConfig)

alertaSomenteFechamento = input.bool(true, "Alertar somente fechamento diário", group=groupConfig)


// ============================================================================
// CONTROLE DIÁRIO
// ============================================================================

groupDiario = "V11.0 DIÁRIO - CONTROLE"

usarSomenteDiario = input.bool(true, "Forçar análise Diário", group=groupDiario)

bloquearIntradiario = input.bool(true, "Bloquear sinais intradiários", group=groupDiario)




// ============================================================================
// SHORT
// ============================================================================

groupShort = "V11.0 DIÁRIO - SHORT PROBABILÍSTICO"

ativarShort = input.bool(true, "Ativar SHORT", group=groupShort)

probMinimaShort = input.float(72.0, "Probabilidade mínima SHORT", minval=50, maxval=99, step=1, group=groupShort)

probMinimaForte = input.float(82.0, "Probabilidade SHORT FORTE", minval=50, maxval=99, step=1, group=groupShort)

probMinimaExtremo = input.float(90.0, "Probabilidade SHORT EXTREMO", minval=60, maxval=99, step=1, group=groupShort)

confluenciaMinima = input.int(7, "Confluência mínima", minval=2, maxval=15, group=groupShort)

usarFiltroSuporte = input.bool(true, "Filtro de suporte", group=groupShort)

distanciaMinimaSuporteATR = input.float(0.60, "Distância mínima do suporte ATR", minval=0.10, maxval=10, step=0.10, group=groupShort)


// ============================================================================
// MODELO PROBABILÍSTICO
// ============================================================================

groupProb = "V11.0 DIÁRIO - MODELO PROBABILÍSTICO"

probSensibilidade = input.float(4.0, "Sensibilidade probabilística", minval=1, maxval=10, step=0.5, group=groupProb)

probSuavizacao = input.int(5, "Suavização probabilidade", minval=2, maxval=30, group=groupProb)

calibracaoPeriodo = input.int(100, "Janela calibração diária", minval=20, maxval=500, group=groupProb)

horizonteCalibracao = input.int(5, "Horizonte calibração diário", minval=1, maxval=30, group=groupProb)

pesoProbModelo = input.float(0.70, "Peso modelo", minval=0, maxval=1, step=0.05, group=groupProb)

pesoProbHistorica = input.float(0.30, "Peso histórico", minval=0, maxval=1, step=0.05, group=groupProb)


// ============================================================================
// CORES
// ============================================================================

corVerde = color.rgb(0, 255, 100)

corVerdeForte = color.rgb(0, 255, 0)

corVermelho = color.rgb(255, 60, 60)

corVermelhoForte = color.rgb(255, 0, 0)

corAmarelo = color.rgb(255, 210, 0)

corCinza = color.rgb(150, 150, 150)

corCiano = color.rgb(0, 220, 255)

corAzul = color.rgb(0, 150, 255)

corBranco = color.white

corRoxo = color.rgb(180, 80, 255)

corFundo = color.rgb(15, 15, 20)

corBorda = color.rgb(70, 70, 80)


// ============================================================================
// DADOS DIÁRIOS
// ============================================================================

[openD, highD, lowD, closeD, volumeD] = request.security(
     syminfo.tickerid,
     "1D",
     [open, high, low, close, volume],
     lookahead=barmerge.lookahead_off)


// ============================================================================
// SÉRIES DIÁRIAS
// ============================================================================

rangeCandle = highD - lowD

rangeSeguro = math.max(rangeCandle, syminfo.mintick)

corpo = math.abs(closeD - openD)

corpoPct = corpo / rangeSeguro

pavioSuperior = highD - math.max(openD, closeD)

pavioInferior = math.min(openD, closeD) - lowD

pavioSuperiorPct = pavioSuperior / rangeSeguro

pavioInferiorPct = pavioInferior / rangeSeguro

direcaoCandle = closeD > openD ? 1 : closeD < openD ? -1 : 0


// ============================================================================
// RETORNOS DIÁRIOS
// ============================================================================

retorno1 = not na(closeD[1]) ? closeD / closeD[1] - 1 : 0.0

retorno3 = not na(closeD[3]) ? closeD / closeD[3] - 1 : 0.0

retorno5 = not na(closeD[5]) ? closeD / closeD[5] - 1 : 0.0

retorno10 = not na(closeD[10]) ? closeD / closeD[10] - 1 : 0.0


// ============================================================================
// MÉDIAS DIÁRIAS
// ============================================================================

mm5 = ta.ema(closeD, 5)

mm10 = ta.ema(closeD, 10)

mm20 = ta.ema(closeD, 20)

mm50 = ta.ema(closeD, 50)

mm100 = ta.ema(closeD, 100)

distMm5 = closeD != 0 ? (closeD / mm5 - 1) * 100 : 0.0

distMm20 = closeD != 0 ? (closeD / mm20 - 1) * 100 : 0.0

distMm50 = closeD != 0 ? (closeD / mm50 - 1) * 100 : 0.0


// ============================================================================
// MICRO-TENDÊNCIA DIÁRIA
// ============================================================================

microFast = ta.ema(closeD, microTrendFast)

microSlow = ta.ema(closeD, microTrendSlow)

microFastAnterior = microFast[1]

microSlowAnterior = microSlow[1]

microTrendBaixa = microFast < microSlow and microFastAnterior <= microSlowAnterior

microTrendAlta = microFast > microSlow and microFastAnterior >= microSlowAnterior

microTrendDist = closeD != 0 ? math.abs(microFast - microSlow) / closeD * 100 : 0.0

microTrendScore = microFast < microSlow ? math.min(100.0, 50.0 + microTrendDist * 100.0) : 0.0


// ============================================================================
// ATR DIÁRIO
// ============================================================================

atrBruto = ta.atr(atrPeriodo)

atrSuaveEMA = ta.ema(atrBruto, suavizacaoPeriodo)

atrSuaveRMA = ta.rma(atrBruto, suavizacaoPeriodo)

atrHarmonizado = atrSuaveEMA * 0.60 + atrSuaveRMA * 0.40

atrSeguro = math.max(atrHarmonizado, syminfo.mintick)

atrMedia = ta.ema(atrHarmonizado, 34)

atrRatio = atrMedia > 0 ? atrHarmonizado / atrMedia : 1.0

atrExpansao = atrHarmonizado > atrHarmonizado[1]

atrContraindo = atrHarmonizado < atrHarmonizado[1]

atrPercentual = closeD != 0 ? atrSeguro / closeD * 100 : 0.0

atrRegime =
     atrRatio >= 1.35 ? 100.0 :
     atrRatio >= 1.15 ? 85.0 :
     atrRatio >= 1.00 ? 70.0 :
     atrRatio >= 0.85 ? 55.0 :
     40.0


// ============================================================================
// VOLUME DIÁRIO
// ============================================================================

volumeMedia = ta.ema(volumeD, volumePeriodo)

volumeRelativo = volumeMedia > 0 ? volumeD / volumeMedia : 1.0

volumeScore =
     volumeRelativo >= 2.50 ? 100.0 :
     volumeRelativo >= 2.00 ? 95.0 :
     volumeRelativo >= 1.50 ? 85.0 :
     volumeRelativo >= 1.20 ? 72.0 :
     volumeRelativo >= 1.00 ? 60.0 :
     40.0


// ============================================================================
// ESTRUTURA DIÁRIA
// ============================================================================

suporte = ta.lowest(lowD[1], estruturaLookback)

resistencia = ta.highest(highD[1], estruturaLookback)

microSuporte = ta.lowest(lowD[1], microLookback)

microResistencia = ta.highest(highD[1], microLookback)

distSuporteATR = (closeD - suporte) / atrSeguro

distResistenciaATR = (resistencia - closeD) / atrSeguro


// ============================================================================
// MICRO-ROMPIMENTO DIÁRIO
// ============================================================================

microBreakoutVenda = closeD < microSuporte

microBreakoutVendaWick = lowD < microSuporte and closeD < openD

microBreakoutForca =
     microBreakoutVenda ? 100.0 :
     microBreakoutVendaWick ? 75.0 :
     closeD < microSuporte + atrSeguro * 0.15 ? 55.0 :
     25.0


// ============================================================================
// ROMPIMENTO ESTRUTURAL DIÁRIO
// ============================================================================

rompimentoEstruturalVenda = closeD < suporte

forcaRompimento =
     rompimentoEstruturalVenda ? 100.0 :
     microBreakoutVenda ? 80.0 :
     microBreakoutVendaWick ? 65.0 :
     30.0


// ============================================================================
// KVO DIÁRIO
// ============================================================================

precoTipico = (highD + lowD + closeD) / 3.0

tendenciaKvo =
     precoTipico > precoTipico[1] ? 1 :
     precoTipico < precoTipico[1] ? -1 :
     0

dmKvo = highD - lowD

dmSeguro = math.max(dmKvo, syminfo.mintick)

vfKvo = volumeD * (2 * (dmKvo / dmSeguro) - 1) * tendenciaKvo

kvo = ta.ema(vfKvo, kvoFast) - ta.ema(vfKvo, kvoSlow)

kvoSignal = ta.ema(kvo, kvoSignalPeriodo)

kvoHist = kvo - kvoSignal

kvoVariacao = kvo - kvo[1]

kvoDesvio = ta.stdev(kvo, 34)

kvoZscore = kvoDesvio > 0 ? kvo / kvoDesvio : 0.0

kvoEstado =
     kvo < kvoSignal and kvoVariacao < 0 ? -1 :
     kvo > kvoSignal and kvoVariacao > 0 ? 1 :
     0

kvoForcaBruta =
     math.abs(kvoHist) /
     (math.abs(kvo) + math.abs(kvoSignal) + 0.000001)

kvoForca = math.min(100.0, kvoForcaBruta * 100.0)

kvoVendaScore =
     kvoEstado < 0 ? kvoForca :
     kvoEstado == 0 ? kvoForca * 0.35 :
     0.0


// ============================================================================
// RSI DIÁRIO
// ============================================================================

rsi = ta.rsi(closeD, 14)

rsiVendaScore =
     rsi <= 25 ? 100.0 :
     rsi <= 35 ? 90.0 :
     rsi <= 45 ? 75.0 :
     rsi < 50 ? 60.0 :
     40.0


// ============================================================================
// MOMENTUM DIÁRIO
// ============================================================================

momentum = ta.roc(closeD, 5)

momentumVendaScore =
     momentum <= -3 ? 100.0 :
     momentum <= -2 ? 90.0 :
     momentum <= -1 ? 80.0 :
     momentum < 0 ? 65.0 :
     35.0


// ============================================================================
// SUAVIZAÇÃO
// ============================================================================

f_harmonizar(src, len) =>
    emaValor = ta.ema(src, len)
    rmaValor = ta.rma(src, len)
    emaValor * 0.60 + rmaValor * 0.40

kvoHarmonizado = f_harmonizar(kvoForca, suavizacaoPeriodo)

momentumHarmonizado = f_harmonizar(momentum, suavizacaoPeriodo)

microScoreHarmonizado = f_harmonizar(microBreakoutForca, suavizacaoPeriodo)


// ============================================================================
// MODELO DIRECIONAL
// ============================================================================

float pontosVenda = 0.0

float pontosCompra = 0.0


if closeD < openD
    pontosVenda += 2
else if closeD > openD
    pontosCompra += 2


if closeD < mm5
    pontosVenda += 1
else
    pontosCompra += 1


if closeD < mm10
    pontosVenda += 1
else
    pontosCompra += 1


if closeD < mm20
    pontosVenda += 1
else
    pontosCompra += 1


if closeD < mm50
    pontosVenda += 1
else
    pontosCompra += 1


if retorno1 < 0
    pontosVenda += 1
else
    pontosCompra += 1


if retorno3 < 0
    pontosVenda += 1
else
    pontosCompra += 1


if retorno5 < 0
    pontosVenda += 1
else
    pontosCompra += 1


if kvoEstado < 0
    pontosVenda += 2
else if kvoEstado > 0
    pontosCompra += 2


if volumeRelativo >= 1.20 and closeD < openD
    pontosVenda += 1


if microFast < microSlow
    pontosVenda += 2
else
    pontosCompra += 2


if microBreakoutVenda
    pontosVenda += 3


if rompimentoEstruturalVenda
    pontosVenda += 3


totalModelo = pontosVenda + pontosCompra

probModeloVenda =
     totalModelo > 0 ?
     pontosVenda / totalModelo * 100 :
     50.0

probModeloCompra =
     totalModelo > 0 ?
     pontosCompra / totalModelo * 100 :
     50.0


// ============================================================================
// MODELO LOGÍSTICO
// ============================================================================

scoreNormalizado = (probModeloVenda - 50.0) / 50.0

scoreKvo = (kvoVendaScore - 50.0) / 50.0

scoreMicro = (microScoreHarmonizado - 50.0) / 50.0

scoreVolume = (volumeScore - 50.0) / 50.0

scoreATR = (atrRegime - 50.0) / 50.0

scoreMomentum =
     momentumHarmonizado < 0 ?
     math.min(1.0, math.abs(momentumHarmonizado) / 5.0) :
     -math.min(1.0, momentumHarmonizado / 5.0)

scoreEstrutura =
     rompimentoEstruturalVenda ? 1.0 :
     microBreakoutVenda ? 0.75 :
     -0.10

scoreTendencia =
     microFast < microSlow ?
     math.min(1.0, 0.50 + microTrendScore / 200.0) :
     -0.50

scoreComposto =
     scoreNormalizado * 0.20 +
     scoreKvo * 0.15 +
     scoreMicro * 0.15 +
     scoreVolume * 0.10 +
     scoreATR * 0.10 +
     scoreMomentum * 0.10 +
     scoreEstrutura * 0.10 +
     scoreTendencia * 0.10

probLogisticaVenda =
     1.0 /
     (1.0 + math.exp(-probSensibilidade * scoreComposto)) *
     100.0


// ============================================================================
// MTF DIÁRIO
// ============================================================================
//
// Em vez de combinar 1m/15m/30m/4H/1D,
// o modelo diário utiliza uma confirmação interna baseada em:
//
// EMA 8
// EMA 21
// EMA 50
// Momentum diário
// KVO proxy diário
// Breakout diário
// ATR diário
//
// Isso evita redundância e mantém a operação exclusivamente no Diário.
// ============================================================================

ema8D = ta.ema(closeD, 8)

ema21D = ta.ema(closeD, 21)

ema50D = ta.ema(closeD, 50)

atrLocal = ta.atr(14)

atrLocalMedia = ta.ema(atrLocal, 34)

atrLocalRatio =
     atrLocalMedia > 0 ?
     atrLocal / atrLocalMedia :
     1.0

microBaixaD =
     ema8D < ema21D and
     ema21D < ema50D

distanciaD =
     closeD != 0 ?
     (ema21D - closeD) / closeD * 100.0 :
     0.0

trendScoreD =
     microBaixaD ?
     math.min(100.0, 55.0 + math.abs(distanciaD) * 10.0) :
     closeD < ema21D ? 55.0 :
     30.0

retornoLocalD =
     not na(closeD[3]) ?
     closeD / closeD[3] - 1.0 :
     0.0

momentumScoreD =
     retornoLocalD < -0.02 ? 100.0 :
     retornoLocalD < -0.01 ? 85.0 :
     retornoLocalD < 0 ? 65.0 :
     30.0

kvoProxyD =
     ta.ema(volumeD * math.sign(ta.change((highD + lowD + closeD) / 3.0)), 13)

kvoProxyMediaD = ta.ema(kvoProxyD, 21)

kvoScoreD =
     kvoProxyD < kvoProxyMediaD ? 85.0 :
     40.0

breakoutD =
     closeD < ta.lowest(lowD[1], 5)

breakoutScoreD =
     breakoutD ? 100.0 :
     closeD < ta.lowest(lowD[1], 8) + atrLocal * 0.15 ? 65.0 :
     30.0

atrScoreD =
     atrLocalRatio >= 1.0 ? 80.0 :
     45.0

scoreDiario =
     trendScoreD * 0.35 +
     momentumScoreD * 0.20 +
     kvoScoreD * 0.20 +
     breakoutScoreD * 0.15 +
     atrScoreD * 0.10

direcaoDiaria =
     scoreDiario >= 55.0 ? -1 :
     1

mtfPercentualVenda = scoreDiario

mtfDirecao = direcaoDiaria

mtfConcordante = mtfDirecao < 0

mtfTexto =
     mtfDirecao < 0 ? "↓ VENDA" :
     mtfDirecao > 0 ? "↑ COMPRA" :
     "→ NEUTRO"


// ============================================================================
// PROBABILIDADE DO MODELO
// ============================================================================

probabilidadeModelo =
     probLogisticaVenda * 0.55 +
     mtfPercentualVenda * 0.25 +
     probModeloVenda * 0.20

probabilidadeModelo :=
     math.min(99.0, math.max(1.0, probabilidadeModelo))


// ============================================================================
// CONFLUÊNCIAS DIÁRIAS
// ============================================================================

int confirmacoes = 0

int divergencias = 0


if direcaoCandle < 0
    confirmacoes += 1
else
    divergencias += 1


if kvoEstado < 0
    confirmacoes += 1
else if kvoEstado > 0
    divergencias += 1


if closeD < mm5
    confirmacoes += 1
else
    divergencias += 1


if closeD < mm10
    confirmacoes += 1
else
    divergencias += 1


if closeD < mm20
    confirmacoes += 1
else
    divergencias += 1


if closeD < mm50
    confirmacoes += 1
else
    divergencias += 1


if retorno1 < 0
    confirmacoes += 1
else
    divergencias += 1


if retorno3 < 0
    confirmacoes += 1
else
    divergencias += 1


if microFast < microSlow
    confirmacoes += 1
else
    divergencias += 1


if microBreakoutVenda
    confirmacoes += 1


if volumeRelativo >= 1.20
    confirmacoes += 1


if mtfPercentualVenda >= 65.0
    confirmacoes += 1
else
    divergencias += 1


if atrRatio >= 1.0
    confirmacoes += 1


// ============================================================================
// CONFLUÊNCIA PERCENTUAL
// ============================================================================

totalConfluencia = confirmacoes + divergencias

confluenciaPercentual =
     totalConfluencia > 0 ?
     confirmacoes / totalConfluencia * 100.0 :
     0.0


// ============================================================================
// BOOST
// ============================================================================

boostConfluencia =
     confirmacoes >= 12 ? 25.0 :
     confirmacoes >= 10 ? 20.0 :
     confirmacoes >= 8 ? 15.0 :
     confirmacoes >= 6 ? 10.0 :
     confirmacoes >= 4 ? 5.0 :
     0.0

boostKvo =
     kvoEstado < 0 and kvoForca >= 70 ? 5.0 :
     kvoEstado < 0 and kvoForca >= 50 ? 3.0 :
     0.0

boostVolume =
     volumeRelativo >= 2.0 ? 5.0 :
     volumeRelativo >= 1.5 ? 3.0 :
     volumeRelativo >= 1.2 ? 2.0 :
     0.0

boostBreakout =
     microBreakoutVenda ? 5.0 :
     microBreakoutVendaWick ? 2.0 :
     0.0

boostDiario =
     mtfPercentualVenda >= 85 ? 8.0 :
     mtfPercentualVenda >= 75 ? 6.0 :
     mtfPercentualVenda >= 65 ? 4.0 :
     0.0

boostTotal =
     math.min(
         boostMaximo,
         boostConfluencia +
         boostKvo +
         boostVolume +
         boostBreakout +
         boostDiario)


// ============================================================================
// PENALTY
// ============================================================================

penalty =
     divergencias * 2.0 +
     (kvoEstado > 0 ? 6.0 : 0.0) +
     (mtfDirecao > 0 ? 8.0 : 0.0) +
     (microFast > microSlow ? 6.0 : 0.0) +
     (atrRatio < 0.75 ? 4.0 : 0.0)

penaltyFinal =
     math.min(penaltyMaximo, penalty)


// ============================================================================
// PROBABILIDADE PRÉ-CALIBRAÇÃO
// ============================================================================

probabilidadePreCalibracao =
     probabilidadeModelo +
     boostTotal * 0.45 -
     penaltyFinal * 0.55

probabilidadePreCalibracao :=
     math.min(
         99.0,
         math.max(
             1.0,
             probabilidadePreCalibracao))


// ============================================================================
// DIREÇÃO
// ============================================================================

direcaoIntensidade =
     probabilidadePreCalibracao >= 50.0 ? -1 :
     1

direcaoTexto =
     direcaoIntensidade < 0 ? "BAIXA" :
     "ALTA"


// ============================================================================
// SINAL BASE PARA CALIBRAÇÃO
// ============================================================================

sinalBaseCalibracao =
     probabilidadePreCalibracao >= 65.0 and
     direcaoIntensidade < 0 and
     mtfPercentualVenda >= 55.0


// ============================================================================
// CALIBRAÇÃO HISTÓRICA DIÁRIA
// ============================================================================

sinalAnterior =
     sinalBaseCalibracao[horizonteCalibracao]

movimentoFavoravel =
     closeD < closeD[horizonteCalibracao]

resultadoHistorico =
     sinalAnterior and movimentoFavoravel

amostraHistorica =
     ta.sma(
         sinalAnterior ? 1.0 : 0.0,
         calibracaoPeriodo) *
     calibracaoPeriodo

acertosHistoricos =
     ta.sma(
         resultadoHistorico ? 1.0 : 0.0,
         calibracaoPeriodo) *
     calibracaoPeriodo

taxaHistorica =
     amostraHistorica >= 5.0 ?
     acertosHistoricos / amostraHistorica * 100.0 :
     probabilidadePreCalibracao

pesoHistoricoEfetivo =
     amostraHistorica >= 10.0 ?
     pesoProbHistorica :
     0.0

pesoModeloEfetivo =
     amostraHistorica >= 10.0 ?
     pesoProbModelo :
     1.0

somaPesos =
     pesoModeloEfetivo +
     pesoHistoricoEfetivo

probabilidadeCalibrada =
     somaPesos > 0 ?
     (
         probabilidadePreCalibracao * pesoModeloEfetivo +
         taxaHistorica * pesoHistoricoEfetivo
     ) / somaPesos :
     probabilidadePreCalibracao


// ============================================================================
// PROBABILIDADE SUAVIZADA
// ============================================================================

probabilidadeSuave =
     ta.ema(
         probabilidadeCalibrada,
         probSuavizacao)

probabilidadeFinal =
     math.min(
         99.0,
         math.max(
             1.0,
             probabilidadeSuave))


// ============================================================================
// INTENSIDADE FINAL
// ============================================================================

intensidadeBase =
     probabilidadeModelo * 0.55 +

f_barra(valor) =>
    tamanho = 10

    v = math.min(
         100.0,
         math.max(
             0.0,
             valor))

    preenchido =
         int(
             math.round(
                 v / 100.0 *
                 tamanho))

    resultado = ""

    for i = 0 to tamanho - 1
        resultado +=
             i < preenchido ?
             "█" :
             "░"

    resultado
