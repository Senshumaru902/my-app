f_limitar(valor, minimo, maximo) =>
    math.max(minimo, math.min(maximo, valor))

f_preco(valor) =>
    na(valor) ? "-" : str.tostring(valor, format.mintick)

// CLASSIFICACAO VOLATILIDADE
// ============================================================================
f_classVolatilidade(score) =>
    score >= 90 ? "EXTREMA" : score >= 75 ? "MUITO FORTE" : score >= 60 ? "FORTE" : score >= 45 ? "MODERADA" : score >= 25 ? "FRACA" : "MUITO FRACA"

f_corVolatilidade(score) =>
    score >= 90 ? color.new(color.red, 0) : score >= 75 ? color.new(color.orange, 10) : score >= 60 ? color.new(color.yellow, 20) : score >= 45 ? color.new(color.blue, 30) : score >= 25 ? color.new(color.gray, 35) : color.new(color.gray, 70)

// ============================================================================
// CLASSIFICACAO FIBONACCI
// ============================================================================
f_classFib(posicao) =>
    posicao <= 23.6 ? "FRACA" : posicao <= 38.2 ? "MODERADA" : posicao <= 61.8 ? "FORTE" : posicao <= 78.6 ? "MUITO FORTE" : "EXTREMA"

f_nivelFib(posicao) =>
    d0 = math.abs(posicao - 0.0)
    d236 = math.abs(posicao - 23.6)
    d382 = math.abs(posicao - 38.2)
    d500 = math.abs(posicao - 50.0)
    d618 = math.abs(posicao - 61.8)
    d786 = math.abs(posicao - 78.6)
    d100 = math.abs(posicao - 100.0)
    menor = math.min(d0, math.min(d236, math.min(d382, math.min(d500, math.min(d618, math.min(d786, d100))))))
    menor == d0 ? "0.0%" : menor == d236 ? "23.6%" : menor == d382 ? "38.2%" : menor == d500 ? "50.0%" : menor == d618 ? "61.8%" : menor == d786 ? "78.6%" : "100.0%"

f_corFib(posicao) =>
    posicao >= 38.2 and posicao <= 61.8 ? color.new(color.green, 20) : posicao >= 23.6 and posicao <= 78.6 ? color.new(color.orange, 25) : color.new(color.red, 30)

// KVO
// ============================================================================
f_kvo() =>
    tp = (high + low + close) / 3.0
    mediaVolume = ta.sma(volume, kvoRapido)
    volumeRelativo = mediaVolume > 0 ? volume / mediaVolume : 1.0
    tendenciaVolume = tp > tp[1] ? 1.0 : tp < tp[1] ? -1.0 : 0.0
    fluxo = tendenciaVolume * volumeRelativo * volume
    kvoRap = ta.ema(fluxo, kvoRapido)
    kvoLen = ta.ema(fluxo, kvoLento)
    kvo = kvoRap - kvoLen
    sinal = ta.ema(kvo, kvoSinal)
    hist = kvo - sinal
    kvoScoreLong = hist > 0 ? 100.0 : 0.0
    kvoScoreShort = hist < 0 ? 100.0 : 0.0
    [kvo, sinal, hist, kvoScoreLong, kvoScoreShort]

// ============================================================================
// SCORE MTF
// ============================================================================
f_scoreMTF() =>
    ema8 = ta.ema(close, emaRapida)
    ema21 = ta.ema(close, emaMedia)
    ema55 = ta.ema(close, emaLenta)
    rsi = ta.rsi(close, rsiPeriodo)
    retorno = close[3] != 0 ? (close - close[3]) / close[3] * 100.0 : 0.0
    volumeMedio = ta.sma(volume, 20)
    volumeRelativo = volumeMedio > 0 ? volume / volumeMedio : 1.0

    scoreLong = 0.0
    if ema8 > ema21
        scoreLong += 25.0
    if ema21 > ema55
        scoreLong += 25.0
    if close > ema8
        scoreLong += 15.0
    if rsi > 50
        scoreLong += 15.0
    if retorno > 0
        scoreLong += 10.0
    if volumeRelativo >= 1.0
        scoreLong += 10.0

    scoreShort = 0.0
    if ema8 < ema21
        scoreShort += 25.0
    if ema21 < ema55
        scoreShort += 25.0
    if close < ema8
        scoreShort += 15.0
    if rsi < 50
        scoreShort += 15.0
    if retorno < 0
        scoreShort += 10.0
    if volumeRelativo >= 1.0
        scoreShort += 10.0

    [scoreLong, scoreShort]

f_modeloLocal() =>
    atrAtual = ta.atr(atrPeriodo)
    atrMedio = ta.sma(atrAtual, atrMediaPeriodo)
    fatorAdaptativo = atrMedio > 0 ? atrAtual / atrMedio : 1.0
    atrAdapt = usarATRAdaptativo ? atrAtual * fatorATR * f_limitar(fatorAdaptativo, 0.75, 1.50) : atrAtual * fatorATR

    ema8 = ta.ema(close, emaRapida)
    ema21 = ta.ema(close, emaMedia)
    ema55 = ta.ema(close, emaLenta)
    rsi = ta.rsi(close, rsiPeriodo)

    volumeMedio = ta.sma(volume, 20)
    volumeRelativo = volumeMedio > 0 ? volume / volumeMedio : 1.0

    corpo = math.abs(close - open)
    rangeCandle = high - low
    forcaCandle = rangeCandle > 0 ? corpo / rangeCandle * 100.0 : 0.0

    retorno1 = close[1] != 0 ? (close - close[1]) / close[1] * 100.0 : 0.0
    retorno3 = close[3] != 0 ? (close - close[3]) / close[3] * 100.0 : 0.0

    tendenciaLong = 0.0
    if ema8 > ema21
        tendenciaLong += 35.0
    if ema21 > ema55
        tendenciaLong += 35.0
    if close > ema8
        tendenciaLong += 30.0

    tendenciaShort = 0.0
    if ema8 < ema21
        tendenciaShort += 35.0
    if ema21 < ema55
        tendenciaShort += 35.0
    if close < ema8
        tendenciaShort += 30.0

    rsiLong = 0.0
    if rsi > 50
        rsiLong := 55.0
    if rsi <= rsiSobrevendido
        rsiLong := 100.0
    if rsi > 50 and rsi < 60
        rsiLong := 70.0

    rsiShort = 0.0
    if rsi < 50
        rsiShort := 55.0
    if rsi >= rsiSobrecomprado
        rsiShort := 100.0
    if rsi < 50 and rsi > 40
        rsiShort := 70.0

    retornoLong = 0.0
    if retorno1 > 0
        retornoLong += 50.0
    if retorno3 > 0
        retornoLong += 50.0

    retornoShort = 0.0
    if retorno1 < 0
        retornoShort += 50.0
    if retorno3 < 0
        retornoShort += 50.0

    candleLong = close > open ? forcaCandle : 0.0
    candleShort = close < open ? forcaCandle : 0.0

    volumeScore = f_limitar(volumeRelativo * 50.0, 0.0, 100.0)

    [kvo, kvoSinalLocal, histKVO, kvoLong, kvoShort] = f_kvo()

    fundo = ta.lowest(low, estruturaLookback)
    topo = ta.highest(high, estruturaLookback)

    distanciaFundoATR = atrAdapt > 0 ? (close - fundo) / atrAdapt : 999.0
    distanciaTopoATR = atrAdapt > 0 ? (topo - close) / atrAdapt : 999.0

    rejeicaoFundo = low <= fundo + atrAdapt * 0.25 and close > low + (high - low) * 0.55
    rejeicaoTopo = high >= topo - atrAdapt * 0.25 and close < low + (high - low) * 0.45

    fundoBase = 0.0
    if distanciaFundoATR <= distanciaMaximaATR
        fundoBase += 40.0
    if rejeicaoFundo
        fundoBase += 35.0
    if rsi <= 50
        fundoBase += 25.0

    forcaFundo = f_limitar(fundoBase, 0.0, 100.0)

    topoBase = 0.0
    if distanciaTopoATR <= distanciaMaximaATR
        topoBase += 40.0
    if rejeicaoTopo
        topoBase += 35.0
    if rsi >= 50
        topoBase += 25.0

    forcaTopo = f_limitar(topoBase, 0.0, 100.0)

    microMax = ta.highest(high, microLookback)
    microMin = ta.lowest(low, microLookback)

    rompimentoLong = close > microMax[1]
    rompimentoShort = close < microMin[1]

    recuperacaoLong = close > open and close > close[1]
    recuperacaoShort = close < open and close < close[1]

    microLong = 0.0
    if recuperacaoLong
        microLong += 35.0
    if rompimentoLong
        microLong += 65.0

    microShort = 0.0
    if recuperacaoShort
        microShort += 35.0
    if rompimentoShort
        microShort += 65.0

    microLong := f_limitar(microLong, 0.0, 100.0)
    microShort := f_limitar(microShort, 0.0, 100.0)

    scoreLong = tendenciaLong * 0.20 + rsiLong * 0.10 + retornoLong * 0.15 + candleLong * 0.15 + volumeScore * 0.10 + kvoLong * 0.15 + forcaFundo * 0.10 + microLong * 0.05
    scoreShort = tendenciaShort * 0.20 + rsiShort * 0.10 + retornoShort * 0.15 + candleShort * 0.15 + volumeScore * 0.10 + kvoShort * 0.15 + forcaTopo * 0.10 + microShort * 0.05

    scoreLong := f_limitar(scoreLong, 0.0, 100.0)
    scoreShort := f_limitar(scoreShort, 0.0, 100.0)

    intensidadeLong = scoreLong * 0.60 + forcaFundo * 0.20 + microLong * 0.20
    intensidadeShort = scoreShort * 0.60 + forcaTopo * 0.20 + microShort * 0.20

    intensidadeLong := f_limitar(intensidadeLong, 0.0, 100.0)
    intensidadeShort := f_limitar(intensidadeShort, 0.0, 100.0)

    confluenciaLong = (tendenciaLong + rsiLong + retornoLong + candleLong + kvoLong + forcaFundo + microLong) / 7.0
    confluenciaShort = (tendenciaShort + rsiShort + retornoShort + candleShort + kvoShort + forcaTopo + microShort) / 7.0

    confluenciaLong := f_limitar(confluenciaLong, 0.0, 100.0)
    confluenciaShort := f_limitar(confluenciaShort, 0.0, 100.0)

    atrVolMedia = ta.sma(atrAtual, volatilidadeLookback)
    razaoVolatilidade = atrVolMedia > 0 ? atrAtual / atrVolMedia : 1.0
    volatilidadeScore = f_limitar(razaoVolatilidade * 50.0, 0.0, 100.0)
    volatilidadePercentual = close != 0 ? atrAtual / close * 100.0 : 0.0

    fibHigh = ta.highest(high, fibLookback)
    fibLow = ta.lowest(low, fibLookback)
    fibRange = fibHigh - fibLow

    fibPosicao = fibRange > 0 ? (close - fibLow) / fibRange * 100.0 : 50.0
    fibPosicao := f_limitar(fibPosicao, 0.0, 100.0)

    fibNivel = f_nivelFib(fibPosicao)
    fibClasse = f_classFib(fibPosicao)

    fib23 = fibHigh - fibRange * 0.236
    fib38 = fibHigh - fibRange * 0.382
    fib50 = fibHigh - fibRange * 0.500
    fib61 = fibHigh - fibRange * 0.618
    fib78 = fibHigh - fibRange * 0.786

    distanciaFib23 = close != 0 ? math.abs(close - fib23) / close * 100.0 : 999.0
    distanciaFib38 = close != 0 ? math.abs(close - fib38) / close * 100.0 : 999.0
    distanciaFib50 = close != 0 ? math.abs(close - fib50) / close * 100.0 : 999.0
    distanciaFib61 = close != 0 ? math.abs(close - fib61) / close * 100.0 : 999.0
    distanciaFib78 = close != 0 ? math.abs(close - fib78) / close * 100.0 : 999.0

    fibProximo = math.min(distanciaFib23, math.min(distanciaFib38, math.min(distanciaFib50, math.min(distanciaFib61, distanciaFib78))))
    fibNivelValido = fibProximo <= fibDistanciaNivel

    [scoreLong, scoreShort, intensidadeLong, intensidadeShort, confluenciaLong, confluenciaShort, atrAdapt, atrAtual, forcaFundo, forcaTopo, microLong, microShort, histKVO, fundo, topo, volatilidadeScore, volatilidadePercentual, fibPosicao, fibNivel, fibClasse, fibNivelValido]

    [score5L, score5S] = request.security(ativo, "5", f_scoreMTF(), barmerge.gaps_off, barmerge.lookahead_off)
    [score15L, score15S] = request.security(ativo, "15", f_scoreMTF(), barmerge.gaps_off, barmerge.lookahead_off)
    [score30L, score30S] = request.security(ativo, "30", f_scoreMTF(), barmerge.gaps_off, barmerge.lookahead_off)
    [score1HL, score1HS] = request.security(ativo, "60", f_scoreMTF(), barmerge.gaps_off, barmerge.lookahead_off)
    [score4HL, score4HS] = request.security(ativo, "240", f_scoreMTF(), barmerge.gaps_off, barmerge.lookahead_off)
    [scoreDL, scoreDS] = request.security(ativo, "D", f_scoreMTF(), barmerge.gaps_off, barmerge.lookahead_off)

    somaPesos = peso1M + peso5M + peso15M + peso30M + peso1H + peso4H + pesoD

    mtfLong = 0.0
    mtfShort = 0.0

    if somaPesos > 0
        mtfLong := (score1L * peso1M + score5L * peso5M + score15L * peso15M + score30L * peso30M + score1HL * peso1H + score4HL * peso4H + scoreDL * pesoD) / somaPesos
        mtfShort := (score1S * peso1M + score5S * peso5M + score15S * peso15M + score30S * peso30M + score1HS * peso1H + score4HS * peso4H + scoreDS * pesoD) / somaPesos

    vantagemMTF = mtfLong - mtfShort

    tendenciaPrincipal = "LATERAL"

    if vantagemMTF >= vantagemTendencia
        tendenciaPrincipal := "ALTA"
    else if vantagemMTF <= -vantagemTendencia
        tendenciaPrincipal := "BAIXA"

    finalLong = localL * 0.55 + mtfLong * 0.45
    finalShort = localS * 0.55 + mtfShort * 0.45

    vantagemLong = finalLong - finalShort
    vantagemShort = finalShort - finalLong

    qualidadeLong = finalLong * 0.30 + intL * 0.15 + confL * 0.15 + mtfLong * 0.15 + fundoForca * 0.10 + microL * 0.10 + f_limitar(vantagemLong * 0.05, 0.0, 5.0)
    qualidadeShort = finalShort * 0.30 + intS * 0.15 + confS * 0.15 + mtfShort * 0.15 + topoForca * 0.10 + microS * 0.10 + f_limitar(vantagemShort * 0.05, 0.0, 5.0)

    qualidadeLong := f_limitar(qualidadeLong, 0.0, 100.0)
    qualidadeShort := f_limitar(qualidadeShort, 0.0, 100.0)

    alinhamentoKvoLong = histAtual > 0
    alinhamentoKvoShort = histAtual < 0

    tendenciaLongOK = tendenciaPrincipal == "ALTA"
    tendenciaShortOK = tendenciaPrincipal == "BAIXA"

    distanciaFundo = close - fundoAtivo
    distanciaTopo = topoAtivo - close

    distanciaFundoATR = atrAdapt > 0 ? distanciaFundo / atrAdapt : 999.0
    distanciaTopoATR = atrAdapt > 0 ? distanciaTopo / atrAdapt : 999.0

    proximidadeFundo = f_limitar(100.0 - distanciaFundoATR / distanciaMaximaATR * 100.0, 0.0, 100.0)
    proximidadeTopo = f_limitar(100.0 - distanciaTopoATR / distanciaMaximaATR * 100.0, 0.0, 100.0)

    distanciaLong = finalLong * 0.25 + intL * 0.15 + confL * 0.15 + f_limitar(vantagemLong * 5.0, 0.0, 100.0) * 0.15 + mtfLong * 0.10 + fundoForca * 0.10 + microL * 0.05 + proximidadeFundo * 0.05
    distanciaShort = finalShort * 0.25 + intS * 0.15 + confS * 0.15 + f_limitar(vantagemShort * 5.0, 0.0, 100.0) * 0.15 + mtfShort * 0.10 + topoForca * 0.10 + microS * 0.05 + proximidadeTopo * 0.05

    distanciaLong := f_limitar(distanciaLong, 0.0, 100.0)
    distanciaShort := f_limitar(distanciaShort, 0.0, 100.0)

    preLong = finalLong >= scorePreMinimo and intL >= intensidadePreMinima and confL >= confluenciaPreMinima and vantagemLong >= vantagemPreMinima and mtfLong >= mtfPreMinimo and fundoForca >= forcaPreMinima and microL >= microPreMinimo and qualidadeLong >= qualidadePreMinima

    preShort = finalShort >= scorePreMinimo and intS >= intensidadePreMinima and confS >= confluenciaPreMinima and vantagemShort >= vantagemPreMinima and mtfShort >= mtfPreMinimo and topoForca >= forcaPreMinima and microS >= microPreMinimo and qualidadeShort >= qualidadePreMinima

    sinalLongBase = finalLong >= scoreMinimo and intL >= intensidadeMinima and confL >= confluenciaMinima and vantagemLong >= vantagemMinima and mtfLong >= mtfMinimo and fundoForca >= forcaFundoTopoMinima and microL >= microMinimo and qualidadeLong >= qualidadeSinalMinima

    sinalShortBase = finalShort >= scoreMinimo and intS >= intensidadeMinima and confS >= confluenciaMinima and vantagemShort >= vantagemMinima and mtfShort >= mtfMinimo and topoForca >= forcaFundoTopoMinima and microS >= microMinimo and qualidadeShort >= qualidadeSinalMinima

    sinalLong = sinalLongBase
    sinalShort = sinalShortBase

    if usarFiltroKVO and not alinhamentoKvoLong
        sinalLong := false

    if usarFiltroKVO and not alinhamentoKvoShort
        sinalShort := false

    if usarFiltroTendencia and not tendenciaLongOK
        sinalLong := false

    if usarFiltroTendencia and not tendenciaShortOK
        sinalShort := false

    if usarFiltroEsticado and distanciaFundoATR > distanciaMaximaATR
        sinalLong := false

    if usarFiltroEsticado and distanciaTopoATR > distanciaMaximaATR
        sinalShort := false

    if usarFiltroVolatilidade and volScore < volatilidadeMinima
        sinalLong := false

    if usarFiltroVolatilidade and volScore < volatilidadeMinima
        sinalShort := false

    if usarFiltroFibonacci and not fibValido
        sinalLong := false

    if usarFiltroFibonacci and not fibValido
        sinalShort := false

    conflito = math.abs(finalLong - finalShort) < vantagemMinima and (sinalLong or sinalShort or preLong or preShort)

    direcao = "NEUTRO"

    if conflito
        direcao := "CONFLITO"
    else if sinalLong
        direcao := "LONG"
    else if sinalShort
        direcao := "SHORT"
    else if preLong and not preShort
        direcao := "PRE-LONG"
    else if preShort and not preLong
        direcao := "PRE-SHORT"
    else if finalLong > finalShort and finalLong >= scorePreMinimo
        direcao := "OBSERVAR LONG"
    else if finalShort > finalLong and finalShort >= scorePreMinimo
        direcao := "OBSERVAR SHORT"

    intensidadeFinal = 0.0
    confluenciaFinal = 0.0
    scoreFinal = 0.0
    mtfFinal = 0.0
    distanciaFinal = 0.0
    qualidadeFinal = 0.0

    if direcao == "LONG" or direcao == "PRE-LONG" or direcao == "OBSERVAR LONG"
        scoreFinal := finalLong
        intensidadeFinal := intL
        confluenciaFinal := confL
        mtfFinal := mtfLong
        distanciaFinal := distanciaLong
        qualidadeFinal := qualidadeLong
    else if direcao == "SHORT" or direcao == "PRE-SHORT" or direcao == "OBSERVAR SHORT"
        scoreFinal := finalShort
        intensidadeFinal := intS
        confluenciaFinal := confS
        mtfFinal := mtfShort
        distanciaFinal := distanciaShort
        qualidadeFinal := qualidadeShort
    else
        if finalLong >= finalShort
            scoreFinal := finalLong
            intensidadeFinal := intL
            confluenciaFinal := confL
            mtfFinal := mtfLong
            distanciaFinal := distanciaLong
            qualidadeFinal := qualidadeLong
        else
            scoreFinal := finalShort
            intensidadeFinal := intS
            confluenciaFinal := confS
            mtfFinal := mtfShort
            distanciaFinal := distanciaShort
            qualidadeFinal := qualidadeShort

    falta = "AGUARDAR"

    if direcao == "LONG"
        if finalLong < scoreMinimo
            falta := "SCORE"
        else if intL < intensidadeMinima
            falta := "INTENSIDADE"
        else if confL < confluenciaMinima
            falta := "CONFLUENCIA"
        else if vantagemLong < vantagemMinima
            falta := "VANTAGEM"
        else if mtfLong < mtfMinimo
            falta := "MTF"
        else if fundoForca < forcaFundoTopoMinima
            falta := "FUNDO"
        else if microL < microMinimo
            falta := "MICRO"
        else if usarFiltroKVO and not alinhamentoKvoLong
            falta := "KVO"
        else if usarFiltroTendencia and not tendenciaLongOK
            falta := "TENDENCIA"
        else if usarFiltroEsticado and distanciaFundoATR > distanciaMaximaATR
            falta := "ESTICADO"
        else if usarFiltroVolatilidade and volScore < volatilidadeMinima
            falta := "VOLATILIDADE"
        else if usarFiltroFibonacci and not fibValido
            falta := "FIBONACCI"
        else
            falta := "OK"

    else if direcao == "SHORT"
        if finalShort < scoreMinimo
            falta := "SCORE"
        else if intS < intensidadeMinima
            falta := "INTENSIDADE"
        else if confS < confluenciaMinima
            falta := "CONFLUENCIA"
        else if vantagemShort < vantagemMinima
            falta := "VANTAGEM"
        else if mtfShort < mtfMinimo
            falta := "MTF"
        else if topoForca < forcaFundoTopoMinima
            falta := "TOPO"
        else if microS < microMinimo
            falta := "MICRO"
        else if usarFiltroKVO and not alinhamentoKvoShort
            falta := "KVO"
        else if usarFiltroTendencia and not tendenciaShortOK
            falta := "TENDENCIA"
        else if usarFiltroEsticado and distanciaTopoATR > distanciaMaximaATR
            falta := "ESTICADO"
        else if usarFiltroVolatilidade and volScore < volatilidadeMinima
            falta := "VOLATILIDADE"
        else if usarFiltroFibonacci and not fibValido
            falta := "FIBONACCI"
        else
            falta := "OK"

    else if direcao == "PRE-LONG"
        if finalLong < scorePreMinimo
            falta := "SCORE"
        else if intL < intensidadePreMinima
            falta := "INTENSIDADE"
        else if confL < confluenciaPreMinima
            falta := "CONFLUENCIA"
        else if vantagemLong < vantagemPreMinima
            falta := "VANTAGEM"
        else if mtfLong < mtfPreMinimo
            falta := "MTF"
        else if fundoForca < forcaPreMinima
            falta := "FUNDO"
        else if microL < microPreMinimo
            falta := "MICRO"
        else
            falta := "CONFIRMACAO"

    else if direcao == "PRE-SHORT"
        if finalShort < scorePreMinimo
            falta := "SCORE"
        else if intS < intensidadePreMinima
            falta := "INTENSIDADE"
        else if confS < confluenciaPreMinima
            falta := "CONFLUENCIA"
        else if vantagemShort < vantagemPreMinima
            falta := "VANTAGEM"
        else if mtfShort < mtfPreMinimo
            falta := "MTF"
        else if topoForca < forcaPreMinima
            falta := "TOPO"
        else if microS < microPreMinimo
            falta := "MICRO"
        else
            falta := "CONFIRMACAO"

    entrada = float(na)
    alvo = float(na)

    if direcao == "LONG" or direcao == "PRE-LONG"
        entrada := close
        alvo := close + atrAdapt * alvoPrincipalATR
    else if direcao == "SHORT" or direcao == "PRE-SHORT"
        entrada := close
        alvo := close - atrAdapt * alvoPrincipalATR

    alvo1Long = close + atrAdapt * alvo1ATR
    alvo2Long = close + atrAdapt * alvo2ATR
    alvo3Long = close + atrAdapt * alvo3ATR
    alvo4Long = close + atrAdapt * alvo4ATR
    alvo5Long = close + atrAdapt * alvo5ATR
    alvo6Long = close + atrAdapt * alvo6ATR
    alvo7Long = close + atrAdapt * alvo7ATR
    alvo8Long = close + atrAdapt * alvo8ATR

    alvo1Short = close - atrAdapt * alvo1ATR
    alvo2Short = close - atrAdapt * alvo2ATR
    alvo3Short = close - atrAdapt * alvo3ATR
    alvo4Short = close - atrAdapt * alvo4ATR
    alvo5Short = close - atrAdapt * alvo5ATR
    alvo6Short = close - atrAdapt * alvo6ATR
    alvo7Short = close - atrAdapt * alvo7ATR
    alvo8Short = close - atrAdapt * alvo8ATR

    sinalLongFinal = sinalLong
    sinalShortFinal = sinalShort

    if somenteBarraConfirmada and not barstate.isconfirmed
        sinalLongFinal := false
        sinalShortFinal := false

    [tendenciaPrincipal, direcao, distanciaFinal, scoreFinal, intensidadeFinal, confluenciaFinal, mtfFinal, falta, entrada, alvo, sinalLongFinal, sinalShortFinal, atrAdapt, alvo1Long, alvo2Long, alvo3Long, alvo4Long, alvo5Long, alvo6Long, alvo7Long, alvo8Long, alvo1Short, alvo2Short, alvo3Short, alvo4Short, alvo5Short, alvo6Short, alvo7Short, alvo8Short, qualidadeLong, qualidadeShort, volScore, volPct, fibPos, fibNivel, fibClasse, fibValido]

f_escreverLinha(tabela, linha, nome, tendencia, direcao, distancia, score, intensidade, confluencia, mtf, falta, entrada, alvo, qualidade, volScore, volPct, fibPos, fibNivel, fibClasse, destaque) =>
    corNome = destaque ? color.new(color.orange, 0) : color.black

    textoVol = str.tostring(volScore, "#.0") + " " + f_classVolatilidade(volScore)
    textoFib = fibNivel + " " + fibClasse

    table.cell(tabela, 0, linha, nome, text_color=color.white, bgcolor=corNome)
    table.cell(tabela, 1, linha, tendencia, text_color=color.white, bgcolor=f_corTendencia(tendencia))
    table.cell(tabela, 2, linha, direcao, text_color=color.white, bgcolor=f_corDirecao(direcao))
    table.cell(tabela, 3, linha, str.tostring(distancia, "#.0"), text_color=color.white, bgcolor=f_corDistancia(distancia))
    table.cell(tabela, 4, linha, str.tostring(score, "#.0"), text_color=color.white, bgcolor=f_corNumerica(score, scoreMinimo))
    table.cell(tabela, 5, linha, str.tostring(intensidade, "#.0"), text_color=color.white, bgcolor=f_corNumerica(intensidade, intensidadeMinima))
    table.cell(tabela, 6, linha, str.tostring(confluencia, "#.0"), text_color=color.white, bgcolor=f_corNumerica(confluencia, confluenciaMinima))
    table.cell(tabela, 7, linha, str.tostring(mtf, "#.0"), text_color=color.white, bgcolor=f_corNumerica(mtf, mtfMinimo))

    if mostrarVolatilidade
        table.cell(tabela, 8, linha, textoVol + " / " + str.tostring(volPct, "#.000") + "%", text_color=color.white, bgcolor=f_corVolatilidade(volScore))
    else
        table.cell(tabela, 8, linha, "-", text_color=color.white, bgcolor=color.black)

    if mostrarFibonacci
        table.cell(tabela, 9, linha, textoFib, text_color=color.white, bgcolor=f_corFib(fibPos))
    else
