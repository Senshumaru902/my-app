
instrument {
    name = "QMS 3S3C ARCO DE ALTA M5 V2",
    short_name = "QMS ARCO ALTA",
    overlay = true
}

-- =========================================
-- CONFIGURACOES
-- =========================================

input_group {
    "INCLINACAO DO ARCO",
    periodo_ema_curta = input(9, "EMA curta", input.integer, 3, 30),
    periodo_ema_longa = input(21, "EMA longa", input.integer, 10, 60),
    periodo_arco = input(3, "Periodo de curvatura", input.integer, 2, 8),
    inclinacao_minima = input(0, "Inclinacao minima", input.integer, 0, 10)
}

input_group {
    "QUALIDADE",
    corpo_nivel4 = input(50, "Corpo minimo N4 (%)", input.integer, 40, 80),
    corpo_nivel5 = input(65, "Corpo minimo N5 (%)", input.integer, 50, 90)
}

input_group {
    "ESTRUTURA",
    periodo_estrutura = input(5, "Periodo estrutural", input.integer, 3, 15)
}

input_group {
    "VISUAL",
    mostrar_nivel4 = input(1, "Mostrar nivel 4", input.integer, 0, 1),
    mostrar_nivel5 = input(1, "Mostrar nivel 5", input.integer, 0, 1)
}

input_group {
    "COR",
    azul_compra = input {
        default = "#2196FF",
        type = input.color
    }
}

-- =========================================
-- MEDIAS
-- =========================================

ema_curta = ema(close, periodo_ema_curta)
ema_longa = ema(close, periodo_ema_longa)

-- =========================================
-- INCLINACAO E CURVATURA
-- =========================================

inclinacao0 = ema_longa - ema_longa[1]
inclinacao1 = ema_longa[1] - ema_longa[2]
inclinacao2 = ema_longa[2] - ema_longa[3]

curvatura0 = inclinacao0 - inclinacao1
curvatura1 = inclinacao1 - inclinacao2

-- Arco ascendente:
-- A media sobe e sua inclinacao aumenta.

arco_alta =
    ema_curta > ema_longa and
    inclinacao0 > inclinacao_minima * 0.0001 and
    inclinacao1 > 0 and
    curvatura0 > 0 and
    curvatura1 > 0

-- Confirmacao da aceleracao

aceleracao_alta =
    inclinacao0 > inclinacao1 and
    inclinacao1 > inclinacao2

-- =========================================
-- TRES SOLDADOS
-- =========================================

soldado1 = close[2] > open[2]
soldado2 = close[1] > open[1]
soldado3 = close > open

progressao_alta =
    close > close[1] and
    close[1] > close[2]

abertura_soldados =
    open[1] >= open[2] and
    open[1] <= close[2] and
    open >= open[1] and
    open <= close[1]

tres_soldados =
    soldado1 and
    soldado2 and
    soldado3 and
    progressao_alta

-- =========================================
-- CORPOS
-- =========================================

amplitude0 = high - low
amplitude1 = high[1] - low[1]
amplitude2 = high[2] - low[2]

corpo0 = abs(close - open)
corpo1 = abs(close[1] - open[1])
corpo2 = abs(close[2] - open[2])

corpos_nivel4 =
    amplitude0 > 0 and
    amplitude1 > 0 and
    amplitude2 > 0 and
    corpo0 >= amplitude0 * corpo_nivel4 / 100 and
    corpo1 >= amplitude1 * corpo_nivel4 / 100 and
    corpo2 >= amplitude2 * corpo_nivel4 / 100

corpos_nivel5 =
    amplitude0 > 0 and
    amplitude1 > 0 and
    amplitude2 > 0 and
    corpo0 >= amplitude0 * corpo_nivel5 / 100 and
    corpo1 >= amplitude1 * corpo_nivel5 / 100 and
    corpo2 >= amplitude2 * corpo_nivel5 / 100

-- =========================================
-- SOMBRAS
-- =========================================

sombra_superior0 = high - max(open, close)
sombra_superior1 = high[1] - max(open[1], close[1])
sombra_superior2 = high[2] - max(open[2], close[2])

sombras_soldados =
    amplitude0 > 0 and
    amplitude1 > 0 and
    amplitude2 > 0 and
    sombra_superior0 <= amplitude0 * 0.25 and
    sombra_superior1 <= amplitude1 * 0.25 and
    sombra_superior2 <= amplitude2 * 0.25

-- =========================================
-- CORRECAO ANTERIOR
-- =========================================

correcao_alta =
    close[3] < open[3] or
    close[4] < open[4] or
    close[5] < open[5]

-- =========================================
-- IMPULSO ANTERIOR
-- =========================================

impulso_alta =
    close[3] > close[6] and
    high[3] > high[6]

-- =========================================
-- ROMPIMENTO
-- =========================================

resistencia = highest(high[3], periodo_estrutura)

rompimento_alta =
    close > resistencia

-- =========================================
-- CLASSIFICACAO
-- =========================================

nivel3_alta =
    tres_soldados and
    corpos_nivel4 and
    arco_alta

nivel4_alta =
    nivel3_alta and
    correcao_alta and
    impulso_alta and
    rompimento_alta

nivel5_alta =
    nivel4_alta and
    corpos_nivel5 and
    sombras_soldados and
    abertura_soldados and
    aceleracao_alta

-- =========================================
-- PLOTAGEM HISTORICA
-- =========================================

plot_shape(
    nivel4_alta and not nivel5_alta and mostrar_nivel4 == 1,
    "ARCO ALTA N4",
    shape_style.circle,
    shape_size.small,
    azul_compra,
    shape_location.belowbar,
    0,
    "N4"
)

plot_shape(
    nivel5_alta and mostrar_nivel5 == 1,
    "ARCO ALTA N5",
    shape_style.circle,
    shape_size.small,
    azul_compra,
    shape_location.belowbar,
    0,
    "N5"
)
