
instrument {
    name = "QMS ESTRELAS SELETIVO PRO",
    short_name = "QMS ESTRELAS PRO",
    overlay = true
}

-- =========================================
-- CONFIGURACOES
-- =========================================

input_group {
    "Seletividade",

    forca_candle = input(
        0.60,
        "Forca do primeiro candle",
        input.double, 0.30, 0.95, 0.05
    ),

    corpo_estrela = input(
        0.30,
        "Corpo maximo do candle central",
        input.double, 0.10, 0.60, 0.05
    ),

    forca_confirmacao = input(
        0.65,
        "Forca do candle de confirmacao",
        input.double, 0.30, 0.95, 0.05
    ),

    recuperacao = input(
        0.60,
        "Nivel de recuperacao",
        input.double, 0.50, 0.90, 0.05
    ),

    rejeicao_minima = input(
        0.25,
        "Rejeicao minima do candle central",
        input.double, 0.10, 0.80, 0.05
    )
}

input_group {
    "Cores",

    cor_compra = input {
        default = "#2196FF",
        type = input.color
    },

    cor_venda = input {
        default = "#A020F0",
        type = input.color
    }
}

-- =========================================
-- CANDLES
-- =========================================

local a2 = open[2]
local f2 = close[2]
local h2 = high[2]
local l2 = low[2]

local a1 = open[1]
local f1 = close[1]
local h1 = high[1]
local l1 = low[1]

local a0 = open
local f0 = close
local h0 = high
local l0 = low

-- =========================================
-- CALCULOS
-- =========================================

local r2 = h2 - l2
local r1 = h1 - l1
local r0 = h0 - l0

local b2 = abs(f2 - a2)
local b1 = abs(f1 - a1)
local b0 = abs(f0 - a0)

local p2 = r2 > 0 and b2 / r2 or 0
local p1 = r1 > 0 and b1 / r1 or 0
local p0 = r0 > 0 and b0 / r0 or 0

local pavio_superior = h1 - max(a1, f1)
local pavio_inferior = min(a1, f1) - l1

local rejeicao_superior =
    r1 > 0 and pavio_superior / r1 or 0

local rejeicao_inferior =
    r1 > 0 and pavio_inferior / r1 or 0

-- =========================================
-- ESTRELA DA MANHA
-- =========================================

local tendencia_baixa =
    f2 < a2 and p2 >= forca_candle

local estrela_central_compra =
    b1 <= b2 * corpo_estrela and
    rejeicao_inferior >= rejeicao_minima and
    l1 <= l2

local confirmacao_alta =
    f0 > a0 and
    p0 >= forca_confirmacao and
    f0 >= (l2 + (h2 - l2) * recuperacao)

local estrela_manha =
    tendencia_baixa and
    estrela_central_compra and
    confirmacao_alta

-- =========================================
-- ESTRELA DA NOITE
-- =========================================

local tendencia_alta =
    f2 > a2 and p2 >= forca_candle

local estrela_central_venda =
    b1 <= b2 * corpo_estrela and
    rejeicao_superior >= rejeicao_minima and
    h1 >= h2

local confirmacao_baixa =
    f0 < a0 and
    p0 >= forca_confirmacao and
    f0 <= (h2 - (h2 - l2) * recuperacao)

local estrela_noite =
    tendencia_alta and
    estrela_central_venda and
    confirmacao_baixa

-- =========================================
-- SINAIS HISTORICOS
-- =========================================

plot_shape(
    estrela_manha,
    "ESTRELA DA MANHA",
    shape_style.triangleup,
    shape_size.large,
    cor_compra,
    shape_location.belowbar,
    0,
    "MANHA",
    cor_compra
)

plot_shape(
    estrela_noite,
    "ESTRELA DA NOITE",
    shape_style.triangledown,
    shape_size.large,
    cor_venda,
    shape_location.abovebar,
    0,
    "NOITE",
    cor_venda
)
