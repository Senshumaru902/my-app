// ============================================================================
// FUNÇÕES BÁSICAS
// ============================================================================

f_clamp(float valor, float minimo, float maximo) =>
    math.max(minimo, math.min(maximo, valor))


f_normalizar(float valor, float minimo, float maximo) =>
    maximo == minimo ? 50.0 : f_clamp((valor - minimo) / (maximo - minimo) * 100.0, 0.0, 100.0)


// ============================================================================
// CLASSIFICAÇÃO DE VOLATILIDADE
// ============================================================================

f_classVolatilidade(float valor) =>
    string resultado = "MUITO FRACA"
    if valor >= 80
        resultado := "EXTREMA"
    else if valor >= 65
        resultado := "MUITO FORTE"
    else if valor >= 50
        resultado := "FORTE"
    else if valor >= 35
        resultado := "MODERADA"
    else if valor >= 20
        resultado := "FRACA"
    resultado


f_corVolatilidade(float valor) =>
    color resultado = color.gray
    if valor >= 80
        resultado := color.red
    else if valor >= 65
        resultado := color.orange
    else if valor >= 50
        resultado := color.yellow
    else if valor >= 35
        resultado := color.aqua
    resultado


// ============================================================================
// FIBONACCI
// ============================================================================

f_classFib(float posicao) =>
    string resultado = "0-23.6"
    if posicao >= 78.6
        resultado := "78.6-100"
    else if posicao >= 61.8
        resultado := "61.8-78.6"
    else if posicao >= 50
        resultado := "50-61.8"
    else if posicao >= 38.2
        resultado := "38.2-50"
    else if posicao >= 23.6
        resultado := "23.6-38.2"
    resultado


f_corFib(float posicao) =>
    color resultado = color.gray
    if posicao >= 38.2 and posicao <= 61.8
        resultado := color.green
    else if posicao >= 23.6 and posicao <= 78.6
        resultado := color.yellow
    else
        resultado := color.orange
    resultado

// ============================================================================
// MODELO LOCAL
// ============================================================================

f_modeloLocal() =>
    float rangeCandle = math.max(high - low, syminfo.mintick)
    float corpo = math.abs(close - open)
    float forcaCandle = f_clamp(corpo / rangeCandle * 100.0, 0.0, 100.0)

    float ema9 = ta.ema(close, emaRapida)
    float ema21 = ta.ema(close, emaMedia)
    float ema50 = ta.ema(close, emaLenta)

    float rsi = ta.rsi(close, rsiPeriodo)

    float fluxo = volume * (2.0 * ((close - low) - (high - close)) / rangeCandle)
    float kvoFastValue = ta.ema(fluxo, kvoFast)
    float kvoSlowValue = ta.ema(fluxo, kvoSlow)
    float kvo = kvoFastValue - kvoSlowValue
    float kvoSignal = ta.ema(kvo, kvoSignalPeriodo)

    float retorno1 = close[1] != 0 ? (close - close[1]) / close[1] * 100.0 : 0.0
    float retorno3 = close[3] != 0 ? (close - close[3]) / close[3] * 100.0 : 0.0

    float tendenciaLong = 50.0
    if ema9 > ema21 and ema21 > ema50
        tendenciaLong := 100.0
    else if ema9 > ema21
        tendenciaLong := 75.0
    else if ema9 > ema50
        tendenciaLong := 60.0
    else
        tendenciaLong := 25.0

    float tendenciaShort = 50.0
    if ema9 < ema21 and ema21 < ema50
        tendenciaShort := 100.0
    else if ema9 < ema21
        tendenciaShort := 75.0
    else if ema9 < ema50
        tendenciaShort := 60.0
    else
        tendenciaShort := 25.0

    float rsiLong = 50.0
    if rsi <= rsiSobrevendido
        rsiLong := 100.0
    else if rsi < 50
        rsiLong := 80.0
    else if rsi < 55
        rsiLong := 60.0
    else
        rsiLong := 25.0

    float rsiShort = 50.0
    if rsi >= rsiSobrecomprado
        rsiShort := 100.0
    else if rsi > 50
        rsiShort := 80.0
    else if rsi > 45
        rsiShort := 60.0
    else
        rsiShort := 25.0

    float kvoLong = kvo > kvoSignal ? 100.0 : kvo > 0 ? 70.0 : 30.0
    float kvoShort = kvo < kvoSignal ? 100.0 : kvo < 0 ? 70.0 : 30.0

    float retornoLong = retorno1 > 0 and retorno3 > 0 ? 100.0 : retorno1 > 0 or retorno3 > 0 ? 65.0 : 25.0
    float retornoShort = retorno1 < 0 and retorno3 < 0 ? 100.0 : retorno1 < 0 or retorno3 < 0 ? 65.0 : 25.0

    float candleLong = close > open ? forcaCandle : forcaCandle * 0.35
    float candleShort = close < open ? forcaCandle : forcaCandle * 0.35

    float scoreLong = tendenciaLong * 0.30 + rsiLong * 0.15 + kvoLong * 0.20 + retornoLong * 0.15 + candleLong * 0.20
    float scoreShort = tendenciaShort * 0.30 + rsiShort * 0.15 + kvoShort * 0.20 + retornoShort * 0.15 + candleShort * 0.20

    [scoreLong, scoreShort, forcaCandle, kvo, kvoSignal, rsi, ema9, ema21, ema50]


// ============================================================================
// SCORE MTF
// ============================================================================

f_scoreMTF() =>
    [localLong, localShort, forcaCandle, kvo, kvoSignal, rsi, ema9, ema21, ema50] = f_modeloLocal()
    [localLong, localShort]


// ============================================================================
// MTF + ATR
// ============================================================================

f_mtfAtr() =>
    [localLong, localShort, forcaCandle, kvo, kvoSignal, rsi, ema9, ema21, ema50] = f_modeloLocal()
    float atrValue = ta.atr(atrPeriodo)
    [localLong, localShort, atrValue]


// ============================================================================
// DADOS 1 MINUTO
// ============================================================================

f_dados1M() =>
    [localLong, localShort, forcaCandle, kvo, kvoSignal, rsi, ema9, ema21, ema50] = f_modeloLocal()

    float atrValue = ta.atr(atrPeriodo)
    float atrMedia = ta.sma(atrValue, atrMediaPeriodo)

    float volatilidade = atrMedia != 0 ? atrValue / atrMedia * 50.0 : 50.0
    volatilidade := f_clamp(volatilidade, 0.0, 100.0)

    float fibHigh = ta.highest(high, fibLookback)
    float fibLow = ta.lowest(low, fibLookback)
    float amplitudeFib = math.max(fibHigh - fibLow, syminfo.mintick)

    float fibPosicao = (close - fibLow) / amplitudeFib * 100.0
    fibPosicao := f_clamp(fibPosicao, 0.0, 100.0)

    float fibNivel = fibPosicao

    [localLong, localShort, forcaCandle, kvo, kvoSignal, rsi, ema9, ema21, ema50, atrValue, atrMedia, volatilidade, fibPosicao, fibNivel]


// ============================================================================
// SCANNER PRINCIPAL
// ============================================================================

f_scanner(string simbolo) =>

    // ------------------------------------------------------------------------
    // 1M
    // ------------------------------------------------------------------------

    [l1, s1, fc1, kvo1, kvoSig1, rsi1, ema9_1, ema21_1, ema50_1, atr1, atrMedio1, vol1, fibPos, fibNivel] = request.security(
         simbolo,
         "1",
         f_dados1M(),
         gaps=barmerge.gaps_off,
         lookahead=barmerge.lookahead_off)

    // ------------------------------------------------------------------------
    // 5M
    // ------------------------------------------------------------------------

    [l5, s5] = request.security(
         simbolo,
         "5",
         f_scoreMTF(),
         gaps=barmerge.gaps_off,
         lookahead=barmerge.lookahead_off)

    // ------------------------------------------------------------------------
    // 15M
    // ------------------------------------------------------------------------

    [l15, s15, atr15] = request.security(
         simbolo,
         "15",
         f_mtfAtr(),
         gaps=barmerge.gaps_off,
         lookahead=barmerge.lookahead_off)

    // ------------------------------------------------------------------------
    // 30M
    // ------------------------------------------------------------------------

    [l30, s30] = request.security(
         simbolo,
         "30",
         f_scoreMTF(),
         gaps=barmerge.gaps_off,
         lookahead=barmerge.lookahead_off)

    // ------------------------------------------------------------------------
    // 1H
    // ------------------------------------------------------------------------

    [l60, s60, atr60] = request.security(
         simbolo,
         "60",
         f_mtfAtr(),
         gaps=barmerge.gaps_off,
         lookahead=barmerge.lookahead_off)

    // ------------------------------------------------------------------------
    // 4H
    // ------------------------------------------------------------------------

    [l240, s240, atr240] = request.security(
         simbolo,
         "240",
         f_mtfAtr(),
         gaps=barmerge.gaps_off,
         lookahead=barmerge.lookahead_off)

    // ------------------------------------------------------------------------
    // DIÁRIO
    // ------------------------------------------------------------------------

    [lD, sD] = request.security(
         simbolo,
         "D",
         f_scoreMTF(),
         gaps=barmerge.gaps_off,
         lookahead=barmerge.lookahead_off)


    // ========================================================================
    // MTF PONDERADO
    // ========================================================================

    float somaPesos = peso1M + peso5M + peso15M + peso30M + peso1H + peso4H + pesoD

    float mtfLong = (
         l1 * peso1M +
         l5 * peso5M +
         l15 * peso15M +
         l30 * peso30M +
         l60 * peso1H +
         l240 * peso4H +
         lD * pesoD) / somaPesos

    float mtfShort = (
         s1 * peso1M +
         s5 * peso5M +
         s15 * peso15M +
         s30 * peso30M +
         s60 * peso1H +
         s240 * peso4H +
         sD * pesoD) / somaPesos


    // ========================================================================
    // ATR COMBINADO
    // ========================================================================

    float atrCombinado = atr15 * 0.35 + atr60 * 0.35 + atr240 * 0.30

    if atrAdaptativo
        atrCombinado := atrCombinado * fatorATR


    // ========================================================================
    // TENDÊNCIA PRINCIPAL
    // ========================================================================

    string tendencia = "LATERAL"

    if ema9_1 > ema21_1 and ema21_1 > ema50_1
        tendencia := "ALTA"
    else if ema9_1 < ema21_1 and ema21_1 < ema50_1
        tendencia := "BAIXA"


    // ========================================================================
    // TENDÊNCIA MTF PRINCIPAL
    // ========================================================================

    float tendenciaMTFLong = 0.0
    float tendenciaMTFShort = 0.0

    if l30 > s30
        tendenciaMTFLong += 25
    if l60 > s60
        tendenciaMTFLong += 30
    if l240 > s240
        tendenciaMTFLong += 30
    if lD > sD
        tendenciaMTFLong += 15

    if s30 > l30
        tendenciaMTFShort += 25
    if s60 > l60
        tendenciaMTFShort += 30
    if s240 > l240
        tendenciaMTFShort += 30
    if sD > lD
        tendenciaMTFShort += 15


    // ========================================================================
    // FUNDO / TOPO
    // ========================================================================

    float fundo = ta.lowest(low, estruturaLookback)
    float topo = ta.highest(high, estruturaLookback)

    float distanciaFundoATR = atrCombinado != 0 ? math.abs(close - fundo) / atrCombinado : 99.0
    float distanciaTopoATR = atrCombinado != 0 ? math.abs(close - topo) / atrCombinado : 99.0

    float proximidadeFundo = f_clamp(100.0 - distanciaFundoATR / math.max(distanciaMaximaATR, 0.1) * 100.0, 0.0, 100.0)
    float proximidadeTopo = f_clamp(100.0 - distanciaTopoATR / math.max(distanciaMaximaATR, 0.1) * 100.0, 0.0, 100.0)

    float forcaFundo = proximidadeFundo

    if rsi1 <= rsiSobrevendido
        forcaFundo += 20
    else if rsi1 < 45
        forcaFundo += 10

    if kvo1 > kvoSig1
        forcaFundo += 15

    if close > open
        forcaFundo += 10

    if close > close[1]
        forcaFundo += 5

    forcaFundo := f_clamp(forcaFundo, 0.0, 100.0)


    float forcaTopo = proximidadeTopo

    if rsi1 >= rsiSobrecomprado
        forcaTopo += 20
    else if rsi1 > 55
        forcaTopo += 10

    if kvo1 < kvoSig1
        forcaTopo += 15

    if close < open
        forcaTopo += 10

    if close < close[1]
        forcaTopo += 5

    forcaTopo := f_clamp(forcaTopo, 0.0, 100.0)


    // ========================================================================
    // MICROESTRUTURA
    // ========================================================================

    float microTopo = ta.highest(high, microLookback)
    float microFundo = ta.lowest(low, microLookback)

    bool microBreakLong = close > microTopo[1]
    bool microBreakShort = close < microFundo[1]

    bool recuperacaoLong = low < low[1] and close > open and close > close[1]
    bool rejeicaoShort = high > high[1] and close < open and close < close[1]

    float microLong = 35.0
    if microBreakLong
        microLong += 35
    if recuperacaoLong
        microLong += 20
    if close > ema9_1
        microLong += 10

    microLong := f_clamp(microLong, 0.0, 100.0)


    float microShort = 35.0
    if microBreakShort
        microShort += 35
    if rejeicaoShort
        microShort += 20
    if close < ema9_1
        microShort += 10

    microShort := f_clamp(microShort, 0.0, 100.0)


    // ========================================================================
    // CONFLUÊNCIA LONG
    // ========================================================================

    float confluenciaLong = 0.0

    if tendencia == "ALTA"
        confluenciaLong += 25

    if kvo1 > kvoSig1
        confluenciaLong += 20

    if rsi1 < 55
        confluenciaLong += 15

    if mtfLong > mtfShort
        confluenciaLong += 20

    if microLong >= 60
        confluenciaLong += 20

    confluenciaLong := f_clamp(confluenciaLong, 0.0, 100.0)


    // ========================================================================
    // CONFLUÊNCIA SHORT
    // ========================================================================

    float confluenciaShort = 0.0

    if tendencia == "BAIXA"
        confluenciaShort += 25

    if kvo1 < kvoSig1
        confluenciaShort += 20

    if rsi1 > 45
        confluenciaShort += 15

    if mtfShort > mtfLong
        confluenciaShort += 20

    if microShort >= 60
        confluenciaShort += 20

    confluenciaShort := f_clamp(confluenciaShort, 0.0, 100.0)


    // ========================================================================
    // SCORE FINAL
    // ========================================================================

    float scoreLong = l1 * 0.50 + mtfLong * 0.50
    float scoreShort = s1 * 0.50 + mtfShort * 0.50


    // ========================================================================
    // VANTAGEM
    // ========================================================================

    float vantagemLong = f_clamp(scoreLong - scoreShort, 0.0, 100.0)
    float vantagemShort = f_clamp(scoreShort - scoreLong, 0.0, 100.0)


    // ========================================================================
    // INTENSIDADE
    // ========================================================================

    float intensidadeLong =
         scoreLong * 0.40 +
         confluenciaLong * 0.20 +
         microLong * 0.15 +
         forcaFundo * 0.25

    float intensidadeShort =
         scoreShort * 0.40 +
         confluenciaShort * 0.20 +
         microShort * 0.15 +
         forcaTopo * 0.25

    intensidadeLong := f_clamp(intensidadeLong, 0.0, 100.0)
    intensidadeShort := f_clamp(intensidadeShort, 0.0, 100.0)


    // ========================================================================
    // QUALIDADE INDEPENDENTE
    // ========================================================================

    float qualidadeLong =
         scoreLong * 0.25 +
         intensidadeLong * 0.20 +
         confluenciaLong * 0.15 +
         mtfLong * 0.15 +
         microLong * 0.10 +
         forcaFundo * 0.15

    float qualidadeShort =
         scoreShort * 0.25 +
         intensidadeShort * 0.20 +
         confluenciaShort * 0.15 +
         mtfShort * 0.15 +
         microShort * 0.10 +
         forcaTopo * 0.15

    qualidadeLong := f_clamp(qualidadeLong, 0.0, 100.0)
    qualidadeShort := f_clamp(qualidadeShort, 0.0, 100.0)


    // ========================================================================
    // VOLATILIDADE
    // ========================================================================

    string classeVolatilidade = f_classVolatilidade(vol1)
    bool volatilidadeOK = not filtroVolatilidade or vol1 >= volatilidadeMinima


    // ========================================================================
    // DISTÂNCIA DA EMA
    // ========================================================================

    float distanciaEMA = atrCombinado != 0 ? math.abs(close - ema21_1) / atrCombinado : 99.0
    float distanciaPercentual = close != 0 ? math.abs(close - ema21_1) / close * 100.0 : 0.0

    string distanciaTexto = "NORMAL"

    if distanciaEMA < 0.50

    // ========================================================================

    // CONDIÇÕES DE PRÉ-SINAL
    // ========================================================================

    bool preLongCond =

         intensidadeLong >= preIntensidadeMinima and
         confluenciaLong >= preConfluenciaMinima and
         vantagemLong >= preVantagemMinima and

         vantagemShort >= preVantagemMinima and
         mtfShort >= preMtfMinimo and
         forcaTopo >= preForcaMinima and
         microShort >= preMicroMinimo and
         qualidadeShort >= qualidadePreMinima and
         volatilidadeOK and
         fibShortOK and
         distanciaOK and
         scoreShort > scoreLong


    // ========================================================================
    // CONDIÇÕES FINAIS
    // ========================================================================

    bool finalLong =
         scoreLong >= scoreMinimo and
         intensidadeLong >= intensidadeMinima and
         confluenciaLong >= confluenciaMinima and
         vantagemLong >= vantagemMinima and
         mtfLong >= mtfMinimo and
         forcaFundo >= forcaFundoTopoMinima and
         microLong >= microMinimo and
         qualidadeLong >= qualidadeSinalMinima and
         tendenciaLongOK and
         mtfLongConfirmado and
         volatilidadeOK and
         fibLongOK and
         distanciaOK and

         vantagemLong > vantagemShort


    bool finalShort =
         scoreShort >= scoreMinimo and
         intensidadeShort >= intensidadeMinima and
         confluenciaShort >= confluenciaMinima and
         vantagemShort >= vantagemMinima and
         mtfShort >= mtfMinimo and
         forcaTopo >= forcaFundoTopoMinima and
         microShort >= microMinimo and
         qualidadeShort >= qualidadeSinalMinima and
         tendenciaShortOK and
         mtfShortConfirmado and
         volatilidadeOK and
         fibShortOK and
         distanciaOK and
         scoreShort > scoreLong and
         vantagemShort > vantagemLong

    else if finalShort

    else if preShortCond

    // ========================================================================

    float scorePrincipal = ehLong ? scoreLong : scoreShort
    float intensidadePrincipal = ehLong ? intensidadeLong : intensidadeShort
    float confluenciaPrincipal = ehLong ? confluenciaLong : confluenciaShort
    float vantagemPrincipal = ehLong ? vantagemLong : vantagemShort
    float mtfPrincipal = ehLong ? mtfLong : mtfShort
    float qualidadePrincipal = ehLong ? qualidadeLong : qualidadeShort

    float alvoPrincipal = ehLong ? close + atrCombinado * targetPrincipalATR : close - atrCombinado * targetPrincipalATR

    // RETORNO
    // ========================================================================

         sinal,
         scorePrincipal,
         intensidadePrincipal,
         confluenciaPrincipal,
         vantagemPrincipal,
         mtfPrincipal,
         qualidadePrincipal,
         classeVolatilidade,
         vol1,
         tendencia,

         distanciaPercentual,
         classeFib,
         fibPos,
         alvoPrincipal
     ]
