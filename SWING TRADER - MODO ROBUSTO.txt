f_clamp(float x, float lo, float hi) =>
    math.max(lo, math.min(hi, x))

f_pts(bool cond, float peso) =>
    cond ? peso : 0.0

f_boolText(bool cond, string sim, string nao) =>
    cond ? sim : nao

// ============================================================================
// FUNÇÃO PRINCIPAL DE ANÁLISE DIÁRIA
//
// Retorna os dados da ÚLTIMA vela diária FECHADA.
//
// [1] = vela diária anterior à atual.
// ============================================================================

f_diario() =>

    // ------------------------------------------------------------------------
    // INDICADORES
    // ------------------------------------------------------------------------

    float ema5 =
         ta.ema(close, 5)

    float ema9 =
         ta.ema(close, 9)

    float ema21 =
         ta.ema(close, 21)

    float ema50 =
         ta.ema(close, 50)

    float rsi =
         ta.rsi(close, 7)

    float atr =
         ta.atr(14)

    [diPlus, diMinus, adx] =
         ta.dmi(14, 14)

    // ------------------------------------------------------------------------
    // MOMENTUM
    // ------------------------------------------------------------------------

    float impulso =
         atr > 0
         ? (close - close[1]) / atr
         : 0.0

    float aceleracao =
         impulso - nz(impulso[1], 0.0)

    // ------------------------------------------------------------------------
    // SUPORTE / RESISTÊNCIA
    // ------------------------------------------------------------------------

    float resistencia =
         ta.highest(high, periodoSR)[1]

    float suporte =
         ta.lowest(low, periodoSR)[1]

    float distanciaResATR =
         atr > 0 and not na(resistencia)
         ? (resistencia - close) / atr
         : 99.0

    float distanciaSupATR =
         atr > 0 and not na(suporte)
         ? (close - suporte) / atr
         : 99.0

    // ------------------------------------------------------------------------
    // RANGE / CANDLE
    // ------------------------------------------------------------------------

    // CORREÇÃO:
    // "range" é palavra reservada no Pine Script.
    // Foi utilizado "rangeCandle".

    float rangeCandle =
         high - low

    float rangeMedio =
         ta.sma(rangeCandle, 20)

    float corpo =
         math.abs(close - open)

    float corpoMedio =
         ta.sma(math.abs(close - open), 20)

    float pavioSup =
         high - math.max(open, close)

    float pavioInf =
         math.min(open, close) - low

    bool candleAlta =
         close > open

    bool candleBaixa =
         close < open

    bool candleForteAlta =
         atr > 0 and
         corpo >= atr * 0.35 and
         candleAlta

    bool candleForteBaixa =
         atr > 0 and
         corpo >= atr * 0.35 and
         candleBaixa

    bool compressao =
         rangeMedio > 0 and
         rangeCandle < rangeMedio * 0.80

    // ------------------------------------------------------------------------
    // ROMPIMENTOS
    // ------------------------------------------------------------------------

    bool rompAlta =
         not na(resistencia) and
         close > resistencia

    bool rompBaixa =
         not na(suporte) and
         close < suporte

    bool preAlta =
         not na(resistencia) and
         atr > 0 and
         high >= resistencia - atr * 0.35 and
         close > open

    bool preBaixa =
         not na(suporte) and
         atr > 0 and
         low <= suporte + atr * 0.35 and
         close < open

    // ------------------------------------------------------------------------
    // RETESTE
    // ------------------------------------------------------------------------

    bool retesteAlta =
         not na(resistencia) and
         low <= resistencia and
         close > resistencia and
         close > open

    bool retesteBaixa =
         not na(suporte) and
         high >= suporte and
         close < suporte and
         close < open

    // ------------------------------------------------------------------------
    // REJEIÇÕES
    // ------------------------------------------------------------------------

    bool rejeicaoSuporte =
         not na(suporte) and
         low <= suporte and
         close > open and
         close > (high + low) / 2

    bool rejeicaoResistencia =
         not na(resistencia) and
         high >= resistencia and
         close < open and
         close < (high + low) / 2

    // ------------------------------------------------------------------------
    // PADRÕES DE CANDLE
    // ------------------------------------------------------------------------

    bool engolfoAlta =
         close > open and
         close[1] < open[1] and
         close >= open[1] and
         open <= close[1]

    bool engolfoBaixa =
         close < open and
         close[1] > open[1] and
         close <= open[1] and
         open >= close[1]

    bool martelo =
         pavioInf >= corpo * 2 and
         pavioSup <= corpo and
         close > open

    bool estrelaCadente =
         pavioSup >= corpo * 2 and
         pavioInf <= corpo and
         close < open

    // ------------------------------------------------------------------------
    // TENDÊNCIA
    // ------------------------------------------------------------------------

    bool tendenciaAlta =
         ema9 > ema21 and
         ema21 > ema50 and
         close >= ema9

    bool tendenciaBaixa =
         ema9 < ema21 and
         ema21 < ema50 and
         close <= ema9

    // ------------------------------------------------------------------------
    // MOMENTUM
    // ------------------------------------------------------------------------

    bool momentumAlta =
         rsi >= 52 and
         impulso > 0 and
         aceleracao >= -0.10 and
         diPlus >= diMinus

    bool momentumBaixa =
         rsi <= 48 and
         impulso < 0 and
         aceleracao <= 0.10 and
         diMinus >= diPlus

    // ------------------------------------------------------------------------
    // ESPAÇO
    // ------------------------------------------------------------------------

    bool espacoCompra =
         rompAlta or
         distanciaResATR >= 0.50

    bool espacoVenda =
         rompBaixa or
         distanciaSupATR >= 0.50

    // ------------------------------------------------------------------------
    // VOLATILIDADE
    // ------------------------------------------------------------------------

    bool volatilidadeBoa =
         atr > 0 and
         rangeMedio > 0 and
         rangeCandle >= rangeMedio * 0.50 and
         rangeCandle <= rangeMedio * 2.50

    // ------------------------------------------------------------------------
    // PADRÕES GRÁFICOS
    // ------------------------------------------------------------------------

    float max5 =
         ta.highest(high, 5)[1]

    float min5 =
         ta.lowest(low, 5)[1]

    float max10 =
         ta.highest(high, 10)[1]

    float min10 =
         ta.lowest(low, 10)[1]

    float max20 =
         ta.highest(high, 20)[1]

    float min20 =
         ta.lowest(low, 20)[1]

    // ------------------------------------------------------------------------
    // TRIÂNGULO / CONSOLIDAÇÃO
    // ------------------------------------------------------------------------

    float amplitude20 =
         max20 - min20

    float amplitude10 =
         max10 - min10

    bool triangulo =
         amplitude20 > 0 and
         amplitude10 < amplitude20 * 0.72 and
         rangeMedio > 0 and
         rangeCandle < rangeMedio * 1.15

    // ------------------------------------------------------------------------
    // CANAL
    // ------------------------------------------------------------------------

    float max8 =
         ta.highest(high, 8)[1]

    float min8 =
         ta.lowest(low, 8)[1]

    float centro8 =
         (max8 + min8) / 2

    bool canalAlta =
         close > centro8 and
         ema9 > ema21

    bool canalBaixa =
         close < centro8 and
         ema9 < ema21

    // ------------------------------------------------------------------------
    // TOPO / FUNDO DUPLO
    // ------------------------------------------------------------------------

    float topoA =
         ta.highest(high, 10)[3]

    float topoB =
         ta.highest(high, 10)[1]

    float fundoA =
         ta.lowest(low, 10)[3]

    float fundoB =
         ta.lowest(low, 10)[1]

    bool topoDuplo =
         topoA > 0 and
         math.abs(topoA - topoB) <= atr * 0.60 and
         close < topoB

    bool fundoDuplo =
         fundoA > 0 and
         math.abs(fundoA - fundoB) <= atr * 0.60 and
         close > fundoB

    // ------------------------------------------------------------------------
    // CUNHA
    // ------------------------------------------------------------------------

    bool cunhaAlta =
         amplitude20 > 0 and
         amplitude10 < amplitude20 * 0.65 and
         ema9 > ema21 and
         close > centro8

    bool cunhaBaixa =
         amplitude20 > 0 and
         amplitude10 < amplitude20 * 0.65 and
         ema9 < ema21 and
         close < centro8

    // ------------------------------------------------------------------------
    // BANDEIRA
    // ------------------------------------------------------------------------

    float impulsoAnterior =
         atr > 0
         ? math.abs(close[5] - close[10]) / atr
         : 0

    bool bandeiraAlta =
         impulsoAnterior >= 1.20 and
         compressao and
         ema9 >= ema21

    bool bandeiraBaixa =
         impulsoAnterior >= 1.20 and
         compressao and
         ema9 <= ema21

    // ------------------------------------------------------------------------
    // AGREGAÇÃO DOS PADRÕES
    // ------------------------------------------------------------------------

    bool padraoAlta =
         usarPadroes and
         (
             (usarDuplo and fundoDuplo) or
             (usarTriangulo and triangulo and rompAlta) or
             (usarCanal and canalAlta) or
             (usarCunha and cunhaAlta) or
             (usarBandeira and bandeiraAlta) or
             retesteAlta or
             engolfoAlta
         )

    bool padraoBaixa =
         usarPadroes and
         (
             (usarDuplo and topoDuplo) or
             (usarTriangulo and triangulo and rompBaixa) or
             (usarCanal and canalBaixa) or
             (usarCunha and cunhaBaixa) or
             (usarBandeira and bandeiraBaixa) or
             retesteBaixa or
             engolfoBaixa
         )

    // ------------------------------------------------------------------------
    // PONTUAÇÃO ESTRUTURAL
    // ------------------------------------------------------------------------

    float estruturaAlta =
         f_clamp(
             f_pts(rompAlta, 25) +
             f_pts(retesteAlta, 15) +
             f_pts(padraoAlta, 20) +
             f_pts(preAlta, 8) +
             f_pts(rejeicaoSuporte, 8) +
             f_pts(engolfoAlta, 7) +
             f_pts(martelo, 5) +
             f_pts(candleForteAlta, 7) +
             f_pts(compressao, 5),
             0,
             100)

    float estruturaBaixa =
         f_clamp(
             f_pts(rompBaixa, 25) +
             f_pts(retesteBaixa, 15) +
             f_pts(padraoBaixa, 20) +
             f_pts(preBaixa, 8) +
             f_pts(rejeicaoResistencia, 8) +
             f_pts(engolfoBaixa, 7) +
             f_pts(estrelaCadente, 5) +
             f_pts(candleForteBaixa, 7) +
             f_pts(compressao, 5),
             0,
             100)

    // ------------------------------------------------------------------------
    // PREVISÃO
    // ------------------------------------------------------------------------

    float previsaoAlta =
         f_clamp(
             f_pts(close > ema5, 7) +
             f_pts(close > ema9, 8) +
             f_pts(ema9 > ema21, 12) +
             f_pts(ema21 > ema50, 10) +
             f_pts(rsi >= 52, 8) +
             f_pts(rsi >= 55, 5) +
             f_pts(impulso > 0, 9) +
             f_pts(aceleracao > 0, 6) +
             f_pts(diPlus > diMinus, 8) +
             f_pts(adx >= 18, 5) +
             f_pts(estruturaAlta >= 40, 8) +
             f_pts(espacoCompra, 8) +
             f_pts(padraoAlta, 6),
             0,
             100)

    float previsaoBaixa =
         f_clamp(
             f_pts(close < ema5, 7) +
             f_pts(close < ema9, 8) +
             f_pts(ema9 < ema21, 12) +
             f_pts(ema21 < ema50, 10) +
             f_pts(rsi <= 48, 8) +
             f_pts(rsi <= 45, 5) +
             f_pts(impulso < 0, 9) +
             f_pts(aceleracao < 0, 6) +
             f_pts(diMinus > diPlus, 8) +
             f_pts(adx >= 18, 5) +
             f_pts(estruturaBaixa >= 40, 8) +
             f_pts(espacoVenda, 8) +
             f_pts(padraoBaixa, 6),
             0,
             100)

    // ------------------------------------------------------------------------
    // SCORE
    // ------------------------------------------------------------------------

    float scoreAlta =
         f_clamp(
             f_pts(tendenciaAlta, 18) +
             f_pts(momentumAlta, 15) +
             f_pts(estruturaAlta >= 40, 13) +
             f_pts(previsaoAlta >= 60, 13) +
             f_pts(padraoAlta, 12) +
             f_pts(espacoCompra, 8) +
             f_pts(rompAlta, 8) +
             f_pts(retesteAlta, 5) +
             f_pts(volatilidadeBoa, 4) +
             f_pts(candleForteAlta or engolfoAlta, 4),
             0,
             100)

    float scoreBaixa =
         f_clamp(
             f_pts(tendenciaBaixa, 18) +
             f_pts(momentumBaixa, 15) +
             f_pts(estruturaBaixa >= 40, 13) +
             f_pts(previsaoBaixa >= 60, 13) +
             f_pts(padraoBaixa, 12) +
             f_pts(espacoVenda, 8) +
             f_pts(rompBaixa, 8) +
             f_pts(retesteBaixa, 5) +
             f_pts(volatilidadeBoa, 4) +
             f_pts(candleForteBaixa or engolfoBaixa, 4),
             0,
             100)

    // ------------------------------------------------------------------------
    // RETORNO
    // ------------------------------------------------------------------------

    [
        close[1],
        ema5[1],
        ema9[1],
        ema21[1],
        ema50[1],
        rsi[1],
        atr[1],
        diPlus[1],
        diMinus[1],
        adx[1],
        suporte[1],
        resistencia[1],
        rompAlta[1],
        rompBaixa[1],
        preAlta[1],
        preBaixa[1],
        retesteAlta[1],
        retesteBaixa[1],
        tendenciaAlta[1],
        tendenciaBaixa[1],
        momentumAlta[1],
        momentumBaixa[1],
        volatilidadeBoa[1],
        padraoAlta[1],
        padraoBaixa[1],
        triangulo[1],
        canalAlta[1],
        canalBaixa[1],
        topoDuplo[1],
        fundoDuplo[1],
        cunhaAlta[1],
        cunhaBaixa[1],
        bandeiraAlta[1],
        bandeiraBaixa[1],
        engolfoAlta[1],
        engolfoBaixa[1],
        martelo[1],
        estrelaCadente[1],
        estruturaAlta[1],
        estruturaBaixa[1],
        previsaoAlta[1],
        previsaoBaixa[1],
        scoreAlta[1],
        scoreBaixa[1],
        distanciaResATR[1],
        distanciaSupATR[1]
    ]

// ============================================================================

[

f_semanal() =>

    float ema9 =
         ta.ema(close, 9)

    float ema21 =
         ta.ema(close, 21)

    [
        ema9[1],
        ema21[1],
        close[1]
    ]

[
    ema9W,
    ema21W,
    closeW
] =
     request.security(
         syminfo.tickerid,
         "W",
         f_semanal(),
         gaps=barmerge.gaps_off,
         lookahead=barmerge.lookahead_off)

bool semanalAlta =
     ema9W > ema21W

bool semanalBaixa =
     ema9W < ema21W

float vantagemPrevisao =
