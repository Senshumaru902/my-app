
instrument {
    name = "QMS BANDEIRAS ALTA E BAIXA V1",
    short_name = "QMS FLAG",
    overlay = true
}

-- =========================================
-- CONFIGURACOES
-- =========================================

periodo_bandeira = input(10, "Velas da bandeira", input.integer, 5, 30)
periodo_impulso = input(12, "Velas da pernada", input.integer, 5, 40)
periodo_atr = input(14, "Periodo ATR", input.integer, 5, 30)

forca_impulso = input(2.0, "Forca minima da pernada (ATR)", input.double, 0.5, 5.0)
max_correcao = input(0.65, "Correcao maxima da pernada", input.double, 0.2, 0.9)

input_group {
    "SUPORTE E RESISTENCIA",

    cor_resistencia = input {
        default = "#FF3030",
        type = input.color
    },

    cor_suporte = input {
        default = "#00C853",
        type = input.color
    },

    largura = input {
        default = 2,
        type = input.line_width
    }
}

input_group {
    "SINAIS",

    cor_alta = input {
        default = "#FFD600",
        type = input.color
    },

    cor_baixa = input {
        default = "#2196FF",
        type = input.color
    }
}

-- =========================================
-- VOLATILIDADE
-- =========================================

atr_valor = sma(tr, periodo_atr)

-- =========================================
-- ESTRUTURA DA BANDEIRA
-- Utiliza velas fechadas
-- =========================================

topo_bandeira = highest(high[1], periodo_bandeira)
fundo_bandeira = lowest(low[1], periodo_bandeira)

amplitude_bandeira = topo_bandeira - fundo_bandeira

-- =========================================
-- PERNADA ANTERIOR
-- =========================================

topo_impulso = highest(high[periodo_bandeira + 1], periodo_impulso)

fundo_impulso = lowest(low[periodo_bandeira + 1], periodo_impulso)

movimento_alta =
    close[periodo_bandeira + 1] -
    close[periodo_bandeira + periodo_impulso + 1]

movimento_baixa =
    close[periodo_bandeira + periodo_impulso + 1] -
    close[periodo_bandeira + 1]

-- =========================================
-- BANDEIRA DE ALTA
-- =========================================

impulso_alta =
    movimento_alta > atr_valor[1] * forca_impulso

correcao_alta =
    fundo_bandeira > topo_impulso -
    movimento_alta * max_correcao

canal_alta =
    close[1] < close[periodo_bandeira] and
    amplitude_bandeira < movimento_alta * 0.80

bandeira_alta =
    impulso_alta and
    correcao_alta and
    canal_alta

-- =========================================
-- BANDEIRA DE BAIXA
-- =========================================

impulso_baixa =
    movimento_baixa > atr_valor[1] * forca_impulso

correcao_baixa =
    topo_bandeira < fundo_impulso +
    movimento_baixa * max_correcao

canal_baixa =
    close[1] > close[periodo_bandeira] and
    amplitude_bandeira < movimento_baixa * 0.80

bandeira_baixa =
    impulso_baixa and
    correcao_baixa and
    canal_baixa

-- =========================================
-- ROMPIMENTOS
-- =========================================

resistencia_anterior = highest(high[2], periodo_bandeira)
suporte_anterior = lowest(low[2], periodo_bandeira)

rompimento_alta =
    bandeira_alta and
    close[1] > resistencia_anterior

rompimento_baixa =
    bandeira_baixa and
    close[1] < suporte_anterior

-- =========================================
-- LINHAS DINAMICAS
-- =========================================

resistencia_alta = iff(bandeira_alta, topo_bandeira, nil)
suporte_alta = iff(bandeira_alta, fundo_bandeira, nil)

resistencia_baixa = iff(bandeira_baixa, topo_bandeira, nil)
suporte_baixa = iff(bandeira_baixa, fundo_bandeira, nil)

plot(
    resistencia_alta,
    "Resistencia Bandeira Alta",
    cor_resistencia,
    largura
)

plot(
    suporte_alta,
    "Suporte Bandeira Alta",
    cor_suporte,
    largura
)

plot(
    resistencia_baixa,
    "Resistencia Bandeira Baixa",
    cor_resistencia,
    largura
)

plot(
    suporte_baixa,
    "Suporte Bandeira Baixa",
    cor_suporte,
    largura
)

-- =========================================
-- MARCACAO DOS ROMPIMENTOS
-- =========================================

plot_shape(
    rompimento_alta,
    "ROMPIMENTO ALTA",
    shape_style.triangleup,
    shape_size.large,
    cor_alta,
    shape_location.belowbar,
    -1,
    "BANDEIRA ALTA",
    cor_alta
)

plot_shape(
    rompimento_baixa,
    "ROMPIMENTO BAIXA",
    shape_style.triangledown,
    shape_size.large,
    cor_baixa,
    shape_location.abovebar,
    -1,
    "BANDEIRA BAIXA",
    cor_baixa
)
