atrMicroMedia = ta.sma(atrMicro, 10)

float atrMicroRatio = 1.0

if atrMicroMedia > 0
    atrMicroRatio := atrMicro / atrMicroMedia

atrMicroExpansao = atrMicro > atrMicro[1]
atrMicroAcelerando = atrMicro > atrMicro[1] and atrMicro[1] > atrMicro[2]


//====================================================
// 16 - EMA MICRO
//====================================================

emaMicroFast = ta.ema(close, microFastLen)
emaMicroMid = ta.ema(close, microMidLen)
emaMicroSlow = ta.ema(close, microSlowLen)

emaMicroUp =
     emaMicroFast > emaMicroMid and
     emaMicroMid > emaMicroSlow

emaMicroDown =
     emaMicroFast < emaMicroMid and
     emaMicroMid < emaMicroSlow

emaMicroSlope =
     emaMicroFast - emaMicroFast[1]

emaMicroSlopePrev =
     emaMicroFast[1] - emaMicroFast[2]

emaMicroAcceleration =
     emaMicroSlope - emaMicroSlopePrev

emaMicroVelocityUp = emaMicroSlope > 0
emaMicroVelocityDown = emaMicroSlope < 0

emaMicroAccelUp = emaMicroAcceleration > 0
emaMicroAccelDown = emaMicroAcceleration < 0

emaMicroSpread =
     math.abs(emaMicroFast - emaMicroSlow)

emaMicroSpreadPrev =
     math.abs(emaMicroFast[1] - emaMicroSlow[1])

emaMicroConvergencia =
     emaMicroSpread < emaMicroSpreadPrev

emaMicroExpansao =
     emaMicroSpread > emaMicroSpreadPrev


//====================================================
// 17 - RSI MICRO
//====================================================

rsiMicro = ta.rsi(close, microRSILen)

rsiMicroVelocity =
     rsiMicro - rsiMicro[1]

rsiMicroVelocityPrev =
     rsiMicro[1] - rsiMicro[2]

rsiMicroAcceleration =
     rsiMicroVelocity - rsiMicroVelocityPrev

rsiMicroUp = rsiMicro > 50
rsiMicroDown = rsiMicro < 50

rsiMicroVelocityUp =
     rsiMicroVelocity > 0

rsiMicroVelocityDown =
     rsiMicroVelocity < 0

rsiMicroAccelUp =
     rsiMicroAcceleration > 0

rsiMicroAccelDown =
     rsiMicroAcceleration < 0

// 18 - DMI MICRO
//====================================================

     ta.dmi(
         microDMILen,
         microDMILen)

dmiMicroUp =
     diPlusMicro > diMinusMicro

dmiMicroDown =
     diMinusMicro > diPlusMicro

dmiMicroSpread =
     diPlusMicro - diMinusMicro

dmiMicroSpreadPrev =
     dmiMicroSpread[1]

dmiMicroAcceleration =
     dmiMicroSpread - dmiMicroSpreadPrev

dmiMicroAccelUp =
     dmiMicroAcceleration > 0

dmiMicroAccelDown =
     dmiMicroAcceleration < 0

adxMicroRising =
     adxMicro > adxMicro[1]


//====================================================
// 19 - VOLUME MICRO
//====================================================

volumeMicroSMA =
     ta.sma(volume, microVolumeLen)

float volumeMicroRatio = 1.0
float volumeMicroPrevRatio = 1.0
float volumeMicroRatio2 = 1.0

if volumeMicroSMA > 0
    volumeMicroRatio := volume / volumeMicroSMA

if volumeMicroSMA[1] > 0
    volumeMicroPrevRatio :=
         volume[1] / volumeMicroSMA[1]

if volumeMicroSMA[2] > 0
    volumeMicroRatio2 :=
         volume[2] / volumeMicroSMA[2]

volumeMicroVelocity =
     volumeMicroRatio -
     volumeMicroPrevRatio

volumeMicroAcceleration =
     volumeMicroVelocity -
     (volumeMicroPrevRatio - volumeMicroRatio2)

volumeMicroUp =
     volumeMicroRatio >= volumeMinimo

volumeMicroStrong =
     volumeMicroRatio >= 1.0

volumeMicroAccelUp =
     volumeMicroAcceleration > 0


//====================================================
// 20 - MOMENTUM MICRO
//====================================================

     close - close[microMomentumLen]

     momentumMicro / atrMicroSafe

     momentumMicroNorm > 0

momentumMicroDown =
     momentumMicroNorm < 0

     momentumMicroNorm >= 0.05

momentumMicroStrongDown =
     momentumMicroNorm <= -0.05

momentumMicroVelocity =
     momentumMicroNorm -
     momentumMicroNorm[1]

momentumMicroVelocityPrev =
     momentumMicroNorm[1] -
     momentumMicroNorm[2]

momentumMicroAcceleration =
     momentumMicroVelocity -
     momentumMicroVelocityPrev

momentumMicroAccelUp =
     momentumMicroAcceleration > 0

momentumMicroAccelDown =
     momentumMicroAcceleration < 0


//====================================================
// 21 - KVO MICRO
//====================================================

dmMicroKVO =
     math.max(
         high - low,
         syminfo.mintick)

var float cmMicroKVO = 0.0

if bar_index == 0
    cmMicroKVO := dmMicroKVO
else
    if hlc3 > hlc3[1]
        cmMicroKVO := cmMicroKVO[1] + dmMicroKVO
    else if hlc3 < hlc3[1]
        cmMicroKVO := cmMicroKVO[1] - dmMicroKVO
    else
        cmMicroKVO := cmMicroKVO[1]

cmMicroSafe =
     math.max(
         math.abs(cmMicroKVO),
         syminfo.mintick)

float trendMicroKVO = 0.0

if hlc3 > hlc3[1]
    trendMicroKVO := 1.0
else if hlc3 < hlc3[1]
    trendMicroKVO := -1.0

vfMicroKVO =
     volume *
     math.abs(
         2.0 *
         (dmMicroKVO / cmMicroSafe) -
         1.0) *
     trendMicroKVO *
     100.0

kvoMicroFast =
     ta.ema(
         vfMicroKVO,
         microKvoFast)

kvoMicroSlow =
     ta.ema(
         vfMicroKVO,
         microKvoSlow)

kvoMicroLine =
     kvoMicroFast -
     kvoMicroSlow

kvoMicroSignalLine =
     ta.ema(
         kvoMicroLine,
         microKvoSignal)

kvoMicroHist =
     kvoMicroLine -
     kvoMicroSignalLine

kvoMicroUp =
     kvoMicroLine >
     kvoMicroSignalLine

kvoMicroDown =
     kvoMicroLine <
     kvoMicroSignalLine

kvoMicroVelocity =
     kvoMicroLine -
     kvoMicroLine[1]

kvoMicroVelocityPrev =
     kvoMicroLine[1] -
     kvoMicroLine[2]

kvoMicroAcceleration =
     kvoMicroVelocity -
     kvoMicroVelocityPrev

kvoMicroAccelUp =
     kvoMicroAcceleration > 0

kvoMicroAccelDown =
     kvoMicroAcceleration < 0

kvoMicroStrongUp =
     kvoMicroHist > 0 and
     kvoMicroVelocity > 0

kvoMicroStrongDown =
     kvoMicroHist < 0 and
     kvoMicroVelocity < 0


//====================================================
// 22 - CANDLE BASE
//====================================================

microRange =
     math.max(
         high - low,
         syminfo.mintick)

microBody =
     math.abs(close - open)

microBodyRatio =
     microBody / microRange

microUpperWick =
     high - math.max(open, close)

microLowerWick =
     math.min(open, close) - low

microClosePosition =
     (close - low) / microRange

microBullBody =
     close > open

microBearBody =
     close < open

microStrongBullCandle =
     microBullBody and
     microBodyRatio >= 0.45 and
     microClosePosition >= 0.65

microStrongBearCandle =
     microBearBody and
     microBodyRatio >= 0.45 and
     microClosePosition <= 0.35

microBullPressure =
     microBullBody and
     microClosePosition >= 0.55

microBearPressure =
     microBearBody and
     microClosePosition <= 0.45

microCandleVelocity =
     (close - open) / microRange

microCandleVelocityPrev =
     (close[1] - open[1]) /
     math.max(
         high[1] - low[1],
         syminfo.mintick)

microCandleAcceleration =
     microCandleVelocity -
     microCandleVelocityPrev

microCandleAccelUp =
     microCandleAcceleration > 0

microCandleAccelDown =
     microCandleAcceleration < 0

     distanciaMicroSuporte <= proximidadeATR


//====================================================
// 25 - ESTRUTURA MICRO
//====================================================

     ta.highest(
         high[1],
         microLookback)

microLowAnterior =
     ta.lowest(
         low[1],
         microLookback)

     high > high[1] and
     high >= microHighAnterior

microLL =
     low < low[1] and
     low <= microLowAnterior

microHL =
     low > low[1]

microLH =
     high < high[1]

microEstruturaAlta =
     (microHH or microHL) and
     close >= emaMicroMid

microEstruturaBaixa =
     (microLL or microLH) and
     close <= emaMicroMid

microEstruturaForteAlta =
     microHH and
     microHL and
     emaMicroUp

microEstruturaForteBaixa =
     microLL and
     microLH and
     emaMicroDown


//====================================================
// 26 - FIGURAS MICRO
//====================================================

     atrMicroSafe *
     toleranciaFiguraATR

microTopoA =
     ta.highest(
         high[1],
         microLookback)

microTopoB =
     ta.highest(
         high[2],
         microLookback)

microFundoA =
     ta.lowest(
         low[1],
         microLookback)

microFundoB =
     ta.lowest(
         low[2],
         microLookback)

microTopoDuplo =
     math.abs(
         microTopoA -
         microTopoB) <=
     microTolFigura

microFundoDuplo =
     math.abs(
         microFundoA -
         microFundoB) <=
     microTolFigura

microTrianguloAsc =
     microTopoA <=
     microTopoB +
     microTolFigura and
     microFundoA >
     microFundoB

microTrianguloDesc =
     microTopoA <
     microTopoB and
     microFundoA >=
     microFundoB -
     microTolFigura

microTrianguloSimetrico =
     microTopoA <
     microTopoB and
     microFundoA >
     microFundoB

microRangeAtual =
     microResistenciaEstrutural -
     microSuporteEstrutural

microConsolidacao =
     microRangeAtual <=
     atrMicroSafe * 1.50

microCanalAlta =
     microConsolidacao and
     close >= emaMicroMid and
     emaMicroUp

microCanalBaixa =
     microConsolidacao and
     close <= emaMicroMid and
     emaMicroDown

     ultimoMicroNivelAlta -
     atrMicroSafe * toleranciaReteste

     usarReteste and
     dentroMicroRetesteBaixa and
     not na(ultimoMicroNivelBaixa) and
