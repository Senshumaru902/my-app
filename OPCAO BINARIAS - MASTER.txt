     high - low

     math.abs(close - open)

     math.min(open, close) -
     low

     math.max(candleRange, syminfo.mintick)

     math.max(candleBody, syminfo.mintick)

     (close - low) /
     safeRange

     close > open

     close < open

     candleBody >=
     atr7 * bodyATRMin and
     candleBody /
     safeRange >=
     bodyRangeMin

     bullishCandle and

     closePosition >=
     closeLongMin and
     upperWick <=
     safeBody * wickBodyMax

     bearishCandle and
     strongBody and

     closeShortMax and
