f_limitar(valor, minimo, maximo) =>
    math.max(minimo, math.min(maximo, valor))

f_pct(valor) =>
    str.tostring(f_limitar(valor, 0, 100), "#.0") + "%"

f_preco(valor) =>
    na(valor) ? "-" : str.tostring(valor, format.mintick)

// ============================================================================
// 10. SCORE MTF
// ============================================================================

f_mtfScore(modo) =>
    atrLocal = ta.atr(atrLength)

    emaFast = ta.ema(close, emaFastLength)
    emaMid = ta.ema(close, emaMidLength)
    emaSlow = ta.ema(close, emaSlowLength)

    rsiLocal = ta.rsi(close, rsiLength)

    volumeMedio = ta.sma(volume, volumeLength)
    volumeRelativo = volumeMedio > 0 ? volume / volumeMedio : 1.0

    corpo = math.abs(close - open)
    rangeCandle = math.max(high - low, syminfo.mintick)
    forcaCandle = f_limitar(corpo / rangeCandle * 100.0, 0, 100)

    retorno1 = close[1] != 0 ? (close - close[1]) / close[1] * 100.0 : 0.0
    retorno3 = close[3] != 0 ? (close - close[3]) / close[3] * 100.0 : 0.0

    tendenciaLong = emaFast > emaMid and emaMid > emaSlow
    tendenciaShort = emaFast < emaMid and emaMid < emaSlow

    forcaTendencia = atrLocal > 0 ? math.abs(emaFast - emaSlow) / atrLocal * 100.0 : 0.0
    forcaTendenciaLimitada = f_limitar(forcaTendencia, 0, 100)

    direcaoVolume = close > close[1] ? 1.0 : close < close[1] ? -1.0 : 0.0

    kvoRaw = ta.ema(volume * direcaoVolume, 34) - ta.ema(volume * direcaoVolume, 55)
    kvoSignal = ta.ema(kvoRaw, 13)

    kvoLong = kvoRaw > kvoSignal
    kvoShort = kvoRaw < kvoSignal

    rsiLongScore = rsiLocal >= 50 ? f_limitar((rsiLocal - 50) * 2.0, 0, 100) : 0.0
    rsiShortScore = rsiLocal <= 50 ? f_limitar((50 - rsiLocal) * 2.0, 0, 100) : 0.0

    retornoLongScore = retorno1 > 0 ? 50.0 + f_limitar(math.abs(retorno1) * 10.0, 0, 50) : 0.0
    retornoShortScore = retorno1 < 0 ? 50.0 + f_limitar(math.abs(retorno1) * 10.0, 0, 50) : 0.0

    retorno3LongScore = retorno3 > 0 ? 50.0 + f_limitar(math.abs(retorno3) * 5.0, 0, 50) : 0.0
    retorno3ShortScore = retorno3 < 0 ? 50.0 + f_limitar(math.abs(retorno3) * 5.0, 0, 50) : 0.0

    volumeScore = f_limitar(volumeRelativo * 50.0, 0, 100)

    tendenciaLongScore = tendenciaLong ? 60.0 + forcaTendenciaLimitada * 0.40 : 20.0 + forcaTendenciaLimitada * 0.20
    tendenciaShortScore = tendenciaShort ? 60.0 + forcaTendenciaLimitada * 0.40 : 20.0 + forcaTendenciaLimitada * 0.20

    kvoLongScore = kvoLong ? 100.0 : 0.0
    kvoShortScore = kvoShort ? 100.0 : 0.0

    candleLongScore = close > open ? forcaCandle : 0.0
    candleShortScore = close < open ? forcaCandle : 0.0

    scoreLong =
         candleLongScore * pesoCandle / 100.0 +
         tendenciaLongScore * pesoTendencia / 100.0 +
         rsiLongScore * pesoRSI / 100.0 +
         kvoLongScore * pesoKVO / 100.0 +
         volumeScore * pesoVolume / 100.0 +
         retornoLongScore * pesoRetorno1 / 100.0 +
         retorno3LongScore * pesoRetorno3 / 100.0 +
         forcaTendenciaLimitada * pesoForca / 100.0

    scoreShort =
         candleShortScore * pesoCandle / 100.0 +
         tendenciaShortScore * pesoTendencia / 100.0 +
         rsiShortScore * pesoRSI / 100.0 +
         kvoShortScore * pesoKVO / 100.0 +
         volumeScore * pesoVolume / 100.0 +
         retornoShortScore * pesoRetorno1 / 100.0 +
         retorno3ShortScore * pesoRetorno3 / 100.0 +
         forcaTendenciaLimitada * pesoForca / 100.0

    somaPesosModelo =
         pesoCandle + pesoTendencia + pesoRSI + pesoKVO +
         pesoVolume + pesoRetorno1 + pesoRetorno3 + pesoForca

    scoreLongNormalizado = somaPesosModelo > 0 ? scoreLong / somaPesosModelo * 100.0 : scoreLong
    scoreShortNormalizado = somaPesosModelo > 0 ? scoreShort / somaPesosModelo * 100.0 : scoreShort

    modo == 1 ? f_limitar(scoreLongNormalizado, 0, 100) :
     modo == -1 ? f_limitar(scoreShortNormalizado, 0, 100) :
     atrLocal

// ============================================================================
// 12. SCORE MTF PONDERADO
// ============================================================================

somaPesos = peso1M + peso5M + peso15M + peso30M + peso1H + peso4H + pesoD

p5 = somaPesos > 0 ? peso5M / somaPesos : 0.0
p15 = somaPesos > 0 ? peso15M / somaPesos : 0.0
p30 = somaPesos > 0 ? peso30M / somaPesos : 0.0
p1H = somaPesos > 0 ? peso1H / somaPesos : 0.0
p4H = somaPesos > 0 ? peso4H / somaPesos : 0.0
pD = somaPesos > 0 ? pesoD / somaPesos : 0.0

scoreMTFLong = f_limitar(
     mtf1Long * p1 + mtf5Long * p5 + mtf15Long * p15 +
     mtf30Long * p30 + mtf1HLong * p1H + mtf4HLong * p4H + mtfDLong * pD, 0, 100)

scoreMTFShort = f_limitar(
     mtf1Short * p1 + mtf5Short * p5 + mtf15Short * p15 +
     mtf30Short * p30 + mtf1HShort * p1H + mtf4HShort * p4H + mtfDShort * pD, 0, 100)

// ============================================================================
// 13. ATR PRINCIPAL
// ============================================================================

somaPesosATR = pesoATR15M + pesoATR1H + pesoATR4H

atrLongo = somaPesosATR > 0 ?
     (atr15M * pesoATR15M + atr1H * pesoATR1H + atr4H * pesoATR4H) / somaPesosATR :
     ta.atr(atrLength)

atrLongoSeguro = math.max(nz(atrLongo, ta.atr(atrLength)), syminfo.mintick)

// ============================================================================
// 14. MODELO LOCAL

atrLocal = ta.atr(atrLength)

emaFastLocal = ta.ema(close, emaFastLength)
emaMidLocal = ta.ema(close, emaMidLength)

rsiLocal = ta.rsi(close, rsiLength)

volumeMedioLocal = ta.sma(volume, volumeLength)
volumeRelativoLocal = volumeMedioLocal > 0 ? volume / volumeMedioLocal : 1.0

corpoLocal = math.abs(close - open)
rangeLocal = math.max(high - low, syminfo.mintick)

forcaCandleLocal = f_limitar(corpoLocal / rangeLocal * 100.0, 0, 100)

retorno1Local = close[1] != 0 ? (close - close[1]) / close[1] * 100.0 : 0.0
retorno3Local = close[3] != 0 ? (close - close[3]) / close[3] * 100.0 : 0.0

tendenciaLongLocal = emaFastLocal > emaMidLocal and emaMidLocal > emaSlowLocal
tendenciaShortLocal = emaFastLocal < emaMidLocal and emaMidLocal < emaSlowLocal

forcaTendenciaLocal = atrLocal > 0 ? math.abs(emaFastLocal - emaSlowLocal) / atrLocal * 100.0 : 0.0
forcaTendenciaLocal := f_limitar(forcaTendenciaLocal, 0, 100)

direcaoVolumeLocal = close > close[1] ? 1.0 : close < close[1] ? -1.0 : 0.0

kvoRawLocal = ta.ema(volume * direcaoVolumeLocal, 34) - ta.ema(volume * direcaoVolumeLocal, 55)
kvoSignalLocal = ta.ema(kvoRawLocal, 13)

kvoLongLocal = kvoRawLocal > kvoSignalLocal
kvoShortLocal = kvoRawLocal < kvoSignalLocal

rsiLongLocal = rsiLocal >= 50 ? f_limitar((rsiLocal - 50) * 2.0, 0, 100) : 0.0
rsiShortLocal = rsiLocal <= 50 ? f_limitar((50 - rsiLocal) * 2.0, 0, 100) : 0.0

retornoLongLocal = retorno1Local > 0 ? 50.0 + f_limitar(math.abs(retorno1Local) * 10.0, 0, 50) : 0.0
retornoShortLocal = retorno1Local < 0 ? 50.0 + f_limitar(math.abs(retorno1Local) * 10.0, 0, 50) : 0.0

retorno3LongLocal = retorno3Local > 0 ? 50.0 + f_limitar(math.abs(retorno3Local) * 5.0, 0, 50) : 0.0
retorno3ShortLocal = retorno3Local < 0 ? 50.0 + f_limitar(math.abs(retorno3Local) * 5.0, 0, 50) : 0.0

volumeLongLocal = volumeRelativoLocal >= 1 ? f_limitar(volumeRelativoLocal * 50.0, 0, 100) : 0.0
volumeShortLocal = volumeRelativoLocal >= 1 ? f_limitar(volumeRelativoLocal * 50.0, 0, 100) : 0.0

tendenciaLongScore = tendenciaLongLocal ? 60.0 + forcaTendenciaLocal * 0.40 : 20.0 + forcaTendenciaLocal * 0.20
tendenciaShortScore = tendenciaShortLocal ? 60.0 + forcaTendenciaLocal * 0.40 : 20.0 + forcaTendenciaLocal * 0.20

kvoLongScore = kvoLongLocal ? 100.0 : 0.0
kvoShortScore = kvoShortLocal ? 100.0 : 0.0

candleLongScore = close > open ? forcaCandleLocal : 0.0
candleShortScore = close < open ? forcaCandleLocal : 0.0

scoreLocalLong =
     candleLongScore * pesoCandle / 100.0 +
     tendenciaLongScore * pesoTendencia / 100.0 +
     rsiLongLocal * pesoRSI / 100.0 +
     kvoLongScore * pesoKVO / 100.0 +
     volumeLongLocal * pesoVolume / 100.0 +
     retornoLongLocal * pesoRetorno1 / 100.0 +
     retorno3LongLocal * pesoRetorno3 / 100.0 +
     forcaTendenciaLocal * pesoForca / 100.0

scoreLocalShort =
     candleShortScore * pesoCandle / 100.0 +
     tendenciaShortScore * pesoTendencia / 100.0 +
     rsiShortLocal * pesoRSI / 100.0 +
     kvoShortScore * pesoKVO / 100.0 +
     volumeShortLocal * pesoVolume / 100.0 +
     retornoShortLocal * pesoRetorno1 / 100.0 +
     retorno3ShortLocal * pesoRetorno3 / 100.0 +
     forcaTendenciaLocal * pesoForca / 100.0

somaPesosLocal = pesoCandle + pesoTendencia + pesoRSI + pesoKVO + pesoVolume + pesoRetorno1 + pesoRetorno3 + pesoForca

scoreLocalLongSeguro = somaPesosLocal > 0 ? f_limitar(scoreLocalLong / somaPesosLocal * 100.0, 0, 100) : 0.0
scoreLocalShortSeguro = somaPesosLocal > 0 ? f_limitar(scoreLocalShort / somaPesosLocal * 100.0, 0, 100) : 0.0

scoreLongSeguro = f_limitar(scoreLocalLongSeguro * 0.50 + scoreMTFLong * 0.50, 0, 100)
scoreShortSeguro = f_limitar(scoreLocalShortSeguro * 0.50 + scoreMTFShort * 0.50, 0, 100)

vantagemLong = scoreLongSeguro - scoreShortSeguro
vantagemShort = scoreShortSeguro - scoreLongSeguro

intensidadeLong = f_limitar(
     scoreLongSeguro * 0.60 +
     forcaTendenciaLocal * 0.20 +
     forcaCandleLocal * 0.10 +
     f_limitar(volumeRelativoLocal * 50.0, 0, 100) * 0.10, 0, 100)

intensidadeShort = f_limitar(
     scoreShortSeguro * 0.60 +
     forcaTendenciaLocal * 0.20 +
     forcaCandleLocal * 0.10 +
     f_limitar(volumeRelativoLocal * 50.0, 0, 100) * 0.10, 0, 100)

// ============================================================================
// 16. CONFLUÊNCIA
// ============================================================================

confluenciaLong = 0.0
confluenciaLong += close > open ? 15 : 0
confluenciaLong += tendenciaLongLocal ? 20 : 0
confluenciaLong += kvoLongLocal ? 15 : 0
confluenciaLong += rsiLocal > 50 ? 10 : 0
confluenciaLong += retorno1Local > 0 ? 10 : 0
confluenciaLong += retorno3Local > 0 ? 10 : 0
confluenciaLong += scoreMTFLong > scoreMTFShort ? 10 : 0
confluenciaLong += volumeRelativoLocal >= 1 ? 10 : 0
confluenciaLong := f_limitar(confluenciaLong, 0, 100)

confluenciaShort = 0.0
confluenciaShort += close < open ? 15 : 0
confluenciaShort += tendenciaShortLocal ? 20 : 0
confluenciaShort += kvoShortLocal ? 15 : 0
confluenciaShort += rsiLocal < 50 ? 10 : 0
confluenciaShort += retorno1Local < 0 ? 10 : 0
confluenciaShort += retorno3Local < 0 ? 10 : 0
confluenciaShort += scoreMTFShort > scoreMTFLong ? 10 : 0
confluenciaShort += volumeRelativoLocal >= 1 ? 10 : 0
confluenciaShort := f_limitar(confluenciaShort, 0, 100)

// ============================================================================
// 17. MICROESTRUTURA
// ============================================================================

microMaxHigh = ta.highest(high[1], microLookback)
microMinLow = ta.lowest(low[1], microLookback)

microBreakLong = close > microMaxHigh
microBreakShort = close < microMinLow

microTendLong = close > emaFastLocal and emaFastLocal > emaMidLocal
microTendShort = close < emaFastLocal and emaFastLocal < emaMidLocal

microLongScore = f_limitar((microBreakLong ? 60.0 : 30.0) + (microTendLong ? 40.0 : 0.0), 0, 100)
microShortScore = f_limitar((microBreakShort ? 60.0 : 30.0) + (microTendShort ? 40.0 : 0.0), 0, 100)

confirmacaoMicroLong = microLongScore >= confirmacaoMicroMin
confirmacaoMicroShort = microShortScore >= confirmacaoMicroMin

// 18. ESTRUTURA

resistencia4H = request.security(syminfo.tickerid, "240", ta.highest(high[1], estruturaLookback), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)
resistenciaD = request.security(syminfo.tickerid, "1D", ta.highest(high[1], estruturaLookback), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)

suporte30 = request.security(syminfo.tickerid, "30", ta.lowest(low[1], estruturaLookback), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)
suporte1H = request.security(syminfo.tickerid, "60", ta.lowest(low[1], estruturaLookback), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)
suporte4H = request.security(syminfo.tickerid, "240", ta.lowest(low[1], estruturaLookback), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)
suporteD = request.security(syminfo.tickerid, "1D", ta.lowest(low[1], estruturaLookback), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)

resistenciaPrincipal = math.min(nz(resistencia30, high), math.min(nz(resistencia1H, high), math.min(nz(resistencia4H, high), nz(resistenciaD, high))))
suportePrincipal = math.max(nz(suporte30, low), math.max(nz(suporte1H, low), math.max(nz(suporte4H, low), nz(suporteD, low))))

rompimentoEstruturalLong = close > resistenciaPrincipal
rompimentoEstruturalShort = close < suportePrincipal

espacoLong = usarRompimento and rompimentoEstruturalLong or ((resistenciaPrincipal - close) / atrLongoSeguro >= 0.5)
espacoShort = usarRompimento and rompimentoEstruturalShort or ((close - suportePrincipal) / atrLongoSeguro >= 0.5)

// ============================================================================

regimeShortOK = not usarFiltroRegime or tendenciaShortLocal or (scoreMTFShort > scoreMTFLong)

// justificar uma direção durante o horizonte escolhido.

     scoreLongSeguro * 0.45 +
     intensidadeLong * 0.20 +
     confluenciaLong * 0.15 +
     microLongScore * 0.10 +
     scoreMTFLong * 0.10, 0, 100)

pressaoShort = f_limitar(
     scoreShortSeguro * 0.45 +
     intensidadeShort * 0.20 +
     confluenciaShort * 0.15 +
     microShortScore * 0.10 +
     scoreMTFShort * 0.10, 0, 100)

movimentoEsperadoATRLong = (pressaoLong / 100.0) * atrLocal * fatorTempo
movimentoEsperadoATRShort = (pressaoShort / 100.0) * atrLocal * fatorTempo

movimentoMinimoPermitido = atrLocal * minMovimentoATR

filtroVencimentoLong = not usarFiltroVencimento or movimentoEsperadoATRLong >= movimentoMinimoPermitido
filtroVencimentoShort = not usarFiltroVencimento or movimentoEsperadoATRShort >= movimentoMinimoPermitido

direcaoShort = scoreShortSeguro > scoreLongSeguro and vantagemShort >= vantagemMinima

filtroMTFShort = scoreMTFShort >= mtfMinShort
