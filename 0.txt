// ============================================================================
// FUNCOES BASICAS
// ============================================================================

f_clamp(float valor, float minimo, float maximo) =>
    math.max(minimo, math.min(maximo, valor))


f_normalizar(float valor, float minimo, float maximo) =>
    maximo == minimo ? 50.0 : f_clamp((valor - minimo) / (maximo - minimo) * 100.0, 0.0, 100.0)


// ============================================================================
// CLASSIFICACAO VOLATILIDADE
// ============================================================================

f_classVolatilidade(float vol) =>
    string resultado = "MUITO FRACA"

    if vol >= 80
        resultado := "EXTREMA"
    else if vol >= 65
        resultado := "MUITO FORTE"
    else if vol >= 50
        resultado := "FORTE"
    else if vol >= 35
        resultado := "MODERADA"
    else if vol >= 20
        resultado := "FRACA"

    resultado


f_corVolatilidade(float vol) =>
    color resultado = color.gray

    if vol >= 80
        resultado := color.red
    else if vol >= 65
        resultado := color.orange
    else if vol >= 50
        resultado := color.yellow
    else if vol >= 35
        resultado := color.aqua

    resultado


// ============================================================================
// CLASSIFICACAO FIBONACCI
// ============================================================================

f_classFib(float posicao) =>
    string resultado = "N/D"

    if not na(posicao)
        if posicao <= 0.236
            resultado := "0-23.6"
        else if posicao <= 0.382
            resultado := "23.6-38.2"
        else if posicao <= 0.500
            resultado := "38.2-50"
        else if posicao <= 0.618
            resultado := "50-61.8"
        else if posicao <= 0.786
            resultado := "61.8-78.6"
        else
            resultado := "78.6-100"

    resultado


f_corFib(float posicao) =>
    color resultado = color.gray

    if not na(posicao)
        if posicao <= 0.382
            resultado := color.lime
        else if posicao <= 0.618
            resultado := color.yellow
        else
            resultado := color.red

    resultado

// ============================================================================
// MODELO LOCAL
// ============================================================================

f_modeloLocal() =>

    float rangeCandle = math.max(high - low, syminfo.mintick)

    float corpo = math.abs(close - open)

    float forcaCandle = f_clamp(corpo / rangeCandle * 100.0, 0.0, 100.0)

    float ema9 = ta.ema(close, emaRapida)
    float ema21 = ta.ema(close, emaMedia)
    float ema50 = ta.ema(close, emaLenta)

    float rsi = ta.rsi(close, rsiPeriodo)

    float fluxo = volume * (2.0 * ((close - low) - (high - close)) / rangeCandle)

    float kvoRap = ta.ema(fluxo, kvoRapido)
    float kvoLen = ta.ema(fluxo, kvoLento)

    float kvo = kvoRap - kvoLen

    float kvoSignal = ta.ema(kvo, kvoSinalPeriodo)

    float retorno1 = nz(close / close[1] - 1.0, 0.0) * 100.0

    float retorno3 = nz(close / close[3] - 1.0, 0.0) * 100.0

    float tendenciaLong = 0.0
    float tendenciaShort = 0.0

    if ema9 > ema21
        tendenciaLong += 25

    if ema9 < ema21
        tendenciaShort += 25

    if ema21 > ema50
        tendenciaLong += 25

    if ema21 < ema50
        tendenciaShort += 25

    float rsiLong = 50.0
    float rsiShort = 50.0

    if rsi <= rsiSobrevendido
        rsiLong := 100.0
    else if rsi < 50
        rsiLong := f_normalizar(50.0 - rsi, 0.0, 15.0)

    if rsi >= rsiSobrecomprado
        rsiShort := 100.0
    else if rsi > 50
        rsiShort := f_normalizar(rsi - 50.0, 0.0, 15.0)

    float kvoLong = 25.0
    float kvoShort = 25.0

    if kvo > kvoSignal
        kvoLong := 100.0

    if kvo < kvoSignal
        kvoShort := 100.0

    float retornoLong = 50.0
    float retornoShort = 50.0

    if retorno1 > 0
        retornoLong += 15
    else
        retornoShort += 15

    if retorno3 > 0
        retornoLong += 15
    else
        retornoShort += 15

    float candleLong = 0.0
    float candleShort = 0.0

    if close > open
        candleLong := forcaCandle

    if close < open
        candleShort := forcaCandle

    float scoreLong = tendenciaLong * 0.30 + rsiLong * 0.15 + kvoLong * 0.20 + retornoLong * 0.15 + candleLong * 0.20

    float scoreShort = tendenciaShort * 0.30 + rsiShort * 0.15 + kvoShort * 0.20 + retornoShort * 0.15 + candleShort * 0.20

    [
         f_clamp(scoreLong, 0.0, 100.0),
         f_clamp(scoreShort, 0.0, 100.0),
         forcaCandle,
         kvo,
         kvoSignal,
         rsi,
         ema9,
         ema21,
         ema50
     ]


// ============================================================================
// SCORE MTF
// ============================================================================

f_scoreMTF() =>

    [localLong, localShort, forcaCandle, kvo, kvoSignal, rsi, ema9, ema21, ema50] = f_modeloLocal()

    [localLong, localShort]


// ============================================================================
// ATR MTF
// ============================================================================

f_mtfAtr() =>

    [localLong, localShort, forcaCandle, kvo, kvoSignal, rsi, ema9, ema21, ema50] = f_modeloLocal()

    float atrAtual = ta.atr(atrPeriodo)

    [localLong, localShort, atrAtual]


// ============================================================================
// DADOS 1M
// ============================================================================

f_dados1M() =>

    [localLong, localShort, forcaCandle, kvo, kvoSignal, rsi, ema9, ema21, ema50] = f_modeloLocal()

    float atrAtual = ta.atr(atrPeriodo)

    float atrMedio = ta.sma(atrAtual, atrMediaPeriodo)

    float volatilidadeBase = 0.0

    if atrMedio > 0
        volatilidadeBase := atrAtual / atrMedio * 50.0

    float volatilidade = f_clamp(volatilidadeBase, 0.0, 100.0)

    float fibHigh = ta.highest(high, fibLookback)

    float fibLow = ta.lowest(low, fibLookback)

    float fibRange = math.max(fibHigh - fibLow, syminfo.mintick)

    float fibPosicao = (close - fibLow) / fibRange

    float fibNivel = fibPosicao * 100.0

    [
         localLong,
         localShort,
         forcaCandle,
         kvo,
         kvoSignal,
         rsi,
         ema9,
         ema21,
         ema50,
         atrAtual,
         atrMedio,
         volatilidade,
         fibPosicao,
         fibNivel
     ]


// ============================================================================
// SCANNER
// ============================================================================

f_scanner(string simbolo) =>

    // ========================================================================
    // 1M
    // ========================================================================

    [localLong, localShort, forcaCandle, kvo, kvoSignal, rsi, ema9, ema21, ema50, atr1M, atrMedio1M, volatilidade1M, fibPosicao, fibNivel] = request.security(simbolo, "1", f_dados1M(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)


    // ========================================================================
    // 5M
    // ========================================================================

    [score5Long, score5Short] = request.security(simbolo, "5", f_scoreMTF(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)


    // ========================================================================
    // 15M
    // ========================================================================

    [score15Long, score15Short, atr15] = request.security(simbolo, "15", f_mtfAtr(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)


    // ========================================================================
    // 30M
    // ========================================================================

    [score30Long, score30Short] = request.security(simbolo, "30", f_scoreMTF(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)


    // ========================================================================
    // 1H
    // ========================================================================

    [score1HLong, score1HShort, atr1H] = request.security(simbolo, "60", f_mtfAtr(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)


    // ========================================================================
    // 4H
    // ========================================================================

    [score4HLong, score4HShort, atr4H] = request.security(simbolo, "240", f_mtfAtr(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)


    // ========================================================================
    // DIARIO
    // ========================================================================

    [scoreDLong, scoreDShort] = request.security(simbolo, "D", f_scoreMTF(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)


    // ========================================================================
    // MTF
    // ========================================================================

    float somaPesos = peso1M + peso5M + peso15M + peso30M + peso1H + peso4H + pesoD

    float mtfLong = 0.0
    float mtfShort = 0.0

    if somaPesos > 0
        mtfLong := (localLong * peso1M + score5Long * peso5M + score15Long * peso15M + score30Long * peso30M + score1HLong * peso1H + score4HLong * peso4H + scoreDLong * pesoD) / somaPesos

        mtfShort := (localShort * peso1M + score5Short * peso5M + score15Short * peso15M + score30Short * peso30M + score1HShort * peso1H + score4HShort * peso4H + scoreDShort * pesoD) / somaPesos


    // ========================================================================
    // ATR
    // ========================================================================

    float atrCombinado = atr15 * 0.35 + atr1H * 0.35 + atr4H * 0.30

    if atrAdaptativo
        atrCombinado := atrCombinado * fatorATR


    // ========================================================================
    // TENDENCIA
    // ========================================================================

    string tendencia = "LATERAL"

    if ema9 > ema21 and ema21 > ema50
        tendencia := "ALTA"
    else if ema9 < ema21 and ema21 < ema50
        tendencia := "BAIXA"


    // ========================================================================
    // FUNDO / TOPO
    // ========================================================================

    float fundo = ta.lowest(low, estruturaLookback)

    float topo = ta.highest(high, estruturaLookback)

    float distanciaFundoATR = 0.0

    float distanciaTopoATR = 0.0

    if atr1M > 0
        distanciaFundoATR := (close - fundo) / atr1M
        distanciaTopoATR := (topo - close) / atr1M

    float proximidadeFundo = f_clamp(100.0 - distanciaFundoATR / math.max(distanciaMaximaATR, 0.1) * 100.0, 0.0, 100.0)

    float proximidadeTopo = f_clamp(100.0 - distanciaTopoATR / math.max(distanciaMaximaATR, 0.1) * 100.0, 0.0, 100.0)


    // ========================================================================
    // FORCA FUNDO
    // ========================================================================

    float forcaFundo = proximidadeFundo

    if rsi <= rsiSobrevendido
        forcaFundo += 20

    if kvo > kvoSignal
        forcaFundo += 15

    if close > open
        forcaFundo += 10

    forcaFundo := f_clamp(forcaFundo, 0.0, 100.0)


    // ========================================================================
    // FORCA TOPO
    // ========================================================================

    float forcaTopo = proximidadeTopo

    if rsi >= rsiSobrecomprado
        forcaTopo += 20

    if kvo < kvoSignal
        forcaTopo += 15

    if close < open
        forcaTopo += 10

    forcaTopo := f_clamp(forcaTopo, 0.0, 100.0)


    // ========================================================================
    // MICROESTRUTURA
    // ========================================================================

    float microHigh = ta.highest(high, microLookback)

    float microLow = ta.lowest(low, microLookback)

    bool microRompimentoLong = close > nz(microHigh[1], microHigh)

    bool microRompimentoShort = close < nz(microLow[1], microLow)

    bool recuperacaoLong = close > open and close > close[1]

    bool rejeicaoShort = close < open and close < close[1]

    float microLong = 0.0

    if microRompimentoLong
        microLong += 50

    if recuperacaoLong
        microLong += 25

    if close > ema9
        microLong += 25

    microLong := f_clamp(microLong, 0.0, 100.0)

    float microShort = 0.0

    if microRompimentoShort
        microShort += 50

    if rejeicaoShort
        microShort += 25

    if close < ema9
        microShort += 25

    microShort := f_clamp(microShort, 0.0, 100.0)


    // ========================================================================
    // CONFLUENCIA LONG
    // ========================================================================

    float confluenciaLong = 0.0

    if tendencia == "ALTA"
        confluenciaLong += 25

    if kvo > kvoSignal
        confluenciaLong += 20

    if rsi < 55
        confluenciaLong += 15

    if mtfLong > mtfShort
        confluenciaLong += 20

    if microLong >= 60
        confluenciaLong += 20

    confluenciaLong := f_clamp(confluenciaLong, 0.0, 100.0)


    // ========================================================================
    // CONFLUENCIA SHORT
    // ========================================================================

    float confluenciaShort = 0.0

    if tendencia == "BAIXA"
        confluenciaShort += 25

    if kvo < kvoSignal
        confluenciaShort += 20

    if rsi > 45
        confluenciaShort += 15

    if mtfShort > mtfLong
        confluenciaShort += 20

    if microShort >= 60
        confluenciaShort += 20

    confluenciaShort := f_clamp(confluenciaShort, 0.0, 100.0)


    // ========================================================================
    // SCORE
    // ========================================================================

    float scoreLong = localLong * 0.50 + mtfLong * 0.50

    float scoreShort = localShort * 0.50 + mtfShort * 0.50

    scoreLong := f_clamp(scoreLong, 0.0, 100.0)

    scoreShort := f_clamp(scoreShort, 0.0, 100.0)


    // ========================================================================
    // VANTAGEM
    // ========================================================================

    float vantagemLong = scoreLong - scoreShort

    float vantagemShort = scoreShort - scoreLong


    // ========================================================================
    // INTENSIDADE
    // ========================================================================

    float intensidadeLong = scoreLong * 0.45 + confluenciaLong * 0.20 + microLong * 0.15 + forcaFundo * 0.20

    float intensidadeShort = scoreShort * 0.45 + confluenciaShort * 0.20 + microShort * 0.15 + forcaTopo * 0.20

    intensidadeLong := f_clamp(intensidadeLong, 0.0, 100.0)

    intensidadeShort := f_clamp(intensidadeShort, 0.0, 100.0)


    // ========================================================================
    // QUALIDADE
    // ========================================================================

    float qualidadeLong = scoreLong * 0.25 + intensidadeLong * 0.20 + confluenciaLong * 0.15 + mtfLong * 0.15 + microLong * 0.10 + forcaFundo * 0.15

    float qualidadeShort = scoreShort * 0.25 + intensidadeShort * 0.20 + confluenciaShort * 0.15 + mtfShort * 0.15 + microShort * 0.10 + forcaTopo * 0.15

    qualidadeLong := f_clamp(qualidadeLong, 0.0, 100.0)

    qualidadeShort := f_clamp(qualidadeShort, 0.0, 100.0)


    // ========================================================================
    // DISTANCIA
    // ========================================================================

    float distanciaEMA21 = 0.0

    if atr1M > 0
        distanciaEMA21 := math.abs(close - ema21) / atr1M

    float distanciaPercentual = 0.0

    if ema21 != 0
        distanciaPercentual := math.abs(close - ema21) / ema21 * 100.0

    string distanciaTexto = "NORMAL"

    if distanciaEMA21 > distanciaMaximaATR

    // ========================================================================

    else if finalShort

    else if preShort

    float intensidadePrincipal = math.max(intensidadeLong, intensidadeShort)

    // ========================================================================

         sinal,
         scorePrincipal,
         intensidadePrincipal,
         confluenciaPrincipal,
         vantagemPrincipal,
         mtfPrincipal,
         qualidadePrincipal,
         classeVolatilidade,
         volatilidade1M,

     ]
