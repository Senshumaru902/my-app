//@version=6
indicator("KVO PRO - WEBHOOK V2", overlay=true, max_labels_count=500)

// ============================================================
// CONFIGURAÇÕES
// ============================================================

grupoCandle = "CANDLE"
grupoKvo = "KVO"
grupoSinal = "WEBHOOK"
grupoVisual = "VISUAL"
grupoPainel = "PAINEL"

// ============================================================
// INPUTS
// ============================================================

corpoMinimo = input.float(0.70, "Corpo mínimo", minval=0.0, maxval=1.0, step=0.05, group=grupoCandle)

kvoFast = input.int(34, "KVO Fast", minval=1, group=grupoKvo)
kvoSlow = input.int(55, "KVO Slow", minval=1, group=grupoKvo)
kvoSignalLen = input.int(13, "KVO Signal", minval=1, group=grupoKvo)

ativarWebhook = input.bool(true, "Ativar Webhook", group=grupoSinal)
somenteFechamento = input.bool(true, "Enviar no fechamento do candle", group=grupoSinal)

mostrarSinais = input.bool(true, "Mostrar BUY/SELL", group=grupoVisual)
mostrarMedias = input.bool(true, "Mostrar MM5/MM10/MM20", group=grupoVisual)

mostrarPainel = input.bool(true, "Mostrar Painel", group=grupoPainel)

posicaoPainel = input.string(
     "top_right",
     "Posição do painel",
     options=["top_right", "top_left", "bottom_right", "bottom_left"],
     group=grupoPainel
)

tamanhoPainel = input.string(
     "Normal",
     "Tamanho",
     options=["Pequeno", "Normal", "Grande"],
     group=grupoPainel
)


// ============================================================
// DADOS BÁSICOS
// ============================================================

abertura = open
maxima = high
minima = low
fechamento = close
volumeAtual = volume

rangeCandle = maxima - minima
corpo = math.abs(fechamento - abertura)
rangeSeguro = math.max(rangeCandle, syminfo.mintick)
corpoPct = corpo / rangeSeguro

direcaoCandle = fechamento > abertura ? 1 : fechamento < abertura ? -1 : 0
candleForte = corpoPct >= corpoMinimo


// ============================================================
// PAVIOS
// ============================================================

pavioSuperior = maxima - math.max(abertura, fechamento)
pavioInferior = math.min(abertura, fechamento) - minima

pavioSuperiorPct = pavioSuperior / rangeSeguro
pavioInferiorPct = pavioInferior / rangeSeguro


// ============================================================
// RETORNOS
// ============================================================

retorno1 = not na(close[1]) ? (fechamento / close[1]) - 1.0 : 0.0
retorno3 = not na(close[3]) ? (fechamento / close[3]) - 1.0 : 0.0
retorno5 = not na(close[5]) ? (fechamento / close[5]) - 1.0 : 0.0


// ============================================================
// MÉDIAS
// ============================================================

mm5 = ta.sma(fechamento, 5)
mm10 = ta.sma(fechamento, 10)
mm20 = ta.sma(fechamento, 20)

distMM5 = not na(mm5) ? fechamento / mm5 - 1.0 : 0.0
distMM10 = not na(mm10) ? fechamento / mm10 - 1.0 : 0.0
distMM20 = not na(mm20) ? fechamento / mm20 - 1.0 : 0.0


// ============================================================
// VOLUME
// ============================================================

volumeMM20 = ta.sma(volumeAtual, 20)

volumeRelativo = not na(volumeMM20) and volumeMM20 != 0 ? volumeAtual / volumeMM20 : 1.0


// ============================================================
// ATR
// ============================================================

atr = ta.atr(14)


// ============================================================
// ESTRUTURA
// ============================================================

suporte20 = ta.lowest(low[1], 20)
resistencia20 = ta.highest(high[1], 20)

distResistencia = not na(atr) and atr != 0 ? (resistencia20 - fechamento) / atr : 0.0
distSuporte = not na(atr) and atr != 0 ? (fechamento - suporte20) / atr : 0.0


// ============================================================
// KVO
// ============================================================

precoTipico = (maxima + minima + fechamento) / 3.0

tendenciaKVO = precoTipico > precoTipico[1] ? 1 : precoTipico < precoTipico[1] ? -1 : 0

dmKVO = maxima - minima


// ============================================================
// CM KVO
// ============================================================

var float cmKVO = 0.0

if bar_index == 0
    cmKVO := dmKVO
else
    if tendenciaKVO == tendenciaKVO[1]
        cmKVO := cmKVO[1] + dmKVO
    else
        cmKVO := dmKVO + dmKVO[1]


// ============================================================
// VOLUME FORCE
// ============================================================

cmSeguro = math.max(math.abs(cmKVO), syminfo.mintick)

vfKVO = volumeAtual * math.abs(2.0 * (dmKVO / cmSeguro) - 1.0) * tendenciaKVO * 100.0


// ============================================================
// KVO
// ============================================================

kvoFastValue = ta.ema(vfKVO, kvoFast)
kvoSlowValue = ta.ema(vfKVO, kvoSlow)

kvo = kvoFastValue - kvoSlowValue
kvoSignal = ta.ema(kvo, kvoSignalLen)
kvoHist = kvo - kvoSignal
kvoVariacao = kvo - kvo[1]


// ============================================================
// KVO Z-SCORE
// ============================================================

kvoMedia = ta.sma(kvo, 34)
kvoDesvio = ta.stdev(kvo, 34)

kvoZscore = not na(kvoDesvio) and kvoDesvio != 0 ? (kvo - kvoMedia) / kvoDesvio : 0.0


// ============================================================
// FORÇA KVO
// ============================================================

kvoForcaBruta = math.abs(kvoHist) / (math.abs(kvo) + math.abs(kvoSignal) + 0.000000001)

kvoForca = math.min(math.max(kvoForcaBruta * 100.0, 0.0), 100.0)


// ============================================================
// ESTADO KVO
// ============================================================

kvoEstado = kvo > kvoSignal and kvoVariacao > 0 ? 1 : kvo < kvoSignal and kvoVariacao < 0 ? -1 : 0


// ============================================================
// FORÇA DO CANDLE
// ============================================================

forcaCorpo = math.min(math.max(corpoPct * 100.0, 0.0), 100.0)


// ============================================================
// MODELO
// ============================================================

pontuacaoModeloAlta = 0.0
pontuacaoModeloBaixa = 0.0

if direcaoCandle > 0
    pontuacaoModeloAlta += 2.0
else if direcaoCandle < 0
    pontuacaoModeloBaixa += 2.0

if retorno1 > 0
    pontuacaoModeloAlta += 1.0
else if retorno1 < 0
    pontuacaoModeloBaixa += 1.0

if retorno3 > 0
    pontuacaoModeloAlta += 1.0
else if retorno3 < 0
    pontuacaoModeloBaixa += 1.0

if fechamento > mm5
    pontuacaoModeloAlta += 1.0
else if fechamento < mm5
    pontuacaoModeloBaixa += 1.0

if fechamento > mm10
    pontuacaoModeloAlta += 1.0
else if fechamento < mm10
    pontuacaoModeloBaixa += 1.0

if fechamento > mm20
    pontuacaoModeloAlta += 1.0
else if fechamento < mm20
    pontuacaoModeloBaixa += 1.0

if kvoEstado > 0
    pontuacaoModeloAlta += 2.0
else if kvoEstado < 0
    pontuacaoModeloBaixa += 2.0


// ============================================================
// SCORE MODELO
// ============================================================

pontuacaoModeloTotal = pontuacaoModeloAlta + pontuacaoModeloBaixa

probAlta = pontuacaoModeloTotal > 0 ? pontuacaoModeloAlta / pontuacaoModeloTotal : 0.5
probBaixa = pontuacaoModeloTotal > 0 ? pontuacaoModeloBaixa / pontuacaoModeloTotal : 0.5

forcaModelo = math.max(probAlta, probBaixa) * 100.0

direcaoModelo = probAlta > probBaixa ? 1 : probBaixa > probAlta ? -1 : 0


// ============================================================
// CONFLUÊNCIA
// ============================================================

confirmacoes = 0
divergencias = 0

if direcaoCandle != 0 and direcaoCandle == direcaoModelo
    confirmacoes += 1
else
    divergencias += 1

if direcaoCandle != 0 and kvoEstado == direcaoCandle
    confirmacoes += 1
else if direcaoCandle != 0 and kvoEstado != 0
    divergencias += 1

if direcaoCandle > 0 and retorno1 > 0
    confirmacoes += 1
else if direcaoCandle < 0 and retorno1 < 0
    confirmacoes += 1
else if retorno1 != 0
    divergencias += 1

if direcaoCandle > 0 and retorno3 > 0
    confirmacoes += 1
else if direcaoCandle < 0 and retorno3 < 0
    confirmacoes += 1
else if retorno3 != 0
    divergencias += 1

if direcaoCandle > 0 and fechamento > mm5
    confirmacoes += 1
else if direcaoCandle < 0 and fechamento < mm5
    confirmacoes += 1
else
    divergencias += 1

if direcaoCandle > 0 and fechamento > mm10
    confirmacoes += 1
else if direcaoCandle < 0 and fechamento < mm10
    confirmacoes += 1
else
    divergencias += 1

if direcaoCandle > 0 and fechamento > mm20
    confirmacoes += 1
else if direcaoCandle < 0 and fechamento < mm20
    confirmacoes += 1
else
    divergencias += 1


// ============================================================
// DIREÇÃO DA INTENSIDADE
// ============================================================

pontuacaoAlta = 0.0
pontuacaoBaixa = 0.0

if direcaoCandle > 0
    pontuacaoAlta += 2.0
else if direcaoCandle < 0
    pontuacaoBaixa += 2.0

if probAlta > probBaixa
    pontuacaoAlta += probAlta * 3.0
else if probBaixa > probAlta
    pontuacaoBaixa += probBaixa * 3.0

if kvoEstado > 0
    pontuacaoAlta += 2.0
else if kvoEstado < 0
    pontuacaoBaixa += 2.0

if retorno1 > 0
    pontuacaoAlta += 1.0
else if retorno1 < 0
    pontuacaoBaixa += 1.0

if retorno3 > 0
    pontuacaoAlta += 1.0
else if retorno3 < 0
    pontuacaoBaixa += 1.0

if fechamento > mm5
    pontuacaoAlta += 1.0
else if fechamento < mm5
    pontuacaoBaixa += 1.0

if fechamento > mm10
    pontuacaoAlta += 1.0
else if fechamento < mm10
    pontuacaoBaixa += 1.0

if fechamento > mm20
    pontuacaoAlta += 1.0
else if fechamento < mm20
    pontuacaoBaixa += 1.0


// ============================================================
// DIREÇÃO
// ============================================================

direcaoIntensidade = pontuacaoAlta > pontuacaoBaixa ? 1 : pontuacaoBaixa > pontuacaoAlta ? -1 : 0


// ============================================================
// CONVICÇÃO
// ============================================================

pontuacaoTotal = pontuacaoAlta + pontuacaoBaixa

confiancaDirecional = pontuacaoTotal > 0 ? math.abs(pontuacaoAlta - pontuacaoBaixa) / pontuacaoTotal * 100.0 : 0.0


// ============================================================
// INTENSIDADE BASE
// ============================================================

intensidadeBase = forcaCorpo * 0.50 + forcaModelo * 0.50


// ============================================================
// BOOST CONFLUÊNCIA
// ============================================================

boostConfluencia = 0.0

if confirmacoes == 1
    boostConfluencia := 2.0
else if confirmacoes == 2
    boostConfluencia := 5.0
else if confirmacoes == 3
    boostConfluencia := 8.0
else if confirmacoes == 4
    boostConfluencia := 12.0
else if confirmacoes == 5
    boostConfluencia := 16.0
else if confirmacoes >= 6
    boostConfluencia := 20.0


// ============================================================
// FATOR MODELO
// ============================================================

fatorModelo = forcaModelo >= 70 ? 1.0 : forcaModelo >= 65 ? 0.90 : forcaModelo >= 60 ? 0.75 : forcaModelo >= 55 ? 0.55 : 0.35


// ============================================================
// FATOR KVO
// ============================================================

fatorKVO = kvoForca >= 60 ? 1.0 : kvoForca >= 40 ? 0.90 : kvoForca >= 20 ? 0.75 : 0.50


// ============================================================
// BOOST
// ============================================================

boost = boostConfluencia * fatorModelo * fatorKVO
boost := math.min(boost, 20.0)


// ============================================================
// BOOST VOLUME
// ============================================================

boostVolume = 0.0

if volumeRelativo >= 1.20
    boostVolume += 2.0

if volumeRelativo >= 1.50
    boostVolume += 2.0

if volumeRelativo >= 2.00
    boostVolume += 2.0

boostVolume := math.min(boostVolume, 6.0)


// ============================================================
// BOOST KVO
// ============================================================

boostKVO = 0.0

if direcaoCandle != 0 and kvoEstado == direcaoCandle
    if kvoForca >= 60
        boostKVO := 5.0
    else if kvoForca >= 40
        boostKVO := 3.0
    else if kvoForca >= 20
        boostKVO := 1.0


// ============================================================
// PENALTY
// ============================================================

penalty = divergencias * 2.0

if direcaoCandle != 0 and kvoEstado != 0 and kvoEstado != direcaoCandle
    penalty += 5.0

if direcaoCandle != 0 and direcaoModelo != direcaoCandle
    penalty += 5.0

penalty := math.min(penalty, 15.0)


// ============================================================
// PENALTY EXTRA
// ============================================================

penaltyExtra = 0.0

if confirmacoes == 0
    penaltyExtra := 8.0
else if confirmacoes == 1 and divergencias >= 4
    penaltyExtra := 7.0
else if confirmacoes == 2 and divergencias >= 4
    penaltyExtra := 5.0
else if confirmacoes == 3 and divergencias >= 4
    penaltyExtra := 3.0

penalty += penaltyExtra
penalty := math.min(penalty, 15.0)


// ============================================================
// BOOST TOTAL
// ============================================================

boostTotal = boost + boostVolume + boostKVO
boostTotal := math.min(boostTotal, 20.0)


// ============================================================
// INTENSIDADE FINAL
// ============================================================

intensidadeFinal = intensidadeBase + boostTotal - penalty

intensidadeFinal := math.max(0.0, math.min(100.0, intensidadeFinal))


// ============================================================
// INTENSIDADE ASSINADA
// ============================================================

intensidadeAssinada = direcaoIntensidade > 0 ? intensidadeFinal : direcaoIntensidade < 0 ? -intensidadeFinal : 0.0


// ============================================================
// CLASSIFICAÇÃO
// ============================================================

classificacao = intensidadeFinal < 40 ? "M FRACA" : intensidadeFinal < 60 ? "FRACA" : intensidadeFinal < 70 ? "MODERADA" : intensidadeFinal < 80 ? "FORTE" : intensidadeFinal < 90 ? "M FORTE" : "EXTREMA"


// ============================================================
// QUALIDADE CONFLUÊNCIA
// ============================================================

qualidadeConfluencia = confirmacoes >= 7 and divergencias == 0 ? "EXTREMA" : confirmacoes >= 6 and divergencias <= 1 ? "M ALTA" : confirmacoes >= 5 and divergencias <= 2 ? "ALTA" : confirmacoes >= 4 and divergencias <= 3 ? "BOA" : confirmacoes >= 3 and divergencias <= 4 ? "MODERADA" : "FRACA"


// ============================================================
// QUALIDADE BOOST
// ============================================================

qualidadeBoost = fatorModelo >= 0.90 and fatorKVO >= 0.90 ? "ALTA" : fatorModelo >= 0.75 and fatorKVO >= 0.75 ? "BOA" : fatorModelo >= 0.55 or fatorKVO >= 0.75 ? "MODERADA" : "BAIXA"


// ============================================================
// DIREÇÃO TEXTO
// ============================================================

direcaoTexto = direcaoIntensidade > 0 ? "ALTA" : direcaoIntensidade < 0 ? "BAIXA" : "NEUTRA"


// ============================================================
// VENCEDOR
// ============================================================

vencedor = probAlta > probBaixa ? "BUY" : probBaixa > probAlta ? "SELL" : "EMPATE"


// ============================================================
// WEBHOOK
// ============================================================

mensagemWebhook = "{" +
     "\"ticker\":\"" + syminfo.ticker + "\"," +
     "\"tickerid\":\"" + syminfo.tickerid + "\"," +
     "\"timeframe\":\"" + timeframe.period + "\"," +
     "\"timestamp\":\"" + str.tostring(time) + "\"," +
     "\"preco\":" + str.tostring(fechamento, "#.########") + "," +
     "\"abertura\":" + str.tostring(abertura, "#.########") + "," +
     "\"maxima\":" + str.tostring(maxima, "#.########") + "," +
     "\"minima\":" + str.tostring(minima, "#.########") + "," +
     "\"volume\":" + str.tostring(volumeAtual, "#.########") + "," +
     "\"volume_relativo\":" + str.tostring(volumeRelativo, "#.####") + "," +
     "\"corpo_pct\":" + str.tostring(corpoPct * 100.0, "#.##") + "," +
     "\"candle_forte\":" + (candleForte ? "true" : "false") + "," +
     "\"direcao_candle\":" + str.tostring(direcaoCandle) + "," +
     "\"direcao\":\"" + direcaoTexto + "\"," +
     "\"buy\":" + str.tostring(probAlta * 100.0, "#.##") + "," +
     "\"sell\":" + str.tostring(probBaixa * 100.0, "#.##") + "," +
     "\"forca_modelo\":" + str.tostring(forcaModelo, "#.##") + "," +
     "\"kvo\":" + str.tostring(kvo, "#.########") + "," +
     "\"kvo_signal\":" + str.tostring(kvoSignal, "#.########") + "," +
     "\"kvo_hist\":" + str.tostring(kvoHist, "#.########") + "," +
     "\"kvo_forca\":" + str.tostring(kvoForca, "#.##") + "," +
     "\"kvo_estado\":" + str.tostring(kvoEstado) + "," +
     "\"retorno_1\":" + str.tostring(retorno1 * 100.0, "#.####") + "," +
     "\"retorno_3\":" + str.tostring(retorno3 * 100.0, "#.####") + "," +
     "\"retorno_5\":" + str.tostring(retorno5 * 100.0, "#.####") + "," +
     "\"mm5\":" + str.tostring(mm5, "#.########") + "," +
     "\"mm10\":" + str.tostring(mm10, "#.########") + "," +
     "\"mm20\":" + str.tostring(mm20, "#.########") + "," +
     "\"confirmacoes\":" + str.tostring(confirmacoes) + "," +
     "\"divergencias\":" + str.tostring(divergencias) + "," +
     "\"qualidade_confluencia\":\"" + qualidadeConfluencia + "\"," +
     "\"intensidade_base\":" + str.tostring(intensidadeBase, "#.##") + "," +
     "\"boost\":" + str.tostring(boost, "#.##") + "," +
     "\"boost_volume\":" + str.tostring(boostVolume, "#.##") + "," +
     "\"boost_kvo\":" + str.tostring(boostKVO, "#.##") + "," +
     "\"boost_total\":" + str.tostring(boostTotal, "#.##") + "," +
     "\"penalty\":" + str.tostring(penalty, "#.##") + "," +
     "\"penalty_extra\":" + str.tostring(penaltyExtra, "#.##") + "," +
     "\"intensidade\":" + str.tostring(intensidadeFinal, "#.##") + "," +
     "\"intensidade_assinada\":" + str.tostring(intensidadeAssinada, "#.##") + "," +
     "\"classificacao\":\"" + classificacao + "\"," +
     "\"qualidade_boost\":\"" + qualidadeBoost + "\"," +
     "\"vencedor\":\"" + vencedor + "\"" +
     "}"


// ============================================================
// ALERTA
// ============================================================

condicaoAlerta = somenteFechamento ? barstate.isconfirmed : true

if ativarWebhook and condicaoAlerta
    alert(mensagemWebhook, alert.freq_once_per_bar_close)


// ============================================================
// VISUAL BUY / SELL
// ============================================================

corDirecao = direcaoIntensidade > 0 ? color.green : direcaoIntensidade < 0 ? color.red : color.gray

plotshape(
     mostrarSinais and direcaoIntensidade > 0 and candleForte,
     title="BUY",
     style=shape.triangleup,
     location=location.belowbar,
     color=color.green,
     size=size.tiny,
     text="BUY",
     textcolor=color.white
)

plotshape(
     mostrarSinais and direcaoIntensidade < 0 and candleForte,
     title="SELL",
     style=shape.triangledown,
     location=location.abovebar,
     color=color.red,
     size=size.tiny,
     text="SELL",
     textcolor=color.white
)


// ============================================================
// COLORAÇÃO DOS CANDLES
// ============================================================

barcolor(
     candleForte ?
         direcaoCandle > 0 ? color.white :
         direcaoCandle < 0 ? color.black :
         color.gray :
     na
)


// ============================================================
// MÉDIAS
// ============================================================

plot(
     mostrarMedias ? mm5 : na,
     title="MM5",
     color=color.new(color.green, 50),
     linewidth=1
)

plot(
     mostrarMedias ? mm10 : na,
     title="MM10",
     color=color.new(color.yellow, 50),
     linewidth=1
)

plot(
     mostrarMedias ? mm20 : na,
     title="MM20",
     color=color.new(color.red, 50),
     linewidth=1
)


// ============================================================
// PAINEL
// ============================================================

posicaoPainelReal =
     posicaoPainel == "top_right" ? position.top_right :
     posicaoPainel == "top_left" ? position.top_left :
     posicaoPainel == "bottom_right" ? position.bottom_right :
     position.bottom_left


tamanhoTexto =
     tamanhoPainel == "Pequeno" ? size.small :
     tamanhoPainel == "Grande" ? size.large :
     size.normal


var table painel = table.new(
     posicaoPainelReal,
     2,
     16,
     bgcolor=color.new(color.black, 10),
     frame_color=color.new(color.white, 50),
     frame_width=1,
     border_color=color.new(color.gray, 70),
     border_width=1
)


// ============================================================
// CORES PAINEL
// ============================================================

corSinal =
     vencedor == "BUY" ? color.green :
     vencedor == "SELL" ? color.red :
     color.gray


corIntensidade =
     intensidadeFinal >= 80 ? color.lime :
     intensidadeFinal >= 70 ? color.green :
     intensidadeFinal >= 60 ? color.yellow :
     intensidadeFinal >= 40 ? color.orange :
     color.red


corKVO =
     kvoEstado > 0 ? color.lime :
     kvoEstado < 0 ? color.red :
     color.gray


corConfluencia =
     confirmacoes >= 5 ? color.lime :
     confirmacoes >= 3 ? color.yellow :
     color.orange


// ============================================================
// ATUALIZAÇÃO DO PAINEL
// ============================================================

if barstate.islast

    if mostrarPainel

        table.cell(painel, 0, 0, "KVO PRO", text_color=color.white, bgcolor=color.new(color.blue, 10), text_size=tamanhoTexto)

        table.cell(painel, 1, 0, syminfo.ticker + " " + timeframe.period, text_color=color.white, bgcolor=color.new(color.blue, 10), text_size=tamanhoTexto)

        table.cell(painel, 0, 1, "SINAL", text_color=color.white, text_size=tamanhoTexto)

        table.cell(painel, 1, 1, vencedor, text_color=color.white, bgcolor=color.new(corSinal, 15), text_size=tamanhoTexto)

        table.cell(painel, 0, 2, "BUY SCORE", text_color=color.white, text_size=tamanhoTexto)

        table.cell(painel, 1, 2, str.tostring(probAlta * 100.0, "#.##") + "%", text_color=color.lime, text_size=tamanhoTexto)

        table.cell(painel, 0, 3, "SELL SCORE", text_color=color.white, text_size=tamanhoTexto)

        table.cell(painel, 1, 3, str.tostring(probBaixa * 100.0, "#.##") + "%", text_color=color.red, text_size=tamanhoTexto)

        table.cell(painel, 0, 4, "INTENSIDADE", text_color=color.white, text_size=tamanhoTexto)

        table.cell(painel, 1, 4, str.tostring(intensidadeFinal, "#.##") + "%", text_color=corIntensidade, text_size=tamanhoTexto)

        table.cell(painel, 0, 5, "CLASSIFICAÇÃO", text_color=color.white, text_size=tamanhoTexto)

        table.cell(painel, 1, 5, classificacao, text_color=corIntensidade, text_size=tamanhoTexto)

        table.cell(painel, 0, 6, "DIREÇÃO", text_color=color.white, text_size=tamanhoTexto)

        table.cell(painel, 1, 6, direcaoTexto, text_color=corDirecao, text_size=tamanhoTexto)
