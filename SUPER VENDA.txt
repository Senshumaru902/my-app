f_mtfAnalise() =>
    ema8 = ta.ema(close, 8)
    ema21 = ta.ema(close, 21)
    ema50 = ta.ema(close, 50)

    atrLocal = ta.atr(14)
    atrLocalMedia = ta.ema(atrLocal, 34)

    atrLocalRatio = atrLocalMedia > 0 ? atrLocal / atrLocalMedia : 1.0

    microBaixa = ema8 < ema21 and ema21 < ema50

    distancia = close != 0 ? (ema21 - close) / close * 100.0 : 0.0

    trendScore = microBaixa ? math.min(100.0, 55.0 + math.abs(distancia) * 10.0) : close < ema21 ? 55.0 : 30.0

    retornoLocal = not na(close[3]) ? close / close[3] - 1.0 : 0.0

    momentumScore = retornoLocal < -0.02 ? 100.0 : retornoLocal < -0.01 ? 85.0 : retornoLocal < 0 ? 65.0 : 30.0

    kvoProxy = ta.ema(volume * math.sign(ta.change(hlc3)), 13)
    kvoProxyMedia = ta.ema(kvoProxy, 21)

    kvoScore = kvoProxy < kvoProxyMedia ? 85.0 : 40.0

    breakout = close < ta.lowest(low[1], 5)

    breakoutScore = breakout ? 100.0 : close < ta.lowest(low[1], 8) + atrLocal * 0.15 ? 65.0 : 30.0

    atrScore = atrLocalRatio >= 1.0 ? 80.0 : 45.0

    score = trendScore * 0.35 + momentumScore * 0.20 + kvoScore * 0.20 + breakoutScore * 0.15 + atrScore * 0.10

    direcao = score >= 55.0 ? -1 : 1

    [direcao, score, atrLocalRatio, breakoutScore, kvoScore]

movimentoFavoravel = close < close[horizonteCalibracao]

resultadoHistorico = sinalAnterior and movimentoFavoravel

amostraHistorica = ta.sma(sinalAnterior ? 1.0 : 0.0, calibracaoPeriodo) * calibracaoPeriodo

acertosHistoricos = ta.sma(resultadoHistorico ? 1.0 : 0.0, calibracaoPeriodo) * calibracaoPeriodo

taxaHistorica = amostraHistorica >= 5.0 ? acertosHistoricos / amostraHistorica * 100.0 : probabilidadePreCalibracao

pesoHistoricoEfetivo = amostraHistorica >= 10.0 ? pesoProbHistorica : 0.0

pesoModeloEfetivo = amostraHistorica >= 10.0 ? pesoProbModelo : 1.0

somaPesos = pesoModeloEfetivo + pesoHistoricoEfetivo

probabilidadeCalibrada = somaPesos > 0 ? (probabilidadePreCalibracao * pesoModeloEfetivo + taxaHistorica * pesoHistoricoEfetivo) / somaPesos : probabilidadePreCalibracao


// ============================================================================
// PROBABILIDADE SUAVIZADA
// ============================================================================

probabilidadeSuave = ta.ema(probabilidadeCalibrada, probSuavizacao)

probabilidadeFinal = math.min(99.0, math.max(1.0, probabilidadeSuave))
