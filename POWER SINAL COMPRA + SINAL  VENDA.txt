f_limitar(valor, minimo, maximo) =>
    math.max(minimo, math.min(maximo, valor))


f_arredondarPreco(valor) =>
    syminfo.mintick > 0 ? math.round(valor / syminfo.mintick) * syminfo.mintick : valor


f_preco(valor) =>
    na(valor) ? "-" : str.tostring(valor, format.mintick)


f_pct(valor) =>
    str.tostring(f_limitar(valor, 0, 100), "#.0") + "%"


// ============================================================================
// FUNCAO MTF
// modo = 1 LONG
// modo = -1 SHORT
// modo = 0 ATR
// ============================================================================

f_mtfScore(modo) =>

    atrLocal = ta.atr(atrLength)

    emaFast = ta.ema(close, emaFastLength)
    emaMid = ta.ema(close, emaMidLength)
    emaSlow = ta.ema(close, emaSlowLength)

    rsiLocal = ta.rsi(close, rsiLength)

    volumeMedio = ta.sma(volume, volumeLength)
    volumeRelativo = volumeMedio > 0 ? volume / volumeMedio : 1.0

    rangeCandle = high - low
    corpoCandle = math.abs(close - open)

    forcaCandle = rangeCandle > 0 ? corpoCandle / rangeCandle * 100.0 : 0.0

    retorno1 = close[1] != 0 ? (close / close[1] - 1.0) * 100.0 : 0.0
    retorno3 = close[3] != 0 ? (close / close[3] - 1.0) * 100.0 : 0.0

    tendenciaAlta = emaFast > emaMid and emaMid > emaSlow
    tendenciaBaixa = emaFast < emaMid and emaMid < emaSlow

    distanciaEMA = atrLocal > 0 ? math.abs(emaFast - emaSlow) / atrLocal : 0.0
    forcaTendencia = f_limitar(distanciaEMA * 25.0, 0, 100)

    volumeScore = f_limitar(volumeRelativo * 50.0, 0, 100)

    candleLong = close > open ? forcaCandle : 0.0
    candleShort = close < open ? forcaCandle : 0.0

    trendLong = tendenciaAlta ? 100.0 : emaFast > emaMid ? 65.0 : 35.0
    trendShort = tendenciaBaixa ? 100.0 : emaFast < emaMid ? 65.0 : 35.0

    rsiLong = f_limitar(rsiLocal, 0, 100)
    rsiShort = f_limitar(100.0 - rsiLocal, 0, 100)

    retornoLong = retorno1 > 0 ? 60.0 + f_limitar(retorno1 * 20.0, 0, 40) : f_limitar(50.0 + retorno1 * 20.0, 0, 100)
    retornoShort = retorno1 < 0 ? 60.0 + f_limitar(math.abs(retorno1) * 20.0, 0, 40) : f_limitar(50.0 - retorno1 * 20.0, 0, 100)

    retorno3Long = retorno3 > 0 ? 60.0 + f_limitar(retorno3 * 10.0, 0, 40) : f_limitar(50.0 + retorno3 * 10.0, 0, 100)
    retorno3Short = retorno3 < 0 ? 60.0 + f_limitar(math.abs(retorno3) * 10.0, 0, 40) : f_limitar(50.0 - retorno3 * 10.0, 0, 100)

    // ------------------------------------------------------------------------
    // KVO
    // ------------------------------------------------------------------------

    tendenciaVolume = close > close[1] ? 1.0 : close < close[1] ? -1.0 : 0.0

    kvoVolume = volume * tendenciaVolume

    kvoFast = ta.ema(kvoVolume, 34)
    kvoSlow = ta.ema(kvoVolume, 55)

    kvoRaw = kvoFast - kvoSlow
    kvoSignal = ta.ema(kvoRaw, 13)

    kvoHist = kvoRaw - kvoSignal

    kvoLong = kvoRaw > kvoSignal ? 100.0 : kvoHist > 0 ? 70.0 : 30.0
    kvoShort = kvoRaw < kvoSignal ? 100.0 : kvoHist < 0 ? 70.0 : 30.0

    kvoLong := kvoHist > kvoHist[1] ? f_limitar(kvoLong + 10.0, 0, 100) : kvoLong
    kvoShort := kvoHist < kvoHist[1] ? f_limitar(kvoShort + 10.0, 0, 100) : kvoShort

    // ------------------------------------------------------------------------
    // SCORE LONG
    // ------------------------------------------------------------------------

    scoreLong = candleLong * 0.15 + trendLong * 0.20 + rsiLong * 0.15 + kvoLong * 0.15 + volumeScore * 0.10 + retornoLong * 0.10 + retorno3Long * 0.05 + forcaTendencia * 0.10

    // ------------------------------------------------------------------------
    // SCORE SHORT
    // ------------------------------------------------------------------------

    scoreShort = candleShort * 0.15 + trendShort * 0.20 + rsiShort * 0.15 + kvoShort * 0.15 + volumeScore * 0.10 + retornoShort * 0.10 + retorno3Short * 0.05 + forcaTendencia * 0.10

    scoreLong := f_limitar(scoreLong, 0, 100)
    scoreShort := f_limitar(scoreShort, 0, 100)

    resultado = modo == 1 ? scoreLong : modo == -1 ? scoreShort : atrLocal

    resultado

retorno3ShortLocal = retorno3Local < 0 ? 60.0 + f_limitar(math.abs(retorno3Local) * 10.0, 0, 40) : f_limitar(50.0 - retorno3Local * 10.0, 0, 100)

scoreLocalShort = candleShortLocal * 0.15 + trendShortLocal * 0.20 + rsiShort * 0.15 + kvoShortLocal * 0.15 + volumeScoreLocal * 0.10 + retornoShortLocal * 0.10 + retorno3ShortLocal * 0.05 + forcaTendenciaLocal * 0.10

intensidadeShort := f_limitar(intensidadeShort, 0, 100)

confluenciaLong := f_limitar(confluenciaLong, 0, 100)

confluenciaShort := f_limitar(confluenciaShort, 0, 100)

microScoreShort := f_limitar(microScoreShort, 0, 100)

espacoShort = suporteMaior < close ? (close - suporteMaior) >= atrLongoSeguro * espacoMinShort : true

entradaValidaShort = usarRompimento ? close <= microMinAnterior or tendenciaShort : tendenciaShort

// ============================================================================

direcaoShort = scoreShortSeguro > scoreLongSeguro

filtroMTFShort = scoreMTFShort >= mtfMinShort and mtfVencedor == -1

// ============================================================================

condicaoShort = ativarShort and scoreShortSeguro >= scoreMinShort and intensidadeShort >= intensidadeMinShort and confluenciaShort >= confluenciaMinShort and vantagemShortFinal >= vantagemMinima and direcaoShort and filtroMTFShort and espacoShort and confirmacaoMicroShort and entradaValidaShort

    direcaoAtual := -1

distanciaEntradaShort = 0.10 - fatorEntradaShort * 0.10

entradaProjetadaShort = close - atrLongoSeguro * distanciaEntradaShort

entradaShortFinal = f_arredondarPreco(entradaProjetadaShort)
