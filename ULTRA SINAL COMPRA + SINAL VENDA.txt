f_limitar(valor, minimo, maximo) =>
    math.max(
         minimo,
         math.min(
             maximo,
             valor))

f_arredondarPreco(valor) =>
    syminfo.mintick > 0 ?
         math.round(valor / syminfo.mintick) * syminfo.mintick :
         valor

f_preco(valor) =>
    na(valor) ?
         "-" :
         str.tostring(
             valor,
             format.mintick)

f_pct(valor) =>
    str.tostring(
         f_limitar(valor, 0, 100),
         "#.0") + "%"

f_mtfScore(modo) =>

    atrLocal = ta.atr(atrLength)

    emaFast = ta.ema(
         close,
         emaFastLength)

    emaMid = ta.ema(
         close,
         emaMidLength)

    emaSlow = ta.ema(
         close,
         emaSlowLength)

    rsiLocal = ta.rsi(
         close,
         rsiLength)

    volumeMedio = ta.sma(
         volume,
         volumeLength)

    volumeRelativo =
         volumeMedio > 0 ?
         volume / volumeMedio :
         1.0

    corpo =
         math.abs(
             close - open)

    rangeCandle =
         math.max(
             high - low,
             syminfo.mintick)

    forcaCandle =
         f_limitar(
             corpo / rangeCandle * 100.0,
             0,
             100)

    retorno1 =
         close[1] != 0 ?
         (close - close[1]) / close[1] * 100.0 :
         0.0

    retorno3 =
         close[3] != 0 ?
         (close - close[3]) / close[3] * 100.0 :
         0.0

    tendenciaLong =
         emaFast > emaMid and
         emaMid > emaSlow

    tendenciaShort =
         emaFast < emaMid and
         emaMid < emaSlow

    forcaTendencia =
         atrLocal > 0 ?
         math.abs(emaFast - emaSlow) / atrLocal * 100.0 :
         0.0

    forcaTendenciaLimitada =
         f_limitar(
             forcaTendencia,
             0,
             100)

    direcaoVolume =
         close > close[1] ?
         1.0 :
         close < close[1] ?
         -1.0 :
         0.0

    kvoRaw =
         ta.ema(
             volume * direcaoVolume,
             34) -
         ta.ema(
             volume * direcaoVolume,
             55)

    kvoSignal =
         ta.ema(
             kvoRaw,
             13)

    kvoLong =
         kvoRaw > kvoSignal

    kvoShort =
         kvoRaw < kvoSignal

    rsiLongScore =
         rsiLocal >= 50 ?
         f_limitar(
             (rsiLocal - 50) * 2.0,
             0,
             100) :
         0.0

    rsiShortScore =
         rsiLocal <= 50 ?
         f_limitar(
             (50 - rsiLocal) * 2.0,
             0,
             100) :
         0.0

    retornoLongScore =
         retorno1 > 0 ?
         50.0 +
         f_limitar(
             math.abs(retorno1) * 10.0,
             0,
             50) :
         0.0

    retornoShortScore =
         retorno1 < 0 ?
         50.0 +
         f_limitar(
             math.abs(retorno1) * 10.0,
             0,
             50) :
         0.0

    retorno3LongScore =
         retorno3 > 0 ?
         50.0 +
         f_limitar(
             math.abs(retorno3) * 5.0,
             0,
             50) :
         0.0

    retorno3ShortScore =
         retorno3 < 0 ?
         50.0 +
         f_limitar(
             math.abs(retorno3) * 5.0,
             0,
             50) :
         0.0

    volumeLongScore =
         volumeRelativo >= 1 ?
         f_limitar(
             volumeRelativo * 50.0,
             0,
             100) :
         0.0

    volumeShortScore =
         volumeRelativo >= 1 ?
         f_limitar(
             volumeRelativo * 50.0,
             0,
             100) :
         0.0

    tendenciaLongScore =
         tendenciaLong ?
         60.0 +
         forcaTendenciaLimitada * 0.40 :
         20.0 +
         forcaTendenciaLimitada * 0.20

    tendenciaShortScore =
         tendenciaShort ?
         60.0 +
         forcaTendenciaLimitada * 0.40 :
         20.0 +
         forcaTendenciaLimitada * 0.20

    kvoLongScore =
         kvoLong ?
         100.0 :
         0.0

    kvoShortScore =
         kvoShort ?
         100.0 :
         0.0

    candleLongScore =
         close > open ?
         forcaCandle :
         0.0

    candleShortScore =
         close < open ?
         forcaCandle :
         0.0

    scoreLong =
         candleLongScore * 0.15 +
         tendenciaLongScore * 0.20 +
         rsiLongScore * 0.15 +
         kvoLongScore * 0.15 +
         volumeLongScore * 0.10 +
         retornoLongScore * 0.10 +
         retorno3LongScore * 0.05 +
         forcaTendenciaLimitada * 0.10

    scoreShort =
         candleShortScore * 0.15 +
         tendenciaShortScore * 0.20 +
         rsiShortScore * 0.15 +
         kvoShortScore * 0.15 +
         volumeShortScore * 0.10 +
         retornoShortScore * 0.10 +
         retorno3ShortScore * 0.05 +
         forcaTendenciaLimitada * 0.10

    modo == 1 ?
         f_limitar(
             scoreLong,
             0,
             100) :
         modo == -1 ?
         f_limitar(
             scoreShort,
             0,
             100) :
         atrLocal
