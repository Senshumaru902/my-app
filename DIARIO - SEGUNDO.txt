f_clamp(float valor, float minimo, float maximo) =>
    math.max(minimo, math.min(maximo, valor))


f_nome(string simbolo) =>
    int p = str.pos(simbolo, ":")
    p >= 0 ? str.substring(simbolo, p + 1, str.length(simbolo)) : simbolo


f_volClasse(float valor) =>
    string r = "MUITO FRACA"
    if valor >= 85
        r := "EXTREMA"
    else if valor >= 70
        r := "MUITO FORTE"
    else if valor >= 55
        r := "FORTE"
    else if valor >= 40
        r := "MODERADA"
    else if valor >= 25
        r := "FRACA"
    r


f_corVol(float valor) =>
    color r = color.gray
    if valor >= 85
        r := color.red
    else if valor >= 70
        r := color.orange
    else if valor >= 55
        r := color.yellow
    else if valor >= 40
        r := color.aqua
    r


f_corSinal(string valor) =>
    color r = color.gray
    if valor == "COMPRA"
        r := color.lime
    else if valor == "VENDA"
        r := color.red
    else if valor == "PRE COMPRA"
        r := color.yellow
    else if valor == "PRE VENDA"
        r := color.orange
    r


f_corTend(string valor) =>
    color r = color.yellow
    if valor == "ALTA"
        r := color.lime
    else if valor == "BAIXA"
        r := color.red
    r


f_pernaClasse(float valor) =>
    string r = "NORMAL"
    if valor >= 85
        r := "EXTREMA"
    else if valor >= 70
        r := "MUITO ESTICADA"
    else if valor >= 55
        r := "ESTICADA"
    else if valor >= 35
        r := "ESTICANDO"
    r


f_corPerna(string valor) =>
    color r = color.green
    if valor == "EXTREMA"
        r := color.red
    else if valor == "MUITO ESTICADA"
        r := color.orange
    else if valor == "ESTICADA"
        r := color.yellow
    else if valor == "ESTICANDO"
        r := color.aqua
    r


f_fibClasse(float valor) =>
    string r = "0-23.6"
    if valor >= 78.6
        r := "78.6-100"
    else if valor >= 61.8
        r := "61.8-78.6"
    else if valor >= 50
        r := "50-61.8"
    else if valor >= 38.2
        r := "38.2-50"
    else if valor >= 23.6
        r := "23.6-38.2"
    r


f_corFib(float valor) =>
    color r = color.orange
    if valor >= 38.2 and valor <= 61.8
        r := color.lime
    else if valor >= 23.6 and valor <= 78.6
        r := color.yellow
    r


f_corStatus(string valor) =>

        r := color.lime

        r := color.yellow
    else if valor == "CAUTELA"

    else if valor == "OBSERVAR"
        r := color.aqua
    r

f_diario() =>

    // ------------------------------------------------------------------------
    // RANGE / CANDLE
    // ------------------------------------------------------------------------

    float rangeCandle = math.max(high - low, syminfo.mintick)
    float corpo = math.abs(close - open)

    float forcaCandle = f_clamp(corpo / rangeCandle * 100.0, 0.0, 100.0)

    float candleLong = close > open ? forcaCandle : forcaCandle * 0.30
    float candleShort = close < open ? forcaCandle : forcaCandle * 0.30


    // ------------------------------------------------------------------------
    // EMAS
    // ------------------------------------------------------------------------

    float ema9 = ta.ema(close, emaRapida)
    float ema21 = ta.ema(close, emaMedia)
    float ema50 = ta.ema(close, emaLenta)


    // ------------------------------------------------------------------------
    // RSI
    // ------------------------------------------------------------------------

    float rsi = ta.rsi(close, rsiPeriodo)


    // ------------------------------------------------------------------------
    // KVO
    // ------------------------------------------------------------------------

    float fluxo = volume * (2.0 * ((close - low) - (high - close)) / rangeCandle)

    float kvoF = ta.ema(fluxo, kvoFast)
    float kvoS = ta.ema(fluxo, kvoSlow)

    float kvo = kvoF - kvoS
    float kvoSignal = ta.ema(kvo, kvoSignalPeriodo)


    // ------------------------------------------------------------------------
    // RETORNOS
    // ------------------------------------------------------------------------

    float retorno1 = close[1] != 0 ? (close - close[1]) / close[1] * 100.0 : 0.0
    float retorno3 = close[3] != 0 ? (close - close[3]) / close[3] * 100.0 : 0.0


    // ------------------------------------------------------------------------
    // TENDENCIA
    // ------------------------------------------------------------------------

    float tendenciaLong = 25.0

    if ema9 > ema21 and ema21 > ema50
        tendenciaLong := 100.0
    else if ema9 > ema21
        tendenciaLong := 75.0
    else if ema9 > ema50
        tendenciaLong := 60.0


    float tendenciaShort = 25.0

    if ema9 < ema21 and ema21 < ema50
        tendenciaShort := 100.0
    else if ema9 < ema21
        tendenciaShort := 75.0
    else if ema9 < ema50
        tendenciaShort := 60.0


    string tendencia = "LATERAL"

    if ema9 > ema21 and ema21 > ema50
        tendencia := "ALTA"
    else if ema9 < ema21 and ema21 < ema50
        tendencia := "BAIXA"


    // ------------------------------------------------------------------------
    // RSI SCORE
    // ------------------------------------------------------------------------

    float rsiLong = 25.0

    if rsi <= rsiSobrevendido
        rsiLong := 100.0
    else if rsi < 45
        rsiLong := 85.0
    else if rsi < 50
        rsiLong := 70.0
    else if rsi < 55
        rsiLong := 55.0


    float rsiShort = 25.0

    if rsi >= rsiSobrecomprado
        rsiShort := 100.0
    else if rsi > 55
        rsiShort := 85.0
    else if rsi > 50
        rsiShort := 70.0
    else if rsi > 45
        rsiShort := 55.0


    // ------------------------------------------------------------------------
    // KVO SCORE
    // ------------------------------------------------------------------------

    float kvoLong = 30.0

    if kvo > kvoSignal and kvo > 0
        kvoLong := 100.0
    else if kvo > kvoSignal
        kvoLong := 80.0
    else if kvo > 0
        kvoLong := 65.0


    float kvoShort = 30.0

    if kvo < kvoSignal and kvo < 0
        kvoShort := 100.0
    else if kvo < kvoSignal
        kvoShort := 80.0
    else if kvo < 0
        kvoShort := 65.0


    // ------------------------------------------------------------------------
    // RETORNO SCORE
    // ------------------------------------------------------------------------

    float retornoLong = 25.0

    if retorno1 > 0 and retorno3 > 0
        retornoLong := 100.0
    else if retorno1 > 0 or retorno3 > 0
        retornoLong := 70.0


    float retornoShort = 25.0

    if retorno1 < 0 and retorno3 < 0
        retornoShort := 100.0
    else if retorno1 < 0 or retorno3 < 0
        retornoShort := 70.0


    // ------------------------------------------------------------------------
    // SCORE BASE
    // ------------------------------------------------------------------------

    float scoreBaseLong = tendenciaLong * 0.30 + rsiLong * 0.15 + kvoLong * 0.20 + retornoLong * 0.15 + candleLong * 0.20

    float scoreBaseShort = tendenciaShort * 0.30 + rsiShort * 0.15 + kvoShort * 0.20 + retornoShort * 0.15 + candleShort * 0.20


    // ------------------------------------------------------------------------
    // ATR
    // ------------------------------------------------------------------------

    float atr = ta.atr(atrPeriodo)
    float atrMedia = ta.sma(atr, atrMediaPeriodo)

    float atrUsado = atr * fatorATR

    // ------------------------------------------------------------------------
    // VOLATILIDADE
    // ------------------------------------------------------------------------

    float volatilidade = atrMedia > 0 ? atr / atrMedia * 50.0 : 50.0

    volatilidade := f_clamp(volatilidade, 0.0, 100.0)


    // ------------------------------------------------------------------------
    // ESTRUTURA
    // ------------------------------------------------------------------------

    float fundo = ta.lowest(low, estruturaLookback)
    float topo = ta.highest(high, estruturaLookback)


    // ------------------------------------------------------------------------
    // DISTANCIA EMA21
    // ------------------------------------------------------------------------

    float distanciaATR = atr > 0 ? math.abs(close - ema21) / atr : 99.0

    string distanciaTexto = "NORMAL"

    if distanciaATR <= 0.50
        distanciaTexto := "PROXIMA"
    else if distanciaATR <= 1.00
        distanciaTexto := "NORMAL"
    else if distanciaATR <= distanciaMaximaATR
        distanciaTexto := "AFASTADA"
    else
        distanciaTexto := "LONGE"


    bool distanciaOK = distanciaATR <= distanciaMaximaATR


    // ------------------------------------------------------------------------
    // FUNDO / TOPO
    // ------------------------------------------------------------------------

    float distanciaFundoATR = atr > 0 ? math.abs(close - fundo) / atr : 99.0
    float distanciaTopoATR = atr > 0 ? math.abs(close - topo) / atr : 99.0

    float proximidadeFundo = f_clamp(100.0 - distanciaFundoATR / distanciaMaximaATR * 100.0, 0.0, 100.0)
    float proximidadeTopo = f_clamp(100.0 - distanciaTopoATR / distanciaMaximaATR * 100.0, 0.0, 100.0)


    // ------------------------------------------------------------------------
    // MICROESTRUTURA
    // ------------------------------------------------------------------------

    float microHigh = ta.highest(high, microLookback)
    float microLow = ta.lowest(low, microLookback)

    bool microBreakLong = close > microHigh[1]
    bool microBreakShort = close < microLow[1]

    bool recuperacaoLong = low < low[1] and close > open and close > close[1]
    bool rejeicaoShort = high > high[1] and close < open and close < close[1]


    float microLong = 35.0

    if microBreakLong
        microLong += 35.0

    if recuperacaoLong
        microLong += 20.0

    if close > ema9
        microLong += 10.0

    microLong := f_clamp(microLong, 0.0, 100.0)


    float microShort = 35.0

    if microBreakShort
        microShort += 35.0

    if rejeicaoShort
        microShort += 20.0

    if close < ema9
        microShort += 10.0

    microShort := f_clamp(microShort, 0.0, 100.0)


    // ------------------------------------------------------------------------
    // FUNDO / TOPO
    // ------------------------------------------------------------------------

    float forcaFundo = proximidadeFundo

    if rsi <= rsiSobrevendido
        forcaFundo += 20.0
    else if rsi < 45
        forcaFundo += 10.0

    if kvo > kvoSignal
        forcaFundo += 15.0

    if close > ema9
        forcaFundo += 10.0

    if recuperacaoLong
        forcaFundo += 15.0

    forcaFundo := f_clamp(forcaFundo, 0.0, 100.0)


    float forcaTopo = proximidadeTopo

    if rsi >= rsiSobrecomprado
        forcaTopo += 20.0
    else if rsi > 55
        forcaTopo += 10.0

    if kvo < kvoSignal
        forcaTopo += 15.0

    if close < ema9
        forcaTopo += 10.0

    if rejeicaoShort
        forcaTopo += 15.0

    forcaTopo := f_clamp(forcaTopo, 0.0, 100.0)


    // ------------------------------------------------------------------------
    // CONFLUENCIA
    // ------------------------------------------------------------------------

    float confluenciaLong = 0.0

    if tendencia == "ALTA"
        confluenciaLong += 25.0

    if kvo > kvoSignal
        confluenciaLong += 20.0

    if rsi < 55
        confluenciaLong += 15.0

    if microLong >= 60
        confluenciaLong += 20.0

    if forcaFundo >= 60
        confluenciaLong += 20.0

    confluenciaLong := f_clamp(confluenciaLong, 0.0, 100.0)


    float confluenciaShort = 0.0

    if tendencia == "BAIXA"
        confluenciaShort += 25.0

    if kvo < kvoSignal
        confluenciaShort += 20.0

    if rsi > 45
        confluenciaShort += 15.0

    if microShort >= 60
        confluenciaShort += 20.0

    if forcaTopo >= 60
        confluenciaShort += 20.0

    confluenciaShort := f_clamp(confluenciaShort, 0.0, 100.0)


    // ------------------------------------------------------------------------
    // SCORE FINAL
    // ------------------------------------------------------------------------

    float scoreLong = scoreBaseLong * 0.55 + forcaFundo * 0.20 + confluenciaLong * 0.15 + microLong * 0.10
    float scoreShort = scoreBaseShort * 0.55 + forcaTopo * 0.20 + confluenciaShort * 0.15 + microShort * 0.10

    scoreLong := f_clamp(scoreLong, 0.0, 100.0)
    scoreShort := f_clamp(scoreShort, 0.0, 100.0)


    // ------------------------------------------------------------------------
    // VANTAGEM
    // ------------------------------------------------------------------------

    float vantagemLong = f_clamp(scoreLong - scoreShort, 0.0, 100.0)
    float vantagemShort = f_clamp(scoreShort - scoreLong, 0.0, 100.0)


    // ------------------------------------------------------------------------
    // INTENSIDADE
    // ------------------------------------------------------------------------

    float intensidadeLong = scoreLong * 0.40 + confluenciaLong * 0.20 + microLong * 0.15 + forcaFundo * 0.25
    float intensidadeShort = scoreShort * 0.40 + confluenciaShort * 0.20 + microShort * 0.15 + forcaTopo * 0.25

    intensidadeLong := f_clamp(intensidadeLong, 0.0, 100.0)
    intensidadeShort := f_clamp(intensidadeShort, 0.0, 100.0)


    // ------------------------------------------------------------------------
    // QUALIDADE
    // ------------------------------------------------------------------------

    float qualidadeLong = scoreLong * 0.30 + intensidadeLong * 0.20 + confluenciaLong * 0.20 + microLong * 0.15 + forcaFundo * 0.15

    float qualidadeShort = scoreShort * 0.30 + intensidadeShort * 0.20 + confluenciaShort * 0.20 + microShort * 0.15 + forcaTopo * 0.15

    qualidadeLong := f_clamp(qualidadeLong, 0.0, 100.0)
    qualidadeShort := f_clamp(qualidadeShort, 0.0, 100.0)


    // ------------------------------------------------------------------------
    // FIBONACCI
    // ------------------------------------------------------------------------

    float fibHigh = ta.highest(high, fibLookback)
    float fibLow = ta.lowest(low, fibLookback)

    float fibAmplitude = math.max(fibHigh - fibLow, syminfo.mintick)

    float fibPos = (close - fibLow) / fibAmplitude * 100.0

    fibPos := f_clamp(fibPos, 0.0, 100.0)

    string fibClasse = f_fibClasse(fibPos)

    bool fibLongOK = true
    bool fibShortOK = true

    if usarFibonacci
        fibLongOK := fibPos <= 65.3
        fibShortOK := fibPos >= 34.7


    // ------------------------------------------------------------------------
    // PERNA
    // ------------------------------------------------------------------------

    float extensaoLongATR = atr > 0 ? (close - ta.lowest(low, pernaLookback)) / atr : 0.0
    float extensaoShortATR = atr > 0 ? (ta.highest(high, pernaLookback) - close) / atr : 0.0


    float pernaLongExt = f_clamp((extensaoLongATR - pernaEsticandoATR) / math.max(pernaExtremaATR - pernaEsticandoATR, 0.1) * 100.0, 0.0, 100.0)

    float pernaShortExt = f_clamp((extensaoShortATR - pernaEsticandoATR) / math.max(pernaExtremaATR - pernaEsticandoATR, 0.1) * 100.0, 0.0, 100.0)


    // ------------------------------------------------------------------------
    // CONTAGEM DE CANDLES
    // ------------------------------------------------------------------------

    int barrasAlta = 0
    int barrasBaixa = 0

    for i = 0 to pernaCandlesMax - 1
        if close[i] > open[i]
            barrasAlta += 1
        if close[i] < open[i]
            barrasBaixa += 1


    float candlesLongScore = f_clamp(barrasAlta / pernaCandlesMax * 100.0, 0.0, 100.0)
    float candlesShortScore = f_clamp(barrasBaixa / pernaCandlesMax * 100.0, 0.0, 100.0)


    // ------------------------------------------------------------------------
    // EXCESSO RSI
    // ------------------------------------------------------------------------

    float excessoRSILong = rsi >= 70 ? 100.0 : rsi >= 65 ? 70.0 : 20.0
    float excessoRSIShort = rsi <= 30 ? 100.0 : rsi <= 35 ? 70.0 : 20.0


    // ------------------------------------------------------------------------
    // PERDA KVO
    // ------------------------------------------------------------------------

    float perdaKVOLong = kvo < kvoSignal ? 70.0 : kvo < 0 ? 45.0 : 10.0
    float perdaKVOShort = kvo > kvoSignal ? 70.0 : kvo > 0 ? 45.0 : 10.0


    // ------------------------------------------------------------------------
    // DISTANCIA COMO RISCO DE PERNA
    // ------------------------------------------------------------------------

    float distanciaScore = f_clamp(distanciaATR / distanciaMaximaATR * 100.0, 0.0, 100.0)


    // ------------------------------------------------------------------------
    // SCORE DA PERNA
    // ------------------------------------------------------------------------

    float pernaScoreLong = pernaLongExt * 0.35 + distanciaScore * 0.15 + candlesLongScore * 0.15 + perdaKVOLong * 0.15 + excessoRSILong * 0.20

    float pernaScoreShort = pernaShortExt * 0.35 + distanciaScore * 0.15 + candlesShortScore * 0.15 + perdaKVOShort * 0.15 + excessoRSIShort * 0.20


    if extensaoLongATR >= pernaExtremaATR
        pernaScoreLong += 10.0
    else if extensaoLongATR >= pernaMuitoEsticadaATR
        pernaScoreLong += 7.0
    else if extensaoLongATR >= pernaEsticadaATR
        pernaScoreLong += 3.0


    if extensaoShortATR >= pernaExtremaATR
        pernaScoreShort += 10.0
    else if extensaoShortATR >= pernaMuitoEsticadaATR
        pernaScoreShort += 7.0
    else if extensaoShortATR >= pernaEsticadaATR
        pernaScoreShort += 3.0


    pernaScoreLong := f_clamp(pernaScoreLong, 0.0, 100.0)
    pernaScoreShort := f_clamp(pernaScoreShort, 0.0, 100.0)


    // ------------------------------------------------------------------------
    // EXAUSTAO
    // ------------------------------------------------------------------------

    bool exaustaoLong = false
    bool exaustaoShort = false


    if pernaScoreLong >= pernaBloqueio
        if extensaoLongATR >= pernaMuitoEsticadaATR and excessoRSILong >= 70
            exaustaoLong := true
        else if extensaoLongATR >= pernaExtremaATR
            exaustaoLong := true


    if pernaScoreShort >= pernaBloqueio
        if extensaoShortATR >= pernaMuitoEsticadaATR and excessoRSIShort >= 70
            exaustaoShort := true
        else if extensaoShortATR >= pernaExtremaATR
            exaustaoShort := true


    bool pernaLongOK = not usarFiltroPerna or not exaustaoLong
    bool pernaShortOK = not usarFiltroPerna or not exaustaoShort


    // ------------------------------------------------------------------------
    // VOLATILIDADE
    // ------------------------------------------------------------------------

    string volClasse = f_volClasse(volatilidade)
