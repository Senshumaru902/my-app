
instrument {
    name = "QMS SUPREMO PRO PADROES V1",
    short_name = "QMS SUPREMO V1",
    overlay = true
}

-- ==========================================
-- CONFIGURACOES
-- ==========================================

periodo_estrutura = input(2, "Sensibilidade dos pivos",
input.integer, 1, 5)

tolerancia = input(25, "Tolerancia dos ombros (%)",
input.integer, 5, 100)

-- ==========================================
-- 1. OMBRO CABECA OMBRO
-- REVERSAO DE ALTA PARA BAIXA
-- ==========================================

o1 = high[12]
cabeca = high[7]
o2 = high[2]

pivo_o1 =
    high[12] > high[13] and
    high[12] > high[11]

pivo_c =
    high[7] > high[8] and
    high[7] > high[6]

pivo_o2 =
    high[2] > high[3] and
    high[2] > high[1]

simetria_oco =
    abs(o1-o2) /
    math.max(abs(o1), 0.000001) <
    tolerancia / 1000

cabeca_maior =
    cabeca > o1 and cabeca > o2

pescoco_oco =
    (low[9] + low[4]) / 2

oco =
    pivo_o1 and pivo_c and pivo_o2 and
    simetria_oco and cabeca_maior

oco_antecipado =
    oco and
    close < pescoco_oco and
    close >= pescoco_oco * 0.995

oco_confirmado =
    oco and
    close < pescoco_oco and
    close[1] >= pescoco_oco

-- ==========================================
-- 2. OCO INVERSO
-- REVERSAO DE BAIXA PARA ALTA
-- ==========================================

oi1 = low[12]
cabeca_i = low[7]
oi2 = low[2]

pivo_i1 =
    low[12] < low[13] and
    low[12] < low[11]

pivo_ci =
    low[7] < low[8] and
    low[7] < low[6]

pivo_i2 =
    low[2] < low[3] and
    low[2] < low[1]

simetria_inv =
    abs(oi1-oi2) /
    math.max(abs(oi1), 0.000001) <
    tolerancia / 1000

cabeca_menor =
    cabeca_i < oi1 and cabeca_i < oi2

pescoco_inv =
    (high[9] + high[4]) / 2

oco_inv =
    pivo_i1 and pivo_ci and pivo_i2 and
    simetria_inv and cabeca_menor

oco_inv_antecipado =
    oco_inv and
    close > pescoco_inv * 0.995 and
    close <= pescoco_inv

oco_inv_confirmado =
    oco_inv and
    close > pescoco_inv and
    close[1] <= pescoco_inv

-- ==========================================
-- 3. TOPO DUPLO M
-- ==========================================

topo1 = high[10]
topo2 = high[4]

m_semelhante =
    abs(topo1-topo2) /
    math.max(abs(topo1), 0.000001) < 0.025

pivo_m1 =
    high[10] > high[11] and
    high[10] > high[9]

pivo_m2 =
    high[4] > high[5] and
    high[4] > high[3]

fundo_m =
    low[7]

padrao_m =
    m_semelhante and pivo_m1 and pivo_m2

venda_m =
    padrao_m and
    close < fundo_m and
    close[1] >= fundo_m

-- ==========================================
-- 4. FUNDO DUPLO W
-- ==========================================

fundo1 = low[10]
fundo2 = low[4]

w_semelhante =
    abs(fundo1-fundo2) /
    math.max(abs(fundo1), 0.000001) < 0.025

pivo_w1 =
    low[10] < low[11] and
    low[10] < low[9]

pivo_w2 =
    low[4] < low[5] and
    low[4] < low[3]

topo_w =
    high[7]

padrao_w =
    w_semelhante and pivo_w1 and pivo_w2

compra_w =
    padrao_w and
    close > topo_w and
    close[1] <= topo_w

-- ==========================================
-- 5. BANDEIRA DE ALTA
-- ==========================================

max_bandeira =
    math.max(high[1],
    math.max(high[2],
    math.max(high[3], high[4])))

min_bandeira =
    math.min(low[1],
    math.min(low[2],
    math.min(low[3], low[4])))

impulso_alta =
    close[6] > close[11] and
    close[6] > open[6]

consolidacao_alta =
    max_bandeira - min_bandeira <
    abs(close[6]-close[11]) * 0.75

bandeira_alta =
    impulso_alta and
    consolidacao_alta

compra_bandeira =
    bandeira_alta and
    close > max_bandeira and
    close[1] <= max_bandeira

-- ==========================================
-- 6. BANDEIRA DE BAIXA
-- ==========================================

impulso_baixa =
    close[6] < close[11] and
    close[6] < open[6]

consolidacao_baixa =
    max_bandeira - min_bandeira <
    abs(close[6]-close[11]) * 0.75

bandeira_baixa =
    impulso_baixa and
    consolidacao_baixa

venda_bandeira =
    bandeira_baixa and
    close < min_bandeira and
    close[1] >= min_bandeira

-- ==========================================
-- 7. ROMPIMENTO E RETESTE
-- ==========================================

resistencia =
    math.max(high[2],
    math.max(high[3],
    math.max(high[4], high[5])))

suporte =
    math.min(low[2],
    math.min(low[3],
    math.min(low[4], low[5])))

rompimento_alta =
    close[1] > resistencia[1] and
    low <= resistencia and
    close > resistencia and
    close > open

rompimento_baixa =
    close[1] < suporte[1] and
    high >= suporte and
    close < suporte and
    close < open

-- ==========================================
-- 8. PULLBACK DE CONTINUACAO
-- ==========================================

pernada_alta =
    close[5] > close[9] and
    close[5] > open[5]

pullback_alta =
    pernada_alta and
    low[2] < close[5] and
    close > open and
    close > close[1]

pernada_baixa =
    close[5] < close[9] and
    close[5] < open[5]

pullback_baixa =
    pernada_baixa and
    high[2] > close[5] and
    close < open and
    close < close[1]

-- ==========================================
-- 9. SINAIS ANTECIPADOS
-- ==========================================

plot_shape(
    oco_antecipado,
    "OCO POSSIVEL",
    shape_style.circle,
    shape_size.small,
    "#FFAA00",
    shape_location.abovebar,
    0,
    "PRE OCO"
)

plot_shape(
    oco_inv_antecipado,
    "OCO INV POSSIVEL",
    shape_style.circle,
    shape_size.small,
    "#00FFFF",
    shape_location.belowbar,
    0,
    "PRE OCO INV"
)

-- ==========================================
-- 10. REVERSOES CONFIRMADAS
-- ==========================================

plot_shape(
    oco_confirmado,
    "OCO VENDA",
    shape_style.arrowdown,
    shape_size.large,
    "#FF3030",
    shape_location.abovebar,
    0,
    "OCO"
)

plot_shape(
    oco_inv_confirmado,
    "OCO INV COMPRA",
    shape_style.arrowup,
    shape_size.large,
    "#00FF66",
    shape_location.belowbar,
    0,
    "OCO INV"
)

-- ==========================================
-- 11. CONTINUACAO CONFIRMADA
-- ==========================================

plot_shape(
    compra_bandeira,
    "BANDEIRA ALTA",
    shape_style.arrowup,
    shape_size.large,
    "#2196FF",
    shape_location.belowbar,
    0,
    "BANDEIRA ALTA"
)

plot_shape(
    venda_bandeira,
    "BANDEIRA BAIXA",
    shape_style.arrowdown,
    shape_size.large,
    "#A020F0",
    shape_location.abovebar,
    0,
    "BANDEIRA BAIXA"
)
