f_clamp(float x, float mn, float mx) =>

    x < mn ? mn : x > mx ? mx : x

f_abs(float x) =>

    x < 0.0 ? -x : x

f_max(float a, float b) =>

    a > b ? a : b

f_min(float a, float b) =>

    a < b ? a : b

f_div(float a, float b) =>

    b == 0.0 ? 0.0 : a / b

bool estruturaBaixa = close < ema21 and ema21 < ema50

if bar_index == 0

if not na(pl1) and not na(pl2) and plBar1 != plBar2

    slopeLow := f_div(pl1 - pl2, float(plBar1 - plBar2))

bool suporteHorizontal = not na(pl1) and not na(pl2) and f_abs(pl1 - pl2) <= toleranciaGeo

bool trianguloDescendente = usarFiguras and suporteHorizontal and lh and slopeHigh < 0.0

bool  trianguloSimetrico = usarFiguras and slopeHigh < 0.0 and slopeLow  > 0.0 and not resistenciaHorizontal and not suporteHorizontal

bool cunhaAscendente = usarFiguras and hh and hl and slopeHigh > 0.0 and slopeLow > 0.0 and slopeLowAbs < slopeHighAbs

bool cunhaDescendente = usarFiguras and lh and ll and slopeHigh < 0.0 and slopeLow < 0.0 and slopeHighAbs < slopeLowAbs

bool  canalAlta = usarFiguras and hh and hl and slopeHigh > 0.0 and  slopeLow > 0.0 and f_abs(slopeHigh - slopeLow) <=  f_max(slopeHighAbs, slopeLowAbs) * 0.60

bool  canalBaixa = usarFiguras and lh and ll and slopeHigh < 0.0 and  slopeLow < 0.0 and f_abs(slopeHigh - slopeLow) <=  f_max(slopeHighAbs, slopeLowAbs) * 0.60

bool topoDuplo = usarFiguras and not na(ph1) and not na(ph2) and f_abs(ph1 - ph2) <= toleranciaGeo

bool fundoDuplo = usarFiguras and not na(pl1) and not na(pl2) and f_abs(pl1 - pl2) <= toleranciaGeo

float faixaEstrutural = faixaAltaAnterior - faixaBaixaAnterior

bool consolidacao = usarFiguras and faixaEstrutural <= atrSafe * 3.0

bool bandeiraAlta = usarFiguras and tendenciaAlta and slopeLow < 0.0

bool bandeiraBaixa = usarFiguras and tendenciaBaixa and slopeHigh > 0.0

bool figuraAlta = trianguloAscendente or cunhaDescendente or canalAlta or fundoDuplo or bandeiraAlta

bool figuraBaixa = trianguloDescendente or cunhaAscendente or canalBaixa or topoDuplo or bandeiraBaixa

else if cunhaAscendente

else if cunhaDescendente

    figuraAtual := "CUNHA DESC."

else if canalAlta

    figuraAtual := "CANAL ALTA"

    figuraAtual := "CANAL BAIXA"

else if consolidacao

float forcaBaixa = f_div(nivelSuporte - close, atrSafe)

bool rompimentoAlta = rompeAlta and forcaAlta >= forcaBreakFinal

bool rompimentoBaixa = rompeBaixa and forcaBaixa >= forcaBreakFinal

    estruturaVendaOK := figuraBaixa or figuraNeutra

bool candleVendaOK = not exigirCandle or bearEngulf or shootingStar or bearPin or bearHarami or (candleBaixa and candleForte)
