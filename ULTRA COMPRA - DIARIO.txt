f_suavizar(src, periodo) =>
    e1 = ta.ema(src, periodo)
    e2 = ta.ema(e1, periodo)
    e3 = ta.ema(e2, periodo)
    (e1 + e2 + e3) / 3.0


f_clamp(v, minimo, maximo) =>
    math.min(math.max(v, minimo), maximo)

// ============================================================================
// DADOS DO CANDLE DIÁRIO
// ============================================================================

rangeCandle = math.max(high - low, syminfo.mintick)

corpo = math.abs(close - open)

corpoPct = corpo / rangeCandle

pavioSuperior = high - math.max(open, close)

pavioInferior = math.min(open, close) - low

direcaoCandle =
     close > open ? 1 :
     close < open ? -1 :
     0

candleForte = corpoPct >= corpoMinimo

candleAltaForte =
     candleForte and direcaoCandle > 0

forcaCorpo =
     f_clamp(corpoPct * 100.0, 0.0, 100.0)

// RETORNOS
// ============================================================================

retorno1 =
     close[1] != 0 ?
     (close / close[1] - 1.0) * 100.0 :
     0.0

retorno3 =
     close[3] != 0 ?
     (close / close[3] - 1.0) * 100.0 :
     0.0

retorno5 =
     close[5] != 0 ?
     (close / close[5] - 1.0) * 100.0 :
     0.0

retorno10 =
     close[10] != 0 ?
     (close / close[10] - 1.0) * 100.0 :
     0.0

// ============================================================================
// ATR DIÁRIO

atrBruto = ta.atr(atrPeriodo)

     atrBruto,
     atrSuavePeriodo)

atrMediaRegime = ta.sma(
     atrSuave,
     atrRegimePeriodo)

atrSeguro =
     math.max(
     nz(atrSuave, atrBruto),
     syminfo.mintick)

atrRegime =
     not na(atrMediaRegime) and atrMediaRegime > 0 ?
     atrSuave / atrMediaRegime :
     1.0

atrExpansaoBase =
     atrSuave[1] != 0 ?
     atrSuave / atrSuave[1] :
     1.0

atrExpansao =
     f_suavizar(
     atrExpansaoBase,
     suavizacaoPeriodo)

atrNormalizado =
     close != 0 ?
     atrSeguro / close * 100.0 :
     0.0

     atrRegime >= 1.35 and
     atrExpansao >= 1.03

atrForte =
     atrRegime >= 1.10 and
     atrExpansao >= 1.00

atrFraco =
     atrRegime <= 0.80

atrBoaCompra =
     atrForte or
     (atrRegime >= 0.90 and atrExpansao >= 1.00)

atrScore =
     atrMuitoForte ? 100.0 :
     atrForte ? 82.0 :
     atrBoaCompra ? 68.0 :
     atrFraco ? 38.0 :
     55.0

     atrMuitoForte ? "MUITO FORTE" :
     atrForte ? "FORTE" :
     atrBoaCompra ? "BOA" :
     atrFraco ? "FRACA" :
     "NORMAL"

// ============================================================================
// KVO DIÁRIO

precoTipico =
     (high + low + close) / 3.0

tendenciaKvo =
     precoTipico > precoTipico[1] ? 1 :
     precoTipico < precoTipico[1] ? -1 :
     0

dmKvo =
     math.max(
     high - low,
     syminfo.mintick)

var float cmKvo = 0.0

if bar_index == 0
    cmKvo := dmKvo
else
    cmKvo :=
         tendenciaKvo == tendenciaKvo[1] ?
         cmKvo[1] + dmKvo :
         dmKvo + dmKvo[1]

cmSeguro =
     math.max(
     math.abs(cmKvo),
     syminfo.mintick)

vfKvo =
     volume *
     math.abs(
     2.0 * (dmKvo / cmSeguro) - 1.0) *
     tendenciaKvo *
     100.0

kvoBruto =
     ta.ema(vfKvo, kvoFast) -
     ta.ema(vfKvo, kvoSlow)

kvo =
     f_suavizar(
     kvoBruto,
     suavizacaoPeriodo)

kvoSignal =
     ta.ema(
     kvo,
     kvoSignalPeriodo)

kvoHist =
     kvo - kvoSignal

kvoVariacao =
     kvo - kvo[1]

kvoDesvio =
     ta.stdev(
     kvo,
     34)

kvoZscore =
     not na(kvoDesvio) and kvoDesvio > 0 ?
     kvo / kvoDesvio :
     0.0

kvoEstadoAlta =
     kvo > kvoSignal and
     kvoVariacao > 0

kvoForcaBruta =
     math.abs(kvoHist) /
     (math.abs(kvo) +
     math.abs(kvoSignal) +
     0.000000001)

kvoForca =
     f_clamp(
     kvoForcaBruta,
     0.0,
     1.0) * 100.0

kvoScoreLong =
     kvoEstadoAlta ?
     50.0 + math.min(kvoForca * 0.50, 50.0) :
     50.0

     f_clamp(
     kvoScoreLong,
     0.0,
     100.0)

     kvoEstadoAlta ?
     "ALTA" :
     "NEUTRO"

// ============================================================================
// FUNÇÃO DE SCORE MACRO

f_macroScoreLong() =>

    e5 = ta.ema(close, 5)

    e20 = ta.ema(close, 20)

    e50 = ta.ema(close, 50)

    e100 = ta.ema(close, 100)

    rsi = ta.rsi(close, 14)

    [macdLine, macdSignal, macdHist] =
         ta.macd(close, 12, 26, 9)

    atrLocal = ta.atr(14)

    atrBase = ta.sma(atrLocal, 50)

    volBase = ta.sma(volume, 20)

    float s = 0.0

    s += close > e5 ? 0.15 : 0.0

    s += e5 > e20 ? 0.20 : 0.0

    s += e20 > e50 ? 0.20 : 0.0

    s += e50 > e100 ? 0.15 : 0.0

    s += rsi >= 55 ? 0.15 : rsi >= 50 ? 0.07 : 0.0

    s += macdHist > 0 ? 0.10 : 0.0

    s += not na(atrBase) and atrLocal >= atrBase ? 0.025 : 0.0

    s += not na(volBase) and volume >= volBase and close > open ? 0.025 : 0.0

    f_clamp(s, 0.0, 1.0)
