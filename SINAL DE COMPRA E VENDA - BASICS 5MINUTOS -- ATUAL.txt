// ============================================================================
// FUNÇÕES
// ============================================================================

f_clamp(valor, minimo, maximo) =>
    math.max(minimo, math.min(maximo, valor))

f_arredondarPreco(preco) =>
    tick = syminfo.mintick
    tick > 0 ? math.round(preco / tick) * tick : preco

f_textoNumero(valor) =>
    na(valor) ? "-" : str.tostring(valor, "#.##")

f_textoPreco(valor) =>
    na(valor) ? "-" : str.tostring(valor, format.mintick)

f_corScore(valor) =>
    valor >= 80.0 ? color.green :
     valor >= 70.0 ? color.lime :
     valor >= 60.0 ? color.orange :
     color.red

f_statusScore(valor) =>
    valor >= 80.0 ? "MUITO FORTE" :
     valor >= 70.0 ? "FORTE" :
     valor >= 60.0 ? "MODERADO" :
     valor >= 50.0 ? "FRACO" :
     "MUITO FRACO"

f_posicaoPainel() =>
    posicaoPainel == "Superior Esquerdo" ? position.top_left :
     posicaoPainel == "Superior Direito" ? position.top_right :
     posicaoPainel == "Inferior Esquerdo" ? position.bottom_left :
     position.bottom_right

f_tamanhoTexto() =>
    tamanhoPainel == "Pequeno" ? size.small :
     tamanhoPainel == "Grande" ? size.large :
     size.normal


// ============================================================================
// 11 - CANDLE
// ============================================================================

rangeCandle = high - low

corpo = math.abs(close - open)

corpoPercentual =
     rangeCandle > 0 ?
     corpo / rangeCandle * 100.0 :
     0.0

candleAlta = close > open
candleBaixa = close < open

candleForte =
     corpoPercentual >= corpoMinimo * 100.0


// ============================================================================
// 12 - RETORNOS
// ============================================================================

retorno1 =
     close[1] > 0 ?
     (close / close[1] - 1.0) * 100.0 :
     0.0

retorno3 =
     close[3] > 0 ?
     (close / close[3] - 1.0) * 100.0 :
     0.0

retorno5 =
     close[5] > 0 ?
     (close / close[5] - 1.0) * 100.0 :
     0.0


// ============================================================================
// 13 - MÉDIAS
// ============================================================================

ema5 = ta.ema(close, 5)
ema10 = ta.ema(close, 10)
ema20 = ta.ema(close, 20)
ema50 = ta.ema(close, 50)
ema100 = ta.ema(close, 100)

tendenciaLongMaior =
     ema20 > ema50 and
     ema50 > ema100 and
     close > ema50

tendenciaShortMaior =
     ema20 < ema50 and
     ema50 < ema100 and
     close < ema50

momentumLong =
     close > ema20 and
     ema20 > ema20[1]

momentumShort =
     close < ema20 and
     ema20 < ema20[1]


// ============================================================================
// 14 - ATR LOCAL
// ============================================================================

atrRaw = ta.atr(atrPeriodo)

atrSuave = ta.ema(
     atrRaw,
     atrSuavePeriodo)

atrBase = ta.sma(
     atrRaw,
     50)

atrSeguro =
     math.max(
         nz(atrSuave, atrRaw),
         syminfo.mintick)

atrRatio =
     atrBase > 0 ?
     atrRaw / atrBase :
     1.0

atrExpansao = atrRatio > 1.05

atrContracao = atrRatio < 0.85

atrScoreLong =
     atrExpansao and candleAlta ? 100.0 :
     atrExpansao ? 70.0 :
     atrContracao ? 45.0 :
     60.0

atrScoreShort =
     atrExpansao and candleBaixa ? 100.0 :
     atrExpansao ? 70.0 :
     atrContracao ? 45.0 :
     60.0


// ============================================================================
// 15 - VOLUME
// ============================================================================

volumeMedia = ta.ema(
     volume,
     volumePeriodo)

volumeRelativo =
     volumeMedia > 0 ?
     volume / volumeMedia :
     1.0

volumeLong =
     f_clamp(
         50.0 +
         (volumeRelativo - 1.0) * 60.0,
         0.0,
         100.0)

volumeShort = volumeLong


// ============================================================================
// 16 - KVO
// ============================================================================

precoTipico =
     (high + low + close) / 3.0

direcaoTipica =
     precoTipico > precoTipico[1] ? 1.0 :
     precoTipico < precoTipico[1] ? -1.0 :
     0.0

dm = volume * direcaoTipica

kvoFastValue =
     ta.ema(dm, kvoFast)

kvoSlowValue =
     ta.ema(dm, kvoSlow)

kvo =
     kvoFastValue - kvoSlowValue

kvoSignal =
     ta.ema(kvo, kvoSignalPeriodo)

kvoHistograma =
     kvo - kvoSignal

kvoVariacao =
     kvo - kvo[1]

kvoDesvio =
     ta.stdev(kvo, 50)

kvoZScore =
     kvoDesvio > 0 ?
     kvoHistograma / kvoDesvio :
     0.0

kvoAlta =
     kvo > kvoSignal and
     kvoVariacao >= 0

kvoBaixa =
     kvo < kvoSignal and
     kvoVariacao <= 0

kvoForcaLong =
     f_clamp(
         50.0 + kvoZScore * 20.0,
         0.0,
         100.0)

kvoForcaShort =
     f_clamp(
         50.0 - kvoZScore * 20.0,
         0.0,
         100.0)

kvoPositivo = kvo > 0
kvoNegativo = kvo < 0


// ============================================================================
// 17 - RSI
// ============================================================================

rsi = ta.rsi(close, 14)

rsiLong =
     rsi >= 50 ?
     f_clamp(
         60.0 + (rsi - 50.0) * 1.20,
         0.0,
         100.0) :
     f_clamp(
         rsi,
         0.0,
         60.0)

rsiShort =
     rsi <= 50 ?
     f_clamp(
         60.0 + (50.0 - rsi) * 1.20,
         0.0,
         100.0) :
     f_clamp(
         100.0 - rsi,
         0.0,
         60.0)


// ============================================================================
// 18 - MICROESTRUTURA
// ============================================================================

microEMAfast =
     ta.ema(
         close,
         microTrendFast)

microEMAslow =
     ta.ema(
         close,
         microTrendSlow)

microSlope =
     microEMAfast -
     microEMAfast[2]

microLong =
     microEMAfast > microEMAslow and
     microSlope > 0

microShort =
     microEMAfast < microEMAslow and
     microSlope < 0

microResistencia =
     ta.highest(
         high[1],
         microLookback)

microSuporte =
     ta.lowest(
         low[1],
         microLookback)

microBreakoutLong =
     close > microResistencia

microBreakoutShort =
     close < microSuporte

microConfirmacaoLong =
     microBreakoutLong and
     volumeRelativo >= microMinimoVolume and
     candleAlta

microConfirmacaoShort =
     microBreakoutShort and
     volumeRelativo >= microMinimoVolume and
     candleBaixa

microScoreLong =
     microConfirmacaoLong ? 100.0 :
     microLong ? 80.0 :
     50.0

microScoreShort =
     microConfirmacaoShort ? 100.0 :
     microShort ? 80.0 :
     50.0


// ============================================================================
// 19 - ESTRUTURA LOCAL
// ============================================================================

resistencia =
     ta.highest(
         high[1],
         estruturaLookback)

suporte =
     ta.lowest(
         low[1],
         estruturaLookback)

distanciaResistenciaATR =
     atrSeguro > 0 ?
     (resistencia - close) / atrSeguro :
     999.0

distanciaSuporteATR =
     atrSeguro > 0 ?
     (close - suporte) / atrSeguro :
     999.0

rompimentoEstruturalLong =
     close > resistencia

rompimentoEstruturalShort =
     close < suporte

// CORREÇÃO IMPORTANTE:
// Se houver rompimento, o espaço é considerado válido.
// O código anterior bloqueava o breakout porque a distância
// passava a ser negativa.

espacoLong =
     rompimentoEstruturalLong or
     distanciaResistenciaATR >= espacoMinLong

espacoShort =
     rompimentoEstruturalShort or
     distanciaSuporteATR >= espacoMinShort

estruturaLong =
     close >= ema20 and
     ema20 >= ema50

estruturaShort =
     close <= ema20 and
     ema20 <= ema50


// ============================================================================
// 20 - MODELO LOCAL
// ============================================================================

localTrendLong =
     tendenciaLongMaior ? 100.0 :
     close > ema50 ? 70.0 :
     35.0

localTrendShort =
     tendenciaShortMaior ? 100.0 :
     close < ema50 ? 70.0 :
     35.0

localMomentumLong =
     momentumLong ? 100.0 :
     close > ema20 ? 65.0 :
     35.0

localMomentumShort =
     momentumShort ? 100.0 :
     close < ema20 ? 65.0 :
     35.0

localRetornoLong =
     retorno1 > 0 and retorno3 > 0 ? 100.0 :
     retorno1 > 0 or retorno3 > 0 ? 65.0 :
     30.0

localRetornoShort =
     retorno1 < 0 and retorno3 < 0 ? 100.0 :
     retorno1 < 0 or retorno3 < 0 ? 65.0 :
     30.0

localKvoLong =
     kvoPositivo and kvoAlta ? 100.0 :
     kvoPositivo ? 70.0 :
     35.0

localKvoShort =
     kvoNegativo and kvoBaixa ? 100.0 :
     kvoNegativo ? 70.0 :
     35.0

scoreLocalLong =
     localTrendLong * 0.25 +
     localMomentumLong * 0.15 +
     localRetornoLong * 0.10 +
     volumeLong * 0.10 +
     localKvoLong * 0.15 +
     rsiLong * 0.10 +
     atrScoreLong * 0.05 +
     microScoreLong * 0.10

scoreLocalShort =
     localTrendShort * 0.25 +
     localMomentumShort * 0.15 +
     localRetornoShort * 0.10 +
     volumeShort * 0.10 +
     localKvoShort * 0.15 +
     rsiShort * 0.10 +
     atrScoreShort * 0.05 +
     microScoreShort * 0.10

// 21 - FUNÇÃO MTF
// ============================================================================

f_mtfScore() =>

    ema8 = ta.ema(close, 8)
    ema21 = ta.ema(close, 21)
    ema50mtf = ta.ema(close, 50)

    rsiLocal = ta.rsi(close, 14)

    atrLocal = ta.atr(14)

    atrBaseLocal =
         ta.sma(
             atrLocal,
             50)

    atrRelLocal =
         atrBaseLocal > 0 ?
         atrLocal / atrBaseLocal :
         1.0

    retornoLocal =
         close[3] > 0 ?
         (close / close[3] - 1.0) * 100.0 :
         0.0

    tendenciaLong =
         ema8 > ema21 and
         ema21 > ema50mtf

    tendenciaShort =
         ema8 < ema21 and
         ema21 < ema50mtf

    tendenciaScoreLong =
         tendenciaLong ? 100.0 :
         tendenciaShort ? 0.0 :
         50.0

    tendenciaScoreShort =
         100.0 -
         tendenciaScoreLong

    rsiScoreLong =
         f_clamp(
             rsiLocal,
             0.0,
             100.0)

    rsiScoreShort =
         f_clamp(
             100.0 - rsiLocal,
             0.0,
             100.0)

    retornoScoreLong =
         f_clamp(
             50.0 +
             retornoLocal * 10.0,
             0.0,
             100.0)

    retornoScoreShort =
         100.0 -
         retornoScoreLong

    atrScore =
         atrRelLocal > 1.05 ? 70.0 :
         atrRelLocal < 0.80 ? 45.0 :
         60.0

    closeEMALong =
         close > ema8 ? 100.0 :
         close < ema8 ? 0.0 :
         50.0

    closeEMAShort =
         100.0 -
         closeEMALong

    scoreLong =
         tendenciaScoreLong * 0.40 +
         rsiScoreLong * 0.20 +
         retornoScoreLong * 0.15 +
         atrScore * 0.10 +
         closeEMALong * 0.15

    scoreShort =
         tendenciaScoreShort * 0.40 +
         rsiScoreShort * 0.20 +
         retornoScoreShort * 0.15 +
         atrScore * 0.10 +
         closeEMAShort * 0.15

    [f_clamp(scoreLong, 0.0, 100.0),
     f_clamp(scoreShort, 0.0, 100.0),
     atrLocal]
