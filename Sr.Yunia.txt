f_limitar(valor, minimo, maximo) =>
    math.max(minimo, math.min(maximo, valor))


f_preco(valor) =>
    na(valor) ? "-" : str.tostring(valor, format.mintick)

// KVO

f_kvo() =>
    tp = (high + low + close) / 3.0

    mediaVolume = ta.sma(volume, kvoRapido)
    volumeRelativo = mediaVolume > 0 ? volume / mediaVolume : 1.0

    tendenciaVolume = tp > tp[1] ? 1.0 : tp < tp[1] ? -1.0 : 0.0

    fluxo = tendenciaVolume * volumeRelativo * volume

    kvoRap = ta.ema(fluxo, kvoRapido)
    kvoLen = ta.ema(fluxo, kvoLento)

    kvo = kvoRap - kvoLen
    sinal = ta.ema(kvo, kvoSinal)
    hist = kvo - sinal

    kvoScoreLong = hist > 0 ? 100.0 : 0.0
    kvoScoreShort = hist < 0 ? 100.0 : 0.0

    [kvo, sinal, hist, kvoScoreLong, kvoScoreShort]

f_modeloLocal() =>
    atrAtual = ta.atr(atrPeriodo)
    atrMedia = ta.sma(atrAtual, atrMediaPeriodo)

    fatorATR = atrMedia > 0 ? atrAtual / atrMedia : 1.0

    atrAdaptado = usarATRAdaptativo ? atrAtual * fatorATR : atrAtual

    ema8 = ta.ema(close, emaRapida)
    ema21 = ta.ema(close, emaMedia)
    ema55 = ta.ema(close, emaLenta)

    rsi = ta.rsi(close, rsiPeriodo)

    volumeMedio = ta.sma(volume, 20)
    volumeRelativo = volumeMedio > 0 ? volume / volumeMedio : 1.0

    rangeCandle = high - low
    corpo = math.abs(close - open)

    forcaCandle = rangeCandle > 0 ? corpo / rangeCandle * 100.0 : 0.0

    retorno1 = close[1] != 0 ? (close - close[1]) / close[1] * 100.0 : 0.0
    retorno3 = close[3] != 0 ? (close - close[3]) / close[3] * 100.0 : 0.0

    // ------------------------------------------------------------------------
    // TENDENCIA
    // ------------------------------------------------------------------------

    tendenciaLong = 0.0

    if ema8 > ema21
        tendenciaLong += 35.0

    if ema21 > ema55
        tendenciaLong += 35.0

    if close > ema8
        tendenciaLong += 30.0

    tendenciaShort = 0.0

    if ema8 < ema21
        tendenciaShort += 35.0

    if ema21 < ema55
        tendenciaShort += 35.0

    if close < ema8
        tendenciaShort += 30.0

    // ------------------------------------------------------------------------
    // RSI
    // ------------------------------------------------------------------------

    rsiLong = 0.0

    if rsi <= rsiSobrevendido
        rsiLong := 100.0
    else if rsi < 50
        rsiLong := 75.0
    else if rsi < 60
        rsiLong := 50.0
    else
        rsiLong := 20.0

    rsiShort = 0.0

    if rsi >= rsiSobrecomprado
        rsiShort := 100.0
    else if rsi > 50
        rsiShort := 75.0
    else if rsi > 40
        rsiShort := 50.0
    else
        rsiShort := 20.0

    // ------------------------------------------------------------------------
    // RETORNOS
    // ------------------------------------------------------------------------

    retornoLong = 0.0

    if retorno1 > 0
        retornoLong += 50.0

    if retorno3 > 0
        retornoLong += 50.0

    retornoShort = 0.0

    if retorno1 < 0
        retornoShort += 50.0

    if retorno3 < 0
        retornoShort += 50.0

    // ------------------------------------------------------------------------
    // CANDLE
    // ------------------------------------------------------------------------

    candleLong = close > open ? forcaCandle : 0.0
    candleShort = close < open ? forcaCandle : 0.0

    // ------------------------------------------------------------------------
    // VOLUME
    // ------------------------------------------------------------------------

    volumeScore = volumeRelativo >= 1.50 ? 100.0 : volumeRelativo >= 1.20 ? 80.0 : volumeRelativo >= 1.00 ? 60.0 : 40.0

    // ------------------------------------------------------------------------
    // FUNDO E TOPO
    // ------------------------------------------------------------------------

    fundoAtual = ta.lowest(low, estruturaLookback)
    topoAtual = ta.highest(high, estruturaLookback)

    distanciaFundo = atrAtual > 0 ? math.abs(close - fundoAtual) / atrAtual : 999.0
    distanciaTopo = atrAtual > 0 ? math.abs(topoAtual - close) / atrAtual : 999.0

    rejeicaoAlta = low < low[1] and close > open and close > low + rangeCandle * 0.60
    rejeicaoBaixa = high > high[1] and close < open and close < high - rangeCandle * 0.60

    forcaFundo = 0.0

    if rejeicaoAlta
        forcaFundo += 35.0

    if close > open
        forcaFundo += 25.0

    if rsi < 45
        forcaFundo += 20.0

    if distanciaFundo <= distanciaMaxATR
        forcaFundo += 20.0

    forcaFundo := f_limitar(forcaFundo, 0, 100)

    forcaTopo = 0.0

    if rejeicaoBaixa
        forcaTopo += 35.0

    if close < open
        forcaTopo += 25.0

    if rsi > 55
        forcaTopo += 20.0

    if distanciaTopo <= distanciaMaxATR
        forcaTopo += 20.0

    forcaTopo := f_limitar(forcaTopo, 0, 100)

    // ------------------------------------------------------------------------
    // MICROESTRUTURA
    // ------------------------------------------------------------------------

    microMax = ta.highest(high, microLookback)
    microMinimo = ta.lowest(low, microLookback)

    microMaxAnterior = microMax[1]
    microMinAnterior = microMinimo[1]

    distanciaBreakLong = atrAtual > 0 ? (microMaxAnterior - close) / atrAtual : 999.0
    distanciaBreakShort = atrAtual > 0 ? (close - microMinAnterior) / atrAtual : 999.0

    microLong = 0.0

    if close > microMaxAnterior
        microLong += 60.0
    else if distanciaBreakLong <= 0.10
        microLong += 45.0
    else if distanciaBreakLong <= 0.25
        microLong += 30.0
    else if distanciaBreakLong <= 0.50
        microLong += 15.0

    if close > open
        microLong += 20.0

    if retorno1 > 0
        microLong += 20.0

    microLong := f_limitar(microLong, 0, 100)

    microShort = 0.0

    if close < microMinAnterior
        microShort += 60.0
    else if distanciaBreakShort <= 0.10
        microShort += 45.0
    else if distanciaBreakShort <= 0.25
        microShort += 30.0
    else if distanciaBreakShort <= 0.50
        microShort += 15.0

    if close < open
        microShort += 20.0

    if retorno1 < 0
        microShort += 20.0

    microShort := f_limitar(microShort, 0, 100)

    // ------------------------------------------------------------------------
    // KVO
    // ------------------------------------------------------------------------

    [kvo, kvoSinalValue, kvoHist, kvoLong, kvoShort] = f_kvo()

    // ------------------------------------------------------------------------
    // SCORE
    // ------------------------------------------------------------------------

    scoreLong = tendenciaLong * 0.20 + rsiLong * 0.10 + retornoLong * 0.15 + candleLong * 0.15 + volumeScore * 0.10 + kvoLong * 0.15 + forcaFundo * 0.10 + microLong * 0.05

    scoreShort = tendenciaShort * 0.20 + rsiShort * 0.10 + retornoShort * 0.15 + candleShort * 0.15 + volumeScore * 0.10 + kvoShort * 0.15 + forcaTopo * 0.10 + microShort * 0.05

    scoreLong := f_limitar(scoreLong * sensibilidade, 0, 100)
    scoreShort := f_limitar(scoreShort * sensibilidade, 0, 100)

    // ------------------------------------------------------------------------
    // INTENSIDADE
    // ------------------------------------------------------------------------

    intensidadeLong = scoreLong * 0.70 + forcaFundo * 0.15 + microLong * 0.15
    intensidadeShort = scoreShort * 0.70 + forcaTopo * 0.15 + microShort * 0.15

    intensidadeLong := f_limitar(intensidadeLong, 0, 100)
    intensidadeShort := f_limitar(intensidadeShort, 0, 100)

    // ------------------------------------------------------------------------
    // CONFLUENCIA
    // ------------------------------------------------------------------------

    confluenciaLong = (tendenciaLong + rsiLong + retornoLong + kvoLong + forcaFundo + microLong) / 6.0
    confluenciaShort = (tendenciaShort + rsiShort + retornoShort + kvoShort + forcaTopo + microShort) / 6.0

    confluenciaLong := f_limitar(confluenciaLong, 0, 100)
    confluenciaShort := f_limitar(confluenciaShort, 0, 100)

    [scoreLong, scoreShort, intensidadeLong, intensidadeShort, confluenciaLong, confluenciaShort, atrAdaptado, atrAtual, forcaFundo, forcaTopo, microLong, microShort]

// SCORE MTF
// ============================================================================

f_scoreMTF() =>
    ema8 = ta.ema(close, emaRapida)
    ema21 = ta.ema(close, emaMedia)
    ema55 = ta.ema(close, emaLenta)

    rsi = ta.rsi(close, rsiPeriodo)

    retorno = close[3] != 0 ? (close - close[3]) / close[3] * 100.0 : 0.0

    volumeMedio = ta.sma(volume, 20)
    volumeRelativo = volumeMedio > 0 ? volume / volumeMedio : 1.0

    scoreLong = 0.0

    if ema8 > ema21
        scoreLong += 25.0

    if ema21 > ema55
        scoreLong += 25.0

    if close > ema8
        scoreLong += 15.0

    if rsi > 50
        scoreLong += 15.0

    if retorno > 0
        scoreLong += 10.0

    if volumeRelativo >= 1.0
        scoreLong += 10.0

    scoreShort = 0.0

    if ema8 < ema21
        scoreShort += 25.0

    if ema21 < ema55
        scoreShort += 25.0

    if close < ema8
        scoreShort += 15.0

    if rsi < 50
        scoreShort += 15.0

    if retorno < 0
        scoreShort += 10.0

    if volumeRelativo >= 1.0
        scoreShort += 10.0

    [scoreLong, scoreShort]

f_scanner(ativo) =>
    [score1L, score1S] = request.security(ativo, "1", f_scoreMTF(), barmerge.gaps_off, barmerge.lookahead_off)
    [score5L, score5S] = request.security(ativo, "5", f_scoreMTF(), barmerge.gaps_off, barmerge.lookahead_off)
    [score15L, score15S] = request.security(ativo, "15", f_scoreMTF(), barmerge.gaps_off, barmerge.lookahead_off)
    [score30L, score30S] = request.security(ativo, "30", f_scoreMTF(), barmerge.gaps_off, barmerge.lookahead_off)
    [score60L, score60S] = request.security(ativo, "60", f_scoreMTF(), barmerge.gaps_off, barmerge.lookahead_off)
    [score240L, score240S] = request.security(ativo, "240", f_scoreMTF(), barmerge.gaps_off, barmerge.lookahead_off)
    [scoreDL, scoreDS] = request.security(ativo, "D", f_scoreMTF(), barmerge.gaps_off, barmerge.lookahead_off)

    [localL, localS, intL, intS, confL, confS, atrAdapt, atrAtual, fundoForca, topoForca, microL, microS] = request.security(ativo, timeframe.period, f_modeloLocal(), barmerge.gaps_off, barmerge.lookahead_off)

    mtfLong = 0.0
    mtfShort = 0.0

    if somaPesos > 0
        mtfLong := (score1L * peso1M + score5L * peso5M + score15L * peso15M + score30L * peso30M + score60L * peso1H + score240L * peso4H + scoreDL * pesoD) / somaPesos
        mtfShort := (score1S * peso1M + score5S * peso5M + score15S * peso15M + score30S * peso30M + score60S * peso1H + score240S * peso4H + scoreDS * pesoD) / somaPesos

    finalLong = localL * 0.55 + mtfLong * 0.45
    finalShort = localS * 0.55 + mtfShort * 0.45

    vantagemLong = finalLong - finalShort
    vantagemShort = finalShort - finalLong

    reqScoreLong = modoSinalAntecipado ? scoreMinLongAntecipado : scoreMinLong
    reqScoreShort = modoSinalAntecipado ? scoreMinShortAntecipado : scoreMinShort

    reqIntLong = modoSinalAntecipado ? intensidadeMinLongAntecipada : intensidadeMinLong
    reqIntShort = modoSinalAntecipado ? intensidadeMinShortAntecipada : intensidadeMinShort

    reqConf = modoSinalAntecipado ? confluenciaMinAntecipada : confluenciaMin
    reqVant = modoSinalAntecipado ? vantagemMinAntecipada : vantagemMin

    reqMTF = modoSinalAntecipado ? mtfMinAntecipado : 45.0

    reqFundo = modoSinalAntecipado ? forcaMinFundoAntecipada : forcaMinFundo
    reqTopo = modoSinalAntecipado ? forcaMinTopoAntecipada : forcaMinTopo

    reqMicro = modoSinalAntecipado ? microMinAntecipado : microMin

    // ------------------------------------------------------------------------
    // DISTANCIA LONG
    // ------------------------------------------------------------------------

    distanciaLong = 0.0

    distanciaLong += f_limitar(finalLong / math.max(reqScoreLong, 1.0) * 100.0, 0, 100) * 0.20
    distanciaLong += f_limitar(intL / math.max(reqIntLong, 1.0) * 100.0, 0, 100) * 0.15
    distanciaLong += f_limitar(confL / math.max(reqConf, 1.0) * 100.0, 0, 100) * 0.15
    distanciaLong += f_limitar(vantagemLong / math.max(reqVant, 1.0) * 100.0, 0, 100) * 0.15
    distanciaLong += f_limitar(mtfLong / math.max(reqMTF, 1.0) * 100.0, 0, 100) * 0.15
    distanciaLong += f_limitar(fundoForca / math.max(reqFundo, 1.0) * 100.0, 0, 100) * 0.10
    distanciaLong += f_limitar(microL / math.max(reqMicro, 1.0) * 100.0, 0, 100) * 0.10

    // ------------------------------------------------------------------------
    // DISTANCIA SHORT
    // ------------------------------------------------------------------------

    distanciaShort = 0.0

    distanciaShort += f_limitar(finalShort / math.max(reqScoreShort, 1.0) * 100.0, 0, 100) * 0.20
    distanciaShort += f_limitar(intS / math.max(reqIntShort, 1.0) * 100.0, 0, 100) * 0.15
    distanciaShort += f_limitar(confS / math.max(reqConf, 1.0) * 100.0, 0, 100) * 0.15
    distanciaShort += f_limitar(vantagemShort / math.max(reqVant, 1.0) * 100.0, 0, 100) * 0.15
    distanciaShort += f_limitar(mtfShort / math.max(reqMTF, 1.0) * 100.0, 0, 100) * 0.15
    distanciaShort += f_limitar(topoForca / math.max(reqTopo, 1.0) * 100.0, 0, 100) * 0.10
    distanciaShort += f_limitar(microS / math.max(reqMicro, 1.0) * 100.0, 0, 100) * 0.10

    distanciaLong := f_limitar(distanciaLong, 0, 100)
    distanciaShort := f_limitar(distanciaShort, 0, 100)

    // ------------------------------------------------------------------------
    // SINAIS
    // ------------------------------------------------------------------------

    sinalLong = finalLong >= reqScoreLong and intL >= reqIntLong and confL >= reqConf and vantagemLong >= reqVant and mtfLong >= reqMTF and fundoForca >= reqFundo and microL >= reqMicro

    sinalShort = finalShort >= reqScoreShort and intS >= reqIntShort and confS >= reqConf and vantagemShort >= reqVant and mtfShort >= reqMTF and topoForca >= reqTopo and microS >= reqMicro

    // ------------------------------------------------------------------------
    // DIRECAO
    // ------------------------------------------------------------------------

    distanciaFinal = math.max(distanciaLong, distanciaShort)

    direcao = "AGUARDAR"

    if sinalLong and not sinalShort
        direcao := "LONG"
    else if sinalShort and not sinalLong
        direcao := "SHORT"
    else if sinalLong and sinalShort
        direcao := finalLong >= finalShort ? "LONG" : "SHORT"
    else
        direcao := distanciaLong >= distanciaShort ? "LONG" : "SHORT"

    // ------------------------------------------------------------------------
    // PRINCIPAL BLOQUEIO
    // ------------------------------------------------------------------------

    falta = "OK"

    if direcao == "LONG"
        if finalLong < reqScoreLong
            falta := "SCORE"
        else if intL < reqIntLong
            falta := "INTENSIDADE"
        else if confL < reqConf
            falta := "CONFLUENCIA"
        else if vantagemLong < reqVant
            falta := "VANTAGEM"
        else if mtfLong < reqMTF
            falta := "MTF"
        else if fundoForca < reqFundo
            falta := "FUNDO"
        else if microL < reqMicro
            falta := "MICRO"
    else
        if finalShort < reqScoreShort
            falta := "SCORE"
        else if intS < reqIntShort
            falta := "INTENSIDADE"
        else if confS < reqConf
            falta := "CONFLUENCIA"
        else if vantagemShort < reqVant
            falta := "VANTAGEM"
        else if mtfShort < reqMTF
            falta := "MTF"
        else if topoForca < reqTopo
            falta := "TOPO"
        else if microS < reqMicro
            falta := "MICRO"

    // ------------------------------------------------------------------------
    // ENTRADA E ALVO
    // ------------------------------------------------------------------------

    entrada = close

    alvo = direcao == "LONG" ? close + atrAdapt * alvoPrincipal : close - atrAdapt * alvoPrincipal

    // ------------------------------------------------------------------------
    // SAIDA FINAL
    // ------------------------------------------------------------------------

    scoreFinal = direcao == "LONG" ? finalLong : finalShort
    intensidadeFinal = direcao == "LONG" ? intL : intS
    confluenciaFinal = direcao == "LONG" ? confL : confS
    mtfFinal = direcao == "LONG" ? mtfLong : mtfShort

    [direcao, distanciaFinal, scoreFinal, intensidadeFinal, confluenciaFinal, mtfFinal, falta, entrada, alvo, sinalLong, sinalShort]

    else if direcao == "SHORT"
        if sinalShort
            cor := color.new(color.red, 5)
        else if distancia >= 80
            cor := color.new(color.red, 25)
        else if distancia >= 60
            cor := color.new(color.red, 45)
        else
            cor := color.new(color.red, 70)

    cor


f_corDistancia(distancia, direcao, sinalLong, sinalShort) =>

        else if distancia >= 60
            cor := color.new(color.red, 40)
        else if distancia >= 40
            cor := color.new(color.yellow, 45)

        else if valor >= 60

        else if valor >= 60
            cor := color.new(color.red, 45)
        else if valor >= 40
            cor := color.new(color.yellow, 50)

f_corFalta(falta, sinalLong, sinalShort) =>
    cor = color.new(color.gray, 75)

    if sinalLong or sinalShort
        cor := color.new(color.green, 5)
    else if falta == "OK"

        cor := color.new(color.yellow, 30)
    else if falta == "MTF"
        cor := color.new(color.blue, 40)
    else if falta == "FUNDO" or falta == "TOPO" or falta == "MICRO"
        cor := color.new(color.purple, 40)

    cor
