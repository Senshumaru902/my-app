f_clamp(valor, minimo, maximo) =>
    math.max(minimo, math.min(maximo, valor))


f_suavizar(valor, periodo) =>
    ema1 = ta.ema(valor, periodo)
    ema2 = ta.ema(ema1, periodo)
    (ema1 + ema2) / 2.0


f_arredondarPreco(valor) =>
    syminfo.mintick > 0 ? math.round(valor / syminfo.mintick) * syminfo.mintick : valor


f_classProb(valor) =>
    valor >= 90 ? "EXTREMA" :
     valor >= 80 ? "MUITO ALTA" :
     valor >= 70 ? "ALTA" :
     valor >= 60 ? "MODERADA" :
     "BAIXA"


f_classIntensidade(valor) =>
    valor >= 90 ? "EXTREMA" :
     valor >= 80 ? "MUITO FORTE" :
     valor >= 70 ? "FORTE" :
     valor >= 55 ? "MODERADA" :
     "FRACA"


f_mtfAnalise() =>
    ema8 = ta.ema(close, 8)
    ema21 = ta.ema(close, 21)
    ema50 = ta.ema(close, 50)

    rsi = ta.rsi(close, 14)

    atrLocal = ta.atr(14)
    atrMedia = ta.sma(atrLocal, 30)

    volMedia = ta.sma(volume, 20)

    retorno = close > 0 and close[3] > 0 ? (close / close[3] - 1.0) * 100.0 : 0.0

    tendenciaLong = 0.0
    tendenciaShort = 0.0

    tendenciaLong += close > ema8 ? 15.0 : 0.0
    tendenciaLong += ema8 > ema21 ? 15.0 : 0.0
    tendenciaLong += ema21 > ema50 ? 15.0 : 0.0
    tendenciaLong += rsi > 50 ? 10.0 : 0.0
    tendenciaLong += rsi > 55 ? 10.0 : 0.0
    tendenciaLong += retorno > 0 ? 10.0 : 0.0
    tendenciaLong += atrLocal >= atrMedia ? 5.0 : 0.0
    tendenciaLong += volume > volMedia ? 10.0 : 0.0

    tendenciaShort += close < ema8 ? 15.0 : 0.0
    tendenciaShort += ema8 < ema21 ? 15.0 : 0.0
    tendenciaShort += ema21 < ema50 ? 15.0 : 0.0
    tendenciaShort += rsi < 50 ? 10.0 : 0.0
    tendenciaShort += rsi < 45 ? 10.0 : 0.0
    tendenciaShort += retorno < 0 ? 10.0 : 0.0
    tendenciaShort += atrLocal >= atrMedia ? 5.0 : 0.0
    tendenciaShort += volume > volMedia ? 10.0 : 0.0

    [f_clamp(tendenciaLong, 0.0, 100.0), f_clamp(tendenciaShort, 0.0, 100.0)]


// ============================================================================
// DADOS DO CANDLE
// ============================================================================

rangeCandle = math.max(high - low, syminfo.mintick)

corpo = math.abs(close - open)

corpoPercentual = rangeCandle > 0 ? corpo / rangeCandle : 0.0

pavioSuperior = high - math.max(open, close)
pavioInferior = math.min(open, close) - low

direcaoCandle = close > open ? 1 : close < open ? -1 : 0

candleForte = corpoPercentual >= corpoMinimo

candleForteLong = candleForte and direcaoCandle > 0
candleForteShort = candleForte and direcaoCandle < 0


// ============================================================================
// RETORNOS
// ============================================================================

retorno1 = close > 0 and close[1] > 0 ? (close / close[1] - 1.0) * 100.0 : 0.0
retorno3 = close > 0 and close[3] > 0 ? (close / close[3] - 1.0) * 100.0 : 0.0
retorno5 = close > 0 and close[5] > 0 ? (close / close[5] - 1.0) * 100.0 : 0.0
retorno10 = close > 0 and close[10] > 0 ? (close / close[10] - 1.0) * 100.0 : 0.0

// ============================================================================
// ATR
// ============================================================================

atrSuave = ta.ema(atrBruto, atrSuavePeriodo)

atrSeguro = nz(atrSuave, atrBruto)

atrRatio = atrMediaRegime > 0 ? atrSeguro / atrMediaRegime : 1.0

atrPercentual = close > 0 ? atrSeguro / close * 100.0 : 0.0

atrExpansao = atrRatio > 1.05
atrContracao = atrRatio < 0.85

atrScore = f_clamp(
     50.0 +
     (atrRatio - 1.0) * 100.0,
     0.0,
     100.0)


// ============================================================================
// VOLUME
// ============================================================================

volumeMedia = ta.ema(volume, volumePeriodo)

volumeRelativo =
     volumeMedia > 0 ?
     volume / volumeMedia :
     1.0

volumeScore = f_clamp(
     50.0 +
     (volumeRelativo - 1.0) * 60.0,
     0.0,
     100.0)

volumeForte = volumeRelativo >= 1.10


// ============================================================================
// KVO
// ============================================================================

precoTipico = (high + low + close) / 3.0

direcaoFluxo = precoTipico > precoTipico[1] ? 1.0 :
     precoTipico < precoTipico[1] ? -1.0 :
     0.0

forcaVolume = volume * direcaoFluxo

kvoRapido = ta.ema(forcaVolume, kvoFast)
kvoLento = ta.ema(forcaVolume, kvoSlow)

kvo = kvoRapido - kvoLento

kvoSignal = ta.ema(kvo, kvoSignalPeriodo)

kvoHistograma = kvo - kvoSignal

kvoVariacao = kvo - kvo[1]

kvoDesvio = ta.stdev(kvo, 50)

kvoZ =
     kvoDesvio > 0 ?
     kvoHistograma / kvoDesvio :
     0.0

kvoLongBase =
     kvo > kvoSignal ? 65.0 : 35.0

kvoShortBase =
     kvo < kvoSignal ? 65.0 : 35.0

kvoLong =
     f_clamp(
         kvoLongBase +
         (kvoVariacao > 0 ? 15.0 : -10.0) +
         (kvoZ > 0 ? 10.0 : -5.0),
         0.0,
         100.0)

kvoShort =
     f_clamp(
         kvoShortBase +
         (kvoVariacao < 0 ? 15.0 : -10.0) +
         (kvoZ < 0 ? 10.0 : -5.0),
         0.0,
         100.0)

kvoBullish = kvo > kvoSignal and kvoVariacao > 0
kvoBearish = kvo < kvoSignal and kvoVariacao < 0

confluenciaLong := f_clamp(confluenciaLong, 0.0, 100.0)

// CONFLUÊNCIA SHORT
// ============================================================================

confluenciaShort = 0.0

confluenciaShort += alinhamentoShort ? 15.0 : 0.0
confluenciaShort += kvoBearish ? 15.0 : 0.0
confluenciaShort += retorno1 < 0 ? 5.0 : 0.0
confluenciaShort += retorno3 < 0 ? 5.0 : 0.0
confluenciaShort += microShort ? 10.0 : 0.0
confluenciaShort += microBreakoutShort ? 10.0 : 0.0
confluenciaShort += volumeForte ? 10.0 : 0.0
confluenciaShort += mtfShort >= 60 ? 10.0 : 0.0
confluenciaShort += candleForteShort ? 5.0 : 0.0
confluenciaShort += atrScore >= 50 ? 5.0 : 0.0

confluenciaShort := f_clamp(confluenciaShort, 0.0, 100.0)


// ============================================================================
// PENALIDADES
// ============================================================================

penaltyLong = 0.0

penaltyLong += kvoBearish ? 12.0 : 0.0
penaltyLong += mtfShort > mtfLong ? 10.0 : 0.0
penaltyLong += microShort ? 8.0 : 0.0
penaltyLong += atrContracao ? 5.0 : 0.0
penaltyLong += distanciaResistencia < 0.50 ? 8.0 : 0.0

penaltyLong := f_clamp(penaltyLong, 0.0, 30.0)


penaltyShort = 0.0

penaltyShort += kvoBullish ? 12.0 : 0.0
penaltyShort += mtfLong > mtfShort ? 10.0 : 0.0
penaltyShort += microLong ? 8.0 : 0.0
penaltyShort += atrContracao ? 5.0 : 0.0
penaltyShort += distanciaSuporte < 0.50 ? 8.0 : 0.0

penaltyShort := f_clamp(penaltyShort, 0.0, 30.0)


// ============================================================================
// PROBABILIDADE LONG
// ============================================================================

probLongBruta =
     modeloLong * 0.25 +
     mtfLong * 0.30 +
     kvoLong * 0.12 +
     microScoreLong * 0.12 +
     confluenciaLong * 0.15 +
     atrScore * 0.06 -
     penaltyLong

bonusLong = 0.0

bonusLong += microConfirmLong ? 3.0 : 0.0
bonusLong += mtfLong >= 70 ? 3.0 : 0.0
bonusLong += kvoLong >= 70 ? 3.0 : 0.0
bonusLong += volumeForte ? 2.0 : 0.0
bonusLong += candleForteLong ? 2.0 : 0.0

probLongFinalBruta = f_clamp(
     probLongBruta + bonusLong,
     1.0,
     99.0)

probLong = f_clamp(
     nz(f_suavizar(probLongFinalBruta, suavizacaoPeriodo), probLongFinalBruta),
     1.0,
     99.0)


// ============================================================================
// PROBABILIDADE SHORT
// ============================================================================

probShortBruta =
     modeloShort * 0.25 +
     mtfShort * 0.30 +
     kvoShort * 0.12 +
     microScoreShort * 0.12 +
     confluenciaShort * 0.15 +
     atrScore * 0.06 -
     penaltyShort

bonusShort = 0.0

bonusShort += microConfirmShort ? 3.0 : 0.0
bonusShort += mtfShort >= 70 ? 3.0 : 0.0
bonusShort += kvoShort >= 70 ? 3.0 : 0.0
bonusShort += volumeForte ? 2.0 : 0.0
bonusShort += candleForteShort ? 2.0 : 0.0

probShortFinalBruta = f_clamp(
     probShortBruta + bonusShort,
     1.0,
     99.0)

probShort = f_clamp(
     nz(f_suavizar(probShortFinalBruta, suavizacaoPeriodo), probShortFinalBruta),
     1.0,
     99.0)


// ============================================================================
// INTENSIDADE DIRECIONAL
// ============================================================================

intensidadeLong =
     modeloLong * 0.25 +
     mtfLong * 0.25 +
     kvoLong * 0.15 +
     confluenciaLong * 0.15 +
     microScoreLong * 0.10 +
     atrScore * 0.10 -
     penaltyLong

intensidadeShort =
     modeloShort * 0.25 +
     mtfShort * 0.25 +
     kvoShort * 0.15 +
     confluenciaShort * 0.15 +
     microScoreShort * 0.10 +
     atrScore * 0.10 -
     penaltyShort

intensidadeLong := f_clamp(intensidadeLong, 0.0, 100.0)
intensidadeShort := f_clamp(intensidadeShort, 0.0, 100.0)

// VANTAGEM
// ============================================================================

vantagemLong = probLong - probShort
vantagemShort = probShort - probLong

vantagemAbsoluta = math.abs(probLong - probShort)

// DIREÇÃO PRINCIPAL
// ============================================================================

direcaoLong =
     ativarLong and
     probLong >= probMinLong and
     probLong > probShort and
     vantagemLong >= vantagemMinima

direcaoShort =
     ativarShort and
     probShort >= probMinShort and
     probShort > probLong and
     vantagemShort >= vantagemMinima

// CONDIÇÕES OPERACIONAIS LONG
// ============================================================================

condLongMTF = mtfLong >= mtfMinLong
condLongKVO = kvoLong >= kvoMinLong
condLongConfluencia = confluenciaLong >= confluenciaMinLong
condLongMicro = microScoreLong >= microScoreMin
condLongEstrutura = filtroResistenciaLong

sinalLong =
     ativarLong and
     direcaoLong and
     condLongMTF and
     condLongKVO and
     condLongConfluencia and
     condLongMicro and
     condLongEstrutura


// ============================================================================
// CONDIÇÕES OPERACIONAIS SHORT
// ============================================================================

condShortMTF = mtfShort >= mtfMinShort
condShortKVO = kvoShort >= kvoMinShort
condShortConfluencia = confluenciaShort >= confluenciaMinShort
condShortMicro = microScoreShort >= microScoreMin
condShortEstrutura = filtroSuporteShort

sinalShort =
     ativarShort and
     direcaoShort and
     condShortMTF and
     condShortKVO and
     condShortConfluencia and
     condShortMicro and
     condShortEstrutura

// FORÇA DO SINAL
// ============================================================================

     probLong * 0.40 +
     intensidadeLong * 0.30 +
     mtfLong * 0.15 +
     confluenciaLong * 0.15

forcaShort =
     probShort * 0.40 +
     intensidadeShort * 0.30 +
     mtfShort * 0.15 +

forcaShort := f_clamp(forcaShort, 0.0, 100.0)

// FATOR DE ENTRADA
//
// Quanto maior a força, mais próximo da região favorável de entrada.
// ============================================================================

forcaDirecional =
     sinalLong ? forcaLong :
     sinalShort ? forcaShort :
     math.max(forcaLong, forcaShort)

     (forcaDirecional - 50.0) / 50.0,
     0.0,
     1.0)

     distanciaEntradaMaxATR -
     (distanciaEntradaMaxATR - distanciaEntradaMinATR) *
     forcaNormalizada

fatorEntradaATR := f_clamp(
     fatorEntradaATR,
     distanciaEntradaMinATR,
     distanciaEntradaMaxATR)

// PREÇOS DE ENTRADA
// ============================================================================

     close + atrSeguro * fatorEntradaATR

precoEntradaShortBruto =

     f_arredondarPreco(precoEntradaLongBruto)

precoEntradaShort =
     f_arredondarPreco(precoEntradaShortBruto)

precoEntradaExato =
     sinalLong ? precoEntradaLong :
     sinalShort ? precoEntradaShort :
     na

// ============================================================================

     0.0

// se o suporte estiver antes do alvo,

     resistencia > precoEntradaLong and

     f_arredondarPreco(resistencia - atrSeguro * 0.10) :

     protegerAlvoEstrutural and
     not na(suporte) and
     suporte < precoEntradaShort and
     suporte > alvoPrincipalShort ?

     alvoPrincipalShort
