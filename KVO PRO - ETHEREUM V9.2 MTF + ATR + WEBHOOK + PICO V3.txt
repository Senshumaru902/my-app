indicator("KVO PRO - ETHEREUM V9.2 MTF + ATR + WEBHOOK + PICO V3", overlay=true, max_labels_count=500)

// ============================================================================
// KVO PRO - ETHEREUM
// V9.2
// INTENSIDADE DIRECIONAL + KVO + BOOST ADAPTATIVO
// + CONFLUÊNCIA
// + MTF
// + ATR
// + WEBHOOK
// + PICO V3
// ============================================================================


// ============================================================================
// CONFIGURAÇÕES
// ============================================================================

groupConfig = "KVO PRO - CONFIGURAÇÕES"

corpoMinimo = input.float(0.70, "Corpo mínimo", minval=0.0, maxval=1.0, step=0.05, group=groupConfig)
atrPeriodo = input.int(14, "ATR período", minval=1, group=groupConfig)
estruturaLookback = input.int(20, "Estrutura", minval=2, group=groupConfig)
confirmacaoLookback = input.int(5, "Confirmação", minval=2, group=groupConfig)

kvoFast = input.int(34, "KVO Fast", minval=1, group=groupConfig)
kvoSlow = input.int(55, "KVO Slow", minval=2, group=groupConfig)
kvoSignalPeriodo = input.int(13, "KVO Signal", minval=1, group=groupConfig)

boostMaximo = input.float(20.0, "Boost máximo", minval=0, maxval=100, group=groupConfig)
penaltyMaximo = input.float(15.0, "Penalty máximo", minval=0, maxval=100, group=groupConfig)

pesoCorpo = input.float(0.50, "Peso corpo", minval=0, maxval=1, step=0.05, group=groupConfig)
pesoModelo = input.float(0.50, "Peso modelo", minval=0, maxval=1, step=0.05, group=groupConfig)

mostrarCandleForca = input.bool(true, "Colorir candles de força", group=groupConfig)
somenteCandleForca = input.bool(false, "Alertar somente candle de força", group=groupConfig)
mostrarTabela = input.bool(true, "Mostrar painel", group=groupConfig)
usarWebhook = input.bool(true, "Ativar webhook", group=groupConfig)
alertaSomenteFechamento = input.bool(true, "Alertar no fechamento do candle", group=groupConfig)


// ============================================================================
// MTF
// ============================================================================

groupMTF = "KVO PRO - MULTI TIMEFRAME"

tf1 = input.timeframe("5", "MTF 1", group=groupMTF)
tf2 = input.timeframe("15", "MTF 2", group=groupMTF)
tf3 = input.timeframe("60", "MTF 3", group=groupMTF)
tf4 = input.timeframe("240", "MTF 4", group=groupMTF)

usarMTF1 = input.bool(true, "Usar MTF 1", group=groupMTF)
usarMTF2 = input.bool(true, "Usar MTF 2", group=groupMTF)
usarMTF3 = input.bool(true, "Usar MTF 3", group=groupMTF)
usarMTF4 = input.bool(true, "Usar MTF 4", group=groupMTF)


// ============================================================================
// CORES
// ============================================================================

corVerde = color.rgb(0, 255, 100)
corVerdeForte = color.rgb(0, 255, 0)
corVermelho = color.rgb(255, 60, 60)
corAmarelo = color.rgb(255, 210, 0)
corCinza = color.rgb(150, 150, 150)
corCiano = color.rgb(0, 220, 255)
corAzul = color.rgb(0, 150, 255)
corBranco = color.white

corFundo = color.rgb(15, 15, 20)
corFundo2 = color.rgb(25, 25, 32)
corBorda = color.rgb(70, 70, 80)


// ============================================================================
// DADOS DO CANDLE
// ============================================================================

abertura = open
maxima = high
minima = low
fechamento = close
volumeAtual = volume

rangeCandle = high - low
corpo = math.abs(close - open)

pavioSuperior = high - math.max(open, close)
pavioInferior = math.min(open, close) - low

rangeSeguro = math.max(rangeCandle, syminfo.mintick)

corpoPct = corpo / rangeSeguro
pavioSuperiorPct = pavioSuperior / rangeSeguro
pavioInferiorPct = pavioInferior / rangeSeguro


// ============================================================================
// DIREÇÃO
// ============================================================================

direcaoCandle = close > open ? 1 : close < open ? -1 : 0

direcaoTexto = direcaoCandle > 0 ? "ALTA" : direcaoCandle < 0 ? "BAIXA" : "NEUTRA"


// ============================================================================
// RETORNOS
// ============================================================================

retorno1 = not na(close[1]) ? close / close[1] - 1.0 : 0.0
retorno3 = not na(close[3]) ? close / close[3] - 1.0 : 0.0
retorno5 = not na(close[5]) ? close / close[5] - 1.0 : 0.0


// ============================================================================
// MÉDIAS
// ============================================================================

mm5 = ta.sma(close, 5)
mm10 = ta.sma(close, 10)
mm20 = ta.sma(close, 20)

distMm5 = not na(mm5) ? close / mm5 - 1.0 : 0.0
distMm10 = not na(mm10) ? close / mm10 - 1.0 : 0.0
distMm20 = not na(mm20) ? close / mm20 - 1.0 : 0.0


// ============================================================================
// VOLUME
// ============================================================================

volumeMm20 = ta.sma(volume, 20)

volumeRelativo = not na(volumeMm20) and volumeMm20 != 0 ?
     volume / volumeMm20 :
     1.0


// ============================================================================
// ATR
// ============================================================================

atr = ta.atr(atrPeriodo)
atrSeguro = math.max(atr, syminfo.mintick)

atrRelativo = not na(close) and close != 0 ?
     atr / close * 100.0 :
     0.0

atrMedia = ta.sma(atr, 20)

atrForte = not na(atrMedia) and atr > atrMedia
atrFraco = not na(atrMedia) and atr < atrMedia

atrDirecao = close > open ? 1 : close < open ? -1 : 0

atrBoaCompra = atrDirecao > 0 and atrForte
atrBoaVenda = atrDirecao < 0 and atrForte

atrClassificacao = atrBoaCompra ?
     "BOA COMPRA" :
     atrBoaVenda ?
     "BOA VENDA" :
     atrForte ?
     "FORTE / NEUTRO" :
     "FRACO"

corAtr = atrBoaCompra ?
     corVerde :
     atrBoaVenda ?
     corVermelho :
     corCinza


// ============================================================================
// ESTRUTURA
// ============================================================================

suporte20 = ta.lowest(low[1], estruturaLookback)
resistencia20 = ta.highest(high[1], estruturaLookback)

suporte5 = ta.lowest(low[1], confirmacaoLookback)
resistencia5 = ta.highest(high[1], confirmacaoLookback)

distResistencia = (resistencia20 - close) / atrSeguro
distSuporte = (close - suporte20) / atrSeguro


// ============================================================================
// KVO
// ============================================================================

precoTipico = (high + low + close) / 3.0

tendenciaKvo =
     precoTipico > precoTipico[1] ? 1 :
     precoTipico < precoTipico[1] ? -1 :
     0

dmKvo = high - low

var float cmKvo = 0.0

if bar_index == 0
    cmKvo := dmKvo
else
    cmKvo :=
         tendenciaKvo == tendenciaKvo[1] ?
         cmKvo[1] + dmKvo :
         dmKvo + dmKvo[1]

cmSeguro = math.max(cmKvo, syminfo.mintick)

vfKvo =
     volume *
     math.abs(2.0 * (dmKvo / cmSeguro) - 1.0) *
     tendenciaKvo *
     100.0

kvo = ta.ema(vfKvo, kvoFast) - ta.ema(vfKvo, kvoSlow)

kvoSignal = ta.ema(kvo, kvoSignalPeriodo)

kvoHist = kvo - kvoSignal

kvoVariacao = kvo - kvo[1]

kvoDesvio = ta.stdev(kvo, 34)

kvoZscore =
     not na(kvoDesvio) and kvoDesvio != 0 ?
     kvo / kvoDesvio :
     0.0

kvoForcaBruta =
     math.abs(kvoHist) /
     (math.abs(kvo) + math.abs(kvoSignal) + 0.000000001)

kvoForca =
     math.min(
         math.max(kvoForcaBruta, 0.0),
         1.0
     ) * 100.0

kvoEstado =
     kvo > kvoSignal and kvoVariacao > 0 ?
     1 :
     kvo < kvoSignal and kvoVariacao < 0 ?
     -1 :
     0

kvoEstadoTexto =
     kvoEstado > 0 ?
     "ALTA" :
     kvoEstado < 0 ?
     "BAIXA" :
     "NEUTRO"

corKvo =
     kvoEstado > 0 ?
     corVerde :
     kvoEstado < 0 ?
     corVermelho :
     corCinza


// ============================================================================
// CANDLE DE FORÇA
// ============================================================================

forcaCorpo =
     math.min(
         math.max(corpoPct * 100.0, 0.0),
         100.0
     )

candleForte = corpoPct >= corpoMinimo


// ============================================================================
// MODELO DIRECIONAL
// ============================================================================

float pontosAltaModelo = 0.0
float pontosBaixaModelo = 0.0

if close > open
    pontosAltaModelo += 2.0
else if close < open
    pontosBaixaModelo += 2.0

if kvoEstado > 0
    pontosAltaModelo += 2.0
else if kvoEstado < 0
    pontosBaixaModelo += 2.0

if retorno1 > 0
    pontosAltaModelo += 1.0
else if retorno1 < 0
    pontosBaixaModelo += 1.0

if retorno3 > 0
    pontosAltaModelo += 1.0
else if retorno3 < 0
    pontosBaixaModelo += 1.0

if close > mm5
    pontosAltaModelo += 1.0
else if close < mm5
    pontosBaixaModelo += 1.0

if close > mm10
    pontosAltaModelo += 1.0
else if close < mm10
    pontosBaixaModelo += 1.0

if close > mm20
    pontosAltaModelo += 1.0
else if close < mm20
    pontosBaixaModelo += 1.0

if volumeRelativo >= 1.20
    if close > open
        pontosAltaModelo += 0.5
    else if close < open
        pontosBaixaModelo += 0.5


// ============================================================================
// NORMALIZAÇÃO DO MODELO
// ============================================================================

pontuacaoModeloTotal = pontosAltaModelo + pontosBaixaModelo

probAlta =
     pontuacaoModeloTotal > 0 ?
     pontosAltaModelo / pontuacaoModeloTotal :
     0.5

probBaixa =
     pontuacaoModeloTotal > 0 ?
     pontosBaixaModelo / pontuacaoModeloTotal :
     0.5

forcaModelo = math.max(probAlta, probBaixa) * 100.0

direcaoModelo =
     probAlta > probBaixa ?
     1 :
     probBaixa > probAlta ?
     -1 :
     0

direcaoModeloTexto =
     direcaoModelo > 0 ?
     "ALTA" :
     direcaoModelo < 0 ?
     "BAIXA" :
     "NEUTRA"


// ============================================================================
// INTENSIDADE BASE
// ============================================================================

intensidadeBase =
     forcaCorpo * pesoCorpo +
     forcaModelo * pesoModelo

intensidadeBase :=
     math.min(
         math.max(intensidadeBase, 0.0),
         100.0
     )


// ============================================================================
// CONFLUÊNCIA
// ============================================================================

int confirmacoes = 0
int divergencias = 0

if direcaoCandle != 0 and direcaoCandle == direcaoModelo
    confirmacoes += 1
else
    divergencias += 1

if direcaoCandle != 0 and kvoEstado == direcaoCandle
    confirmacoes += 1
else if direcaoCandle != 0 and kvoEstado != 0
    divergencias += 1

if direcaoCandle > 0 and retorno1 > 0
    confirmacoes += 1
else if direcaoCandle < 0 and retorno1 < 0
    confirmacoes += 1
else if retorno1 != 0
    divergencias += 1

if direcaoCandle > 0 and retorno3 > 0
    confirmacoes += 1
else if direcaoCandle < 0 and retorno3 < 0
    confirmacoes += 1
else if retorno3 != 0
    divergencias += 1

if direcaoCandle > 0 and close > mm5
    confirmacoes += 1
else if direcaoCandle < 0 and close < mm5
    confirmacoes += 1
else
    divergencias += 1

if direcaoCandle > 0 and close > mm10
    confirmacoes += 1
else if direcaoCandle < 0 and close < mm10
    confirmacoes += 1
else
    divergencias += 1

if direcaoCandle > 0 and close > mm20
    confirmacoes += 1
else if direcaoCandle < 0 and close < mm20
    confirmacoes += 1
else
    divergencias += 1


// ============================================================================
// DIREÇÃO DA INTENSIDADE
// ============================================================================

float pontuacaoAlta = 0.0
float pontuacaoBaixa = 0.0

if direcaoCandle > 0
    pontuacaoAlta += 2.0
else if direcaoCandle < 0
    pontuacaoBaixa += 2.0

if probAlta > probBaixa
    pontuacaoAlta += probAlta * 3.0
else if probBaixa > probAlta
    pontuacaoBaixa += probBaixa * 3.0

if kvoEstado > 0
    pontuacaoAlta += 2.0
else if kvoEstado < 0
    pontuacaoBaixa += 2.0

if retorno1 > 0
    pontuacaoAlta += 1.0
else if retorno1 < 0
    pontuacaoBaixa += 1.0

if retorno3 > 0
    pontuacaoAlta += 1.0
else if retorno3 < 0
    pontuacaoBaixa += 1.0

if close > mm5
    pontuacaoAlta += 1.0
else if close < mm5
    pontuacaoBaixa += 1.0

if close > mm10
    pontuacaoAlta += 1.0
else if close < mm10
    pontuacaoBaixa += 1.0

if close > mm20
    pontuacaoAlta += 1.0
else if close < mm20
    pontuacaoBaixa += 1.0


// ============================================================================
// DIREÇÃO PREDOMINANTE
// ============================================================================

direcaoIntensidade =
     pontuacaoAlta > pontuacaoBaixa ?
     1 :
     pontuacaoBaixa > pontuacaoAlta ?
     -1 :
     0

direcaoIntensidadeTexto =
     direcaoIntensidade > 0 ?
     "ALTA" :
     direcaoIntensidade < 0 ?
     "BAIXA" :
     "NEUTRA"


// ============================================================================
// CONVICÇÃO DIRECIONAL
// ============================================================================

pontuacaoTotal = pontuacaoAlta + pontuacaoBaixa

diferencaDirecional =
     math.abs(
         pontuacaoAlta - pontuacaoBaixa
     )

confiancaDirecional =
     pontuacaoTotal > 0 ?
     diferencaDirecional / pontuacaoTotal * 100.0 :
     0.0


// ============================================================================
// BOOST CONFLUÊNCIA
// ============================================================================

float boostConfluenciaBruto = 0.0

if confirmacoes == 0
    boostConfluenciaBruto := 0.0
else if confirmacoes == 1
    boostConfluenciaBruto := 2.0
else if confirmacoes == 2
    boostConfluenciaBruto := 5.0
else if confirmacoes == 3
    boostConfluenciaBruto := 8.0
else if confirmacoes == 4
    boostConfluenciaBruto := 12.0
else if confirmacoes == 5
    boostConfluenciaBruto := 16.0
else
    boostConfluenciaBruto := boostMaximo


// ============================================================================
// FATOR MODELO
// ============================================================================

float fatorModeloBoost = 0.35

if forcaModelo >= 70
    fatorModeloBoost := 1.00
else if forcaModelo >= 65
    fatorModeloBoost := 0.90
else if forcaModelo >= 60
    fatorModeloBoost := 0.75
else if forcaModelo >= 55
    fatorModeloBoost := 0.55


// ============================================================================
// FATOR KVO
// ============================================================================

float fatorKvoBoost = 0.50

if kvoForca >= 60
    fatorKvoBoost := 1.00
else if kvoForca >= 40
    fatorKvoBoost := 0.90
else if kvoForca >= 20
    fatorKvoBoost := 0.75


// ============================================================================
// BOOST AJUSTADO
// ============================================================================

boostConfluenciaAjustado =
     boostConfluenciaBruto *
     fatorModeloBoost *
     fatorKvoBoost

boostConfluenciaAjustado :=
     math.min(
         boostConfluenciaAjustado,
         boostMaximo
     )


// ============================================================================
// BOOST VOLUME
// ============================================================================

float boostVolume = 0.0

if volumeRelativo >= 1.20
    boostVolume += 2.0

if volumeRelativo >= 1.50
    boostVolume += 2.0

if volumeRelativo >= 2.00
    boostVolume += 2.0

boostVolume :=
     math.min(
         boostVolume,
         6.0
     )


// ============================================================================
// BOOST KVO
// ============================================================================

float boostKvo = 0.0

if direcaoCandle != 0 and kvoEstado == direcaoCandle
    if kvoForca >= 60
        boostKvo := 5.0
    else if kvoForca >= 40
        boostKvo := 3.0
    else if kvoForca >= 20
        boostKvo := 1.0


// ============================================================================
// BOOST TOTAL
// ============================================================================

boostTotal =
     boostConfluenciaAjustado +
     boostVolume +
     boostKvo

boostTotal :=
     math.min(
         boostTotal,
         boostMaximo
     )


// ============================================================================
// CLASSIFICAÇÃO DO BOOST
// ============================================================================

string qualidadeBoost = "BAIXA"

if fatorModeloBoost >= 0.90 and fatorKvoBoost >= 0.90
    qualidadeBoost := "ALTA"
else if fatorModeloBoost >= 0.75 and fatorKvoBoost >= 0.75
    qualidadeBoost := "BOA"
else if fatorModeloBoost >= 0.55 or fatorKvoBoost >= 0.75
    qualidadeBoost := "MODERADA"


// ============================================================================
// COR DO BOOST
// ============================================================================

corBoostResultado =
     qualidadeBoost == "ALTA" ?
     corVerde :
     qualidadeBoost == "MODERADA" ?
     corCiano :
     corCinza


// ============================================================================
// PENALTY
// ============================================================================

float penalty = 0.0

penalty += divergencias * 2.0

if direcaoCandle != 0 and kvoEstado != 0 and kvoEstado != direcaoCandle
    penalty += 5.0

if direcaoCandle != 0 and direcaoModelo != 0 and direcaoModelo != direcaoCandle
    penalty += 5.0

penalty :=
     math.min(
         penalty,
         penaltyMaximo
     )


// ============================================================================
// PENALTY EXTRA
// ============================================================================

float penaltyConfluenciaExtra = 0.0

if confirmacoes == 0
    penaltyConfluenciaExtra := 8.0
else if confirmacoes == 1 and divergencias >= 4
    penaltyConfluenciaExtra := 7.0
else if confirmacoes == 2 and divergencias >= 4
    penaltyConfluenciaExtra := 5.0
else if confirmacoes == 3 and divergencias >= 4
    penaltyConfluenciaExtra := 3.0

penalty += penaltyConfluenciaExtra

penalty :=
     math.min(
         penalty,
         penaltyMaximo
     )


// ============================================================================
// INTENSIDADE FINAL
// ============================================================================

intensidadeFinal =
     intensidadeBase +
     boostTotal -
     penalty

intensidadeFinal :=
     math.min(
         math.max(intensidadeFinal, 0.0),
         100.0
     )


// ============================================================================
// INTENSIDADE ASSINADA
// ============================================================================

intensidadeAssinada =
     direcaoIntensidade > 0 ?
     intensidadeFinal :
     direcaoIntensidade < 0 ?
     -intensidadeFinal :
     0.0


// ============================================================================
// CLASSIFICAÇÃO
// ============================================================================

string classificacao = "M FRACA"

if intensidadeFinal >= 90
    classificacao := "EXTREMA"
else if intensidadeFinal >= 80
    classificacao := "M FORTE"
else if intensidadeFinal >= 70
    classificacao := "FORTE"
else if intensidadeFinal >= 60
    classificacao := "MODERADA"
else if intensidadeFinal >= 40
    classificacao := "FRACA"


// ============================================================================
// QUALIDADE CONFLUÊNCIA
// ============================================================================

string qualidadeConfluencia = "FRACA"

if confirmacoes >= 7 and divergencias == 0
    qualidadeConfluencia := "EXTREMA"
else if confirmacoes >= 6 and divergencias <= 1
    qualidadeConfluencia := "M ALTA"
else if confirmacoes >= 5 and divergencias <= 2
    qualidadeConfluencia := "ALTA"
else if confirmacoes >= 4 and divergencias <= 3
    qualidadeConfluencia := "BOA"
else if confirmacoes >= 3 and divergencias <= 4
    qualidadeConfluencia := "MODERADA"
else if confirmacoes <= 1 and divergencias >= 4
    qualidadeConfluencia := "M FRACA"


// ============================================================================
// NÍVEL MODELO
// ============================================================================

string nivelModelo = "FRACO"

if forcaModelo >= 80
    nivelModelo := "M FORTE"
else if forcaModelo >= 70
    nivelModelo := "FORTE"
else if forcaModelo >= 60
    nivelModelo := "MODERADO"


// ============================================================================
// BUY / SELL
// ============================================================================

string vencedor = "EMPATE"

if probAlta > probBaixa
    vencedor := "BUY"
else if probBaixa > probAlta
    vencedor := "SELL"


// ============================================================================
// CORES
// ============================================================================

corDirecao =
     direcaoIntensidade > 0 ?
     corVerde :
     direcaoIntensidade < 0 ?
     corVermelho :
     corCinza

corModelo =
     direcaoModelo > 0 ?
     corVerde :
     direcaoModelo < 0 ?
     corVermelho :
     corCinza

corVencedor =
     vencedor == "BUY" ?
     corVerde :
     vencedor == "SELL" ?
     corVermelho :
     corCinza

corClassificacao =
     intensidadeFinal >= 90 ?
     corVerdeForte :
     intensidadeFinal >= 70 ?
     corVerde :
     intensidadeFinal >= 60 ?
     corAmarelo :
     intensidadeFinal >= 40 ?
     corAmarelo :
     corCinza


// ============================================================================
// COR DA CONFLUÊNCIA
// EXTREMA = CINZA
// ============================================================================

corConfluencia =
     qualidadeConfluencia == "EXTREMA" ?
     corCinza :
     qualidadeConfluencia == "M ALTA" ?
     corVerde :
     qualidadeConfluencia == "ALTA" ?
     corVerde :
     qualidadeConfluencia == "BOA" ?
     corAmarelo :
     qualidadeConfluencia == "MODERADA" ?
     corAmarelo :
     corVermelho


// ============================================================================
// COR DA CONFLUÊNCIA NO RESULTADO
// ============================================================================

corConfResultado =
     qualidadeConfluencia == "ALTA" or
     qualidadeConfluencia == "M ALTA" ?
     corVerde :
     corCinza


// ============================================================================
// COR DO VOLUME
// AZUL SOMENTE QUANDO > 0.35
// ============================================================================

corVolume =
     volumeRelativo > 0.35 ?
     corCiano :
     corCinza


// ============================================================================
// COR DO CANDLE
// ============================================================================

corResultadoCandle =
     candleForte ?
     direcaoCandle > 0 ?
     corVerde :
     direcaoCandle < 0 ?
     corVermelho :
     corCinza :
     corCinza


// ============================================================================
// MTF - FUNÇÃO DE DIREÇÃO
// ============================================================================

f_direcaoMTF() =>
    close > open ? 1 : close < open ? -1 : 0


// ============================================================================
// MTF
// ============================================================================

mtf1 = request.security(
     syminfo.tickerid,
     tf1,
     f_direcaoMTF(),
     lookahead=barmerge.lookahead_off
)

mtf2 = request.security(
     syminfo.tickerid,
     tf2,
     f_direcaoMTF(),
     lookahead=barmerge.lookahead_off
)

mtf3 = request.security(
     syminfo.tickerid,
     tf3,
     f_direcaoMTF(),
     lookahead=barmerge.lookahead_off
)

mtf4 = request.security(
     syminfo.tickerid,
     tf4,
     f_direcaoMTF(),
     lookahead=barmerge.lookahead_off
)


// ============================================================================
// CONTAGEM MTF
// ============================================================================

int mtfAlta = 0
int mtfBaixa = 0
int mtfAtivos = 0

if usarMTF1
    mtfAtivos += 1
    if mtf1 > 0
        mtfAlta += 1
    else if mtf1 < 0
        mtfBaixa += 1

if usarMTF2
    mtfAtivos += 1
    if mtf2 > 0
        mtfAlta += 1
    else if mtf2 < 0
        mtfBaixa += 1

if usarMTF3
    mtfAtivos += 1
    if mtf3 > 0
        mtfAlta += 1
    else if mtf3 < 0
        mtfBaixa += 1

if usarMTF4
    mtfAtivos += 1
    if mtf4 > 0
        mtfAlta += 1
    else if mtf4 < 0
        mtfBaixa += 1


// ============================================================================
// DIREÇÃO MTF
// ============================================================================

direcaoMTF =
     mtfAlta > mtfBaixa ?
     1 :
     mtfBaixa > mtfAlta ?
     -1 :
     0

mtfConcordante =
     direcaoIntensidade != 0 and
     direcaoMTF == direcaoIntensidade

mtfPercentual =
     mtfAtivos > 0 ?
     math.max(mtfAlta, mtfBaixa) / mtfAtivos * 100.0 :
     0.0

mtfTexto =
     direcaoMTF > 0 ?
     "↑ COMPRA" :
     direcaoMTF < 0 ?
     "↓ VENDA" :
     "→ NEUTRO"

corMTF =
     direcaoMTF > 0 ?
     corVerde :
     direcaoMTF < 0 ?
     corVermelho :
     corCinza


// ============================================================================
// CLASSIFICAÇÃO MTF
// ============================================================================

string classificacaoMTF = "NEUTRO"

if mtfAtivos > 0
    if mtfPercentual >= 75
        classificacaoMTF := "FORTE"
    else if mtfPercentual >= 50
        classificacaoMTF := "MODERADO"
    else
        classificacaoMTF := "FRACO"


// ============================================================================
// QUADRADO DE CONFIRMAÇÃO
// ============================================================================

criterioCandleCompra =
     candleForte and
     direcaoCandle > 0

criterioCandleVenda =
     candleForte and
     direcaoCandle < 0

criterioIntensidadeCompra =
     intensidadeFinal >= 70 and
     direcaoIntensidade > 0

criterioIntensidadeVenda =
     intensidadeFinal >= 70 and
     direcaoIntensidade < 0

criterioMTFCompra =
     mtfAtivos > 0 and
     direcaoMTF > 0 and
     mtfConcordante

criterioMTFVenda =
     mtfAtivos > 0 and
     direcaoMTF < 0 and
     mtfConcordante

criterioATRCompra = atrBoaCompra
criterioATRVenda = atrBoaVenda

criterioConfCompra =
     qualidadeConfluencia == "ALTA" or
     qualidadeConfluencia == "M ALTA" or
     qualidadeConfluencia == "EXTREMA"

criterioConfVenda =
     qualidadeConfluencia == "ALTA" or
     qualidadeConfluencia == "M ALTA" or
     qualidadeConfluencia == "EXTREMA"

criterioBoostCompra =
     qualidadeBoost == "ALTA" or
     qualidadeBoost == "MODERADA"

criterioBoostVenda =
     qualidadeBoost == "ALTA" or
     qualidadeBoost == "MODERADA"

criterioBuy =
     vencedor == "BUY"

criterioSell =
     vencedor == "SELL"


// ============================================================================
// SINAL FINAL DO QUADRADO
// ============================================================================

sinalCompra =
     criterioCandleCompra and
     criterioIntensidadeCompra and
     criterioMTFCompra and
     criterioATRCompra and
     criterioConfCompra and
     criterioBoostCompra and
     criterioBuy

sinalVenda =
     criterioCandleVenda and
     criterioIntensidadeVenda and
     criterioMTFVenda and
     criterioATRVenda and
     criterioConfVenda and
     criterioBoostVenda and
     criterioSell


// ============================================================================
// TEXTO DO QUADRADO
// ============================================================================

textoQuadrado =
     sinalCompra ?
     "■ COMPRA" :
     sinalVenda ?
     "■ VENDA" :
     "■"

corQuadrado =
     sinalCompra ?
     corVerde :
     sinalVenda ?
     corVermelho :
     corCinza


// ============================================================================
// PICO V3 - CLASSIFICAÇÃO CONTÍNUA
// ============================================================================
// O PICO nunca fica "SEM PICO".
// A classificação é sempre direcionada para ALTA ou VENDA.
//
// Níveis:
// FORMAÇÃO = início de concentração de força
// MODERADO = força relevante
// FORTE = pico de força
// EXTREMO = pico excepcional
//
// O topo/fundo anterior funciona como reforço e não como requisito.
// ============================================================================

groupPico = "KVO PRO - PICO V3"

picoLookback = input.int(
     10,
     "Lookback do pico",
     minval=3,
     maxval=100,
     group=groupPico
)

picoMinimoModerado = input.float(
     65.0,
     "Score mínimo PICO MODERADO",
     minval=40.0,
     maxval=100.0,
     step=1.0,
     group=groupPico
)

picoMinimoForte = input.float(
     76.0,
     "Score mínimo PICO FORTE",
     minval=50.0,
     maxval=100.0,
     step=1.0,
     group=groupPico
)

picoMinimoExtremo = input.float(
     90.0,
     "Score mínimo PICO EXTREMO",
     minval=60.0,
     maxval=100.0,
     step=1.0,
     group=groupPico
)


// ============================================================================
// ACELERAÇÃO
// ============================================================================

aceleracaoIntensidade =
     not na(intensidadeFinal[1]) ?
     intensidadeFinal - intensidadeFinal[1] :
     0.0

aceleracaoKvo =
     not na(kvoForca[1]) ?
     kvoForca - kvoForca[1] :
     0.0


// ============================================================================
// ESTRUTURA DO PREÇO
// ============================================================================

novoTopoPico =
     not na(ta.highest(high[1], picoLookback)) and
     high >= ta.highest(high[1], picoLookback)

novoFundoPico =
     not na(ta.lowest(low[1], picoLookback)) and
     low <= ta.lowest(low[1], picoLookback)


// ============================================================================
// DISTÂNCIA DA ESTRUTURA
// ============================================================================

distTopoPico =
     not na(ta.highest(high[1], picoLookback)) ?
     math.max(
         0.0,
         (high - ta.highest(high[1], picoLookback)) /
         atrSeguro
     ) :
     0.0

distFundoPico =
     not na(ta.lowest(low[1], picoLookback)) ?
     math.max(
         0.0,
         (ta.lowest(low[1], picoLookback) - low) /
         atrSeguro
     ) :
     0.0


// ============================================================================
// SCORE DE INTENSIDADE
// ============================================================================

picoScoreIntensidadeAlta =
     intensidadeFinal >= 95 ?
     100.0 :
     intensidadeFinal >= 90 ?
     95.0 :
     intensidadeFinal >= 80 ?
     85.0 :
     intensidadeFinal >= 70 ?
     75.0 :
     intensidadeFinal >= 60 ?
     65.0 :
     intensidadeFinal

picoScoreIntensidadeVenda =
     picoScoreIntensidadeAlta


// ============================================================================
// SCORE KVO
// ============================================================================

picoScoreKvoAlta =
     kvoEstado > 0 ?
     kvoForca :
     kvoEstado == 0 ?
     kvoForca * 0.35 :
     0.0

picoScoreKvoVenda =
     kvoEstado < 0 ?
     kvoForca :
     kvoEstado == 0 ?
     kvoForca * 0.35 :
     0.0


// ============================================================================
// SCORE VOLUME
// ============================================================================

picoScoreVolume =
     volumeRelativo >= 2.50 ?
     100.0 :
     volumeRelativo >= 2.00 ?
     95.0 :
     volumeRelativo >= 1.50 ?
     85.0 :
     volumeRelativo >= 1.20 ?
     70.0 :
     volumeRelativo >= 1.00 ?
     55.0 :
     35.0


// ============================================================================
// SCORE CANDLE
// ============================================================================

     direcaoCandle > 0 ?
     forcaCorpo :
     direcaoCandle == 0 ?
     forcaCorpo * 0.35 :
     0.0

picoScoreCandleVenda =
     direcaoCandle < 0 ?
     forcaCorpo :
     direcaoCandle == 0 ?
     forcaCorpo * 0.35 :
     0.0


// ============================================================================
// SCORE ATR
// ============================================================================

     atrForte ?
     100.0 :
     60.0


// ============================================================================
// SCORE ACELERAÇÃO
// ============================================================================

     math.min(
         math.max(
             50.0 +
             aceleracaoIntensidade * 5.0 +
             aceleracaoKvo * 0.50,
             0.0
         ),
         100.0
     )


// ============================================================================
// SCORE MTF
// ============================================================================

picoScoreMtfAlta =
     mtfAtivos > 0 and direcaoMTF > 0 ?
     mtfPercentual :
     mtfAtivos > 0 and direcaoMTF == 0 ?
     mtfPercentual * 0.35 :
     35.0

picoScoreMtfVenda =
     mtfAtivos > 0 and direcaoMTF < 0 ?
     mtfPercentual :
     mtfAtivos > 0 and direcaoMTF == 0 ?
     mtfPercentual * 0.35 :
     35.0


// ============================================================================
// BÔNUS DE ROMPIMENTO
// ============================================================================

     novoTopoPico ?
     math.min(
         10.0 + distTopoPico * 5.0,
         15.0
     ) :
     0.0

bonusFundoPico =
     novoFundoPico ?
     math.min(
         10.0 + distFundoPico * 5.0,
         15.0
     ) :
     0.0


// ============================================================================
// SCORE FINAL ALTA
// ============================================================================

picoScoreAlta =
     picoScoreIntensidadeAlta * 0.25 +
     picoScoreKvoAlta * 0.20 +
     picoScoreVolume * 0.15 +
     picoScoreCandleAlta * 0.15 +
     picoScoreAtr * 0.10 +
     picoScoreAceleracao * 0.10 +
     picoScoreMtfAlta * 0.05 +
     bonusTopoPico

     math.min(
         math.max(
             picoScoreAlta,
             0.0
         ),
         100.0
     )

// SCORE FINAL VENDA

     picoScoreIntensidadeVenda * 0.25 +
     picoScoreKvoVenda * 0.20 +
     picoScoreVolume * 0.15 +
     picoScoreCandleVenda * 0.15 +
     picoScoreAtr * 0.10 +
     picoScoreAceleracao * 0.10 +
     picoScoreMtfVenda * 0.05 +
     bonusFundoPico

     math.min(
         math.max(
             picoScoreVenda,
             0.0
         ),
         100.0
     )

     not picoExtremoAlta and
     not picoForteAlta and
     not picoModeradoAlta

     not picoModeradoVenda

         exaustaoAlta ?
         "PICO FORTE ALTA | EXAUSTÃO" :
         "PICO FORTE ALTA"

    classificacaoPico :=
         exaustaoVenda ?
         "PICO FORTE VENDA | EXAUSTÃO" :
         "PICO FORTE VENDA"

    classificacaoPico :=
         exaustaoAlta ?
         "PICO MODERADO ALTA | EXAUSTÃO" :
         "PICO MODERADO ALTA"

else if picoModeradoVenda
    classificacaoPico :=
         exaustaoVenda ?

    resultado = ""

    for i = 0 to tamanho - 1
        resultado += i < preenchido ? "█" : "░"

    resultado
