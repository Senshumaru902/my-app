f_suavizar(src, periodo) =>
    e1 = ta.ema(src, periodo)
    e2 = ta.ema(e1, periodo)
    e3 = ta.ema(e2, periodo)
    (e1 + e2 + e3) / 3.0

f_clamp(v, minimo, maximo) =>
    math.min(math.max(v, minimo), maximo)

// ============================================================================
// DADOS DO CANDLE
// ============================================================================

rangeCandle = math.max(high - low, syminfo.mintick)
corpo = math.abs(close - open)
corpoPct = corpo / rangeCandle
pavioSuperior = high - math.max(open, close)
pavioInferior = math.min(open, close) - low
direcaoCandle = close > open ? 1 : close < open ? -1 : 0
candleForte = corpoPct >= corpoMinimo
candleAltaForte = candleForte and direcaoCandle > 0
forcaCorpo = f_clamp(corpoPct * 100.0, 0.0, 100.0)

// ============================================================================
// RETORNOS / MÉDIAS / VOLUME
// ============================================================================

retorno1 = close[1] != 0 ? (close / close[1] - 1.0) * 100.0 : 0.0
retorno3 = close[3] != 0 ? (close / close[3] - 1.0) * 100.0 : 0.0
retorno5 = close[5] != 0 ? (close / close[5] - 1.0) * 100.0 : 0.0

mm5 = ta.ema(close, 5)
mm10 = ta.ema(close, 10)
mm20 = ta.ema(close, 20)
mm50 = ta.ema(close, 50)
mm100 = ta.ema(close, 100)

volumeMedio = ta.sma(volume, 20)
volumeRelativo = not na(volumeMedio) and volumeMedio > 0 ? volume / volumeMedio : 1.0
volumeSuave = f_suavizar(volumeRelativo, suavizacaoPeriodo)

// ============================================================================

atrBruto = ta.atr(atrPeriodo)
atrSuave = ta.ema(atrBruto, atrSuavePeriodo)
atrMediaRegime = ta.sma(atrSuave, atrRegimePeriodo)
atrSeguro = math.max(nz(atrSuave, atrBruto), syminfo.mintick)
atrRegime = not na(atrMediaRegime) and atrMediaRegime > 0 ? atrSuave / atrMediaRegime : 1.0
atrExpansaoBase = atrSuave[1] != 0 ? atrSuave / atrSuave[1] : 1.0
atrExpansao = f_suavizar(atrExpansaoBase, suavizacaoPeriodo)
atrNormalizado = close != 0 ? atrSeguro / close * 100.0 : 0.0
atrMuitoForte = atrRegime >= 1.35 and atrExpansao >= 1.03
atrForte = atrRegime >= 1.10 and atrExpansao >= 1.00
atrFraco = atrRegime <= 0.80
atrBoaCompra = atrForte or (atrRegime >= 0.90 and atrExpansao >= 1.00)
atrScore = atrMuitoForte ? 100.0 : atrForte ? 82.0 : atrBoaCompra ? 68.0 : atrFraco ? 38.0 : 55.0
atrClassificacao = atrMuitoForte ? "MUITO FORTE" : atrForte ? "FORTE" : atrBoaCompra ? "BOA" : atrFraco ? "FRACA" : "NORMAL"
corAtr = atrScore >= 80 ? corVerde : atrScore >= 60 ? corCiano : corCinza

resistencia20 = ta.highest(high[1], estruturaLookback)

resistencia5 = ta.highest(high[1], confirmacaoLookback)

distanciaResistenciaAtual = not na(resistencia20) and resistencia20 > close ? (resistencia20 - close) / atrSeguro : 999.0
resistenciaDisponivel = not na(resistencia20) and resistencia20 > close
resistenciaProxima = resistenciaDisponivel and distanciaResistenciaAtual < distanciaMinimaResistenciaATR
filtroResistenciaLong = not usarFiltroResistencia or not resistenciaProxima

// ============================================================================
// KVO SUAVIZADO
// ============================================================================

precoTipico = (high + low + close) / 3.0
tendenciaKvo = precoTipico > precoTipico[1] ? 1 : precoTipico < precoTipico[1] ? -1 : 0
dmKvo = math.max(high - low, syminfo.mintick)

var float cmKvo = 0.0
if bar_index == 0
    cmKvo := dmKvo
else
    cmKvo := tendenciaKvo == tendenciaKvo[1] ? cmKvo[1] + dmKvo : dmKvo + dmKvo[1]

cmSeguro = math.max(math.abs(cmKvo), syminfo.mintick)
vfKvo = volume * math.abs(2.0 * (dmKvo / cmSeguro) - 1.0) * tendenciaKvo * 100.0
kvoBruto = ta.ema(vfKvo, kvoFast) - ta.ema(vfKvo, kvoSlow)

kvoEstadoAlta = kvo > kvoSignal and kvoVariacao > 0
kvoForcaBruta = math.abs(kvoHist) / (math.abs(kvo) + math.abs(kvoSignal) + 0.000000001)
kvoForca = f_clamp(kvoForcaBruta, 0.0, 1.0) * 100.0
kvoScoreLong = kvoEstadoAlta ? 50.0 + math.min(kvoForca * 0.50, 50.0) : 50.0
kvoScoreLong := f_clamp(kvoScoreLong, 0.0, 100.0)

microTendenciaAlta = microEmaFast > microEmaSlow and microSlope > 0
microTendenciaFormacao = microEmaFast > microEmaSlow or microSlope > 0
microTrendScore = microTendenciaAlta ? 100.0 : microTendenciaFormacao ? 68.0 : 45.0

if microTendenciaAlta
    microScoreAlta += 25.0
else if microTendenciaFormacao

if microVolumeConfirmado and microRompimentoAlta

if microATRConfirmado

microLongConfirmado = microRompimentoAlta and microScoreAlta >= microScoreMinimo
microFormacaoAlta = microRompimentoAlta or microTendenciaAlta
microTexto = microLongConfirmado ? "MICRO LONG" : microRompimentoAlta ? "MICRO ROMPIMENTO" : microTendenciaAlta ? "MICRO TENDÊNCIA" : microTendenciaFormacao ? "MICRO FORMAÇÃO" : "AGUARDANDO"
corMicro = microLongConfirmado ? corVerdeForte : microFormacaoAlta ? corVerde : corCinza

if close > mm20 and mm20 > mm50

confluencia := f_clamp(confluencia, 0.0, 100.0)
qualidadeConfluencia = confluencia >= 85 ? "EXTREMA" : confluencia >= 75 ? "MUITO ALTA" : confluencia >= 65 ? "ALTA" : confluencia >= 50 ? "BOA" : confluencia >= 35 ? "MODERADA" : "FRACA"

// ============================================================================
// MTF - TENDÊNCIA + RSI + MACD + ATR + VOLUME
// ============================================================================

f_mtfScoreLong() =>
    e5 = ta.ema(close, 5)
    e20 = ta.ema(close, 20)
    e50 = ta.ema(close, 50)
    rsi = ta.rsi(close, 14)
    [macdLine, macdSignal, macdHist] = ta.macd(close, 12, 26, 9)
    atrLocal = ta.atr(14)
    atrBase = ta.sma(atrLocal, 50)
    volBase = ta.sma(volume, 20)
    float s = 0.0
    s += close > e5 ? 0.18 : 0.0
    s += e5 > e20 ? 0.22 : 0.0
    s += e20 > e50 ? 0.22 : 0.0
    s += rsi >= 55 ? 0.15 : rsi >= 50 ? 0.07 : 0.0
    s += macdHist > 0 ? 0.15 : 0.0
    s += not na(atrBase) and atrLocal >= atrBase ? 0.04 : 0.0
    s += not na(volBase) and volume >= volBase and close > open ? 0.04 : 0.0
    f_clamp(s, 0.0, 1.0)

mtf1mRaw = request.security(syminfo.tickerid, "1", f_mtfScoreLong(), lookahead=barmerge.lookahead_off)
mtf15mRaw = request.security(syminfo.tickerid, "15", f_mtfScoreLong(), lookahead=barmerge.lookahead_off)
mtf30mRaw = request.security(syminfo.tickerid, "30", f_mtfScoreLong(), lookahead=barmerge.lookahead_off)
mtf4hRaw = request.security(syminfo.tickerid, "240", f_mtfScoreLong(), lookahead=barmerge.lookahead_off)
mtfDailyRaw = request.security(syminfo.tickerid, "1D", f_mtfScoreLong(), lookahead=barmerge.lookahead_off)

mtf1m = f_suavizar(mtf1mRaw, suavizacaoPeriodo)
mtf15m = f_suavizar(mtf15mRaw, suavizacaoPeriodo)
mtf30m = f_suavizar(mtf30mRaw, suavizacaoPeriodo)
mtf4h = f_suavizar(mtf4hRaw, suavizacaoPeriodo)
mtfDaily = f_suavizar(mtfDailyRaw, suavizacaoPeriodo)

prob1m = f_clamp(mtf1m * 100.0, 0.0, 100.0)
prob15m = f_clamp(mtf15m * 100.0, 0.0, 100.0)
prob30m = f_clamp(mtf30m * 100.0, 0.0, 100.0)
prob4h = f_clamp(mtf4h * 100.0, 0.0, 100.0)
probDaily = f_clamp(mtfDaily * 100.0, 0.0, 100.0)

float mtfSoma = 0.0
float mtfPeso = 0.0
if usarMTF1
    mtfSoma += mtf1m * 0.10
    mtfPeso += 0.10
if usarMTF15
    mtfSoma += mtf15m * 0.15
    mtfPeso += 0.15
if usarMTF30
    mtfSoma += mtf30m * 0.15
    mtfPeso += 0.15
if usarMTF240
    mtfSoma += mtf4h * 0.25
    mtfPeso += 0.25
if usarMTFD
    mtfSoma += mtfDaily * 0.35
    mtfPeso += 0.35

mtfScoreLong = mtfPeso > 0 ? mtfSoma / mtfPeso : 0.0
mtfPercentualLong = f_clamp(mtfScoreLong * 100.0, 0.0, 100.0)
mtfConcordanteLong = mtfPercentualLong >= longMinimoMTF
mtfTexto = mtfConcordanteLong ? "MTF LONG" : "MTF EM FORMAÇÃO"
corMTF = mtfConcordanteLong ? corVerde : corCinza

// ============================================================================
// INTENSIDADE LONG
// ============================================================================

intensidadeBase = forcaCorpo * 0.15 + probModeloAltaSuave * 0.25 + kvoScoreLong * 0.15 + mtfPercentualLong * 0.25 + confluencia * 0.10 + atrScore * 0.10

boostMicro = microLongConfirmado ? 8.0 : microTendenciaAlta ? 4.0 : 0.0
boostVolume = volumeSuave >= 1.50 and direcaoCandle > 0 ? 4.0 : volumeSuave >= 1.20 and direcaoCandle > 0 ? 2.0 : 0.0
boostHarmonia = mtfPercentualLong >= 75 and kvoEstadoAlta and microTendenciaAlta ? 5.0 : 0.0
penaltyExaustao = close > mm20 and corpoPct < 0.35 and pavioSuperior > corpo ? 8.0 : 0.0

intensidadeFinal = f_clamp(intensidadeBase + boostMicro + boostVolume + boostHarmonia - penaltyExaustao, 0.0, 100.0)
intensidadeSuave = f_suavizar(intensidadeFinal, suavizacaoPeriodo)
classificacao = intensidadeSuave >= 90 ? "EXTREMA" : intensidadeSuave >= 80 ? "MUITO FORTE" : intensidadeSuave >= 70 ? "FORTE" : intensidadeSuave >= 60 ? "MODERADA" : intensidadeSuave >= 40 ? "FRACA" : "MUITO FRACA"
corClassificacao = intensidadeSuave >= 80 ? corVerdeForte : intensidadeSuave >= 70 ? corVerde : intensidadeSuave >= 60 ? corCiano : intensidadeSuave >= 40 ? corAmarelo : corCinza

// ============================================================================
// PICO LONG / IMPULSO LONG
// ============================================================================

novoTopo = not na(ta.highest(high[1], estruturaLookback)) and high > ta.highest(high[1], estruturaLookback)
impulsoAlta = retorno1 > 0 and retorno3 > 0 and close > mm20
exaustaoAlta = close > mm20 and corpoPct < 0.35 and pavioSuperior > corpo

float picoScoreAlta = 0.0
if novoTopo
    picoScoreAlta += 30.0
if impulsoAlta
    picoScoreAlta += 25.0
if volumeRelativo >= 1.20 and direcaoCandle > 0
    picoScoreAlta += 15.0
if kvoEstadoAlta
    picoScoreAlta += 15.0
if atrBoaCompra and direcaoCandle > 0
    picoScoreAlta += 15.0
if microTendenciaAlta
    picoScoreAlta += 10.0
picoScoreAlta := f_clamp(picoScoreAlta, 0.0, 100.0)

picoExtremoAlta = picoScoreAlta >= 85 and not exaustaoAlta
picoForteAlta = picoScoreAlta >= 70 and not picoExtremoAlta
picoModeradoAlta = picoScoreAlta >= 55 and not picoForteAlta and not picoExtremoAlta
classificacaoPico = picoExtremoAlta ? "PICO EXTREMO LONG" : picoForteAlta ? "PICO FORTE LONG" : picoModeradoAlta ? "PICO MODERADO LONG" : "PICO EM FORMAÇÃO"
corPico = picoExtremoAlta ? corVerdeForte : picoForteAlta ? corVerde : picoModeradoAlta ? corCiano : corCinza

// ============================================================================
// PREVISÃO PROBABILÍSTICA LONG
// ============================================================================

// Pesos normalizados: 20% local + 35% MTF + 15% micro + 10% KVO +
// 8% ATR + 12% confluência.
probAtrLong = atrMuitoForte ? 1.00 : atrForte ? 0.88 : atrBoaCompra ? 0.72 : atrFraco ? 0.35 : 0.55
probMicroLong = microScoreAlta / 100.0
probMtfLong = mtfPercentualLong / 100.0
probKvoLong = kvoScoreLong / 100.0
probConfLong = confluencia / 100.0
probLocalLong = probModeloAltaSuave / 100.0

probabilidadeBaseLong = (probLocalLong * 0.20 + probMtfLong * 0.35 + probMicroLong * 0.15 + probKvoLong * 0.10 + probAtrLong * 0.08 + probConfLong * 0.12) * 100.0

// Harmonização: média de EMA, EMA da EMA e EMA da EMA da EMA.
probabilidadeLong = f_suavizar(probabilidadeBaseLong, suavizacaoPeriodo)

// Bônus somente quando os componentes bullish estão simultaneamente alinhados.
if microLongConfirmado and mtfPercentualLong >= 70 and kvoEstadoAlta
    probabilidadeLong += 4.0
if picoForteAlta and mtfPercentualLong >= 70
    probabilidadeLong += 2.0
if exaustaoAlta
    probabilidadeLong -= 6.0

probabilidadeLong := f_clamp(probabilidadeLong, 0.0, 99.0)
probabilidadeLongSuave = f_suavizar(probabilidadeLong, suavizacaoPeriodo)
probabilidadeLongSuave := f_clamp(probabilidadeLongSuave, 0.0, 99.0)
confiancaLong = probabilidadeLongSuave

// ============================================================================
// SCORE LONG
// ============================================================================

float scoreLong = 0.0
if probModeloAltaSuave >= 60
