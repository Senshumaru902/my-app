//@version=6
strategy(
     "V40.17 CORRIGIDA - PREVISÃO DIÁRIA + PADRÕES GRÁFICOS",
     overlay=true,
     pyramiding=0,
     process_orders_on_close=false,
     calc_on_every_tick=true,
     calc_on_order_fills=false,
     initial_capital=100,
     default_qty_type=strategy.percent_of_equity,
     default_qty_value=1)

// ============================================================================
// V40.17 CORRIGIDA
// PREVISÃO DIÁRIA + PADRÕES GRÁFICOS
//
// PRINCÍPIOS:
// 1. Motor principal calculado no DIÁRIO.
// 2. Semanal utilizado como confirmação/filtro.
// 3. O gráfico pode estar em qualquer timeframe.
// 4. Sinais NÃO são desenhados no gráfico.
// 5. Informações principais ficam no painel MODO FÁCIL.
// 6. Strategy Tester continua funcionando.
// 7. Duração da operação é contabilizada em DIAS, e não em candles do gráfico.
// ============================================================================


// ============================================================================
// FUNÇÕES BÁSICAS
// ============================================================================

f_clamp(float x, float lo, float hi) =>
    math.max(lo, math.min(hi, x))

f_pts(bool cond, float peso) =>
    cond ? peso : 0.0


// ============================================================================
// GRUPOS DE INPUTS
// ============================================================================

G_MODO   = "V40.17 - MODO"
G_PADROES = "V40.17 - PADRÕES GRÁFICOS"
G_FILTROS = "V40.17 - FILTROS"
G_EXEC   = "V40.17 - EXECUÇÃO"
G_VISUAL = "V40.17 - VISUAL"
G_PAINEL = "V40.17 - PAINEL"


// ============================================================================
// MODO
// ============================================================================

ativar = input.bool(
     true,
     "Ativar algoritmo",
     group=G_MODO)

modo = input.string(
     "Equilibrado",
     "Modo",
     options=["Conservador", "Equilibrado", "Rápido"],
     group=G_MODO)

periodoSR = input.int(
     20,
     "Período suporte/resistência",
     minval=10,
     maxval=100,
     group=G_MODO)

scoreBase = input.float(
     70.0,
     "Score mínimo base",
     minval=45,
     maxval=95,
     step=0.5,
     group=G_MODO)

vantagemBase = input.float(
     10.0,
     "Vantagem mínima base",
     minval=3,
     maxval=40,
     step=0.5,
     group=G_MODO)


// ============================================================================
// PADRÕES
// ============================================================================

usarPadroes = input.bool(
     true,
     "Usar padrões gráficos",
     group=G_PADROES)

exigirPadrao = input.bool(
     false,
     "Exigir padrão para entrada",
     group=G_PADROES)

usarDuplo = input.bool(
     true,
     "Usar topo/fundo duplo",
     group=G_PADROES)

usarTriangulo = input.bool(
     true,
     "Usar triângulo",
     group=G_PADROES)

     true,
     "Usar canal",

usarCunha = input.bool(
     true,
     "Usar cunha",
     group=G_PADROES)

usarBandeira = input.bool(
     true,
     "Usar bandeira",
     group=G_PADROES)


// ============================================================================
// FILTROS
// ============================================================================

exigirTendencia = input.bool(
     true,
     "Exigir tendência",
     group=G_FILTROS)

exigirMomentum = input.bool(
     true,
     "Exigir momentum",
     group=G_FILTROS)

exigirEstrutura = input.bool(
     true,
     "Exigir estrutura",
     group=G_FILTROS)

usarRompimento = input.bool(
     true,
     "Usar rompimento",
     group=G_FILTROS)

usarReteste = input.bool(
     true,
     "Usar reteste",
     group=G_FILTROS)

usarSemanal = input.bool(
     true,
     "Usar confirmação semanal",
     group=G_FILTROS)

usarVolatilidade = input.bool(
     true,
     "Usar filtro de volatilidade",
     group=G_FILTROS)

executarStrategy = input.bool(

mostrarSinais = input.bool(

mostrarPainel = input.bool(

// ============================================================================
// PERFIL DINÂMICO
// ============================================================================

scoreMin =
     modo == "Conservador"
     ? math.max(scoreBase, 76.0)
     : modo == "Rápido"
     ? math.min(scoreBase, 62.0)
     : scoreBase

vantagemMin =
     modo == "Conservador"
     ? math.max(vantagemBase, 14.0)
     : modo == "Rápido"
     ? math.min(vantagemBase, 7.0)
     : vantagemBase


// ============================================================================
// MOTOR DIÁRIO
// ============================================================================

f_diario() =>

    // ------------------------------------------------------------------------
    // MÉDIAS
    // ------------------------------------------------------------------------

    ema5 = ta.ema(close, 5)
    ema9 = ta.ema(close, 9)
    ema21 = ta.ema(close, 21)
    ema50 = ta.ema(close, 50)

    // ------------------------------------------------------------------------
    // MOMENTUM
    // ------------------------------------------------------------------------

    rsi = ta.rsi(close, 7)

    atr = ta.atr(14)

    [diPlus, diMinus, adx] = ta.dmi(14, 14)

    impulso = atr > 0
         ? (close - close[1]) / atr
         : 0.0

    aceleracao = impulso - impulso[1]

    // ------------------------------------------------------------------------
    // SUPORTE / RESISTÊNCIA
    // ------------------------------------------------------------------------

    suporte = ta.lowest(low, periodoSR)[1]
    resistencia = ta.highest(high, periodoSR)[1]

    distanciaSup =
         atr > 0
         ? (close - suporte) / atr
         : 0.0

    distanciaRes =
         atr > 0
         ? (resistencia - close) / atr
         : 0.0

    // ------------------------------------------------------------------------
    // CANDLE
    // ------------------------------------------------------------------------

    rangeCandle = high - low

    mediaRange = ta.sma(
         high - low,
         20)

    corpo = math.abs(close - open)

    mediaCorpo = ta.sma(
         math.abs(close - open),
         20)

    pavioSuperior =
         high - math.max(open, close)

    pavioInferior =
         math.min(open, close) - low

    candleAlta = close > open
    candleBaixa = close < open

    candleForteAlta =
         candleAlta and
         atr > 0 and
         corpo >= atr * 0.35

    candleForteBaixa =
         candleBaixa and
         atr > 0 and
         corpo >= atr * 0.35

    compressao =
         mediaRange > 0 and
         rangeCandle < mediaRange * 0.80

    volatilidadeBoa =
         mediaRange > 0 and
         rangeCandle >= mediaRange * 0.50 and
         rangeCandle <= mediaRange * 2.50

    // ------------------------------------------------------------------------
    // ROMPIMENTOS
    // ------------------------------------------------------------------------

    rompAlta =
         close > resistencia

    rompBaixa =
         close < suporte

    // ------------------------------------------------------------------------
    // PRÉ-ROMPIMENTO
    // ------------------------------------------------------------------------

    preAlta =
         resistencia > 0 and
         atr > 0 and
         high >= resistencia - atr * 0.35 and
         close > open

    preBaixa =
         suporte > 0 and
         atr > 0 and
         low <= suporte + atr * 0.35 and
         close < open

    // ------------------------------------------------------------------------
    // RETESTE
    // ------------------------------------------------------------------------

    retesteAlta =
         resistencia > 0 and
         low <= resistencia and
         close > resistencia and
         close > open

    retesteBaixa =
         suporte > 0 and
         high >= suporte and
         close < suporte and
         close < open

    // ------------------------------------------------------------------------
    // REJEIÇÃO
    // ------------------------------------------------------------------------

    rejeicaoAlta =
         suporte > 0 and
         atr > 0 and
         low <= suporte + atr * 0.25 and
         close > open and
         pavioInferior >= corpo

    rejeicaoBaixa =
         resistencia > 0 and
         atr > 0 and
         high >= resistencia - atr * 0.25 and
         close < open and
         pavioSuperior >= corpo

    // ------------------------------------------------------------------------
    // PADRÕES DE CANDLE
    // ------------------------------------------------------------------------

    engolfoAlta =
         close > open and
         close[1] < open[1] and
         open <= close[1] and
         close >= open[1]

    engolfoBaixa =
         close < open and
         close[1] > open[1] and
         open >= close[1] and
         close <= open[1]

    martelo =
         pavioInferior > corpo * 2.0 and
         pavioSuperior <= corpo * 0.80 and
         close >= open

    estrelaCadente =
         pavioSuperior > corpo * 2.0 and
         pavioInferior <= corpo * 0.80 and
         close <= open

    // ------------------------------------------------------------------------
    // TENDÊNCIA
    // ------------------------------------------------------------------------

    tendenciaAlta =
         ema9 > ema21 and
         ema21 > ema50 and
         close >= ema9

    tendenciaBaixa =
         ema9 < ema21 and
         ema21 < ema50 and
         close <= ema9

    tendenciaNeutra =
         not tendenciaAlta and
         not tendenciaBaixa

    // ------------------------------------------------------------------------
    // MOMENTUM
    // ------------------------------------------------------------------------

    momentumAlta =
         rsi >= 52 and
         impulso > 0 and
         aceleracao >= -0.10 and
         diPlus >= diMinus

    momentumBaixa =
         rsi <= 48 and
         impulso < 0 and
         aceleracao <= 0.10 and
         diMinus >= diPlus

    // ------------------------------------------------------------------------
    // ESPAÇO
    // ------------------------------------------------------------------------

    espacoAlta =
         rompAlta or
         distanciaRes >= 0.50

    espacoBaixa =
         rompBaixa or
         distanciaSup >= 0.50

    // ------------------------------------------------------------------------
    // PADRÕES ESTRUTURAIS
    // ------------------------------------------------------------------------

    max5 = ta.highest(high, 5)[1]
    min5 = ta.lowest(low, 5)[1]

    max10 = ta.highest(high, 10)[1]
    min10 = ta.lowest(low, 10)[1]

    max20 = ta.highest(high, 20)[1]
    min20 = ta.lowest(low, 20)[1]

    amplitude10 = max10 - min10
    amplitude20 = max20 - min20

    triangulo =
         amplitude20 > 0 and
         amplitude10 < amplitude20 * 0.72 and
         rangeCandle <= mediaRange * 1.15

    centro8 =
         (ta.highest(high, 8)[1] +
          ta.lowest(low, 8)[1]) / 2.0

    canalAlta =
         close > centro8 and
         ema9 >= ema21

    canalBaixa =
         close < centro8 and
         ema9 <= ema21

    topoA =
         ta.highest(high, 10)[3]

    topoB =
         ta.highest(high, 10)[1]

    fundoA =
         ta.lowest(low, 10)[3]

    fundoB =
         ta.lowest(low, 10)[1]

    duploTopo =
         atr > 0 and
         math.abs(topoA - topoB) <= atr * 0.60 and
         close < topoB

    duploFundo =
         atr > 0 and
         math.abs(fundoA - fundoB) <= atr * 0.60 and
         close > fundoB

    cunhaAlta =
         amplitude20 > 0 and
         amplitude10 < amplitude20 * 0.72 and
         tendenciaAlta and
         close >= centro8

    cunhaBaixa =
         amplitude20 > 0 and
         amplitude10 < amplitude20 * 0.72 and
         tendenciaBaixa and
         close <= centro8

    impulsoAnterior =
         atr > 0
         ? math.abs(close[5] - close[10]) / atr
         : 0.0

    bandeiraAlta =
         impulsoAnterior >= 1.20 and
         compressao and
         ema9 >= ema21 and
         close >= centro8

    bandeiraBaixa =
         impulsoAnterior >= 1.20 and
         compressao and
         ema9 <= ema21 and
         close <= centro8

    // ------------------------------------------------------------------------
    // PADRÃO DE COMPRA
    // ------------------------------------------------------------------------

    padraoAlta =
         usarPadroes and
         (
             (usarDuplo and duploFundo) or
             (usarTriangulo and triangulo and rompAlta) or
             (usarCanal and canalAlta) or
             (usarCunha and cunhaAlta) or
             (usarBandeira and bandeiraAlta) or
             retesteAlta or
             engolfoAlta
         )

    // ------------------------------------------------------------------------
    // PADRÃO DE VENDA
    // ------------------------------------------------------------------------

    padraoBaixa =
         usarPadroes and
         (
             (usarDuplo and duploTopo) or
             (usarTriangulo and triangulo and rompBaixa) or
             (usarCanal and canalBaixa) or
             (usarCunha and cunhaBaixa) or
             (usarBandeira and bandeiraBaixa) or
             retesteBaixa or
             engolfoBaixa
         )

    // ------------------------------------------------------------------------
    // ESTRUTURA
    // ------------------------------------------------------------------------

    estruturaAlta =
         f_pts(rompAlta, 30) +
         f_pts(retesteAlta, 25) +
         f_pts(padraoAlta, 20) +
         f_pts(preAlta, 10) +
         f_pts(rejeicaoAlta, 10) +
         f_pts(engolfoAlta, 5) +
         f_pts(martelo, 5) +
         f_pts(candleForteAlta, 5) +
         f_pts(compressao, 3)

    estruturaBaixa =
         f_pts(rompBaixa, 30) +
         f_pts(retesteBaixa, 25) +
         f_pts(padraoBaixa, 20) +
         f_pts(preBaixa, 10) +
         f_pts(rejeicaoBaixa, 10) +
         f_pts(engolfoBaixa, 5) +
         f_pts(estrelaCadente, 5) +
         f_pts(candleForteBaixa, 5) +
         f_pts(compressao, 3)

    estruturaAlta := f_clamp(estruturaAlta, 0, 100)
    estruturaBaixa := f_clamp(estruturaBaixa, 0, 100)

    // ------------------------------------------------------------------------
    // PREVISÃO ALTA
    // ------------------------------------------------------------------------

    previsaoAlta =
         f_pts(ema5 > ema9, 8) +
         f_pts(ema9 > ema21, 12) +
         f_pts(ema21 > ema50, 10) +
         f_pts(rsi >= 52, 10) +
         f_pts(impulso > 0, 10) +
         f_pts(aceleracao >= 0, 7) +
         f_pts(diPlus > diMinus, 10) +
         f_pts(adx >= 20, 7) +
         estruturaAlta * 0.16 +
         f_pts(espacoAlta, 5) +
         f_pts(padraoAlta, 5)

    // ------------------------------------------------------------------------
    // PREVISÃO BAIXA
    // ------------------------------------------------------------------------

    previsaoBaixa =
         f_pts(ema5 < ema9, 8) +
         f_pts(ema9 < ema21, 12) +
         f_pts(ema21 < ema50, 10) +
         f_pts(rsi <= 48, 10) +
         f_pts(impulso < 0, 10) +
         f_pts(aceleracao <= 0, 7) +
         f_pts(diMinus > diPlus, 10) +
         f_pts(adx >= 20, 7) +
         estruturaBaixa * 0.16 +
         f_pts(espacoBaixa, 5) +
         f_pts(padraoBaixa, 5)

    previsaoAltaFinal =
         f_clamp(previsaoAlta, 0, 100)

    previsaoBaixaFinal =
         f_clamp(previsaoBaixa, 0, 100)

    // ------------------------------------------------------------------------
    // SCORE FINAL DE COMPRA
    // ------------------------------------------------------------------------

    scoreAlta =
         f_pts(tendenciaAlta, 18) +
         f_pts(momentumAlta, 15) +
         estruturaAlta * 0.20 +
         previsaoAltaFinal * 0.22 +
         f_pts(padraoAlta, 10) +
         f_pts(espacoAlta, 5) +
         f_pts(rompAlta, 5) +
         f_pts(retesteAlta, 5) +
         f_pts(volatilidadeBoa, 3) +
         f_pts(candleForteAlta, 2)

    // ------------------------------------------------------------------------
    // SCORE FINAL DE VENDA
    // ------------------------------------------------------------------------

    scoreBaixa =
         f_pts(tendenciaBaixa, 18) +
         f_pts(momentumBaixa, 15) +
         estruturaBaixa * 0.20 +
         previsaoBaixaFinal * 0.22 +
         f_pts(padraoBaixa, 10) +
         f_pts(espacoBaixa, 5) +
         f_pts(rompBaixa, 5) +
         f_pts(retesteBaixa, 5) +
         f_pts(volatilidadeBoa, 3) +
         f_pts(candleForteBaixa, 2)

    scoreAltaFinal =
         f_clamp(scoreAlta, 0, 100)

    scoreBaixaFinal =
         f_clamp(scoreBaixa, 0, 100)

    // ------------------------------------------------------------------------
    // SAÍDA DA FUNÇÃO
    // É UTILIZADO [1] PARA TRABALHAR COM O ÚLTIMO DIÁRIO FECHADO.
    // ------------------------------------------------------------------------

    [
         close[1],
         ema5[1],
         ema9[1],
         ema21[1],
         ema50[1],
         rsi[1],
         atr[1],
         diPlus[1],
         diMinus[1],
         adx[1],
         impulso[1],
         aceleracao[1],
         suporte[1],
         resistencia[1],
         distanciaSup[1],
         distanciaRes[1],
         estruturaAlta[1],
         estruturaBaixa[1],
         previsaoAltaFinal[1],
         previsaoBaixaFinal[1],
         scoreAltaFinal[1],
         scoreBaixaFinal[1],
         tendenciaAlta[1],
         tendenciaBaixa[1],
         momentumAlta[1],
         momentumBaixa[1],
         rompAlta[1],
         rompBaixa[1],
         preAlta[1],
         preBaixa[1],
         retesteAlta[1],
         retesteBaixa[1],
         padraoAlta[1],
         padraoBaixa[1],
         duploFundo[1],
         duploTopo[1],
         triangulo[1],
         canalAlta[1],
         canalBaixa[1],
         cunhaAlta[1],
         cunhaBaixa[1],
         bandeiraAlta[1],
         bandeiraBaixa[1],
         rejeicaoAlta[1],
         rejeicaoBaixa[1],
         volatilidadeBoa[1]
    ]


// ============================================================================
// REQUEST DIÁRIO
// ============================================================================

[
     closeD,
     ema5D,
     ema9D,
     ema21D,
     ema50D,
     rsiD,
     atrD,
     diPlusD,
     diMinusD,
     adxD,
     impulsoD,
     aceleracaoD,
     suporteD,
     resistenciaD,
     distanciaSupD,
     distanciaResD,
     estruturaAltaD,
     estruturaBaixaD,
     previsaoAltaD,
     previsaoBaixaD,
     scoreAltaD,
     scoreBaixaD,
     tendenciaAltaD,
     tendenciaBaixaD,
     momentumAltaD,
     momentumBaixaD,
     rompAltaD,
     rompBaixaD,
     preAltaD,
     preBaixaD,
     retesteAltaD,
     retesteBaixaD,
     padraoAltaD,
     padraoBaixaD,
     duploFundoD,
     duploTopoD,
     trianguloD,
     canalAltaD,
     canalBaixaD,
     cunhaAltaD,
     cunhaBaixaD,
     bandeiraAltaD,
     bandeiraBaixaD,
     rejeicaoAltaD,
     rejeicaoBaixaD,
     volatilidadeBoaD
] =
     request.security(
         syminfo.tickerid,
         "D",
         f_diario(),
         gaps=barmerge.gaps_off,
         lookahead=barmerge.lookahead_off)


// ============================================================================
// MOTOR SEMANAL
// ============================================================================

f_semanal() =>
    ema9 = ta.ema(close, 9)
    ema21 = ta.ema(close, 21)

    [
         ema9[1],
         ema21[1],
         close[1]
    ]

[
     ema9W,
     ema21W,
     closeW
] =
     request.security(
         syminfo.tickerid,
         "W",
         f_semanal(),
         gaps=barmerge.gaps_off,
         lookahead=barmerge.lookahead_off)

semanalAlta =
     ema9W > ema21W

semanalBaixa =
     ema9W < ema21W


// ============================================================================
// CONFLUÊNCIA
// ============================================================================

vantagemPrevisao =
     previsaoAltaD - previsaoBaixaD

previsaoCompra =
     previsaoAltaD >= previsaoBaixaD + vantagemMin

previsaoVenda =
     previsaoBaixaD >= previsaoAltaD + vantagemMin


// ============================================================================
// FILTROS DE COMPRA
// ============================================================================

filtroTendenciaCompra =
     not exigirTendencia or tendenciaAltaD

filtroMomentumCompra =
     not exigirMomentum or momentumAltaD

filtroEstruturaCompra =
     not exigirEstrutura or estruturaAltaD >= 45

filtroRompimentoCompra =
     not usarRompimento or
     rompAltaD or
     preAltaD or
     retesteAltaD

filtroRetesteCompra =
     not usarReteste or
     retesteAltaD or
     rompAltaD or
     preAltaD

filtroSemanalCompra =
     not usarSemanal or semanalAlta

filtroVolatilidadeCompra =
     not usarVolatilidade or volatilidadeBoaD

filtroPadraoCompra =
     not exigirPadrao or padraoAltaD


// ============================================================================
// FILTROS DE VENDA
// ============================================================================

filtroTendenciaVenda =
     not exigirTendencia or tendenciaBaixaD

filtroMomentumVenda =
     not exigirMomentum or momentumBaixaD

filtroEstruturaVenda =
     not exigirEstrutura or estruturaBaixaD >= 45

filtroRompimentoVenda =
     not usarRompimento or
     rompBaixaD or
     preBaixaD or
     retesteBaixaD

filtroRetesteVenda =
     not usarReteste or
     retesteBaixaD or
     rompBaixaD or
     preBaixaD

filtroSemanalVenda =
     not usarSemanal or semanalBaixa

filtroVolatilidadeVenda =
     not usarVolatilidade or volatilidadeBoaD

filtroPadraoVenda =
     not exigirPadrao or padraoBaixaD


// ============================================================================
// CANDIDATO DE COMPRA
// ============================================================================

candidatoCompra =
     ativar and
     scoreAltaD >= scoreMin and
     previsaoCompra and
     filtroTendenciaCompra and
     filtroMomentumCompra and
     filtroEstruturaCompra and
     filtroRompimentoCompra and
     filtroRetesteCompra and
     filtroSemanalCompra and
     filtroVolatilidadeCompra and
     filtroPadraoCompra


// ============================================================================
// CANDIDATO DE VENDA
// ============================================================================

candidatoVenda =
     ativar and
     scoreBaixaD >= scoreMin and
     previsaoVenda and
     filtroTendenciaVenda and
     filtroMomentumVenda and
     filtroEstruturaVenda and
     filtroRompimentoVenda and
     filtroRetesteVenda and
     filtroSemanalVenda and
     filtroVolatilidadeVenda and
     filtroPadraoVenda


// ============================================================================
// CONFLITO
// ============================================================================

conflito =
     candidatoCompra and candidatoVenda


// ============================================================================
// DECISÃO
// ============================================================================

compra =
     candidatoCompra and not conflito

venda =
     candidatoVenda and not conflito

aguarda =
     not compra and not venda

direcao =
     conflito
     ? "CONFLITO"
     : compra
     ? "COMPRA"
     : venda
