f_clamp(float valor, float minimo, float maximo) =>
    math.max(minimo, math.min(maximo, valor))

f_classVolatilidade(float valor) =>
    string resultado = "MUITO FRACA"
    if valor >= 80
        resultado := "EXTREMA"
    else if valor >= 65
        resultado := "MUITO FORTE"
    else if valor >= 50
        resultado := "FORTE"
    else if valor >= 35
        resultado := "MODERADA"
    else if valor >= 20
        resultado := "FRACA"
    resultado

f_corVolatilidade(float valor) =>
    color resultado = color.gray
    if valor >= 80
        resultado := color.red
    else if valor >= 65
        resultado := color.orange
    else if valor >= 50
        resultado := color.yellow
    else if valor >= 35
        resultado := color.aqua
    resultado

f_classFib(float posicao) =>
    string resultado = "0-23.6"
    if posicao >= 78.6
        resultado := "78.6-100"
    else if posicao >= 61.8

    else if posicao >= 50
        resultado := "50-61.8"
    else if posicao >= 38.2
        resultado := "38.2-50"
    else if posicao >= 23.6
        resultado := "23.6-38.2"

f_padroesGrafico() =>
    float atr = ta.atr(atrPeriodo)
    float tolerancia = math.max(atr * 0.35, syminfo.mintick * 5.0)
    int metade = int(math.max(math.floor(padraoLookback / 2), 4))

    float lowRecente = ta.lowest(low, metade)
    float lowAnterior = ta.lowest(low, metade)[metade]
    float highRecente = ta.highest(high, metade)
    float highAnterior = ta.highest(high, metade)[metade]

    bool fundoDuplo = math.abs(lowRecente - lowAnterior) <= tolerancia and close > ta.sma(close, 5)
    bool topoDuplo = math.abs(highRecente - highAnterior) <= tolerancia and close < ta.sma(close, 5)

    bool rompimentoAlta = close > ta.highest(high, 10)[1]
    bool rompimentoBaixa = close < ta.lowest(low, 10)[1]

    float rangeAtual = math.max(high - low, syminfo.mintick)
    float rangeMedio = math.max(ta.sma(high - low, 10), syminfo.mintick)
    bool compressao = rangeAtual < rangeMedio * 0.72

    bool engolfoAlta = close > open and close[1] < open[1] and close >= open[1] and open <= close[1]
    bool engolfoBaixa = close < open and close[1] > open[1] and close <= open[1] and open >= close[1]

    string padrao = "SEM PADRAO"
    float confianca = 0.0

    if usarPadroes
        if fundoDuplo
            padrao := "DUPLO FUNDO"
            confianca := 86.0
        else if topoDuplo
            padrao := "DUPLO TOPO"
            confianca := 86.0
        else if rompimentoAlta
            padrao := "ROMPIMENTO ALTA"
            confianca := 82.0
        else if rompimentoBaixa
            padrao := "ROMPIMENTO BAIXA"
            confianca := 82.0
        else if engolfoAlta
            padrao := "ENGOLFO ALTA"
            confianca := 76.0
        else if engolfoBaixa
            padrao := "ENGOLFO BAIXA"
            confianca := 76.0
        else if compressao
            padrao := "COMPRESSAO"
            confianca := 64.0

    if confianca < padraoMinConfianca
        padrao := "SEM PADRAO"
        confianca := 0.0

    [padrao, confianca]

f_modeloLocal() =>
    float rangeCandle = math.max(high - low, syminfo.mintick)
    float corpo = math.abs(close - open)
    float forcaCandle = f_clamp(corpo / rangeCandle * 100.0, 0.0, 100.0)

    float ema9 = ta.ema(close, emaRapida)
    float ema21 = ta.ema(close, emaMedia)
    float ema50 = ta.ema(close, emaLenta)

    float rsi = ta.rsi(close, rsiPeriodo)

    float fluxo = volume * (2.0 * ((close - low) - (high - close)) / rangeCandle)
    float kvoFastValue = ta.ema(fluxo, kvoFast)
    float kvoSlowValue = ta.ema(fluxo, kvoSlow)
    float kvo = kvoFastValue - kvoSlowValue
    float kvoSignal = ta.ema(kvo, kvoSignalPeriodo)

    float retorno1 = 0.0
    if close[1] != 0
        retorno1 := (close - close[1]) / close[1] * 100.0

    float retorno3 = 0.0
    if close[3] != 0
        retorno3 := (close - close[3]) / close[3] * 100.0

    float tendenciaLong = 25.0
    if ema9 > ema21 and ema21 > ema50
        tendenciaLong := 100.0
    else if ema9 > ema21
        tendenciaLong := 75.0
    else if ema9 > ema50
        tendenciaLong := 60.0

    float tendenciaShort = 25.0
    if ema9 < ema21 and ema21 < ema50
        tendenciaShort := 100.0
    else if ema9 < ema21
        tendenciaShort := 75.0
    else if ema9 < ema50
        tendenciaShort := 60.0

    float rsiLong = 25.0
    if rsi <= rsiSobrevendido
        rsiLong := 100.0
    else if rsi < 50
        rsiLong := 80.0
    else if rsi < 55
        rsiLong := 60.0

    float rsiShort = 25.0
    if rsi >= rsiSobrecomprado
        rsiShort := 100.0
    else if rsi > 50
        rsiShort := 80.0
    else if rsi > 45
        rsiShort := 60.0

    float kvoLong = 30.0
    if kvo > kvoSignal
        kvoLong := 100.0
    else if kvo > 0
        kvoLong := 70.0

    float kvoShort = 30.0
    if kvo < kvoSignal
        kvoShort := 100.0
    else if kvo < 0
        kvoShort := 70.0

    float retornoLong = 25.0
    if retorno1 > 0 and retorno3 > 0
        retornoLong := 100.0
    else if retorno1 > 0 or retorno3 > 0
        retornoLong := 65.0

    float retornoShort = 25.0
    if retorno1 < 0 and retorno3 < 0
        retornoShort := 100.0
    else if retorno1 < 0 or retorno3 < 0
        retornoShort := 65.0

    float candleLong = forcaCandle * 0.35
    if close > open
        candleLong := forcaCandle

    float candleShort = forcaCandle * 0.35
    if close < open
        candleShort := forcaCandle

    float scoreLong = tendenciaLong * 0.30 + rsiLong * 0.15 + kvoLong * 0.20 + retornoLong * 0.15 + candleLong * 0.20
    float scoreShort = tendenciaShort * 0.30 + rsiShort * 0.15 + kvoShort * 0.20 + retornoShort * 0.15 + candleShort * 0.20

    [scoreLong, scoreShort, forcaCandle, kvo, kvoSignal, rsi, ema9, ema21, ema50]

f_scoreMTF() =>
    [scoreLong, scoreShort, forcaCandle, kvo, kvoSignal, rsi, ema9, ema21, ema50] = f_modeloLocal()
    [scoreLong, scoreShort]

f_dados1M() =>
    [padraoTexto, padraoConfianca] = f_padroesGrafico()
    [localLong, localShort, forcaCandle, kvo, kvoSignal, rsi, ema9, ema21, ema50] = f_modeloLocal()

    float atrValue = ta.atr(atrPeriodo)
    float atrMedia = ta.sma(atrValue, atrMediaPeriodo)

    float volatilidade = 50.0
    if atrMedia != 0
        volatilidade := atrValue / atrMedia * 50.0

    volatilidade := f_clamp(volatilidade, 0.0, 100.0)

    float fibHigh = ta.highest(high, fibLookback)
    float fibLow = ta.lowest(low, fibLookback)
    float fibAmplitude = math.max(fibHigh - fibLow, syminfo.mintick)

    float fibPosicao = (close - fibLow) / fibAmplitude * 100.0
    fibPosicao := f_clamp(fibPosicao, 0.0, 100.0)

    float estruturaFundo = ta.lowest(low, estruturaLookback)
    float estruturaTopo = ta.highest(high, estruturaLookback)

    float microTopo = ta.highest(high, microLookback)
    float microFundo = ta.lowest(low, microLookback)

    bool microBreakLong = close > microTopo[1]
    bool microBreakShort = close < microFundo[1]

    bool recuperacaoLong = low < low[1] and close > open and close > close[1]
    bool rejeicaoMicroShort = high > high[1] and close < open and close < close[1]

    int candlesAlta = 0
    int candlesBaixa = 0

    int barrasDesdeBaixa = ta.barssince(close <= open)
    int barrasDesdeAlta = ta.barssince(close >= open)

    if close > open
        candlesAlta := nz(barrasDesdeBaixa) + 1

    if close < open
        candlesBaixa := nz(barrasDesdeAlta) + 1

    float retorno1 = 0.0
    if close[1] != 0
        retorno1 := (close - close[1]) / close[1] * 100.0

    float retornoAnterior = 0.0
    if close[2] != 0
        retornoAnterior := (close[1] - close[2]) / close[2] * 100.0

    float aceleracaoLong = 0.0
    if retorno1 > 0
        aceleracaoLong := retorno1 - retornoAnterior

    float aceleracaoShort = 0.0
    if retorno1 < 0
        aceleracaoShort := math.abs(retorno1) - math.abs(retornoAnterior)

    float baseAceleracao = 0.0001
    if close != 0
        baseAceleracao := atrValue / close * 100.0

    baseAceleracao := math.max(baseAceleracao, 0.0001)

    float aceleracaoLongScore = f_clamp(aceleracaoLong / baseAceleracao * 100.0, 0.0, 100.0)
    float aceleracaoShortScore = f_clamp(aceleracaoShort / baseAceleracao * 100.0, 0.0, 100.0)

    float kvoVariacao = kvo - nz(kvo[1])

    float kvoPerdaLong = 0.0
    if kvo > 0 and kvoVariacao < 0
        kvoPerdaLong := 100.0
    else if kvo > 0
        kvoPerdaLong := 25.0

    float kvoPerdaShort = 0.0
    if kvo < 0 and kvoVariacao > 0
        kvoPerdaShort := 100.0
    else if kvo < 0
        kvoPerdaShort := 25.0

    float rsiExcessoLong = 0.0
    if rsi > rsiSobrecomprado
        rsiExcessoLong := f_clamp((rsi - rsiSobrecomprado) / math.max(80.0 - rsiSobrecomprado, 0.1) * 100.0, 0.0, 100.0)

    float rsiExcessoShort = 0.0
    if rsi < rsiSobrevendido
        rsiExcessoShort := f_clamp((rsiSobrevendido - rsi) / math.max(rsiSobrevendido - 20.0, 0.1) * 100.0, 0.0, 100.0)

    // ========================================================================
    // DISTANCIA LOCAL 1M
    // ========================================================================

    float distanciaEMAATR = 0.0
    if atrValue > 0
        distanciaEMAATR := math.abs(close - ema21) / atrValue

    float distanciaEMAScore = f_clamp((distanciaEMAATR - 0.50) / 1.50 * 100.0, 0.0, 100.0)

    // ========================================================================
    // EXTENSAO DA PERNA
    // ========================================================================

    float extensaoLongATR = 0.0
    if atrValue > 0
        extensaoLongATR := math.max(close - estruturaFundo, 0.0) / atrValue

    float extensaoShortATR = 0.0
    if atrValue > 0
        extensaoShortATR := math.max(estruturaTopo - close, 0.0) / atrValue

    float sombraSuperior = math.max(high - math.max(open, close), 0.0)
    float sombraInferior = math.max(math.min(open, close) - low, 0.0)

    float rangeCandle = math.max(high - low, syminfo.mintick)

    float rejeicaoPavioLong = sombraSuperior / rangeCandle * 100.0
    float rejeicaoPavioShort = sombraInferior / rangeCandle * 100.0

    [padraoTexto, padraoConfianca, localLong, localShort, forcaCandle, kvo, kvoSignal, rsi, ema9, ema21, ema50, atrValue, atrMedia, volatilidade, fibPosicao, close, estruturaFundo, estruturaTopo, candlesAlta, candlesBaixa, retorno1, aceleracaoLong, aceleracaoShort, aceleracaoLongScore, aceleracaoShortScore, kvoPerdaLong, kvoPerdaShort, rsiExcessoLong, rsiExcessoShort, distanciaEMAATR, distanciaEMAScore, extensaoLongATR, extensaoShortATR, rejeicaoPavioLong, rejeicaoPavioShort, microBreakLong, microBreakShort, recuperacaoLong, rejeicaoMicroShort]

f_scanner(string simbolo) =>
    [padrao1, padraoConf1, l1, s1, fc1, kvo1, kvoSig1, rsi1, ema9_1, ema21_1, ema50_1, atr1, atrMedio1, vol1, fibPos, preco1, fundo1, topo1, candlesAlta1, candlesBaixa1, retorno1, aceleracaoLong, aceleracaoShort, aceleracaoLongScore, aceleracaoShortScore, kvoPerdaLong, kvoPerdaShort, rsiExcessoLong, rsiExcessoShort, distanciaEMAATR, distanciaEMAScore, extensaoLongATR, extensaoShortATR, rejeicaoPavioLong, rejeicaoPavioShort, microBreakLong, microBreakShort, recuperacaoLong, rejeicaoMicroShort] = request.security(simbolo, "1", f_dados1M(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)

    [l5, s5] = request.security(simbolo, "5", f_scoreMTF(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)
    [l15, s15, atr15] = request.security(simbolo, "15", f_mtfAtr(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)
    [l30, s30] = request.security(simbolo, "30", f_scoreMTF(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)
    [l60, s60, atr60] = request.security(simbolo, "60", f_mtfAtr(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)
    [l240, s240, atr240] = request.security(simbolo, "240", f_mtfAtr(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)
    [lD, sD] = request.security(simbolo, "D", f_scoreMTF(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)

    float somaPesos = peso1M + peso5M + peso15M + peso30M + peso1H + peso4H + pesoD

    float mtfLong = (l1 * peso1M + l5 * peso5M + l15 * peso15M + l30 * peso30M + l60 * peso1H + l240 * peso4H + lD * pesoD) / somaPesos
    float mtfShort = (s1 * peso1M + s5 * peso5M + s15 * peso15M + s30 * peso30M + s60 * peso1H + s240 * peso4H + sD * pesoD) / somaPesos

    float atrCombinado = atr15 * 0.35 + atr60 * 0.35 + atr240 * 0.30

    if atrAdaptativo
        atrCombinado := atrCombinado * fatorATR

    // ========================================================================
    // TENDENCIA
    // ========================================================================

    string tendencia = "LATERAL"

    if ema9_1 > ema21_1 and ema21_1 > ema50_1
        tendencia := "ALTA"
    else if ema9_1 < ema21_1 and ema21_1 < ema50_1
        tendencia := "BAIXA"

    float tendenciaMTFLong = 0.0

    if l30 > s30
        tendenciaMTFLong += 25.0
    if l60 > s60
        tendenciaMTFLong += 30.0
    if l240 > s240
        tendenciaMTFLong += 30.0
    if lD > sD
        tendenciaMTFLong += 15.0

    float tendenciaMTFShort = 0.0

    if s30 > l30
        tendenciaMTFShort += 25.0
    if s60 > l60
        tendenciaMTFShort += 30.0
    if s240 > l240
        tendenciaMTFShort += 30.0
    if sD > lD
        tendenciaMTFShort += 15.0

    // FUNDO / TOPO
    // ========================================================================

    float distanciaFundoATR = 99.0
    if atrCombinado > 0
        distanciaFundoATR := math.abs(preco1 - fundo1) / atrCombinado

    float distanciaTopoATR = 99.0
    if atrCombinado > 0
        distanciaTopoATR := math.abs(preco1 - topo1) / atrCombinado

    float proximidadeFundo = f_clamp(100.0 - distanciaFundoATR / math.max(distanciaMaximaATR, 0.1) * 100.0, 0.0, 100.0)
    float proximidadeTopo = f_clamp(100.0 - distanciaTopoATR / math.max(distanciaMaximaATR, 0.1) * 100.0, 0.0, 100.0)

    float forcaFundo = proximidadeFundo

    if rsi1 <= rsiSobrevendido
        forcaFundo += 20.0
    else if rsi1 < 45
        forcaFundo += 10.0

    if kvo1 > kvoSig1
        forcaFundo += 15.0

    if preco1 > ema9_1

    if recuperacaoLong

    if rsi1 >= rsiSobrecomprado

    else if rsi1 > 55
        forcaTopo += 10.0

    if kvo1 < kvoSig1
        forcaTopo += 15.0

    if preco1 < ema9_1
        forcaTopo += 10.0

    if rejeicaoMicroShort

    forcaTopo := f_clamp(forcaTopo, 0.0, 100.0)

    // ========================================================================
    // MICROESTRUTURA
    // ========================================================================

    float microLong = 35.0

    if microBreakLong

    if recuperacaoLong
        microLong += 20.0
    if preco1 > ema9_1
        microLong += 10.0

    microLong := f_clamp(microLong, 0.0, 100.0)

    float microShort = 35.0

    if microBreakShort
        microShort += 35.0
    if rejeicaoMicroShort
        microShort += 20.0
    if preco1 < ema9_1
        microShort += 10.0

    // ========================================================================
    // CONFLUENCIA
    // ========================================================================

    float confluenciaLong = 0.0

    if tendencia == "ALTA"
        confluenciaLong += 25.0
    if kvo1 > kvoSig1
        confluenciaLong += 20.0
    if rsi1 < 55
        confluenciaLong += 15.0
    if mtfLong > mtfShort
        confluenciaLong += 20.0
    if microLong >= 60

    float confluenciaShort = 0.0

    if tendencia == "BAIXA"

    if kvo1 < kvoSig1

    if rsi1 > 45
        confluenciaShort += 15.0
    if mtfShort > mtfLong
        confluenciaShort += 20.0
    if microShort >= 60
        confluenciaShort += 20.0

    confluenciaShort := f_clamp(confluenciaShort, 0.0, 100.0)

    // ========================================================================
    // SCORE
    // ========================================================================

    float scoreLong = l1 * 0.50 + mtfLong * 0.50
    float scoreShort = s1 * 0.50 + mtfShort * 0.50

    float vantagemLong = f_clamp(scoreLong - scoreShort, 0.0, 100.0)
    float vantagemShort = f_clamp(scoreShort - scoreLong, 0.0, 100.0)

    // ========================================================================
    // INTENSIDADE
    // ========================================================================

    float intensidadeLong = scoreLong * 0.40 + confluenciaLong * 0.20 + microLong * 0.15 + forcaFundo * 0.25
    float intensidadeShort = scoreShort * 0.40 + confluenciaShort * 0.20 + microShort * 0.15 + forcaTopo * 0.25

    intensidadeShort := f_clamp(intensidadeShort, 0.0, 100.0)

    // QUALIDADE
    // ========================================================================

    float qualidadeShort = scoreShort * 0.25 + intensidadeShort * 0.20 + confluenciaShort * 0.15 + mtfShort * 0.15 + microShort * 0.10 + forcaTopo * 0.15

    qualidadeShort := f_clamp(qualidadeShort, 0.0, 100.0)

    // VOLATILIDADE
    // ========================================================================

    float pernaScoreShort = extensaoShortScore * 0.30 + distanciaEMAScore * 0.18 + candlesShortScore * 0.12 + riscoAceleracaoShort * 0.15 + kvoPerdaShort * 0.15 + rsiExcessoShort * 0.10

        pernaScoreLong += 12.0
    else if extensaoLongATR >= pernaMuitoEsticadaATR
        pernaScoreLong += 8.0
    else if extensaoLongATR >= pernaEsticadaATR
        pernaScoreLong += 4.0

    if extensaoShortATR >= pernaExtremaATR
        pernaScoreShort += 12.0
    else if extensaoShortATR >= pernaMuitoEsticadaATR
        pernaScoreShort += 8.0
    else if extensaoShortATR >= pernaEsticadaATR
        pernaScoreShort += 4.0

    if extensaoLongATR >= pernaEsticadaATR and rejeicaoPavioLong >= 40.0
        pernaScoreLong += 5.0

    if extensaoShortATR >= pernaEsticadaATR and rejeicaoPavioShort >= 40.0
        pernaScoreShort += 5.0

    pernaScoreLong := f_clamp(pernaScoreLong, 0.0, 100.0)
    pernaScoreShort := f_clamp(pernaScoreShort, 0.0, 100.0)

    // EXAUSTAO
    // ========================================================================

    bool exaustaoLong = false

    if pernaScoreLong >= pernaBloqueioScore

        else if extensaoLongATR >= pernaExtremaATR and distanciaEMAScore >= 60.0

    bool exaustaoShort = false

    if pernaScoreShort >= pernaBloqueioScore
        if extensaoShortATR >= pernaMuitoEsticadaATR and riscoAceleracaoShort >= 45.0
            exaustaoShort := true
        else if rsiExcessoShort >= 65.0 and kvoPerdaShort >= 50.0
            exaustaoShort := true
        else if extensaoShortATR >= pernaExtremaATR and distanciaEMAScore >= 60.0
            exaustaoShort := true

    bool pernaShortOK = true

    if usarFiltroPernaEsticada
        pernaLongOK := not exaustaoLong
        pernaShortOK := not exaustaoShort

    bool pernaPrincipalLong = scoreLong >= scoreShort
    float pernaScorePrincipal = pernaPrincipalLong ? pernaScoreLong : pernaScoreShort
    string classePernaPrincipal = f_classPerna(pernaScorePrincipal)
    float extensaoPernaPrincipalATR = pernaPrincipalLong ? extensaoLongATR : extensaoShortATR

    // FIBONACCI
    // ========================================================================

    string classeFib = f_classFib(fibPos)

    bool fibLongOK = true
    bool fibShortOK = true

    if usarFibonacci
        fibLongOK := fibPos <= 65.3
        fibShortOK := fibPos >= 34.7

    // ========================================================================
    // CONFIRMACAO MTF
    // ========================================================================

    bool mtfLongConfirmado = l15 >= s15 and l30 >= s30 and l60 >= s60 and l240 >= s240
    bool mtfShortConfirmado = s15 >= l15 and s30 >= l30 and s60 >= l60 and s240 >= l240

    bool tendenciaLongOK = tendencia == "ALTA" and tendenciaMTFLong >= 60.0
    bool tendenciaShortOK = tendencia == "BAIXA" and tendenciaMTFShort >= 60.0

    // ========================================================================
    // PRE SINAL
    // ========================================================================

    bool preLongCond = scoreLong >= preScoreMinimo and intensidadeLong >= preIntensidadeMinima and confluenciaLong >= preConfluenciaMinima and vantagemLong >= preVantagemMinima and mtfLong >= preMtfMinimo and forcaFundo >= preForcaMinima and microLong >= preMicroMinimo and qualidadeLong >= qualidadePreMinima and volatilidadeOK and fibLongOK and distanciaOK and pernaLongOK and scoreLong > scoreShort

    // ========================================================================
    // SINAL FINAL
    // ========================================================================

    bool finalLong = scoreLong >= scoreMinimo and intensidadeLong >= intensidadeMinima and confluenciaLong >= confluenciaMinima and vantagemLong >= vantagemMinima and mtfLong >= mtfMinimo and forcaFundo >= forcaFundoTopoMinima and microLong >= microMinimo and qualidadeLong >= qualidadeSinalMinima and tendenciaLongOK and mtfLongConfirmado and volatilidadeOK and fibLongOK and distanciaOK and pernaLongOK and scoreLong > scoreShort

    bool finalShort = scoreShort >= scoreMinimo and intensidadeShort >= intensidadeMinima and confluenciaShort >= confluenciaMinima and vantagemShort >= vantagemMinima and mtfShort >= mtfMinimo and forcaTopo >= forcaFundoTopoMinima and microShort >= microMinimo and qualidadeShort >= qualidadeSinalMinima and tendenciaShortOK and mtfShortConfirmado and volatilidadeOK and fibShortOK and distanciaOK and pernaShortOK and scoreShort > scoreLong

    // ========================================================================
    // SINAL
    // ========================================================================

    string sinal = "AGUARDAR"

    if finalLong
        sinal := "COMPRA"
    else if finalShort
        sinal := "VENDA"
    else if preLongCond
        sinal := "PRE COMPRA"
    else if preShortCond
        sinal := "PRE VENDA"

    // ========================================================================
    // PRINCIPAL
    // ========================================================================

    float scorePrincipal = scoreLong
    float intensidadePrincipal = intensidadeLong
    float confluenciaPrincipal = confluenciaLong
    float vantagemPrincipal = vantagemLong
    float mtfPrincipal = mtfLong
    float qualidadePrincipal = qualidadeLong
    float alvoPrincipal = preco1 + atrCombinado * targetPrincipalATR

    if scoreShort > scoreLong
        scorePrincipal := scoreShort
        intensidadePrincipal := intensidadeShort
        confluenciaPrincipal := confluenciaShort
        vantagemPrincipal := vantagemShort
        mtfPrincipal := mtfShort
        qualidadePrincipal := qualidadeShort
        alvoPrincipal := preco1 - atrCombinado * targetPrincipalATR

    [sinal, scorePrincipal, intensidadePrincipal, confluenciaPrincipal, vantagemPrincipal, mtfPrincipal, qualidadePrincipal, classeVolatilidade, vol1, tendencia, distanciaTexto, distanciaEMA, classeFib, fibPos, alvoPrincipal, classePernaPrincipal, pernaScorePrincipal, extensaoPernaPrincipalATR, padrao1, padraoConf1]

// ============================================================================

        result := position.middle_right
    else if valor == "Direita Inferior"
        result := position.bottom_right
    else if valor == "Esquerda Superior"
        result := position.top_left
    else if valor == "Esquerda Meio"
        result := position.middle_left
    else

    result

f_celula(table tabela, int coluna, int linha, string texto, color fundo, color textoCor) =>
    table.cell(tabela, coluna, linha, texto, bgcolor=fundo, text_color=textoCor, text_size=size.small)

f_corPadrao(string padrao) =>
    color resultado = color.gray
    if str.contains(padrao, "FUNDO") or str.contains(padrao, "ALTA")
        resultado := color.lime
    else if str.contains(padrao, "TOPO") or str.contains(padrao, "BAIXA")
        resultado := color.red
    else if padrao == "COMPRESSAO"
        resultado := color.yellow
    resultado

f_statusLeitura(string sinal, float score, string distancia, string perna, string padrao) =>
    string resultado = "AGUARDAR"
    if sinal == "COMPRA" or sinal == "VENDA"
        resultado := "SINAL"
    else if sinal == "PRE COMPRA" or sinal == "PRE VENDA"
        resultado := "PRE-SINAL"
    else if perna == "EXTREMA" or perna == "MUITO ESTICADA"
        resultado := "CAUTELA"
    else if distancia == "LONGE"
        resultado := "DISTANTE"
    else if padrao != "SEM PADRAO" and score >= 65
        resultado := "OBSERVAR"
    else if score >= 65
        resultado := "OBSERVAR"
    resultado

f_corStatus(string status) =>
    color resultado = color.gray
    if status == "SINAL"
        resultado := color.lime
    else if status == "PRE-SINAL"
        resultado := color.yellow
    else if status == "CAUTELA"
        resultado := color.orange
    else if status == "OBSERVAR"
        resultado := color.aqua
    resultado
