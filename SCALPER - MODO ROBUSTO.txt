f_limitar(float valor, float minimo, float maximo) =>
    math.max(minimo, math.min(maximo, valor))


f_pontos(bool condicao, float pontos) =>
    condicao ? pontos : 0.0


f_textoDirecao(int direcao) =>
    direcao == 1 ? "COMPRA" :
     direcao == -1 ? "VENDA" :
     "NEUTRO"


f_corDirecao(int direcao) =>
    direcao == 1 ? color.lime :
     direcao == -1 ? color.red :
     color.gray


f_textoForca(float valor) =>
    valor >= 80 ? "EXTREMA" :
     valor >= 70 ? "MUITO FORTE" :
     valor >= 60 ? "FORTE" :
     valor >= 50 ? "MODERADA" :
     valor >= 40 ? "FRACA" :
     "MUITO FRACA"


f_corForca(float valor) =>
    valor >= 70 ? color.lime :
     valor >= 55 ? color.green :
     valor >= 45 ? color.yellow :
     valor >= 30 ? color.orange :
     color.red


f_textoQualidade(float valor) =>
    valor >= 80 ? "EXCELENTE" :
     valor >= 70 ? "MUITO BOA" :
     valor >= 60 ? "BOA" :
     valor >= 50 ? "REGULAR" :
     valor >= 40 ? "FRACA" :
     "MUITO FRACA"

f_direcaoTF() =>
    ema8TF = ta.ema(close, 8)
    ema21TF = ta.ema(close, 21)
    rsiTF = ta.rsi(close, 10)

    close > ema8TF and
     ema8TF > ema21TF and
     rsiTF >= 50 ? 1 :
     close < ema8TF and
     ema8TF < ema21TF and
     rsiTF <= 50 ? -1 :
     0


f_forcaTF() =>
    ema8TF = ta.ema(close, 8)
    ema21TF = ta.ema(close, 21)
    atrTF = ta.atr(14)

    distanciaTF =
         atrTF > 0 ?
         math.abs(ema8TF - ema21TF) / atrTF :
         0.0

    f_limitar(
         distanciaTF * 35 + 35,
         0,
         100)

     math.max(
         high - low,
         syminfo.mintick)

     math.abs(close - open)

corpoPercentual =
     rangeCandle > 0 ?
     corpoCandle / rangeCandle * 100 :
     0

pavioSuperior =
     high - math.max(open, close)

pavioInferior =
     math.min(open, close) - low

forcaCandleAlta =
     close > open ?
     f_limitar(
         corpoPercentual +
         pavioInferior /
         math.max(rangeCandle, syminfo.mintick) * 20,
         0,
         100) :
     0

forcaCandleBaixa =
     close < open ?
     f_limitar(
         corpoPercentual +
         pavioSuperior /
         math.max(rangeCandle, syminfo.mintick) * 20,
         0,
         100) :
     0

// ============================================================================
// 21. KVO
// ============================================================================

rangeKvo =
     math.max(
         high - low,
         syminfo.mintick)

tpKvo =
     (high + low + close) / 3

tpAnterior =
     nz(
         tpKvo[1],
         tpKvo)

tendenciaKvo =
     tpKvo > tpAnterior ?
     1.0 :
     -1.0

forcaKvo =
     volume *
     (
         math.abs(tpKvo - tpAnterior) /
         rangeKvo
     ) *
     tendenciaKvo

kvoRapidoValor =
     ta.ema(
         forcaKvo,
         kvoRapido)

kvoLentoValor =
     ta.ema(
         forcaKvo,
         kvoLento)

kvoLinha =
     kvoRapidoValor -
     kvoLentoValor

kvoSinalValor =
     ta.ema(
         kvoLinha,
         kvoSinal)

kvoHistograma =
     kvoLinha -
     kvoSinalValor

kvoAlta =
     kvoLinha >
     kvoSinalValor

kvoBaixa =
     kvoLinha <
     kvoSinalValor

kvoCruzAlta =
     ta.crossover(
         kvoLinha,
         kvoSinalValor)

kvoCruzBaixa =
     ta.crossunder(
         kvoLinha,
         kvoSinalValor)

kvoVariacao =
     kvoLinha -
     nz(kvoLinha[1])

kvoForca =
     f_limitar(
         50 +
         kvoVariacao /
         math.max(
             math.abs(kvoLinha) + 1,
             1) * 100,
         0,
         100)


// ============================================================================
// 22. SUPORTE / RESISTÊNCIA
// ============================================================================

resistencia =
     ta.highest(
         high,
         periodoSR)[1]

suporte =
     ta.lowest(
         low,
         periodoSR)[1]

distanciaSuporte =
     math.abs(
         close - suporte)

distanciaResistencia =
     math.abs(
         resistencia - close)

pertoSuporte =
     not na(suporte) and
     distanciaSuporte <=
     atr * proximidadeSRATR

pertoResistencia =
     not na(resistencia) and
     distanciaResistencia <=
     atr * proximidadeSRATR

rompeuResistencia =
     not na(resistencia) and
     close > resistencia

rompeuSuporte =
     not na(suporte) and
     close < suporte

         gaps=barmerge.gaps_off,
         lookahead=barmerge.lookahead_off)


pesoTotal =
     pesoTF1 +
     pesoTF2 +
     pesoTF3 +
     pesoTF4 +
     pesoTF5 +
     pesoTF6 +
     pesoTF7

float somaMTF = 0.0

somaMTF += mtfDir1 * pesoTF1
somaMTF += mtfDir2 * pesoTF2
somaMTF += mtfDir3 * pesoTF3
somaMTF += mtfDir4 * pesoTF4
somaMTF += mtfDir5 * pesoTF5
somaMTF += mtfDir6 * pesoTF6
somaMTF += mtfDir7 * pesoTF7

mtfDirecao =
     pesoTotal > 0 ?
     somaMTF / pesoTotal :
     0

mtfCompra =
     f_limitar(
         50 + mtfDirecao * 50,
         0,
         100)

mtfVenda =
     f_limitar(
         50 - mtfDirecao * 50,
         0,
         100)

mtfForca =
     f_limitar(
         (
             mtfForca1 * pesoTF1 +
             mtfForca2 * pesoTF2 +
             mtfForca3 * pesoTF3 +
             mtfForca4 * pesoTF4 +
             mtfForca5 * pesoTF5 +
             mtfForca6 * pesoTF6 +
             mtfForca7 * pesoTF7
         ) /
         math.max(pesoTotal, 1),
         0,
         100)


// ============================================================================
// 29. CONFLUÊNCIA
// ============================================================================

float confluenciaCompra = 0.0

confluenciaCompra += f_pontos(
     tendenciaAlta,
     20)

confluenciaCompra += f_pontos(
     kvoAlta,
     15)

confluenciaCompra += f_pontos(
     rsiCompra,
     10)

confluenciaCompra += f_pontos(
     pressaoAlta,
     10)

confluenciaCompra += f_pontos(
     microAlta,
     10)

confluenciaCompra += f_pontos(
     retornoAlta,
     10)

confluenciaCompra += f_pontos(
     mtfCompra >= 55,
     15)

confluenciaCompra += f_pontos(
     pertoSuporte,
     10)

confluenciaCompra := f_limitar(
     confluenciaCompra,
     0,
     100)


float confluenciaVenda = 0.0

confluenciaVenda += f_pontos(
     tendenciaBaixa,
     20)

confluenciaVenda += f_pontos(
     kvoBaixa,
     15)

confluenciaVenda += f_pontos(
     rsiVenda,
     10)

confluenciaVenda += f_pontos(
     pressaoBaixa,
     10)

confluenciaVenda += f_pontos(
     microBaixa,
     10)

confluenciaVenda += f_pontos(
     retornoBaixa,
     10)

confluenciaVenda += f_pontos(
     mtfVenda >= 55,
     15)

confluenciaVenda += f_pontos(
     pertoResistencia,
     10)

confluenciaVenda := f_limitar(
     confluenciaVenda,
     0,
     100)


// ============================================================================
// 30. SCORE FINAL
// ============================================================================

float scoreCompra =
     usarMTF ?
     compraLocal * 0.55 +
     mtfCompra * 0.45 :
     compraLocal

float scoreVenda =
     usarMTF ?
     vendaLocal * 0.55 +
     mtfVenda * 0.45 :
     vendaLocal

scoreCompra := f_limitar(
     scoreCompra,
     0,
     100)

scoreVenda := f_limitar(
     scoreVenda,
     0,
     100)

vantagem =
     math.abs(
         scoreCompra -
         scoreVenda)

direcaoAtual =
     scoreCompra > scoreVenda ?
     1 :
     scoreVenda > scoreCompra ?
     -1 :
     0

scoreDominante =
     direcaoAtual == 1 ?
     scoreCompra :
     direcaoAtual == -1 ?
     scoreVenda :
     50


// ============================================================================
// 31. FORÇA
// ============================================================================

forcaCompra =
     f_limitar(
         scoreCompra * 0.45 +
         confluenciaCompra * 0.25 +
         mtfForca * 0.15 +
         forcaCandleAlta * 0.15,
         0,
         100)

forcaVenda =
     f_limitar(
         scoreVenda * 0.45 +
         confluenciaVenda * 0.25 +
         mtfForca * 0.15 +
         forcaCandleBaixa * 0.15,
         0,
         100)

forcaDominante =
     direcaoAtual == 1 ?
     forcaCompra :
     direcaoAtual == -1 ?
     forcaVenda :
     0


// ============================================================================
// 32. PADRÃO ATUAL
// ============================================================================

padraoAtual =
     f_textoPadrao(
         fundoEstrutural,
         topoEstrutural,
         rompimentoAlta,
         rompimentoBaixa,
         recuperacaoFundo,
         rejeicaoAlta,
         rejeicaoTopo)


// ============================================================================
// 33. OPORTUNIDADE
// ============================================================================

oportunidadeCompra =
     scoreCompra >= 80 and
     forcaCompra >= 70 and
     confluenciaCompra >= 60 ?
     "MUITO FORTE" :
     scoreCompra >= 70 and
     forcaCompra >= 60 ?
     "FORTE" :
     scoreCompra >= 60 ?
     "MODERADA" :
     "FRACA"

oportunidadeVenda =
     scoreVenda >= 80 and
     forcaVenda >= 70 and
     confluenciaVenda >= 60 ?
     "MUITO FORTE" :
     scoreVenda >= 70 and
     forcaVenda >= 60 ?
     "FORTE" :
     scoreVenda >= 60 ?
     "MODERADA" :
     "FRACA"


// ============================================================================
// 34. PREVISÃO ATUAL
// ============================================================================

previsaoAltaAtual =
     f_limitar(
         scoreCompra * 0.55 +
         confluenciaCompra * 0.25 +
         mtfCompra * 0.20,
         0,
         100)

previsaoBaixaAtual =
     f_limitar(
         scoreVenda * 0.55 +
         confluenciaVenda * 0.25 +
         mtfVenda * 0.20,
         0,
         100)


// ============================================================================
// 35. PRÓXIMA PREVISÃO
//
// IMPORTANTE:
// time_close representa o horário de fechamento da vela atual.
// Esse é o próximo momento de confirmação/previsão para o próximo candle.
//
// O horário é convertido explicitamente para São Paulo.
// ============================================================================

proximaPrevisaoTime = time_close

     na(proximaPrevisaoTime) ?
     "--:--" :
     str.format_time(
         proximaPrevisaoTime,
         "HH:mm",
         "America/Sao_Paulo")


// ============================================================================
// 36. CONDIÇÕES DE SINAL
// ============================================================================

     forcaCompra >= forcaMinima and
     confluenciaCompra >= confluenciaMinima and

     scoreVenda > scoreCompra and
     not pertoSuporte
