f_limitar(valor, minimo, maximo) =>
    math.max(minimo, math.min(maximo, valor))


f_pct(valor) =>
    str.tostring(f_limitar(valor, 0, 100), "#.0") + "%"


f_preco(valor) =>
    na(valor) ? "-" : str.tostring(valor, format.mintick)

// 12 - SCORE LONG MTF
// ============================================================================

f_scoreLong() =>
    atr = ta.atr(atrLength)

    emaFast = ta.ema(close, emaFastLength)

    emaMid = ta.ema(close, emaMidLength)

    emaSlow = ta.ema(close, emaSlowLength)

    rsi = ta.rsi(close, rsiLength)

    volumeMedio = ta.sma(volume, volumeLength)

    volumeRel = volumeMedio > 0 ? volume / volumeMedio : 1.0

    corpo = math.abs(close - open)

    rangeCandle = math.max(high - low, syminfo.mintick)

    forcaCandle = f_limitar(corpo / rangeCandle * 100.0, 0, 100)

    retorno1 = close[1] != 0 ? (close - close[1]) / close[1] * 100.0 : 0.0

    retorno3 = close[3] != 0 ? (close - close[3]) / close[3] * 100.0 : 0.0

    tendencia = emaFast > emaMid and emaMid > emaSlow

    distanciaEMA = atr > 0 ? math.abs(emaFast - emaSlow) / atr : 0.0

    tendenciaScore = tendencia ? f_limitar(55.0 + distanciaEMA * 30.0, 0, 100) : f_limitar(25.0 + distanciaEMA * 15.0, 0, 100)

    rsiScore = rsi >= 50 ? f_limitar((rsi - 50.0) * 2.0, 0, 100) : 0.0

    retornoScore = retorno1 > 0 ? f_limitar(50.0 + math.abs(retorno1) * 10.0, 0, 100) : 0.0

    retorno3Score = retorno3 > 0 ? f_limitar(50.0 + math.abs(retorno3) * 5.0, 0, 100) : 0.0

    volumeScore = f_limitar(volumeRel * 50.0, 0, 100)

    direcao = close > close[1] ? 1.0 : close < close[1] ? -1.0 : 0.0

    kvo = ta.ema(volume * direcao, kvoFast) - ta.ema(volume * direcao, kvoSlow)

    kvoSignal = ta.ema(kvo, kvoSignalLength)

    kvoScore = kvo > kvoSignal ? 100.0 : 0.0

    candleScore = close > open ? forcaCandle : 0.0

    score = candleScore * 0.15 + tendenciaScore * 0.20 + rsiScore * 0.15 + kvoScore * 0.15 + volumeScore * 0.10 + retornoScore * 0.10 + retorno3Score * 0.05 + forcaCandle * 0.10

    f_limitar(score, 0, 100)

f_scoreShort() =>
    atr = ta.atr(atrLength)

    emaSlow = ta.ema(close, emaSlowLength)

    volumeMedio = ta.sma(volume, volumeLength)

    volumeRel = volumeMedio > 0 ? volume / volumeMedio : 1.0

    corpo = math.abs(close - open)

    rangeCandle = math.max(high - low, syminfo.mintick)

    forcaCandle = f_limitar(corpo / rangeCandle * 100.0, 0, 100)

    retorno1 = close[1] != 0 ? (close - close[1]) / close[1] * 100.0 : 0.0

    retorno3 = close[3] != 0 ? (close - close[3]) / close[3] * 100.0 : 0.0

    tendencia = emaFast < emaMid and emaMid < emaSlow

    distanciaEMA = atr > 0 ? math.abs(emaFast - emaSlow) / atr : 0.0

    tendenciaScore = tendencia ? f_limitar(55.0 + distanciaEMA * 30.0, 0, 100) : f_limitar(25.0 + distanciaEMA * 15.0, 0, 100)

    rsiScore = rsi <= 50 ? f_limitar((50.0 - rsi) * 2.0, 0, 100) : 0.0

    retornoScore = retorno1 < 0 ? f_limitar(50.0 + math.abs(retorno1) * 10.0, 0, 100) : 0.0

    retorno3Score = retorno3 < 0 ? f_limitar(50.0 + math.abs(retorno3) * 5.0, 0, 100) : 0.0

    volumeScore = f_limitar(volumeRel * 50.0, 0, 100)

    direcao = close > close[1] ? 1.0 : close < close[1] ? -1.0 : 0.0

    kvo = ta.ema(volume * direcao, kvoFast) - ta.ema(volume * direcao, kvoSlow)

    kvoSignal = ta.ema(kvo, kvoSignalLength)

    kvoScore = kvo < kvoSignal ? 100.0 : 0.0

    candleScore = close < open ? forcaCandle : 0.0

    score = candleScore * 0.15 + tendenciaScore * 0.20 + rsiScore * 0.15 + kvoScore * 0.15 + volumeScore * 0.10 + retornoScore * 0.10 + retorno3Score * 0.05 + forcaCandle * 0.10

    f_limitar(score, 0, 100)

kvoVirandoBaixa = kvo < kvo[1] and kvo[1] >= kvo[2]

// ============================================================================

fundo = ta.lowest(low, lookbackFundo)

topo = ta.highest(high, lookbackTopo)

distanciaTopo = topo - close

distanciaFundoATR = atrSeguro > 0 ? distanciaFundo / atrSeguro : 999.0

distanciaTopoATR = atrSeguro > 0 ? distanciaTopo / atrSeguro : 999.0

// ============================================================================

rsiFundo = rsi <= 50 ? f_limitar((50.0 - rsi) * 2.0, 0, 100) : 0.0

recuperacaoFundo = close > open and close > close[1] ? 100.0 : close > close[1] ? 65.0 : 25.0

kvoFundo = kvoVirandoAlta ? 100.0 : kvoLong ? 65.0 : 20.0

fundoScore = f_limitar(fundoBase * 0.40 + rsiFundo * 0.20 + recuperacaoFundo * 0.20 + kvoFundo * 0.20, 0, 100)


// ============================================================================
// 22 - FORCA DO TOPO
// ============================================================================

topoBase = distanciaTopoATR <= distanciaTopoATRMax ? f_limitar(100.0 - distanciaTopoATR / distanciaTopoATRMax * 60.0, 0, 100) : 0.0

rsiTopo = rsi >= 50 ? f_limitar((rsi - 50.0) * 2.0, 0, 100) : 0.0

rejeicaoTopo = close < open and close < close[1] ? 100.0 : close < close[1] ? 65.0 : 25.0

kvoTopo = kvoVirandoBaixa ? 100.0 : kvoShort ? 65.0 : 20.0

topoScore = f_limitar(topoBase * 0.40 + rsiTopo * 0.20 + rejeicaoTopo * 0.20 + kvoTopo * 0.20, 0, 100)


// ============================================================================
// 23 - MICROESTRUTURA
// ============================================================================

microHigh = ta.highest(high[1], 5)

microLow = ta.lowest(low[1], 5)

microBreakAlta = close > microHigh

microBreakBaixa = close < microLow

microRecuperacao = close > open and close > close[1]

microRejeicao = close < open and close < close[1]

microLongScore = f_limitar((microBreakAlta ? 65.0 : 30.0) + (microRecuperacao ? 35.0 : 0.0), 0, 100)

microShortScore = f_limitar((microBreakBaixa ? 65.0 : 30.0) + (microRejeicao ? 35.0 : 0.0), 0, 100)


// ============================================================================
// 24 - RSI ADAPTATIVO
// ============================================================================

rsiLongScore = rsi <= rsiSobrevendido ? 100.0 : rsi < 50 ? f_limitar((50.0 - rsi) * 2.0, 0, 100) : f_limitar((rsi - 50.0) * 0.50, 0, 25)

rsiShortScore = rsi >= rsiSobrecomprado ? 100.0 : rsi > 50 ? f_limitar((rsi - 50.0) * 2.0, 0, 100) : f_limitar((50.0 - rsi) * 0.50, 0, 25)


// ============================================================================
// 25 - TENDENCIA
// ============================================================================

tendenciaShortScore = tendenciaShort ? f_limitar(65.0 + forcaTendencia * 0.35, 0, 100) : f_limitar(30.0 + forcaTendencia * 0.20, 0, 100)


// ============================================================================
// 26 - CONFLUENCIA
// ============================================================================

confluenciaLong += fundoScore >= forcaFundoMin ? 20.0 : 0.0

confluenciaLong += rsiLongScore >= 50 ? 15.0 : 0.0

confluenciaLong += kvoLong ? 15.0 : 0.0

confluenciaLong += tendenciaLong ? 15.0 : 0.0

confluenciaLong += microLongScore >= 50 ? 15.0 : 0.0

confluenciaLong += scoreMTFLong > scoreMTFShort ? 20.0 : 0.0

confluenciaLong := f_limitar(confluenciaLong, 0, 100)


confluenciaShort = 0.0

confluenciaShort += topoScore >= forcaTopoMin ? 20.0 : 0.0

confluenciaShort += rsiShortScore >= 50 ? 15.0 : 0.0

confluenciaShort += kvoShort ? 15.0 : 0.0

confluenciaShort += tendenciaShort ? 15.0 : 0.0

confluenciaShort += microShortScore >= 50 ? 15.0 : 0.0

confluenciaShort += scoreMTFShort > scoreMTFLong ? 20.0 : 0.0

confluenciaShort := f_limitar(confluenciaShort, 0, 100)


// ============================================================================
// 27 - SCORE FINAL
