// ============================================================================
// DADOS DO CANDLE
// ============================================================================

abertura = open
maxima = high
minima = low
fechamento = close
volumeAtual = volume

rangeCandle = high - low
corpo = math.abs(close - open)

pavioSuperior = high - math.max(open, close)
pavioInferior = math.min(open, close) - low

rangeSeguro = math.max(rangeCandle, syminfo.mintick)

corpoPct = corpo / rangeSeguro

pavioSuperiorPct = pavioSuperior / rangeSeguro
pavioInferiorPct = pavioInferior / rangeSeguro


// ============================================================================
// DIREÇÃO
// ============================================================================

direcaoCandle =
     close > open ? 1 :
     close < open ? -1 :
     0

direcaoTexto =
     direcaoCandle > 0 ?
     "ALTA" :
     direcaoCandle < 0 ?
     "BAIXA" :
     "NEUTRA"


// ============================================================================
// RETORNOS
// ============================================================================

retorno1 =
     not na(close[1]) ?
     close / close[1] - 1.0 :
     0.0

retorno3 =
     not na(close[3]) ?
     close / close[3] - 1.0 :
     0.0

retorno5 =
     not na(close[5]) ?
     close / close[5] - 1.0 :
     0.0


// ============================================================================
// MÉDIAS
// ============================================================================

mm5 = ta.sma(close, 5)
mm10 = ta.sma(close, 10)
mm20 = ta.sma(close, 20)

distMm5 =
     not na(mm5) ?
     close / mm5 - 1.0 :
     0.0

distMm10 =
     not na(mm10) ?
     close / mm10 - 1.0 :
     0.0

distMm20 =
     not na(mm20) ?
     close / mm20 - 1.0 :
     0.0


// ============================================================================
// VOLUME
// ============================================================================

volumeMm20 = ta.sma(volume, 20)

volumeRelativo =
     not na(volumeMm20) and volumeMm20 != 0 ?
     volume / volumeMm20 :
     1.0


// ============================================================================
// ATR
// ============================================================================

atr = ta.atr(atrPeriodo)

atrSeguro = math.max(
     atr,
     syminfo.mintick
)

atrRelativo =
     not na(close) and close != 0 ?
     atr / close * 100.0 :
     0.0

atrMedia = ta.sma(atr, 20)

atrForte =
     not na(atrMedia) and atr > atrMedia

atrFraco =
     not na(atrMedia) and atr < atrMedia

atrDirecao =
     close > open ? 1 :
     close < open ? -1 :
     0

atrBoaCompra =
     atrDirecao > 0 and atrForte

atrBoaVenda =
     atrDirecao < 0 and atrForte

atrClassificacao =
     atrBoaCompra ?
     "BOA COMPRA" :
     atrBoaVenda ?
     "BOA VENDA" :
     atrForte ?
     "FORTE / NEUTRO" :
     "FRACO"

corAtr =
     atrBoaCompra ?
     corVerde :
     atrBoaVenda ?
     corVermelho :
     corCinza


// ============================================================================
// ESTRUTURA
// ============================================================================

suporte20 = ta.lowest(
     low[1],
     estruturaLookback
)

resistencia20 = ta.highest(
     high[1],
     estruturaLookback
)

suporte5 = ta.lowest(
     low[1],
     confirmacaoLookback
)

resistencia5 = ta.highest(
     high[1],
     confirmacaoLookback
)

distResistencia =
     (resistencia20 - close) /
     atrSeguro

distSuporte =
     (close - suporte20) /
     atrSeguro


// ============================================================================
// KVO
// ============================================================================

precoTipico =
     (high + low + close) / 3.0

tendenciaKvo =
     precoTipico > precoTipico[1] ?
     1 :
     precoTipico < precoTipico[1] ?
     -1 :
     0

dmKvo = high - low

var float cmKvo = 0.0

if bar_index == 0
    cmKvo := dmKvo
else
    cmKvo :=
         tendenciaKvo == tendenciaKvo[1] ?
         cmKvo[1] + dmKvo :
         dmKvo + dmKvo[1]

cmSeguro = math.max(
     cmKvo,
     syminfo.mintick
)

vfKvo =
     volume *
     math.abs(
         2.0 *
         (dmKvo / cmSeguro) -
         1.0
     ) *
     tendenciaKvo *
     100.0

kvo =
     ta.ema(vfKvo, kvoFast) -
     ta.ema(vfKvo, kvoSlow)

kvoSignal =
     ta.ema(
         kvo,
         kvoSignalPeriodo
     )

kvoHist =
     kvo - kvoSignal

kvoVariacao =
     kvo - kvo[1]

kvoDesvio =
     ta.stdev(
         kvo,
         34
     )

kvoZscore =
     not na(kvoDesvio) and kvoDesvio != 0 ?
     kvo / kvoDesvio :
     0.0

kvoForcaBruta =
     math.abs(kvoHist) /
     (
         math.abs(kvo) +
         math.abs(kvoSignal) +
         0.000000001
     )

kvoForca =
     math.min(
         math.max(
             kvoForcaBruta,
             0.0
         ),
         1.0
     ) * 100.0

kvoEstado =
     kvo > kvoSignal and kvoVariacao > 0 ?
     1 :
     kvo < kvoSignal and kvoVariacao < 0 ?
     -1 :
     0

kvoEstadoTexto =
     kvoEstado > 0 ?
     "ALTA" :
     kvoEstado < 0 ?
     "BAIXA" :
     "NEUTRO"

corKvo =
     kvoEstado > 0 ?
     corVerde :
     kvoEstado < 0 ?
     corVermelho :
     corCinza


// ============================================================================
// CANDLE DE FORÇA
// ============================================================================

forcaCorpo =
     math.min(
         math.max(
             corpoPct * 100.0,
             0.0
         ),
         100.0
     )

candleForte =
     corpoPct >= corpoMinimo


// ============================================================================
// MODELO DIRECIONAL
// ============================================================================

float pontosAltaModelo = 0.0
float pontosBaixaModelo = 0.0

if close > open
    pontosAltaModelo += 2.0
else if close < open
    pontosBaixaModelo += 2.0

if kvoEstado > 0
    pontosAltaModelo += 2.0
else if kvoEstado < 0
    pontosBaixaModelo += 2.0

if retorno1 > 0
    pontosAltaModelo += 1.0
else if retorno1 < 0
    pontosBaixaModelo += 1.0

if retorno3 > 0
    pontosAltaModelo += 1.0
else if retorno3 < 0
    pontosBaixaModelo += 1.0

if close > mm5
    pontosAltaModelo += 1.0
else if close < mm5
    pontosBaixaModelo += 1.0

if close > mm10
    pontosAltaModelo += 1.0
else if close < mm10
    pontosBaixaModelo += 1.0

if close > mm20
    pontosAltaModelo += 1.0
else if close < mm20
    pontosBaixaModelo += 1.0

if volumeRelativo >= 1.20
    if close > open
        pontosAltaModelo += 0.5
    else if close < open
        pontosBaixaModelo += 0.5


// ============================================================================
// NORMALIZAÇÃO
// ============================================================================

pontuacaoModeloTotal =
     pontosAltaModelo +
     pontosBaixaModelo

probAlta =
     pontuacaoModeloTotal > 0 ?
     pontosAltaModelo /
     pontuacaoModeloTotal :
     0.5

probBaixa =
     pontuacaoModeloTotal > 0 ?
     pontosBaixaModelo /
     pontuacaoModeloTotal :
     0.5

forcaModelo =
     math.max(
         probAlta,
         probBaixa
     ) * 100.0

direcaoModelo =
     probAlta > probBaixa ?
     1 :
     probBaixa > probAlta ?
     -1 :
     0

direcaoModeloTexto =
     direcaoModelo > 0 ?
     "ALTA" :
     direcaoModelo < 0 ?
     "BAIXA" :
     "NEUTRA"


// ============================================================================
// INTENSIDADE BASE
// ============================================================================

intensidadeBase =
     forcaCorpo * pesoCorpo +
     forcaModelo * pesoModelo

intensidadeBase :=
     math.min(
         math.max(
             intensidadeBase,
             0.0
         ),
         100.0
     )


// ============================================================================
// CONFLUÊNCIA
// ============================================================================

int confirmacoes = 0
int divergencias = 0

if direcaoCandle != 0 and direcaoCandle == direcaoModelo
    confirmacoes += 1
else
    divergencias += 1

if direcaoCandle != 0 and kvoEstado == direcaoCandle
    confirmacoes += 1
else if direcaoCandle != 0 and kvoEstado != 0
    divergencias += 1

if direcaoCandle > 0 and retorno1 > 0
    confirmacoes += 1
else if direcaoCandle < 0 and retorno1 < 0
    confirmacoes += 1
else if retorno1 != 0
    divergencias += 1

if direcaoCandle > 0 and retorno3 > 0
    confirmacoes += 1
else if direcaoCandle < 0 and retorno3 < 0
    confirmacoes += 1
else if retorno3 != 0
    divergencias += 1

if direcaoCandle > 0 and close > mm5
    confirmacoes += 1
else if direcaoCandle < 0 and close < mm5
    confirmacoes += 1
else
    divergencias += 1

if direcaoCandle > 0 and close > mm10
    confirmacoes += 1
else if direcaoCandle < 0 and close < mm10
    confirmacoes += 1
else
    divergencias += 1

if direcaoCandle > 0 and close > mm20
    confirmacoes += 1
else if direcaoCandle < 0 and close < mm20
    confirmacoes += 1
else
    divergencias += 1


// ============================================================================
// DIREÇÃO DA INTENSIDADE
// ============================================================================

float pontuacaoAlta = 0.0
float pontuacaoBaixa = 0.0

if direcaoCandle > 0
    pontuacaoAlta += 2.0
else if direcaoCandle < 0
    pontuacaoBaixa += 2.0

if probAlta > probBaixa
    pontuacaoAlta += probAlta * 3.0
else if probBaixa > probAlta
    pontuacaoBaixa += probBaixa * 3.0

if kvoEstado > 0
    pontuacaoAlta += 2.0
else if kvoEstado < 0
    pontuacaoBaixa += 2.0

if retorno1 > 0
    pontuacaoAlta += 1.0
else if retorno1 < 0
    pontuacaoBaixa += 1.0

if retorno3 > 0
    pontuacaoAlta += 1.0
else if retorno3 < 0
    pontuacaoBaixa += 1.0

if close > mm5
    pontuacaoAlta += 1.0
else if close < mm5
    pontuacaoBaixa += 1.0

if close > mm10
    pontuacaoAlta += 1.0
else if close < mm10
    pontuacaoBaixa += 1.0

if close > mm20
    pontuacaoAlta += 1.0
else if close < mm20
    pontuacaoBaixa += 1.0


// ============================================================================
// DIREÇÃO PREDOMINANTE
// ============================================================================

direcaoIntensidade =
     pontuacaoAlta > pontuacaoBaixa ?
     1 :
     pontuacaoBaixa > pontuacaoAlta ?
     -1 :
     0

direcaoIntensidadeTexto =
     direcaoIntensidade > 0 ?
     "ALTA" :
     direcaoIntensidade < 0 ?
     "BAIXA" :
     "NEUTRA"


// ============================================================================
// CONVICÇÃO DIRECIONAL
// ============================================================================

pontuacaoTotal =
     pontuacaoAlta +
     pontuacaoBaixa

diferencaDirecional =
     math.abs(
         pontuacaoAlta -
         pontuacaoBaixa
     )

confiancaDirecional =
     pontuacaoTotal > 0 ?
     diferencaDirecional /
     pontuacaoTotal *
     100.0 :
     0.0


// ============================================================================
// BOOST CONFLUÊNCIA
// ============================================================================

float boostConfluenciaBruto = 0.0

if confirmacoes == 0
    boostConfluenciaBruto := 0.0
else if confirmacoes == 1
    boostConfluenciaBruto := 2.0
else if confirmacoes == 2
    boostConfluenciaBruto := 5.0
else if confirmacoes == 3
    boostConfluenciaBruto := 8.0
else if confirmacoes == 4
    boostConfluenciaBruto := 12.0
else if confirmacoes == 5
    boostConfluenciaBruto := 16.0
else
    boostConfluenciaBruto := boostMaximo


// ============================================================================
// FATOR MODELO
// ============================================================================

float fatorModeloBoost = 0.35

if forcaModelo >= 70
    fatorModeloBoost := 1.00
else if forcaModelo >= 65
    fatorModeloBoost := 0.90
else if forcaModelo >= 60
    fatorModeloBoost := 0.75
else if forcaModelo >= 55
    fatorModeloBoost := 0.55


// ============================================================================
// FATOR KVO
// ============================================================================

float fatorKvoBoost = 0.50

if kvoForca >= 60
    fatorKvoBoost := 1.00
else if kvoForca >= 40
    fatorKvoBoost := 0.90
else if kvoForca >= 20
    fatorKvoBoost := 0.75


// ============================================================================
// BOOST AJUSTADO
// ============================================================================

boostConfluenciaAjustado =
     boostConfluenciaBruto *
     fatorModeloBoost *
     fatorKvoBoost

boostConfluenciaAjustado :=
     math.min(
         boostConfluenciaAjustado,
         boostMaximo
     )


// ============================================================================
// BOOST VOLUME
// ============================================================================

float boostVolume = 0.0

if volumeRelativo >= 1.20
    boostVolume += 2.0

if volumeRelativo >= 1.50
    boostVolume += 2.0

if volumeRelativo >= 2.00
    boostVolume += 2.0

boostVolume :=
     math.min(
         boostVolume,
         6.0
     )


// ============================================================================
// BOOST KVO
// ============================================================================

float boostKvo = 0.0

if direcaoCandle != 0 and kvoEstado == direcaoCandle
    if kvoForca >= 60
        boostKvo := 5.0
    else if kvoForca >= 40
        boostKvo := 3.0
    else if kvoForca >= 20
        boostKvo := 1.0


// ============================================================================
// BOOST TOTAL
// ============================================================================

boostTotal =
     boostConfluenciaAjustado +
     boostVolume +
     boostKvo

boostTotal :=
     math.min(
         boostTotal,
         boostMaximo
     )


// ============================================================================
// QUALIDADE BOOST
// ============================================================================

string qualidadeBoost = "BAIXA"

if fatorModeloBoost >= 0.90 and fatorKvoBoost >= 0.90
    qualidadeBoost := "ALTA"
else if fatorModeloBoost >= 0.75 and fatorKvoBoost >= 0.75
    qualidadeBoost := "BOA"
else if fatorModeloBoost >= 0.55 or fatorKvoBoost >= 0.75
    qualidadeBoost := "MODERADA"


corBoostResultado =
     qualidadeBoost == "ALTA" ?
     corVerde :
     qualidadeBoost == "MODERADA" ?
     corCiano :
     corCinza


// ============================================================================
// PENALTY
// ============================================================================

float penalty = 0.0

penalty += divergencias * 2.0

if direcaoCandle != 0 and kvoEstado != 0 and kvoEstado != direcaoCandle
    penalty += 5.0

if direcaoCandle != 0 and direcaoModelo != 0 and direcaoModelo != direcaoCandle
    penalty += 5.0

penalty :=
     math.min(
         penalty,
         penaltyMaximo
     )


// ============================================================================
// PENALTY EXTRA
// ============================================================================

float penaltyConfluenciaExtra = 0.0

if confirmacoes == 0
    penaltyConfluenciaExtra := 8.0
else if confirmacoes == 1 and divergencias >= 4
    penaltyConfluenciaExtra := 7.0
else if confirmacoes == 2 and divergencias >= 4
    penaltyConfluenciaExtra := 5.0
else if confirmacoes == 3 and divergencias >= 4
    penaltyConfluenciaExtra := 3.0

penalty += penaltyConfluenciaExtra

penalty :=
     math.min(
         penalty,
         penaltyMaximo
     )


// ============================================================================
// INTENSIDADE FINAL
// ============================================================================

intensidadeFinal =
     intensidadeBase +
     boostTotal -
     penalty

intensidadeFinal :=
     math.min(
         math.max(
             intensidadeFinal,
             0.0
         ),
         100.0
     )


// ============================================================================
// INTENSIDADE ASSINADA
// ============================================================================

intensidadeAssinada =
     direcaoIntensidade > 0 ?
     intensidadeFinal :
     direcaoIntensidade < 0 ?
     -intensidadeFinal :
     0.0


// ============================================================================
// CLASSIFICAÇÃO
// ============================================================================

string classificacao = "M FRACA"

if intensidadeFinal >= 90
    classificacao := "EXTREMA"
else if intensidadeFinal >= 80
    classificacao := "M FORTE"
else if intensidadeFinal >= 70
    classificacao := "FORTE"
else if intensidadeFinal >= 60
    classificacao := "MODERADA"
else if intensidadeFinal >= 40
    classificacao := "FRACA"


// ============================================================================
// QUALIDADE CONFLUÊNCIA
// ============================================================================

string qualidadeConfluencia = "FRACA"

if confirmacoes >= 7 and divergencias == 0
    qualidadeConfluencia := "EXTREMA"
else if confirmacoes >= 6 and divergencias <= 1
    qualidadeConfluencia := "M ALTA"
else if confirmacoes >= 5 and divergencias <= 2
    qualidadeConfluencia := "ALTA"
else if confirmacoes >= 4 and divergencias <= 3
    qualidadeConfluencia := "BOA"
else if confirmacoes >= 3 and divergencias <= 4
    qualidadeConfluencia := "MODERADA"
else if confirmacoes <= 1 and divergencias >= 4
    qualidadeConfluencia := "M FRACA"


// ============================================================================
// NÍVEL MODELO
// ============================================================================

string nivelModelo = "FRACO"

if forcaModelo >= 80
    nivelModelo := "M FORTE"
else if forcaModelo >= 70
    nivelModelo := "FORTE"
else if forcaModelo >= 60
    nivelModelo := "MODERADO"


// ============================================================================
// BUY / SELL
// ============================================================================

string vencedor = "BUY"


// ============================================================================
// CORES
// ============================================================================

corDirecao =
     direcaoIntensidade > 0 ?
     corVerde :
     direcaoIntensidade < 0 ?
     corVermelho :
     corCinza

corModelo =
     direcaoModelo > 0 ?
     corVerde :
     direcaoModelo < 0 ?
     corVermelho :
     corCinza

corVencedor = corVerde

corClassificacao =
     intensidadeFinal >= 90 ?
     corVerdeForte :
     intensidadeFinal >= 70 ?
     corVerde :
     intensidadeFinal >= 60 ?
     corAmarelo :
     intensidadeFinal >= 40 ?
     corAmarelo :
     corCinza


// ============================================================================
// COR CONFLUÊNCIA

f_barra(valor) =>
    tamanho = 8
    v = math.min(
         math.max(
             valor,
             0.0
         ),
         100.0
    )

    preenchido =
         int(
             math.round(
                 v /
                 100.0 *
                 tamanho
             )
         )

    resultado = ""

    for i = 0 to tamanho - 1
        resultado +=
             i < preenchido ?
             "█" :
             "░"

    resultado
